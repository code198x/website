### A Glimpse of What's Coming

Here's a raster interrupt - the foundation of every impressive C64 effect:

```6502
; This is Tier 2 material - don't worry if it seems alien
irq_handler:
    pha
    txa
    pha
    tya
    pha
    
    lda #$1b        ; Wait for line $1b
    cmp $d012
    bne *-3
    
    ; Now we're at exactly raster line $1b
    lda #1          ; Change border color
    sta $d020
    
    ; Do cycle-exact timing here
    nop
    nop
    nop
    
    lda #0          ; Change it back
    sta $d020
    
    lda #1          ; Acknowledge interrupt
    sta $d019
    
    pla
    tay
    pla
    tax
    pla
    rti             ; Return from interrupt
```

This simple routine creates a colored line across the screen - the beginning of split-screen effects, parallax scrolling, and impossible graphics. In Tier 2, you'll understand every line.

### The Sprite Multiplexer Preview

```6502
; Professional games display 32+ sprites using 8 hardware sprites
; Here's the concept (simplified):
sprite_multiplex:
    ; Sort sprites by Y position
    jsr sort_sprites
    
    ; Set up first 8 sprites
    ldx #0
setup_loop:
    lda sprite_y,x
    sta $d001,x*2       ; Hardware sprite Y
    lda sprite_x,x
    sta $d000,x*2       ; Hardware sprite X
    inx
    cpx #8
    bne setup_loop
    
    ; Wait for sprite to pass
    ; Reuse hardware sprite for next virtual sprite
    ; This happens multiple times per frame!
    rts
```

### Memory Banking Magic

```6502
; The C64 has 64KB RAM but the 6502 can only see 64KB total
; Banking lets you swap memory regions:
switch_bank:
    lda $01         ; Processor port
    and #%11111100  ; Clear banking bits
    ora #%00000101  ; RAM visible at $a000-$bfff
    sta $01
    ; Now you have 8KB more RAM visible!
    rts
```

### Tier 2 Preview: Your First Raster Effect

```6502
; This is what you'll understand completely in Tier 2
; A simple raster bar effect - the gateway to C64 magic

        sei             ; Disable interrupts
        
        lda #$7f
        sta $dc0d       ; Disable CIA interrupts
        sta $dd0d
        
        lda #$01
        sta $d01a       ; Enable raster interrupt
        
        lda #<irq       ; Set interrupt vector
        sta $0314
        lda #>irq
        sta $0315
        
        lda #$00        ; Trigger at line 0
        sta $d012
        lda $d011
        and #$7f
        sta $d011
        
        cli             ; Enable interrupts
        rts

irq:
        lda $d012       ; Wait for exact line
wait:   cmp $d012
        beq wait
        
        ; Now we're synchronized to the electron beam!
        ldx #0
raster_loop:
        lda colors,x    ; Color table
        sta $d020       ; Border
        sta $d021       ; Background
        
        ; Waste exactly 63 cycles for next line
        ldy #9
delay:  dey
        bne delay
        nop
        
        inx
        cpx #32         ; 32 lines of colors
        bne raster_loop
        
        lda #$01        ; Acknowledge interrupt
        sta $d019
        
        jmp $ea31       ; System IRQ handler

colors:
        .byte $06,$06,$04,$04,$0e,$0e,$03,$03
        .byte $01,$01,$03,$03,$0e,$0e,$04,$04
        .byte $04,$04,$0e,$0e,$03,$03,$01,$01
        .byte $03,$03,$0e,$0e,$04,$04,$06,$06
```

When this runs, you'll see smooth color gradients flowing down your screen - impossible without cycle-exact timing. This is just the beginning.
