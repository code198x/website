# Advanced Graphics Techniques

Push the VIC-II chip beyond its limits. Create high-resolution graphics, multicolor bitmaps, and effects that seem impossible on 1982 hardware.

## Bitmap Mode

Full screen pixel control:

```asm
; Enable bitmap mode
; Bitmap at $2000, screen at $0400
setup_bitmap:
    ; Set bitmap address
    lda #$3B        ; Bitmap mode on
    sta $D011

    lda #$18        ; Bitmap at $2000, screen at $0400
    sta $D018

    ; Clear bitmap (8000 bytes)
    lda #0
    ldx #0
clear_bitmap:
    sta $2000,x
    sta $2100,x
    sta $2200,x
    sta $2300,x
    sta $2400,x
    sta $2500,x
    sta $2600,x
    sta $2700,x
    inx
    bne clear_bitmap

    ; Continue for all 8000 bytes...
    rts

; Plot pixel in bitmap
; X = x coordinate (0-319)
; Y = y coordinate (0-199)
plot_pixel:
    ; Save coordinates
    stx plot_x
    sty plot_y

    ; Calculate byte address
    ; Address = $2000 + (Y & $F8) * 40 + (Y & 7) + (X & $1F8)

    tya
    and #$F8        ; Y & $F8
    sta temp
    asl
    asl             ; ×4
    clc
    adc temp        ; ×5
    asl
    asl
    asl             ; ×40 total
    sta bitmap_addr

    tya
    and #$07        ; Y & 7
    clc
    adc bitmap_addr
    sta bitmap_addr

    txa
    and #$F8        ; X & $1F8
    clc
    adc bitmap_addr
    sta bitmap_addr
    lda #$20        ; High byte
    adc #0
    sta bitmap_addr+1

    ; Calculate bit position
    txa
    and #$07        ; X & 7
    tax
    lda bit_table,x

    ; Set pixel
    ldy #0
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts

bit_table:
    .byte $80,$40,$20,$10,$08,$04,$02,$01

plot_x:       .byte 0
plot_y:       .byte 0
bitmap_addr:  .word 0
temp:         .byte 0
```

## Line Drawing

Bresenham's algorithm for lines:

```asm
; Draw line from (x0,y0) to (x1,y1)
draw_line:
    ; Calculate deltas
    lda x1
    sec
    sbc x0
    bpl dx_positive

    ; dx is negative
    eor #$FF
    clc
    adc #1
    sta dx
    lda #$FF
    sta x_inc
    jmp calc_dy

dx_positive:
    sta dx
    lda #1
    sta x_inc

calc_dy:
    lda y1
    sec
    sbc y0
    bpl dy_positive

    ; dy is negative
    eor #$FF
    clc
    adc #1
    sta dy
    lda #$FF
    sta y_inc
    jmp start_line

dy_positive:
    sta dy
    lda #1
    sta y_inc

start_line:
    ; Bresenham's algorithm
    lda dx
    cmp dy
    bcs x_major

y_major:
    ; dy > dx
    lda dy
    lsr
    sta error

y_loop:
    ldx x0
    ldy y0
    jsr plot_pixel

    lda y0
    cmp y1
    beq line_done

    lda error
    sec
    sbc dx
    sta error
    bpl no_x_step_y

    lda error
    clc
    adc dy
    sta error

    lda x0
    clc
    adc x_inc
    sta x0

no_x_step_y:
    lda y0
    clc
    adc y_inc
    sta y0
    jmp y_loop

x_major:
    ; dx >= dy
    lda dx
    lsr
    sta error

x_loop:
    ldx x0
    ldy y0
    jsr plot_pixel

    lda x0
    cmp x1
    beq line_done

    lda error
    sec
    sbc dy
    sta error
    bpl no_y_step_x

    lda error
    clc
    adc dx
    sta error

    lda y0
    clc
    adc y_inc
    sta y0

no_y_step_x:
    lda x0
    clc
    adc x_inc
    sta x0
    jmp x_loop

line_done:
    rts

x0:     .byte 0
y0:     .byte 0
x1:     .byte 100
y1:     .byte 50
dx:     .byte 0
dy:     .byte 0
x_inc:  .byte 0
y_inc:  .byte 0
error:  .byte 0
```

## Multicolor Bitmap

160×200 with 4 colors per cell:

```asm
; Multicolor bitmap mode
setup_multicolor:
    lda #$3B        ; Bitmap mode
    sta $D011

    lda #$18        ; Multicolor on
    sta $D016

    lda #$18        ; Bitmap at $2000
    sta $D018

    ; Set up color RAM
    ; Each cell can have:
    ; - Background color ($D021)
    ; - Upper nibble of screen RAM
    ; - Lower nibble of screen RAM
    ; - Color RAM

    ldx #0
set_colors:
    lda #$12        ; Colors 1 and 2
    sta $0400,x
    sta $0500,x
    sta $0600,x
    sta $06E8,x

    lda #$07        ; Color 3
    sta $D800,x
    sta $D900,x
    sta $DA00,x
    sta $DAE8,x

    inx
    bne set_colors
    rts

; Plot multicolor pixel (2×1 pixels)
plot_multicolor:
    ; X = x coordinate (0-159)
    ; Y = y coordinate (0-199)
    ; A = color (0-3)

    pha             ; Save color

    ; Calculate byte position
    txa
    asl             ; ×2 for hires coordinates
    tax
    jsr calculate_bitmap_addr

    ; Get bit pair position
    txa
    lsr
    and #$03        ; Which pair (0-3)
    tax

    pla             ; Get color
    asl
    asl
    cpx #3
    beq pair3
    cpx #2
    beq pair2
    cpx #1
    beq pair1

pair0:
    ; Bits 6-7
    asl
    asl
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts

pair1:
    ; Bits 4-5
    asl
    asl
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts

pair2:
    ; Bits 2-3
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts

pair3:
    ; Bits 0-1
    lsr
    lsr
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts
```

## Character Mode Graphics

Extended color and multicolor text:

```asm
; Extended color mode (4 background colors)
extended_color_mode:
    lda $D011
    ora #$40        ; ECM bit
    sta $D011

    ; Set 4 background colors
    lda #$00        ; Black
    sta $D021
    lda #$01        ; White
    sta $D022
    lda #$02        ; Red
    sta $D023
    lda #$05        ; Green
    sta $D024

    ; Character codes 0-63 use different backgrounds
    ; 00-3F: Background 0
    ; 40-7F: Background 1
    ; 80-BF: Background 2
    ; C0-FF: Background 3
    rts

; Multicolor character mode
multicolor_char_mode:
    lda $D016
    ora #$10        ; MCM bit
    sta $D016

    ; Characters with color > 7 are multicolor
    ldx #0
setup_mc_chars:
    lda #$08        ; Make multicolor
    sta $D800,x
    sta $D900,x
    sta $DA00,x
    sta $DAE8,x
    inx
    bne setup_mc_chars

    ; Set shared colors
    lda #$01        ; White
    sta $D022
    lda #$02        ; Red
    sta $D023
    rts
```

## Sprite Overlays

Combine sprites for high-res overlays:

```asm
; Overlay sprites for extra colors
sprite_overlay:
    ; Use two sprites at same position
    ; Different colors = more colors!

    ; Sprite 0: Base layer
    lda #100
    sta $D000       ; X position
    lda #100
    sta $D001       ; Y position
    lda #$01        ; White
    sta $D027

    ; Sprite 1: Overlay (use OR/AND logic)
    lda #100
    sta $D002
    lda #100
    sta $D003
    lda #$02        ; Red
    sta $D028

    ; Priority: sprite 1 behind sprite 0
    lda #$01
    sta $D01B

    ; Create composite colors
    rts

; Sprite-based text overlay
sprite_text:
    ; 8 sprites = 24×21 pixels each
    ; Can create 192×21 pixel text banner

    ldx #0
position_text_sprites:
    txa
    asl
    asl
    asl             ; ×8
    asl
    asl             ; ×32
    clc
    adc #24         ; Start X
    sta $D000,x

    lda #50         ; Y position
    sta $D001,x

    ; Point to character sprite
    txa
    clc
    adc #$80        ; Sprite pointers
    sta $07F8,x

    inx
    inx             ; Next sprite
    cpx #16
    bne position_text_sprites

    ; Enable all sprites
    lda #$FF
    sta $D015
    rts
```

## Fill Algorithms

Fast area filling:

```asm
; Stack-based flood fill
flood_fill:
    ; X,Y = start position
    ; A = fill color

    sta fill_color

    ; Get original color
    jsr get_pixel
    sta original_color
    cmp fill_color
    beq fill_done   ; Already filled

    ; Push starting point
    txa
    pha
    tya
    pha

fill_loop:
    ; Pop point from stack
    tsx
    cpx #$FF
    beq fill_done   ; Stack empty

    pla
    tay
    pla
    tax

    ; Check if we should fill
    jsr get_pixel
    cmp original_color
    bne fill_loop   ; Different color

    ; Fill this pixel
    lda fill_color
    jsr set_pixel

    ; Push adjacent pixels
    ; Right
    inx
    cpx #160        ; Boundary check
    bcs skip_right
    txa
    pha
    tya
    pha
skip_right:

    ; Left
    dex
    dex
    bpl push_left
    inx
    jmp skip_left
push_left:
    txa
    pha
    tya
    pha
skip_left:
    inx             ; Restore X

    ; Down
    iny
    cpy #200
    bcs skip_down
    txa
    pha
    tya
    pha
skip_down:

    ; Up
    dey
    dey
    bpl push_up
    iny
    jmp fill_loop
push_up:
    txa
    pha
    tya
    pha
    iny             ; Restore Y

    jmp fill_loop

fill_done:
    rts

fill_color:     .byte 0
original_color: .byte 0
```

## Circle and Ellipse Drawing

Draw perfect curves using Bresenham's algorithm:

```asm
; Bresenham circle algorithm
draw_circle:
    ; center_x, center_y = center point
    ; radius = radius

    lda #0
    sta x_pos
    lda radius
    sta y_pos

    ; Decision parameter
    lda #3
    sec
    sbc radius
    sbc radius
    sta decision

circle_loop:
    ; Draw 8 symmetric points
    jsr plot_circle_points

    ; Check if done
    lda x_pos
    cmp y_pos
    bcs circle_done

    ; Update decision parameter
    lda decision
    bmi circle_case1

circle_case2:
    ; decision >= 0
    dec y_pos

    ; decision = decision + 4*(x-y) + 10
    lda x_pos
    sec
    sbc y_pos
    asl
    asl
    clc
    adc #10
    clc
    adc decision
    sta decision
    jmp circle_continue

circle_case1:
    ; decision < 0
    ; decision = decision + 4*x + 6
    lda x_pos
    asl
    asl
    clc
    adc #6
    clc
    adc decision
    sta decision

circle_continue:
    inc x_pos
    jmp circle_loop

circle_done:
    rts

plot_circle_points:
    ; Plot 8 symmetric points
    ; (x,y), (y,x), (-x,y), (y,-x)
    ; (-x,-y), (-y,-x), (x,-y), (-y,x)

    ; Point 1: (cx+x, cy+y)
    lda center_x
    clc
    adc x_pos
    tax
    lda center_y
    clc
    adc y_pos
    tay
    jsr plot_pixel

    ; Point 2: (cx+y, cy+x)
    lda center_x
    clc
    adc y_pos
    tax
    lda center_y
    clc
    adc x_pos
    tay
    jsr plot_pixel

    ; Point 3: (cx-x, cy+y)
    lda center_x
    sec
    sbc x_pos
    tax
    lda center_y
    clc
    adc y_pos
    tay
    jsr plot_pixel

    ; Point 4: (cx-y, cy+x)
    lda center_x
    sec
    sbc y_pos
    tax
    lda center_y
    clc
    adc x_pos
    tay
    jsr plot_pixel

    ; Point 5: (cx+x, cy-y)
    lda center_x
    clc
    adc x_pos
    tax
    lda center_y
    sec
    sbc y_pos
    tay
    jsr plot_pixel

    ; Point 6: (cx+y, cy-x)
    lda center_x
    clc
    adc y_pos
    tax
    lda center_y
    sec
    sbc x_pos
    tay
    jsr plot_pixel

    ; Point 7: (cx-x, cy-y)
    lda center_x
    sec
    sbc x_pos
    tax
    lda center_y
    sec
    sbc y_pos
    tay
    jsr plot_pixel

    ; Point 8: (cx-y, cy-x)
    lda center_x
    sec
    sbc y_pos
    tax
    lda center_y
    sec
    sbc x_pos
    tay
    jsr plot_pixel

    rts

center_x: .byte 160
center_y: .byte 100
radius:   .byte 50
x_pos:    .byte 0
y_pos:    .byte 0
decision: .byte 0

; Draw filled circle
draw_filled_circle:
    lda #0
    sta x_pos
    lda radius
    sta y_pos

    lda #3
    sec
    sbc radius
    sbc radius
    sta decision

filled_circle_loop:
    ; Draw horizontal lines instead of points
    jsr draw_circle_lines

    lda x_pos
    cmp y_pos
    bcs filled_circle_done

    lda decision
    bmi filled_case1

    dec y_pos
    lda x_pos
    sec
    sbc y_pos
    asl
    asl
    clc
    adc #10
    clc
    adc decision
    sta decision
    jmp filled_continue

filled_case1:
    lda x_pos
    asl
    asl
    clc
    adc #6
    clc
    adc decision
    sta decision

filled_continue:
    inc x_pos
    jmp filled_circle_loop

filled_circle_done:
    rts

draw_circle_lines:
    ; Draw horizontal lines for filled circle

    ; Line from (cx-x) to (cx+x) at cy+y
    lda center_x
    sec
    sbc x_pos
    sta x0
    lda center_x
    clc
    adc x_pos
    sta x1
    lda center_y
    clc
    adc y_pos
    sta y0
    sta y1
    jsr draw_line

    ; Line from (cx-x) to (cx+x) at cy-y
    lda center_y
    sec
    sbc y_pos
    sta y0
    sta y1
    jsr draw_line

    ; Line from (cx-y) to (cx+y) at cy+x
    lda center_x
    sec
    sbc y_pos
    sta x0
    lda center_x
    clc
    adc y_pos
    sta x1
    lda center_y
    clc
    adc x_pos
    sta y0
    sta y1
    jsr draw_line

    ; Line from (cx-y) to (cx+y) at cy-x
    lda center_y
    sec
    sbc x_pos
    sta y0
    sta y1
    jsr draw_line

    rts
```

## 3D Wireframe

Simple 3D projection:

```asm
; 3D to 2D projection
; Project 3D point (x,y,z) to screen
project_3d:
    ; Perspective projection
    ; screen_x = x * 128 / (z + 128) + 160
    ; screen_y = y * 128 / (z + 128) + 100

    ; Calculate divisor (z + 128)
    lda z_coord
    clc
    adc #128
    sta divisor

    ; Project X
    lda x_coord
    ldx #128
    jsr multiply
    ldx divisor
    jsr divide
    clc
    adc #160        ; Center X
    sta screen_x

    ; Project Y
    lda y_coord
    ldx #128
    jsr multiply
    ldx divisor
    jsr divide
    clc
    adc #100        ; Center Y
    sta screen_y

    rts

; Draw wireframe cube
draw_cube:
    ; Define cube vertices
    ; Front face
    lda #-20
    sta x_coord
    sta y_coord
    lda #20
    sta z_coord
    jsr project_3d
    ldx screen_x
    ldy screen_y
    jsr plot_pixel

    ; ... project and draw all 8 vertices

    ; Draw edges
    ; ... draw 12 edges connecting vertices

    rts

; Rotate cube
rotate_cube:
    ; Simple rotation around Y axis
    ldx #0
rotate_vertex:
    ; x' = x * cos - z * sin
    ; z' = x * sin + z * cos

    lda cube_x,x
    sta old_x
    lda cube_z,x
    sta old_z

    ; Simplified with fixed angle
    lda old_x
    sec
    sbc old_z
    lsr             ; Approximate cos/sin
    sta cube_x,x

    lda old_z
    clc
    adc old_x
    lsr
    sta cube_z,x

    inx
    cpx #8          ; 8 vertices
    bne rotate_vertex
    rts

x_coord:    .byte 0
y_coord:    .byte 0
z_coord:    .byte 0
divisor:    .byte 0
screen_x:   .byte 0
screen_y:   .byte 0
old_x:      .byte 0
old_z:      .byte 0

cube_x: .res 8
cube_y: .res 8
cube_z: .res 8
```

## Texture Mapping

Simple texture mapping on sprites:

```asm
; Texture mapping for sprites
texture_map:
    ; Map 8x8 texture to sprite

    ; Source texture pointer
    lda #<texture_data
    sta tex_ptr
    lda #>texture_data
    sta tex_ptr+1

    ; Destination sprite pointer
    lda #<sprite_data
    sta spr_ptr
    lda #>sprite_data
    sta spr_ptr+1

    ; Scale factors (fixed point 8.8)
    lda #$0100      ; 1.0 scale
    sta scale_x
    lda #$0100
    sta scale_y

    ldy #0          ; Sprite Y
tex_y_loop:
    ; Calculate texture Y
    tya
    ldx scale_y
    jsr multiply_scale
    sta tex_y

    ldx #0          ; Sprite X
tex_x_loop:
    ; Calculate texture X
    txa
    ldx scale_x
    jsr multiply_scale
    sta tex_x

    ; Get texture pixel
    jsr get_texture_pixel

    ; Set sprite pixel
    jsr set_sprite_pixel

    inx
    cpx #24         ; Sprite width
    bne tex_x_loop

    iny
    cpy #21         ; Sprite height
    bne tex_y_loop

    rts

get_texture_pixel:
    ; Get pixel from texture
    lda tex_y
    and #$07        ; Wrap to 8x8
    asl
    asl
    asl             ; *8
    clc
    adc tex_x
    and #$07
    tay
    lda (tex_ptr),y
    rts

set_sprite_pixel:
    ; Set pixel in sprite
    ; Calculate byte position
    tya
    sta temp_y
    txa
    lsr
    lsr
    lsr             ; /8 for byte
    sta temp_byte

    ; Calculate bit position
    txa
    and #$07
    tax
    lda bit_masks,x
    sta temp_mask

    ; Calculate sprite offset
    lda temp_y
    asl
    clc
    adc temp_y      ; *3 (3 bytes per row)
    clc
    adc temp_byte
    tay

    ; Set or clear bit
    lda sprite_pixel
    beq clear_pixel

    lda (spr_ptr),y
    ora temp_mask
    sta (spr_ptr),y
    rts

clear_pixel:
    lda temp_mask
    eor #$FF
    sta temp_mask
    lda (spr_ptr),y
    and temp_mask
    sta (spr_ptr),y
    rts

tex_ptr:    .word 0
spr_ptr:    .word 0
tex_x:      .byte 0
tex_y:      .byte 0
scale_x:    .word $0100
scale_y:    .word $0100
temp_y:     .byte 0
temp_byte:  .byte 0
temp_mask:  .byte 0
sprite_pixel: .byte 0

bit_masks:
    .byte $80,$40,$20,$10,$08,$04,$02,$01

texture_data:
    ; 8x8 checkerboard pattern
    .byte %10101010
    .byte %01010101
    .byte %10101010
    .byte %01010101
    .byte %10101010
    .byte %01010101
    .byte %10101010
    .byte %01010101

sprite_data: .res 63
```

## Particle Systems

Create dynamic particle effects:

```asm
; Particle system
MAX_PARTICLES = 32

particle_x:     .res MAX_PARTICLES
particle_y:     .res MAX_PARTICLES
particle_vx:    .res MAX_PARTICLES  ; Velocity X
particle_vy:    .res MAX_PARTICLES  ; Velocity Y
particle_life:  .res MAX_PARTICLES
particle_color: .res MAX_PARTICLES
particle_active:.res MAX_PARTICLES

init_particles:
    ldx #0
init_part_loop:
    lda #0
    sta particle_active,x
    sta particle_life,x
    inx
    cpx #MAX_PARTICLES
    bne init_part_loop
    rts

spawn_particle:
    ; Find inactive particle
    ldx #0
find_inactive:
    lda particle_active,x
    beq found_inactive
    inx
    cpx #MAX_PARTICLES
    bne find_inactive
    rts             ; No free particles

found_inactive:
    ; Initialize particle
    lda spawn_x
    sta particle_x,x
    lda spawn_y
    sta particle_y,x

    ; Random velocity
    lda $D41B       ; Random
    and #$07
    sec
    sbc #4          ; -4 to +3
    sta particle_vx,x

    lda $D41B
    and #$07
    sec
    sbc #8          ; Upward bias
    sta particle_vy,x

    ; Set life and color
    lda #60         ; 60 frames
    sta particle_life,x
    lda spawn_color
    sta particle_color,x

    lda #1
    sta particle_active,x
    rts

update_particles:
    ldx #0
update_part_loop:
    lda particle_active,x
    beq next_particle

    ; Update position
    lda particle_x,x
    clc
    adc particle_vx,x
    sta particle_x,x

    lda particle_y,x
    clc
    adc particle_vy,x
    sta particle_y,x

    ; Apply gravity
    inc particle_vy,x

    ; Decrease life
    dec particle_life,x
    bne particle_alive

    ; Particle died
    lda #0
    sta particle_active,x
    jmp next_particle

particle_alive:
    ; Fade color based on life
    lda particle_life,x
    lsr
    lsr
    lsr             ; /8
    sta particle_color,x

next_particle:
    inx
    cpx #MAX_PARTICLES
    bne update_part_loop
    rts

draw_particles:
    ldx #0
draw_part_loop:
    lda particle_active,x
    beq skip_particle

    ; Convert to screen coordinates
    lda particle_x,x
    lsr             ; /2 for screen
    sta temp_x
    lda particle_y,x
    lsr
    sta temp_y

    ; Check bounds
    lda temp_x
    cmp #160
    bcs skip_particle
    lda temp_y
    cmp #200
    bcs skip_particle

    ; Plot particle
    stx saved_x
    ldx temp_x
    ldy temp_y
    jsr plot_pixel
    ldx saved_x

skip_particle:
    inx
    cpx #MAX_PARTICLES
    bne draw_part_loop
```

## Double Buffering

Smooth animation with page flipping:

```asm
; Double buffer system
front_buffer:   .byte 0  ; 0=$0400, 1=$0800
back_buffer:    .byte 1

setup_double_buffer:
    ; Clear both screens
    ldx #0
clear_buffers:
    lda #$20
    sta $0400,x
    sta $0500,x
    sta $0600,x
    sta $06E8,x
    sta $0800,x
    sta $0900,x
    sta $0A00,x
    sta $0AE8,x
    inx
    bne clear_buffers

    ; Set initial display
    lda #$15        ; Show $0400
    sta $D018
    rts

flip_buffers:
    ; Swap buffers
    lda front_buffer
    eor #$01
    sta front_buffer
    lda back_buffer
    eor #$01
    sta back_buffer

    ; Update display
    lda front_buffer
    beq show_0400

    lda #$25        ; Show $0800
    sta $D018
    rts

show_0400:
    lda #$15        ; Show $0400
    sta $D018
    rts

get_back_buffer:
    ; Return address of back buffer
    lda back_buffer
    beq buffer_0400

    lda #$08
    sta buffer_ptr+1
    lda #$00
    sta buffer_ptr
    rts

buffer_0400:
    lda #$04
    sta buffer_ptr+1
    lda #$00
    sta buffer_ptr
    rts

buffer_ptr: .word 0

; Draw to back buffer
draw_to_back:
    jsr get_back_buffer

    ; Draw operations here
    ldy #0
draw_loop:
    lda graphics_data,y
    sta (buffer_ptr),y
    iny
    bne draw_loop

    ; When done, flip
    jsr flip_buffers
    rts

graphics_data: .res 256
```

## Copper-style Effects

Mimic Amiga copper with raster tricks:

```asm
; Copper list emulation
copper_list:
    .byte 50, $20, $02  ; Line 50: Border red
    .byte 100, $20, $07 ; Line 100: Border yellow
    .byte 150, $20, $05 ; Line 150: Border green
    .byte 200, $20, $06 ; Line 200: Border blue
    .byte $FF           ; End marker

copper_index: .byte 0

copper_irq:
    pha
    txa
    pha

    ; Get current copper command
    ldx copper_index
    lda copper_list,x
    cmp #$FF
    beq copper_reset

    ; Wait for raster line
    sta $D012

    ; Get register
    inx
    lda copper_list,x
    sta copper_reg

    ; Get value
    inx
    lda copper_list,x
    ldx copper_reg
    cpx #$20
    beq set_d020
    cpx #$21
    beq set_d021
    jmp copper_next

set_d020:
    sta $D020
    jmp copper_next

set_d021:
    sta $D021

copper_next:
    ; Advance to next command
    lda copper_index
    clc
    adc #3
    sta copper_index

    ; Set next interrupt
    ldx copper_index
    lda copper_list,x
    cmp #$FF
    beq copper_reset
    sta $D012
    jmp copper_done

copper_reset:
    lda #0
    sta copper_index
    lda copper_list
    sta $D012

copper_done:
    asl $D019
    pla
    tax
    pla
    rti

copper_reg: .byte 0

; Advanced copper with gradients
copper_gradient:
    pha

    ; Smooth color gradient
    lda $D012
    lsr
    lsr
    and #$0F
    tax
    lda gradient_table,x
    sta $D020
    sta $D021

    ; Next line
    inc $D012
    lda $D012
    cmp #250
    bcc gradient_continue
    lda #50
    sta $D012

gradient_continue:
    asl $D019
    pla
    rti

gradient_table:
    .byte $00,$09,$02,$08
    .byte $0A,$0F,$07,$01
    .byte $01,$07,$0F,$0A
    .byte $08,$02,$09,$00
```
