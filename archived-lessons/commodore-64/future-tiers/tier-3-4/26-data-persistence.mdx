# Data Persistence and Storage

Games without saves are just temporary experiences. The ability to preserve progress, maintain high scores, and load saved games transformed gaming from brief sessions into ongoing adventures. Today you'll learn how C64 games stored data on disk and tape.

## What You'll Build

This lesson teaches you to create:

1. **High score system** that persists between sessions
2. **Save game functionality** with multiple slots
3. **Level data loading** from disk files
4. **Configuration storage** for game settings

These techniques powered everything from "Ultima" save games to "Summer Games" world records.

## Understanding C64 Storage

The C64 supports multiple storage devices:

- **1541 Disk Drive**: 170KB per side, intelligent drive with its own CPU
- **Datasette**: Cassette tape storage, slow but reliable
- **Cartridges**: Instant loading but read-only
- **REU**: RAM Expansion Unit for fast temporary storage

### The 1541 Disk Drive

The 1541 isn't just a storage device - it's a computer with:

- 6502 CPU (same as the C64!)
- 2KB of RAM
- Sophisticated DOS in ROM
- Serial communication with the C64

This intelligence enables custom fast loaders and copy protection schemes.

## KERNAL I/O System

The C64's KERNAL ROM provides standardized I/O routines that work across all devices:

```asm
; KERNAL I/O vectors
SETLFS = $FFBA  ; Set logical file parameters
SETNAM = $FFBD  ; Set filename
OPEN   = $FFC0  ; Open file
CLOSE  = $FFC3  ; Close file
CHKIN  = $FFC6  ; Set input channel
CHKOUT = $FFC9  ; Set output channel
CLRCHN = $FFCC  ; Clear channels
CHRIN  = $FFCF  ; Read byte
CHROUT = $FFD2  ; Write byte
LOAD   = $FFD5  ; Load file
SAVE   = $FFD8  ; Save file
GETIN  = $FFE4  ; Get byte (non-blocking)
READST = $FFB7  ; Read I/O status

; Device numbers
DEVICE_KEYBOARD = 0
DEVICE_TAPE = 1
DEVICE_SCREEN = 3
DEVICE_PRINTER = 4
DEVICE_DISK = 8
```

## Building a High Score System

Let's create a complete high score table with disk persistence:

```asm
; High Score Table System
; Maintains top 10 scores with names

    * = $0801
    ; BASIC stub
    !byte $0b,$08,$0a,$00,$9e
    !text "2061"
    !byte $00,$00,$00

    * = $080d

; Constants
NUM_SCORES = 10
NAME_LENGTH = 8
SCORE_LENGTH = 3    ; 3 bytes for 24-bit score

; High score table structure
; Each entry: 8 bytes name + 3 bytes score = 11 bytes
highscore_table:
    ; Initialize with default scores
    !text "STEVE   "
    !byte $00,$10,$00  ; 4096
    !text "JASON   "
    !byte $00,$08,$00  ; 2048
    !text "SARAH   "
    !byte $00,$04,$00  ; 1024
    !text "MIKE    "
    !byte $00,$02,$00  ; 512
    !text "LISA    "
    !byte $00,$01,$00  ; 256
    !text "DAVE    "
    !byte $80,$00,$00  ; 128
    !text "ANNA    "
    !byte $40,$00,$00  ; 64
    !text "TOM     "
    !byte $20,$00,$00  ; 32
    !text "EMMA    "
    !byte $10,$00,$00  ; 16
    !text "C64     "
    !byte $08,$00,$00  ; 8

; Current game score
current_score: !byte 0,0,0
player_name: !text "        "

main:
    ; Load high scores from disk
    jsr load_highscores

    ; Display the table
    jsr display_highscores

    ; Simulate getting a new score
    lda #$00
    sta current_score
    lda #$05        ; Score = 1280
    sta current_score+1
    lda #$00
    sta current_score+2

    ; Check if it's a high score
    jsr check_highscore
    bcc no_highscore

    ; Get player name
    jsr enter_name

    ; Insert into table
    jsr insert_highscore

    ; Save updated table
    jsr save_highscores

    ; Redisplay
    jsr display_highscores

no_highscore:
    rts

; Load high scores from disk
load_highscores:
    ; Set file parameters
    lda #1          ; Logical file number
    ldx #DEVICE_DISK
    ldy #0          ; No command
    jsr SETLFS

    ; Set filename
    lda #highscore_filename_end - highscore_filename
    ldx #<highscore_filename
    ldy #>highscore_filename
    jsr SETNAM

    ; Try to open file
    jsr OPEN
    bcs load_error  ; File doesn't exist

    ; Select file for input
    ldx #1
    jsr CHKIN

    ; Read all scores
    ldx #0
read_scores:
    jsr CHRIN       ; Read byte
    sta highscore_table,x
    inx
    cpx #(NUM_SCORES * 11)
    bne read_scores

    ; Close file
    lda #1
    jsr CLOSE
    jsr CLRCHN

    clc             ; Success
    rts

load_error:
    ; File doesn't exist - use defaults
    sec
    rts

highscore_filename:
    !text "HIGHSCORES,S,R"
highscore_filename_end:

; Save high scores to disk
save_highscores:
    ; Delete old file first (scratch)
    jsr scratch_file

    ; Set file parameters for write
    lda #1
    ldx #DEVICE_DISK
    ldy #1          ; Write mode
    jsr SETLFS

    ; Set filename (with write flag)
    lda #save_filename_end - save_filename
    ldx #<save_filename
    ldy #>save_filename
    jsr SETNAM

    ; Open file
    jsr OPEN
    bcs save_error

    ; Select file for output
    ldx #1
    jsr CHKOUT

    ; Write all scores
    ldx #0
write_scores:
    lda highscore_table,x
    jsr CHROUT
    inx
    cpx #(NUM_SCORES * 11)
    bne write_scores

    ; Close file
    lda #1
    jsr CLOSE
    jsr CLRCHN

    clc
    rts

save_error:
    ; Handle disk error
    jsr show_disk_error
    sec
    rts

save_filename:
    !text "HIGHSCORES,S,W"
save_filename_end:

; Scratch (delete) existing file
scratch_file:
    lda #15         ; Command channel
    ldx #DEVICE_DISK
    ldy #15
    jsr SETLFS

    ; Scratch command
    lda #scratch_cmd_end - scratch_cmd
    ldx #<scratch_cmd
    ldy #>scratch_cmd
    jsr SETNAM

    jsr OPEN
    lda #15
    jsr CLOSE
    rts

scratch_cmd:
    !text "S0:HIGHSCORES"
scratch_cmd_end:

; Check if current score qualifies
; Output: C=1 if high score, X=position
check_highscore:
    ldx #0
check_loop:
    ; Calculate offset to score
    txa
    pha

    ; Multiply X by 11 for entry offset
    lda #0
    sta temp
    sta temp+1
    txa
    asl             ; *2
    asl             ; *4
    asl             ; *8
    sta temp
    txa
    asl             ; *2
    clc
    adc temp        ; *10
    clc
    adc x_reg       ; *11
    tax

    ; Compare scores (3 bytes, MSB first)
    lda current_score+2
    cmp highscore_table+8,x
    bcc not_higher
    bne is_higher

    lda current_score+1
    cmp highscore_table+9,x
    bcc not_higher
    bne is_higher

    lda current_score
    cmp highscore_table+10,x
    bcc not_higher

is_higher:
    pla
    tax
    sec             ; Found position
    rts

not_higher:
    pla
    tax
    inx
    cpx #NUM_SCORES
    bne check_loop

    clc             ; Not a high score
    rts

x_reg: .byte 0
temp: .word 0

; Insert score at position X
insert_highscore:
    stx insert_pos

    ; Shift lower scores down
    ldx #NUM_SCORES-2
shift_loop:
    cpx insert_pos
    bcc insert_now

    ; Calculate source and dest offsets
    txa
    pha

    ; Copy entry X to X+1
    lda #0
    sta src_offset
    sta src_offset+1
    txa
    ; Multiply by 11
    asl
    asl
    asl             ; *8
    sta src_offset
    txa
    asl             ; *2
    clc
    adc src_offset
    clc
    adc x_reg       ; *11
    sta src_offset

    ; Dest = src + 11
    clc
    adc #11
    sta dst_offset

    ; Copy 11 bytes
    ldy #0
copy_entry:
    ldx src_offset
    lda highscore_table,x
    ldx dst_offset
    sta highscore_table,x
    inc src_offset
    inc dst_offset
    iny
    cpy #11
    bne copy_entry

    pla
    tax
    dex
    jmp shift_loop

insert_now:
    ; Insert new entry at position
    ldx insert_pos

    ; Calculate offset
    lda #0
    sta temp
    txa
    asl
    asl
    asl             ; *8
    sta temp
    txa
    asl             ; *2
    clc
    adc temp
    clc
    adc x_reg       ; *11
    tax

    ; Copy name
    ldy #0
copy_name:
    lda player_name,y
    sta highscore_table,x
    inx
    iny
    cpy #NAME_LENGTH
    bne copy_name

    ; Copy score
    lda current_score
    sta highscore_table,x
    inx
    lda current_score+1
    sta highscore_table,x
    inx
    lda current_score+2
    sta highscore_table,x

    rts

insert_pos: .byte 0
src_offset: .byte 0
dst_offset: .byte 0

; Display high score table
display_highscores:
    ; Clear screen
    lda #147        ; Clear screen
    jsr CHROUT

    ; Title
    ldx #0
show_title:
    lda title_text,x
    beq show_scores
    jsr CHROUT
    inx
    jmp show_title

show_scores:
    ldx #0
    stx entry_num

display_loop:
    ; Position
    lda entry_num
    clc
    adc #'1'
    jsr CHROUT
    lda #'.'
    jsr CHROUT
    lda #' '
    jsr CHROUT

    ; Calculate table offset
    lda entry_num
    asl
    asl
    asl             ; *8
    sta temp
    lda entry_num
    asl             ; *2
    clc
    adc temp
    clc
    adc entry_num   ; *11
    tax

    ; Show name
    ldy #0
show_name:
    lda highscore_table,x
    jsr CHROUT
    inx
    iny
    cpy #NAME_LENGTH
    bne show_name

    lda #' '
    jsr CHROUT

    ; Show score (convert to decimal)
    lda highscore_table,x
    sta score_bytes
    inx
    lda highscore_table,x
    sta score_bytes+1
    inx
    lda highscore_table,x
    sta score_bytes+2

    jsr print_score

    ; New line
    lda #13
    jsr CHROUT

    inc entry_num
    lda entry_num
    cmp #NUM_SCORES
    bne display_loop

    rts

entry_num: .byte 0
score_bytes: .byte 0,0,0

title_text:
    !text "     HIGH SCORES",13,13,0

; Enter player name
enter_name:
    ; Prompt
    ldx #0
show_prompt:
    lda name_prompt,x
    beq get_name
    jsr CHROUT
    inx
    jmp show_prompt

get_name:
    ldx #0
name_loop:
    jsr GETIN
    beq name_loop   ; Wait for key

    cmp #13         ; Return?
    beq name_done

    cmp #20         ; Delete?
    bne add_char

    cpx #0
    beq name_loop
    dex
    lda #' '
    sta player_name,x
    jsr CHROUT
    jmp name_loop

add_char:
    cpx #NAME_LENGTH
    beq name_loop

    sta player_name,x
    jsr CHROUT
    inx
    jmp name_loop

name_done:
    ; Pad with spaces
pad_loop:
    cpx #NAME_LENGTH
    beq done_padding
    lda #' '
    sta player_name,x
    inx
    jmp pad_loop

done_padding:
    lda #13
    jsr CHROUT
    rts

name_prompt:
    !text 13,"ENTER YOUR NAME: ",0

; Print score in decimal
print_score:
    ; Simple hex to decimal for display
    ; (Real implementation would be more sophisticated)
    lda score_bytes+2
    jsr print_hex
    lda score_bytes+1
    jsr print_hex
    lda score_bytes
    jsr print_hex
    rts

print_hex:
    pha
    lsr
    lsr
    lsr
    lsr
    jsr print_nibble
    pla
    and #$0F
    jsr print_nibble
    rts

print_nibble:
    cmp #10
    bcc is_digit
    clc
    adc #'A'-10
    jmp print_it
is_digit:
    clc
    adc #'0'
print_it:
    jsr CHROUT
    rts
```

## Complete Save Game System

Now let's build a comprehensive save/load system with multiple slots:

```asm
; Save Game System with Multiple Slots
; Supports 4 save slots with validation

; Save slot structure (256 bytes each)
; Bytes 0-1: Magic number ($C0,$64)
; Byte 2: Version number
; Byte 3: Checksum
; Bytes 4-255: Game data

SAVE_MAGIC1 = $C0
SAVE_MAGIC2 = $64
SAVE_VERSION = $01
SAVE_SIZE = 256
NUM_SLOTS = 4

; Save game to slot (0-3)
; Input: A = slot number
save_game_slot:
    sta current_slot

    ; Build filename (SAVE0, SAVE1, etc.)
    clc
    adc #'0'
    sta slot_filename+4

    ; Pack game state
    jsr pack_save_data

    ; Calculate checksum
    jsr calc_save_checksum

    ; Delete old save
    jsr delete_save_slot

    ; Write new save
    lda #2
    ldx #DEVICE_DISK
    ldy #2
    jsr SETLFS

    lda #slot_write_end - slot_filename
    ldx #<slot_filename
    ldy #>slot_filename
    jsr SETNAM

    jsr OPEN
    bcs save_slot_error

    ldx #2
    jsr CHKOUT

    ; Write save data
    ldx #0
write_save_loop:
    lda save_buffer,x
    jsr CHROUT
    inx
    bne write_save_loop

    lda #2
    jsr CLOSE
    jsr CLRCHN

    ; Show success message
    jsr show_save_success
    clc
    rts

save_slot_error:
    jsr show_save_error
    sec
    rts

current_slot: .byte 0
slot_filename:
    !text "SAVE0,S,W"
slot_write_end:

; Load game from slot
; Input: A = slot number
load_game_slot:
    sta current_slot

    ; Build filename
    clc
    adc #'0'
    sta load_slot_name+4

    ; Open file
    lda #2
    ldx #DEVICE_DISK
    ldy #2
    jsr SETLFS

    lda #load_slot_end - load_slot_name
    ldx #<load_slot_name
    ldy #>load_slot_name
    jsr SETNAM

    jsr OPEN
    bcs load_slot_error

    ldx #2
    jsr CHKIN

    ; Read save data
    ldx #0
read_save_loop:
    jsr CHRIN
    sta save_buffer,x
    inx
    bne read_save_loop

    lda #2
    jsr CLOSE
    jsr CLRCHN

    ; Validate save
    jsr validate_save_data
    bcs invalid_save

    ; Unpack game state
    jsr unpack_save_data

    ; Show success
    jsr show_load_success
    clc
    rts

load_slot_error:
    ; Slot empty or disk error
    jsr show_load_error
    sec
    rts

invalid_save:
    jsr show_corrupt_save
    sec
    rts

load_slot_name:
    !text "SAVE0,S,R"
load_slot_end:

; Pack game state into save buffer
pack_save_data:
    ; Header
    lda #SAVE_MAGIC1
    sta save_buffer
    lda #SAVE_MAGIC2
    sta save_buffer+1
    lda #SAVE_VERSION
    sta save_buffer+2
    ; Checksum will go in byte 3

    ; Game data starts at byte 4
    ldx #4

    ; Player state
    lda player_x
    sta save_buffer,x
    inx
    lda player_y
    sta save_buffer,x
    inx
    lda player_health
    sta save_buffer,x
    inx
    lda player_lives
    sta save_buffer,x
    inx

    ; Inventory (16 bytes)
    ldy #0
pack_inventory:
    lda inventory,y
    sta save_buffer,x
    inx
    iny
    cpy #16
    bne pack_inventory

    ; Level progress
    lda current_level
    sta save_buffer,x
    inx
    lda level_flags
    sta save_buffer,x
    inx
    lda level_flags+1
    sta save_buffer,x
    inx

    ; Score (3 bytes)
    lda score
    sta save_buffer,x
    inx
    lda score+1
    sta save_buffer,x
    inx
    lda score+2
    sta save_buffer,x
    inx

    ; Time played (2 bytes)
    lda time_played
    sta save_buffer,x
    inx
    lda time_played+1
    sta save_buffer,x
    inx

    ; Game flags (8 bytes)
    ldy #0
pack_flags:
    lda game_flags,y
    sta save_buffer,x
    inx
    iny
    cpy #8
    bne pack_flags

    ; Custom data area
    ; ... add more as needed

    rts

; Unpack save data to game state
unpack_save_data:
    ; Skip header (already validated)
    ldx #4

    ; Player state
    lda save_buffer,x
    sta player_x
    inx
    lda save_buffer,x
    sta player_y
    inx
    lda save_buffer,x
    sta player_health
    inx
    lda save_buffer,x
    sta player_lives
    inx

    ; Inventory
    ldy #0
unpack_inventory:
    lda save_buffer,x
    sta inventory,y
    inx
    iny
    cpy #16
    bne unpack_inventory

    ; Level progress
    lda save_buffer,x
    sta current_level
    inx
    lda save_buffer,x
    sta level_flags
    inx
    lda save_buffer,x
    sta level_flags+1
    inx

    ; Score
    lda save_buffer,x
    sta score
    inx
    lda save_buffer,x
    sta score+1
    inx
    lda save_buffer,x
    sta score+2
    inx

    ; Time played
    lda save_buffer,x
    sta time_played
    inx
    lda save_buffer,x
    sta time_played+1
    inx

    ; Game flags
    ldy #0
unpack_flags:
    lda save_buffer,x
    sta game_flags,y
    inx
    iny
    cpy #8
    bne unpack_flags

    rts

; Calculate checksum
calc_save_checksum:
    lda #0
    sta checksum

    ldx #4          ; Start after header
calc_loop:
    lda save_buffer,x
    eor checksum
    sta checksum
    inx
    bne calc_loop

    ; Store in header
    lda checksum
    sta save_buffer+3
    rts

; Validate save data
validate_save_data:
    ; Check magic number
    lda save_buffer
    cmp #SAVE_MAGIC1
    bne invalid
    lda save_buffer+1
    cmp #SAVE_MAGIC2
    bne invalid

    ; Check version
    lda save_buffer+2
    cmp #SAVE_VERSION
    bne invalid

    ; Verify checksum
    lda #0
    sta checksum
    ldx #4
verify_loop:
    lda save_buffer,x
    eor checksum
    sta checksum
    inx
    bne verify_loop

    lda checksum
    cmp save_buffer+3
    bne invalid

    clc             ; Valid
    rts

invalid:
    sec             ; Invalid
    rts

checksum: .byte 0

; Delete save slot
delete_save_slot:
    lda #15
    ldx #DEVICE_DISK
    ldy #15
    jsr SETLFS

    ; Build scratch command
    lda current_slot
    clc
    adc #'0'
    sta delete_cmd+8

    lda #delete_cmd_end - delete_cmd
    ldx #<delete_cmd
    ldy #>delete_cmd
    jsr SETNAM

    jsr OPEN
    lda #15
    jsr CLOSE
    rts

delete_cmd:
    !text "S0:SAVE0"
delete_cmd_end:

; Save buffer and game variables
save_buffer: .res 256

; Game state variables (examples)
player_x: .byte 10
player_y: .byte 10
player_health: .byte 100
player_lives: .byte 3
inventory: .res 16
current_level: .byte 1
level_flags: .word 0
score: .byte 0,0,0
time_played: .word 0
game_flags: .res 8
```

## Disk Error Handling

Proper error handling is crucial for a professional experience:

```asm
; Read disk error channel
read_disk_error:
    lda #15         ; Error channel
    ldx #DEVICE_DISK
    ldy #15
    jsr SETLFS

    lda #0          ; No filename
    jsr SETNAM

    jsr OPEN

    ldx #15
    jsr CHKIN

    ; Read error message
    ldx #0
read_error_loop:
    jsr CHRIN
    sta error_buffer,x
    inx
    cmp #13         ; CR
    bne read_error_loop

    lda #0
    sta error_buffer-1,x

    lda #15
    jsr CLOSE
    jsr CLRCHN

    ; Parse error code
    lda error_buffer
    sec
    sbc #'0'
    asl
    asl
    asl
    asl             ; *10
    sta error_code
    lda error_buffer+1
    sec
    sbc #'0'
    clc
    adc error_code
    sta error_code

    rts

error_buffer: .res 40
error_code: .byte 0

; Common disk errors
; 00 = OK
; 20 = Block header not found
; 21 = No sync character
; 22 = Data block not present
; 23 = Checksum error in data
; 25 = Write verify error
; 26 = Write protect on
; 27 = Checksum error in header
; 29 = Disk ID mismatch
; 30 = General syntax error
; 62 = File not found
; 63 = File exists
; 70 = No channel available
; 72 = Disk full
; 73 = DOS version mismatch
; 74 = Drive not ready
```

## Fast Loader Implementation

The 1541's intelligence allows custom fast loaders that bypass the slow serial protocol:

```asm
; Simple fast load concept
; This runs on the drive's CPU!

drive_code:
    ; This code is uploaded to the 1541
    ; It reads sectors directly and sends
    ; data using custom protocol

    ; Drive memory map:
    ; $0000-$07FF: RAM
    ; $C000-$FFFF: ROM

    ; Read track/sector
    lda #18         ; Directory track
    sta $08         ; Track
    lda #1          ; First sector
    sta $09         ; Sector

    ; Execute ROM routine to read
    jsr $F50A       ; Read sector

    ; Send data to C64 fast
    ldx #0
send_fast:
    lda $0700,x     ; Buffer data
    sta $1800       ; Port register
    ; Custom timing for speed
    inx
    bne send_fast

    rts

; This is a simplified example
; Real fast loaders are much more complex!
```

## Level Data Loading

Loading level data efficiently:

```asm
; Load level from disk
; Input: A = level number
load_level:
    ; Convert level number to filename
    clc
    adc #'0'
    sta level_name+5

    ; Load to specific address
    lda #0          ; Use LOAD vector
    ldx #DEVICE_DISK
    ldy #1          ; Load to address
    jsr SETLFS

    lda #level_name_end - level_name

    ldy #>level_name
    jsr SETNAM

    lda #0          ; LOAD
    ldx #<level_data
    ldy #>level_data
    jsr LOAD
    bcs load_level_error

    ; Process loaded data
    jsr decompress_level
    jsr init_level
```

## Tape Storage

For completeness, here's cassette tape I/O:

```asm
; Save to tape
save_to_tape:
    lda #1          ; File number
    ldx #DEVICE_TAPE
    ldy #0
    jsr SETLFS

    lda #tape_name_end - tape_name
    ldx #<tape_name
    ldy #>tape_name
    jsr SETNAM

    ; Save memory range
    lda #<data_start
    sta $C1
    lda #>data_start
    sta $C2

    ldx #<data_end
    ldy #>data_end
    lda #$C1        ; Pointer location
    jsr SAVE

    rts

tape_name:
    !text "GAMESAVE"
tape_name_end:

    ; Your data here
```

## Historical Context

**"Ultima IV" (1985)** featured one of the most sophisticated save systems on the C64, tracking character stats, world state, and moral choices across multiple disk sides.

**"Pirates!" (1987)** by Sid Meier saved entire careers - your pirate's adventures, crew, ships, and reputation persisted across gaming sessions.

**"Wasteland" (1988)** required a separate save disk and modified the game disks themselves, permanently changing the game world based on player actions.

**Fast Loaders** like Epyx Fastload and Action Replay revolutionized C64 gaming by reducing load times from minutes to seconds using custom drive code.
