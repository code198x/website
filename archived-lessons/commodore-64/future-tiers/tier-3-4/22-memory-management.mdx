# Memory Management Mastery

The C64's 64KB seems tiny today, but clever memory management can make it feel infinite. Learn to squeeze every byte, swap banks, and organize memory for maximum efficiency.

## C64 Memory Map

Understanding the complete memory layout:

```asm
; C64 Memory Map
; $0000-$00FF - Zero page (fast access)
; $0100-$01FF - Stack
; $0200-$02FF - OS input buffer/free
; $0300-$03FF - OS vectors/free
; $0400-$07FF - Screen RAM (1000 bytes)
; $0800-$9FFF - BASIC RAM (38KB when BASIC on)
; $A000-$BFFF - BASIC ROM (8KB, can be switched out)
; $C000-$CFFF - Free RAM (4KB)
; $D000-$DFFF - I/O area (VIC, SID, CIA, can be switched)
; $E000-$FFFF - KERNAL ROM (8KB, can be switched out)

; Memory configuration register
; $01 - Processor port
; Bits 0-2 control memory banking:
; %x00 - RAM everywhere
; %x01 - RAM at A000-BFFF and E000-FFFF
; %x10 - RAM at A000-BFFF, KERNAL at E000-FFFF
; %x11 - BASIC at A000-BFFF, KERNAL at E000-FFFF
; %1xx - I/O at D000-DFFF
; %0xx - RAM at D000-DFFF

; Switch out BASIC for more RAM
disable_basic:
    lda $01
    and #$FE        ; Clear bit 0
    sta $01
    ; Now have RAM from $A000-$BFFF
    rts

; Switch out KERNAL too (careful!)
all_ram:
    sei             ; Disable interrupts first!
    lda #$35        ; Default with I/O
    sta $01
    and #$FD        ; Clear bit 1
    sta $01
    ; Now have 62KB RAM (except I/O)
    rts

; Restore normal memory
restore_memory:
    lda #$37        ; BASIC + KERNAL + I/O
    sta $01
    cli
    rts
```

### C64 Memory Banks

| Address Range | Name      | Type    | Description                            |
| ------------- | --------- | ------- | -------------------------------------- |
| $0000-$00FF   | Zero Page | RAM     | Fast access memory for variables       |
| $0100-$01FF   | Stack     | RAM     | System stack                           |
| $0400-$07FF   | Screen    | RAM     | Screen memory (1K)                     |
| $0800-$9FFF   | Program   | RAM     | Main program space                     |
| $A000-$BFFF   | BASIC ROM | ROM/RAM | BASIC interpreter (can be banked out)  |
| $D000-$DFFF   | I/O       | I/O/RAM | Hardware registers (can be banked out) |
| $E000-$FFFF   | KERNAL    | ROM/RAM | KERNAL ROM (can be banked out)         |

## Dynamic Memory Allocation

Simple heap management:

```asm
; Heap allocator
heap_start:     .word $C000  ; Start of heap
heap_ptr:       .word $C000  ; Current position
heap_end:       .word $D000  ; End of heap

; Allocate memory
; A/Y = size needed
; Returns: A/Y = pointer to memory (or 0 if failed)
malloc:
    ; Save size
    sta req_size_lo
    sty req_size_hi

    ; Check if enough space
    clc
    lda heap_ptr
    adc req_size_lo
    sta new_ptr_lo
    lda heap_ptr+1
    adc req_size_hi
    sta new_ptr_hi

    ; Compare with heap end
    cmp heap_end+1
    bcc space_ok
    bne no_space
    lda new_ptr_lo
    cmp heap_end
    bcs no_space

space_ok:
    ; Return current pointer
    lda heap_ptr
    ldy heap_ptr+1

    ; Update heap pointer
    ldx new_ptr_lo
    stx heap_ptr
    ldx new_ptr_hi
    stx heap_ptr+1

    rts

no_space:
    lda #0
    tay
    rts

; Free all memory (reset heap)
free_all:
    lda heap_start
    sta heap_ptr
    lda heap_start+1
    sta heap_ptr+1
    rts

req_size_lo:    .byte 0
req_size_hi:    .byte 0
new_ptr_lo:     .byte 0
new_ptr_hi:     .byte 0

; Memory block header structure
; For more complex allocation with free()
BLOCK_SIZE      = 0  ; 2 bytes
BLOCK_NEXT      = 2  ; 2 bytes pointer
BLOCK_FLAGS     = 4  ; 1 byte (bit 7 = used)
BLOCK_DATA      = 5  ; Start of user data

; Linked list allocator
init_heap:
    ; Create first free block
    lda heap_start
    sta $FB
    lda heap_start+1
    sta $FC

    ; Size = total heap size
    ldy #BLOCK_SIZE
    lda #<4000      ; Example: 4000 bytes
    sta ($FB),y
    iny
    lda #>4000
    sta ($FB),y

    ; Next = null
    ldy #BLOCK_NEXT
    lda #0
    sta ($FB),y
    iny
    sta ($FB),y

    ; Flags = free
    ldy #BLOCK_FLAGS
    lda #0
    sta ($FB),y
    rts
```

## Memory Pools

Fixed-size allocation for efficiency:

```asm
; Object pools for sprites, bullets, etc
POOL_SIZE       = 16
OBJECT_SIZE     = 8

; Bullet pool
bullet_pool:    .res POOL_SIZE * OBJECT_SIZE
bullet_free:    .res POOL_SIZE  ; Free list

init_bullet_pool:
    ; Mark all as free
    ldx #POOL_SIZE-1
    lda #$FF
init_pool_loop:
    sta bullet_free,x
    dex
    bpl init_pool_loop
    rts

alloc_bullet:
    ; Find free slot
    ldx #0
find_free:
    lda bullet_free,x
    bne found_free
    inx
    cpx #POOL_SIZE
    bne find_free

    ; No free slots
    ldx #$FF
    rts

found_free:
    ; Mark as used
    lda #0
    sta bullet_free,x

    ; Calculate address
    txa
    asl
    asl
    asl             ; ×8 for object size
    clc
    adc #<bullet_pool
    tay
    lda #>bullet_pool
    adc #0

    ; Return Y/A = address, X = index
    rts

free_bullet:
    ; X = bullet index
    lda #$FF
    sta bullet_free,x

    ; Optional: clear the data
    txa
    asl
    asl
    asl
    tax
    ldy #OBJECT_SIZE-1
    lda #0
clear_bullet:
    sta bullet_pool,x
    inx
    dey
    bpl clear_bullet
    rts

; Multi-size pool system
; ======================

POOL_8_SIZE   = 32    ; 32 8-byte objects
POOL_16_SIZE  = 16    ; 16 16-byte objects
POOL_32_SIZE  = 8     ; 8 32-byte objects
POOL_64_SIZE  = 4     ; 4 64-byte objects

pool_8:   .res POOL_8_SIZE * 8
pool_16:  .res POOL_16_SIZE * 16
pool_32:  .res POOL_32_SIZE * 32
pool_64:  .res POOL_64_SIZE * 64

pool_8_mask:  .res POOL_8_SIZE/8    ; Bit mask for free slots
pool_16_mask: .res POOL_16_SIZE/8
pool_32_mask: .res POOL_32_SIZE/8
pool_64_mask: .res POOL_64_SIZE/8

init_all_pools:
    ; Initialize all pool masks to $ff (all free)
    lda #$ff
    ldx #(POOL_8_SIZE/8)-1
.init8:
    sta pool_8_mask,x
    dex
    bpl .init8

    ldx #(POOL_16_SIZE/8)-1
.init16:
    sta pool_16_mask,x
    dex
    bpl .init16

    ; etc...
    rts

; Smart allocation - choose right pool
smart_alloc:
    ; A = size needed
    cmp #9
    bcc alloc_from_8
    cmp #17
    bcc alloc_from_16
    cmp #33
    bcc alloc_from_32
    cmp #65
    bcc alloc_from_64

    ; Too big for pools, use heap
    jmp malloc

alloc_from_8:
    ldx #<pool_8
    ldy #>pool_8
    lda #<pool_8_mask
    sta mask_ptr
    lda #>pool_8_mask
    sta mask_ptr+1
    lda #8
    jmp alloc_from_pool

alloc_from_16:
    ldx #<pool_16
    ldy #>pool_16
    lda #<pool_16_mask
    sta mask_ptr
    lda #>pool_16_mask
    sta mask_ptr+1
    lda #16
    jmp alloc_from_pool

alloc_from_pool:
    ; Generic pool allocator
    ; Find first free bit in mask
    sta object_size
    stx pool_base
    sty pool_base+1

    ldy #0
.find_byte:
    lda (mask_ptr),y
    bne .found_byte
    iny
    cpy #4              ; Max 32 objects
    bne .find_byte

    ; Pool full
    ldx #$ff
    rts

.found_byte:
    ; Find first set bit
    ldx #0
.find_bit:
    lsr
    bcs .found_bit
    inx
    cpx #8
    bne .find_bit

.found_bit:
    ; Calculate object index
    tya
    asl
    asl
    asl                 ; Byte * 8
    sta temp
    txa
    clc
    adc temp            ; + bit = index

    ; Clear bit in mask
    tax
    lda bit_clear_table,x
    and (mask_ptr),y
    sta (mask_ptr),y

    ; Calculate object address
    txa
    ldx object_size
.mul_loop:
    asl
    dex
    bne .mul_loop

    clc
    adc pool_base
    tay
    lda pool_base+1
    adc #0
    tax
    tya

    rts

mask_ptr:     .word 0
pool_base:    .word 0
object_size:  .byte 0
temp:         .byte 0

bit_clear_table:
    .byte $fe,$fd,$fb,$f7,$ef,$df,$bf,$7f
```

## Bank Switching

Access all 64KB:

```asm
; VIC bank switching (16KB banks)
; VIC can see one of 4 banks
set_vic_bank:
    ; A = bank (0-3)
    sta current_bank

    ; Set DD00 for VIC bank
    lda $DD00
    and #$FC        ; Clear bits 0-1
    ora current_bank
    eor #$03        ; Invert (hardware quirk)
    sta $DD00

    ; Update screen and charset pointers
    lda current_bank
    cmp #0
    beq bank0
    cmp #1
    beq bank1
    cmp #2
    beq bank2

bank3:  ; $C000-$FFFF
    lda #$3C        ; Screen at $F000, chars at $E000
    jmp set_pointers
bank2:  ; $8000-$BFFF
    lda #$2C        ; Screen at $B000, chars at $A000
    jmp set_pointers
bank1:  ; $4000-$7FFF
    lda #$1C        ; Screen at $7000, chars at $6000
    jmp set_pointers
bank0:  ; $0000-$3FFF (default)
    lda #$15        ; Screen at $0400, chars at $1000

set_pointers:
    sta $D018
    rts

current_bank: .byte 0

; Double buffering with bank switching
swap_screens:
    lda current_screen
    eor #$01
    sta current_screen

    beq show_screen1

show_screen2:
    lda #$80        ; Screen 2 at $8400
    sta $D018
    rts

show_screen1:
    lda #$10        ; Screen 1 at $0400
    sta $D018
    rts

current_screen: .byte 0
```

## Compression Techniques

Store more in less space:

```asm
; RLE decompression to save space
decompress_rle:
    ; $FB-FC = source
    ; $FD-FE = destination

    ldy #0
decomp_loop:
    lda ($FB),y     ; Get command byte
    bmi rle_run     ; Negative = run

    ; Literal copy
    tax
    inx             ; Count is value + 1
literal_copy:
    iny
    lda ($FB),y
    sta ($FD),y
    dex
    bne literal_copy
    jmp advance_dest

rle_run:
    and #$7F        ; Remove run flag
    tax
    inx             ; Count
    iny
    lda ($FB),y     ; Value to repeat
run_loop:
    sta ($FD),y
    inc $FD
    bne no_carry_d
    inc $FE
no_carry_d:
    dex
    bne run_loop

advance_dest:
    ; Update pointers...
    jmp decomp_loop

; Dictionary compression for text
dictionary:
    .text " the "   ; 0
    .text "ing "    ; 1
    .text "you "    ; 2
    .text " and "   ; 3
    ; etc...

decompress_text:
    ldy #0
text_loop:
    lda ($FB),y
    bpl normal_char

    ; Dictionary reference
    and #$7F
    asl
    asl             ; ×4 for 4-char entries
    tax

    ; Copy dictionary entry
    lda dictionary,x
    jsr output_char
    lda dictionary+1,x
    jsr output_char
    lda dictionary+2,x
    jsr output_char
    lda dictionary+3,x
    jsr output_char
    jmp next_char

normal_char:
    jsr output_char

next_char:
    iny
    jmp text_loop
```

## Memory-Mapped Structures

Efficient data organization:

```asm
; Structure of Arrays (SoA) vs Array of Structures (AoS)

; Bad: Array of Structures (cache unfriendly)
; enemy_data:
;   .byte x,y,health,type  ; Enemy 0
;   .byte x,y,health,type  ; Enemy 1
;   ...

; Good: Structure of Arrays (cache friendly)
enemy_x:      .res MAX_ENEMIES
enemy_y:      .res MAX_ENEMIES
enemy_health: .res MAX_ENEMIES
enemy_type:   .res MAX_ENEMIES

; Update all X positions efficiently
update_all_x:
    ldx #MAX_ENEMIES-1
update_x_loop:
    inc enemy_x,x
    dex
    bpl update_x_loop
    rts

; Align data for speed
.align 256          ; Page-align for no page crossing
sprite_data:
    .res 64         ; Sprite shape data

; Zero-page shadows of frequently accessed data
.alias player_x_zp = $F0
.alias player_y_zp = $F1
.alias score_zp    = $F2  ; 3 bytes

; Update shadows
sync_to_zp:
    lda player_x
    sta player_x_zp
    lda player_y
    sta player_y_zp
    ldx #2
sync_score:
    lda score,x
    sta score_zp,x
    dex
    bpl sync_score
    rts
```

```asm

OVERLAY_ADDR = $8000  ; Load overlays here

overlay_table:

    .word boss_overlay

load_overlay:
    ; A = overlay ID
    asl

    lda overlay_table,x

    rts

load_addr: .word 0

; Overlay header format

```

## Complete Memory System

Full memory management example:

```asm
* = $0801
    .byte $0C,$08,$0A,$00,$9E,$32,$30,$36,$31,$00,$00,$00

* = $080D
start:
    ; Optimize memory layout
    jsr setup_memory

    ; Initialize subsystems
    jsr init_pools
    jsr init_heap

main_loop:
    jsr game_update

    ; Memory pressure?
    lda heap_ptr+1
    cmp #$CF        ; Near end?
    bcc memory_ok

    ; Compact/garbage collect
    jsr compact_memory

memory_ok:
    jmp main_loop

setup_memory:
    ; Disable BASIC for more RAM
    sei
    lda #$36        ; I/O visible, BASIC out
    sta $01
    cli

    ; Set up memory regions
    ; $0800-$2FFF: Main code
    ; $3000-$3FFF: Sprite data
    ; $4000-$7FFF: Level data
    ; $8000-$9FFF: Music
    ; $A000-$BFFF: Overlays
    ; $C000-$CFFF: Heap
    ; $D000-$DFFF: I/O
    ; $E000-$FFFF: Reserved

    rts

compact_memory:
    ; Simple mark-and-sweep
    jsr mark_used_blocks
    jsr sweep_free_blocks
    jsr defragment
    rts

; Memory statistics
show_memory_stats:
    ; Calculate free memory
    sec
    lda heap_end
    sbc heap_ptr
    sta free_lo
    lda heap_end+1
    sbc heap_ptr+1
    sta free_hi

    ; Display
    ldx #0
    lda #'F'
    sta $0400,x
    lda #'R'
    sta $0401,x
    lda #'E'
    sta $0402,x
    lda #'E'
    sta $0403,x
    lda #':'
    sta $0404,x

    lda free_hi
    jsr show_hex
    lda free_lo
    jsr show_hex
    rts

free_lo: .byte 0
free_hi: .byte 0

; Page-aligned buffers
.align 256
buffer1: .res 256
.align 256
buffer2: .res 256
```

## Key Concepts

1. **Memory Banking**: Switch ROM/RAM to access all 64KB
2. **Heap Management**: Dynamic allocation for flexible memory use
3. **Object Pools**: Fixed-size allocations for performance
4. **Compression**: Store more data in limited space
5. **Overlay Systems**: Load code/data on demand
