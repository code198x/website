# Testing and Quality Assurance

Shipping a bug-free game on the C64 was challenging. No debuggers, no stack traces, no exception handling. Just you, the metal, and careful testing. The games that succeeded - "Impossible Mission," "Boulder Dash," "Wizball" - had rigorous testing behind them.

## What You'll Build

This lesson teaches you to create:
1. **Automated test framework** for assembly code
2. **Memory integrity checker** to catch corruption
3. **Diagnostic display system** for debugging
4. **Regression test suite** to prevent bug reintroduction

These techniques ensure your game runs flawlessly on real hardware.

## The Challenge of C64 Testing

Testing on the C64 faced unique challenges:
- No debugging tools (unless you had an expensive monitor cartridge)
- Limited memory for test code
- No error messages or stack traces
- Bugs could corrupt memory silently
- Race conditions with interrupts

Professional developers created their own testing tools. Let's build them.

## Building a Test Framework

First, create a systematic testing system:

```asm
; Complete test framework for C64 games
; Run comprehensive tests and report results

    * = $0801
    ; BASIC stub
    !byte $0b,$08,$0a,$00,$9e
    !text "2061"
    !byte $00,$00,$00

    * = $080d

; Test tracking
tests_total:    .byte 0
tests_passed:   .byte 0
tests_failed:   .byte 0

; Test results buffer
MAX_FAILURES = 10

failure_msgs:   .res MAX_FAILURES * 2  ; Pointers to messages

main:
    jsr init_testing
    jsr run_all_tests
    jsr display_results
    rts

; Initialize test system
init_testing:
    lda #0
    sta tests_total
    sta tests_passed
    sta tests_failed

    ; Clear screen
    lda #147
    jsr $FFD2
    
    ; Display header
    ldx #0

    lda header_text,x
    beq header_done
    jsr $FFD2
    inx
    jmp show_header
header_done:
    rts

    !text "C64 GAME TEST SUITE",13
    !text "===================",13,13,0

; Run all test suites
run_all_tests:
    ; Core functionality
    lda #<suite_core_name
    sta current_suite
    lda #>suite_core_name
    sta current_suite+1
    jsr display_suite_name
    jsr test_core_functions
    
    ; Collision detection
    lda #<suite_collision_name
    sta current_suite
    lda #>suite_collision_name
    sta current_suite+1
    jsr display_suite_name

    sta current_suite

    sta current_suite
    lda #>suite_boundary_name
    sta current_suite+1
    jsr display_suite_name
    jsr test_boundaries
    
    ; Performance
    lda #<suite_perf_name
    sta current_suite
    lda #>suite_perf_name
    sta current_suite+1
    jsr display_suite_name
    jsr test_performance
    
    rts

suite_collision_name: !text "COLLISION DETECTION",0
suite_memory_name:    !text "MEMORY INTEGRITY",0
suite_boundary_name:  !text "BOUNDARY CONDITIONS",0

    ldy #0

    beq suite_shown
    jsr $FFD2
    iny
    jmp show_suite
suite_shown:
    lda #':'
    jsr $FFD2
    lda #13
    jsr $FFD2
    rts

; Test core game functions
test_core_functions:
    ; Test initialization
    lda #<test_init_name
    sta current_test
    lda #>test_init_name
    sta current_test+1
    
    jsr game_init
    
    ; Verify initial state
    lda player_lives
    cmp #3
    bne test_init_fail
    
    lda score
    ora score+1
    ora score+2
    bne test_init_fail
    
    jsr test_pass
    jmp test_movement

test_init_fail:
    lda #<msg_init_fail
    ldy #>msg_init_fail
    jsr test_fail

test_movement:
    ; Test player movement
    lda #<test_move_name
    sta current_test
    lda #>test_move_name
    sta current_test+1
    
    ; Setup
    lda #100
    sta player_x
    sta player_y
    
    ; Test right movement
    lda #$F7        ; Right bit clear
    sta $DC00
    jsr update_player
    
    lda player_x
    cmp #101
    bcc move_fail
    
    ; Test left movement
    lda #$FB        ; Left bit clear
    sta $DC00
    jsr update_player
    
    lda player_x
    cmp #101
    bcs move_fail
    
    jsr test_pass
    jmp test_scoring

move_fail:
    lda #<msg_move_fail
    ldy #>msg_move_fail
    jsr test_fail

test_scoring:
    ; Test score system
    lda #<test_score_name
    sta current_test
    lda #>test_score_name
    sta current_test+1
    
    ; Reset score
    lda #0
    sta score
    sta score+1
    sta score+2
    
    ; Add points
    lda #100
    jsr add_score
    
    lda score
    cmp #100
    bne score_fail
    
    ; Test overflow
    lda #$FF
    sta score
    sta score+1
    sta score+2
    
    lda #1
    jsr add_score
    
    ; Should handle overflow gracefully
    lda score+2
    cmp #$FF
    bne score_fail
    
    jsr test_pass
    rts

score_fail:
    lda #<msg_score_fail
    ldy #>msg_score_fail
    jsr test_fail
    rts

test_init_name:  !text "Initialization",0
test_move_name:  !text "Movement",0
test_score_name: !text "Scoring",0

msg_move_fail:   !text "Movement incorrect",0

test_collision_system:
    ; Test basic sprite collision
    lda #<test_coll_name
    sta current_test
    lda #>test_coll_name
    sta current_test+1
    
    ; Place sprites overlapping
    lda #100
    sta sprite1_x
    sta sprite1_y
    lda #105        ; Within 8 pixels
    sta sprite2_x
    sta sprite2_y
    
    jsr check_sprite_collision
    bcs coll_test1_pass
    
    lda #<msg_coll_miss
    ldy #>msg_coll_miss
    jsr test_fail
    jmp test_no_collision

coll_test1_pass:
    jsr test_pass

test_no_collision:
    ; Test non-collision
    lda #<test_nocoll_name
    sta current_test
    lda #>test_nocoll_name
    sta current_test+1
    
    ; Place sprites apart
    lda #100
    sta sprite1_x
    sta sprite1_y
    lda #200        ; Far apart
    sta sprite2_x
    sta sprite2_y
    
    jsr check_sprite_collision
    bcc nocoll_test_pass
    
    lda #<msg_false_coll
    ldy #>msg_false_coll
    jsr test_fail
    rts

nocoll_test_pass:
    jsr test_pass
    rts

test_coll_name:   !text "Collision detect",0
test_nocoll_name: !text "No false collision",0
msg_coll_miss:    !text "Missed collision",0
msg_false_coll:   !text "False collision",0

; Memory integrity tests
test_memory_integrity:
    ; Test stack operations
    lda #<test_stack_name
    sta current_test
    lda #>test_stack_name
    sta current_test+1
    
    ; Save stack pointer
    tsx
    stx saved_sp
    
    ; Push test pattern
    lda #$AA
    pha
    lda #$55
    pha
    
    ; Pop and verify
    pla
    cmp #$55
    bne stack_fail
    pla
    cmp #$AA
    bne stack_fail
    
    ; Verify stack pointer restored
    tsx
    cpx saved_sp
    bne stack_fail
    
    jsr test_pass
    jmp test_zp_integrity

stack_fail:
    lda #<msg_stack_fail
    ldy #>msg_stack_fail
    jsr test_fail

test_zp_integrity:
    ; Test zero page preservation
    lda #<test_zp_name
    sta current_test
    lda #>test_zp_name
    sta current_test+1
    
    ; Fill zero page test area with pattern
    ldx #$10
    lda #$5A
fill_zp:
    sta $F0,x
    dex
    bpl fill_zp
    
    ; Run game functions
    jsr game_update
    
    ; Verify pattern preserved
    ldx #$10
check_zp:
    lda $F0,x
    cmp #$5A
    bne zp_corrupt
    dex
    bpl check_zp
    
    jsr test_pass
    jmp test_buffer_overflow

zp_corrupt:
    lda #<msg_zp_corrupt
    ldy #>msg_zp_corrupt
    jsr test_fail

test_buffer_overflow:
    ; Test buffer boundaries
    lda #<test_buffer_name
    sta current_test
    lda #>test_buffer_name
    sta current_test+1
    
    ; Place guard bytes
    lda #$DE
    sta buffer_start-1
    lda #$AD
    sta buffer_end
    
    ; Fill buffer
    ldx #BUFFER_SIZE-1
    lda #$00
fill_buffer:
    sta buffer_start,x
    dex
    bpl fill_buffer
    
    ; Verify guards intact
    lda buffer_start-1
    cmp #$DE
    bne buffer_overflow
    lda buffer_end
    cmp #$AD
    bne buffer_overflow
    
    jsr test_pass
    rts

buffer_overflow:
    lda #<msg_buffer_over
    ldy #>msg_buffer_over
    jsr test_fail
    rts

saved_sp:         .byte 0
test_stack_name:  !text "Stack integrity",0
test_zp_name:     !text "Zero page safety",0
test_buffer_name: !text "Buffer bounds",0
msg_stack_fail:   !text "Stack corrupted",0
msg_zp_corrupt:   !text "Zero page corrupted",0
msg_buffer_over:  !text "Buffer overflow",0

BUFFER_SIZE = 256
buffer_start: .res BUFFER_SIZE
buffer_end:   .byte 0

; Boundary condition tests
test_boundaries:
    ; Test screen boundaries
    lda #<test_scrn_name
    sta current_test
    lda #>test_scrn_name
    sta current_test+1
    
    ; Test left edge
    lda #0
    sta player_x
    jsr move_player_left
    
    lda player_x
    cmp #0          ; Should stop at 0
    beq left_bound_ok
    
    lda #<msg_bound_left
    ldy #>msg_bound_left
    jsr test_fail
    jmp test_right_bound

left_bound_ok:
    jsr test_pass

test_right_bound:
    ; Test right edge
    lda #<test_right_name
    sta current_test
    lda #>test_right_name
    sta current_test+1
    
    lda #255
    sta player_x
    jsr move_player_right
    
    lda player_x
    cmp #255        ; Should stop at 255
    beq right_bound_ok
    
    lda #<msg_bound_right
    ldy #>msg_bound_right
    jsr test_fail
    jmp test_array_bounds

right_bound_ok:
    jsr test_pass

test_array_bounds:
    ; Test array indexing
    lda #<test_array_name
    sta current_test
    lda #>test_array_name
    sta current_test+1
    
    ; Test with max index
    ldx #MAX_ENEMIES-1
    lda enemy_x,x     ; Should not crash
    
    ; Test wraparound protection
    ldx #255
    jsr get_enemy_safe  ; Should handle invalid index
    
    jsr test_pass
    rts

test_scrn_name:   !text "Screen left",0
test_right_name:  !text "Screen right",0
test_array_name:  !text "Array bounds",0
msg_bound_left:   !text "Left bound failed",0
msg_bound_right:  !text "Right bound failed",0

; Performance tests
test_performance:
    ; Test frame rate
    lda #<test_fps_name
    sta current_test
    lda #>test_fps_name
    sta current_test+1
    
    ; Measure frames in fixed time
    lda #0
    sta frame_count
    sta frame_count+1
    
    ; Setup timer
    lda #$FF
    sta $DC04       ; Timer A low
    sta $DC05       ; Timer A high
    
    lda #$01
    sta $DC0E       ; Start timer
    
perf_loop:
    jsr game_update
    jsr wait_frame
    
    ; Increment counter
    inc frame_count
    bne check_timer
    inc frame_count+1
    
check_timer:
    ; Check if time elapsed
    lda $DC05
    cmp #$80        ; Half time
    bcs perf_loop
    
    ; Stop timer
    lda #$00
    sta $DC0E
    
    ; Check frame count
    lda frame_count+1
    bne good_fps        ; >256 frames
    lda frame_count
    cmp #50            ; At least 50 frames
    bcs good_fps
    
    lda #<msg_slow_fps
    ldy #>msg_slow_fps
    jsr test_fail
    rts

good_fps:
    jsr test_pass
    rts

frame_count:     .word 0
test_fps_name:   !text "Frame rate",0
msg_slow_fps:    !text "FPS too low",0

; Test pass/fail handlers
test_pass:
    inc tests_passed
    inc tests_total
    
    ; Display checkmark
    lda #$D6        ; Checkmark character
    jsr $FFD2
    lda #' '
    jsr $FFD2
    
    ; Display test name
    ldy #0
show_test_pass:
    lda (current_test),y
    beq test_pass_done
    jsr $FFD2
    iny
    jmp show_test_pass
test_pass_done:
    lda #13
    jsr $FFD2
    rts

test_fail:
    ; A/Y = error message pointer
    sta temp_ptr
    sty temp_ptr+1
    
    inc tests_failed
    inc tests_total
    
    ; Store failure message
    ldx failure_count
    cpx #MAX_FAILURES
    beq skip_store
    
    lda temp_ptr
    sta failure_msgs,x
    inx
    lda temp_ptr+1
    sta failure_msgs,x
    inc failure_count

skip_store:
    ; Display X
    lda #'X'
    jsr $FFD2
    lda #' '
    jsr $FFD2
    
    ; Display test name
    ldy #0
show_test_fail:
    lda (current_test),y
    beq test_fail_msg
    jsr $FFD2
    iny
    jmp show_test_fail
    
test_fail_msg:
    lda #' '
    jsr $FFD2
    lda #'-'
    jsr $FFD2
    lda #' '
    jsr $FFD2
    
    ; Display error
    ldy #0
show_error:
    lda (temp_ptr),y
    beq test_fail_done
    jsr $FFD2
    iny
    jmp show_error
test_fail_done:
    lda #13
    jsr $FFD2
    rts

temp_ptr: .word 0

; Display final results
display_results:
    ; Summary line
    lda #13
    jsr $FFD2
    ldx #0
show_summary:
    lda summary_text,x
    beq show_totals
    jsr $FFD2
    inx
    jmp show_summary

show_totals:
    ; Convert numbers to ASCII
    lda tests_passed
    jsr print_number
    
    lda #'/'
    jsr $FFD2
    
    lda tests_total
    jsr print_number
    
    ldx #0
show_passed:
    lda passed_text,x
    beq check_failures
    jsr $FFD2
    inx
    jmp show_passed

check_failures:
    lda tests_failed
    beq all_passed
    
    ; Show failures
    lda #13
    jsr $FFD2
    ldx #0
show_failed:
    lda failed_text,x
    beq list_failures
    jsr $FFD2
    inx
    jmp show_failed

list_failures:
    ; List failure messages
    ldx #0
    stx temp
list_loop:
    lda temp
    cmp failure_count
    beq done_results
    
    asl
    tax
    lda failure_msgs,x
    sta temp_ptr
    lda failure_msgs+1,x
    sta temp_ptr+1
    
    lda #' '
    jsr $FFD2
    lda #'-'
    jsr $FFD2
    lda #' '
    jsr $FFD2
    
    ldy #0
show_fail_msg:
    lda (temp_ptr),y
    beq next_fail
    jsr $FFD2
    iny
    jmp show_fail_msg
    
next_fail:
    lda #13
    jsr $FFD2
    inc temp
    jmp list_loop

all_passed:
    ldx #0
show_success:
    lda success_text,x
    beq done_results
    jsr $FFD2
    inx
    jmp show_success

done_results:
    rts

temp: .byte 0

summary_text: !text 13,"RESULTS: ",0
passed_text:  !text " tests passed",13,0
failed_text:  !text "FAILURES:",13,0
success_text: !text 13,"ALL TESTS PASSED!",13,0

; Print number in decimal
print_number:
    ; Simple 2-digit display
    ldx #'0'
    sec
tens_loop:
    sbc #10
    bcc print_ones
    inx
    jmp tens_loop
print_ones:

    pha
    txa
    cmp #'0'
    beq skip_tens
    jsr $FFD2
skip_tens:
    pla
    jsr $FFD2
    rts
```

## Diagnostic Display System

Create visual debugging tools:

```asm
; Visual debugging display
; Shows game state in real-time

; Debug display area (bottom of screen)
DEBUG_LINE = $0400 + (24 * 40)

; Initialize debug display
init_debug:
    ; Draw separator line
    ldx #39
    lda #$C0        ; Horizontal line
sep_loop:
    sta $0400 + (23 * 40),x
    lda #1          ; White
    sta $D800 + (23 * 40),x
    dex
    bpl sep_loop
    rts

; Update debug display
update_debug:
    ; Player position
    lda #'X'
    sta DEBUG_LINE
    lda #':'
    sta DEBUG_LINE+1
    
    lda player_x
    lsr
    lsr
    lsr
    lsr
    tax
    lda hex_chars,x
    sta DEBUG_LINE+2
    
    lda player_x
    and #$0F
    tax
    lda hex_chars,x
    sta DEBUG_LINE+3
    
    ; Player Y
    lda #' '
    sta DEBUG_LINE+4
    lda #'Y'
    sta DEBUG_LINE+5
    lda #':'
    sta DEBUG_LINE+6
    
    lda player_y
    lsr
    lsr
    lsr
    lsr
    tax
    lda hex_chars,x
    sta DEBUG_LINE+7
    
    lda player_y
    and #$0F
    tax
    lda hex_chars,x
    sta DEBUG_LINE+8
    
    ; Memory free
    lda #' '
    sta DEBUG_LINE+10
    lda #'M'
    sta DEBUG_LINE+11
    lda #':'
    sta DEBUG_LINE+12
    
    jsr calc_free_memory
    jsr display_hex_word
    
    ; Frame counter
    lda #' '
    sta DEBUG_LINE+17
    lda #'F'
    sta DEBUG_LINE+18
    lda #':'
    sta DEBUG_LINE+19
    
    inc frame_counter
    lda frame_counter
    jsr display_hex_byte
    
    ; Sprite collision register
    lda #' '
    sta DEBUG_LINE+22
    lda #'C'
    sta DEBUG_LINE+23
    lda #':'
    sta DEBUG_LINE+24
    
    lda $D01E       ; Sprite collision
    jsr display_hex_byte
    
    rts

hex_chars: !text "0123456789ABCDEF"
frame_counter: .byte 0

; Calculate free memory
calc_free_memory:
    ; Simple check - distance from heap to stack
    lda #$FF
    sec
    tsx
    stx temp
    sbc temp
    sta free_mem_hi
    
    lda heap_ptr
    sta free_mem_lo
    rts

free_mem_lo: .byte 0
free_mem_hi: .byte 0
heap_ptr:    .word $2000

; Display hex byte at current position
display_hex_byte:
    pha
    lsr
    lsr
    lsr
    lsr
    tax
    lda hex_chars,x
    jsr $FFD2
    
    pla
    and #$0F
    tax
    lda hex_chars,x
    jsr $FFD2
    rts

; Memory watch system
watch_addresses:
    .word $D020     ; Border color
    .word $D021     ; Background
    .word player_x
    .word player_y
    .word 0         ; End marker

; Display memory watches
show_watches:
    ldx #0
    ldy #0
watch_loop:
    lda watch_addresses,x
    sta temp_ptr
    inx
    lda watch_addresses,x
    sta temp_ptr+1
    inx
    
    ; Check for end
    ora temp_ptr
    beq watches_done
    
    ; Display address
    lda temp_ptr+1
    jsr display_hex_byte
    lda temp_ptr
    jsr display_hex_byte
    
    lda #':'
    jsr $FFD2
    
    ; Display value
    lda (temp_ptr),y
    jsr display_hex_byte
    
    lda #' '
    jsr $FFD2
    
    jmp watch_loop
    
watches_done:
    rts
```

## Stress Testing

Push the system to its limits:

```asm
; Stress test suite
; Tests system under heavy load

stress_test_all:
    jsr stress_test_sprites
    jsr stress_test_collisions
    jsr stress_test_memory
    jsr stress_test_interrupts
    rts

; Maximum sprite stress test
stress_test_sprites:
    ; Enable all sprites
    lda #$FF
    sta $D015
    
    ; Position sprites to cause maximum overlap
    ldx #0
    lda #100
position_stress:
    sta $D000,x     ; X position
    clc
    adc #5          ; Slight offset
    inx
    sta $D000,x     ; Y position
    inx
    cpx #16
    bne position_stress
    
    ; Run game loop under stress
    ldy #0
stress_sprite_loop:
    jsr game_update
    
    ; Check for corruption
    lda $D015
    cmp #$FF
    bne sprite_corruption
    
    iny
    bne stress_sprite_loop
    
    ; Test passed
    rts

sprite_corruption:
    ; Log error
    lda #<msg_sprite_corrupt
    ldy #>msg_sprite_corrupt
    jsr log_error
    rts

msg_sprite_corrupt: !text "Sprite register corrupted",0

; Collision stress test
stress_test_collisions:
    ; Create maximum collision scenarios
    ldx #0
setup_collision_stress:
    lda #1
    sta enemy_active,x
    
    ; Cluster enemies
    txa
    asl
    asl
    clc
    adc #50
    sta enemy_x,x
    
    lda #100
    sta enemy_y,x
    
    inx
    cpx #MAX_ENEMIES
    bne setup_collision_stress
    
    ; Place player in middle
    lda #100
    sta player_x
    sta player_y
    
    ; Run collision detection repeatedly
    ldy #0
collision_stress_loop:
    jsr check_all_collisions
    
    ; Verify data integrity
    ldx #0
verify_enemies:
    lda enemy_active,x
    cmp #2
    bcs enemy_corrupt
    
    inx

    bne verify_enemies
    
    iny
    bne collision_stress_loop
    
    rts

enemy_corrupt:
    lda #<msg_enemy_corrupt
    ldy #>msg_enemy_corrupt
    jsr log_error
    rts

msg_enemy_corrupt: !text "Enemy data corrupted",0

; Memory allocation stress test
stress_test_memory:
    ; Repeatedly allocate and free
    ldx #0
mem_stress_loop:

    tya

    ; Write test pattern
    ldy #0
    lda #$AA
pattern_loop:
    sta (temp_ptr),y
    iny
    cpy #64
    bne pattern_loop
    
    inx
    cpx #64
    bne mem_stress_loop

    ldx #0
free_loop:
    lda allocated_blocks,x
    sta temp_ptr
    lda allocated_blocks+64,x
    sta temp_ptr+1
    
    jsr free
    
    inx
    cpx #64
    bne free_loop
    
    rts

allocation_failed:
    pla
    lda #<msg_alloc_fail
    ldy #>msg_alloc_fail
    jsr log_error
    rts

allocated_blocks: .res 128
msg_alloc_fail: !text "Memory allocation failed",0

; Interrupt stress test
stress_test_interrupts:
    ; Enable multiple interrupt sources
    lda #$01
    sta $D01A       ; Raster IRQ
    lda #$FF
    sta $DC0D       ; Timer IRQ
    
    ; Rapid interrupt triggering
    lda #0
    sta $D012       ; Every line
    
    ; Run for extended period
    ldy #0
    ldx #0
irq_stress_loop:

    jsr complex_calculation
    
    ; Check stack didn't overflow
    tsx

    iny
    bne irq_stress_loop
    inx

    bne irq_stress_loop

    rts

stack_overflow:

    ldy #>msg_stack_over
    jsr log_error
    rts

msg_stack_over: !text "Stack overflow detected",0

complex_calculation:
    ; Simulate complex work
    ldx #100
calc_loop:
    lda $D41B       ; Random
    adc accumulator
    sta accumulator
    dex
    bne calc_loop
    rts

accumulator: .byte 0
```

## Error Logging System

Track and report errors:

```asm
; Error logging system
MAX_LOG_ENTRIES = 20
log_count:      .byte 0
log_buffer:     .res MAX_LOG_ENTRIES * 32

; Log error message
; A/Y = pointer to message
log_error:
    sta temp_ptr
    sty temp_ptr+1
    
    ; Check if log full
    lda log_count
    cmp #MAX_LOG_ENTRIES
    bcs log_full
    
    ; Calculate buffer position
    ldx log_count
    lda #0
    sta temp2
    txa
    asl
    asl
    asl
    asl
    asl             ; *32
    tax
    
    ; Copy message
    ldy #0
copy_log:
    lda (temp_ptr),y
    sta log_buffer,x
    beq log_copied
    inx
    iny
    cpy #31
    bne copy_log
    
log_copied:
    inc log_count
    
log_full:
    rts

temp2: .byte 0

; Display error log
show_error_log:
    lda #13
    jsr $FFD2
    
    ldx #0
log_header:
    lda error_log_header,x
    beq show_entries
    jsr $FFD2
    inx
    jmp log_header

show_entries:
    lda log_count
    beq no_errors
    
    ldx #0
    stx temp
    
show_log_loop:
    ; Calculate entry position
    lda temp
    asl
    asl
    asl
    asl
    asl
    tax
    
    ; Display entry
    ldy #0
show_entry:
    lda log_buffer,x
    beq next_entry
    jsr $FFD2
    inx
    iny
    cpy #32
    bne show_entry
    
next_entry:
    lda #13
    jsr $FFD2
    
    inc temp
    lda temp
    cmp log_count
    bne show_log_loop
    
    rts

no_errors:
    ldx #0
show_no_errors:
    lda no_errors_msg,x
    beq done_log
    jsr $FFD2
    inx
    jmp show_no_errors

done_log:
    rts

no_errors_msg:    !text "No errors logged",13,0
```

## Testing Best Practices

1. **Test early and often** - Don't wait until the end
2. **Automate everything** - Manual testing misses bugs
3. **Test edge cases** - Normal gameplay isn't enough
4. **Keep regression tests** - Old bugs love to return
5. **Monitor performance** - Slowdowns kill games
6. **Test on real hardware** - Emulators aren't perfect
