```nasm
; Prism Quest - Achievement Prisms System
    ORG $8000

start:
    ; Initialize complete Spectrum environment
    CALL clear_complete_screen

    ; Set up enhanced dimensional world for achievement tracking
    CALL init_complete_dimensional_state
    CALL init_entity_multiplication_system

    ; Initialize new achievement prism systems
    CALL init_achievement_prism_system
    CALL init_mastery_tracking_system
    CALL init_prism_calculation_system
    CALL init_achievement_display_system

    ; Create achievement-ready dimensional world
    CALL create_achievement_dimensional_world

    ; Achievement Prisms main loop
achievement_prism_loop:
    ; Process all navigator input
    CALL read_complete_input

    ; Update primary navigator and entities
    CALL update_primary_navigator
    CALL update_navigator_entities
    CALL update_spectral_entities

    ; Process achievement systems
    CALL process_achievement_events
    CALL update_achievement_calculations
    CALL update_mastery_tracking
    CALL update_prism_progression

    ; Process all interactions with achievement tracking
    CALL process_multi_entity_collisions_with_achievements
    CALL update_dimensional_achievement_tracking

    ; Update visual and audio systems
    CALL update_achievement_visual_effects
    CALL update_achievement_display_system
    CALL update_dimensional_audio

    ; Check for achievement-based victory conditions
    CALL check_achievement_victory_conditions

    JR achievement_prism_loop

; Initialize achievement prism system using mathematical operations
init_achievement_prism_system:
    ; Set up achievement calculation systems
    CALL init_prism_calculation_framework
    CALL init_achievement_mathematics
    CALL init_mastery_arithmetic
    CALL init_progression_tracking

    ; Initialize achievement prism counters
    LD A, $00           ; Clear all achievement counters
    LD (dimensional_prisms_collected), A
    LD (spectral_interactions), A
    LD (navigator_rescues), A
    LD (mastery_achievements), A

    ; Initialize achievement calculation variables
    LD (temp_achievement_value), A
    LD (last_achievement_value), A
    LD (achievement_multiplier), A

    ; Set initial achievement parameters
    LD A, $01           ; Base achievement multiplier
    LD (achievement_multiplier), A
    LD A, $00           ; No streak initially
    LD (achievement_streak), A
    LD (streak_timer), A

    RET

; Initialize prism calculation framework for mathematical operations
init_prism_calculation_framework:
    ; Set up achievement calculation lookup tables
    LD B, 16            ; Initialize 16 achievement values
    LD HL, base_achievement_table
    LD DE, achievement_point_values

init_achievement_table_loop:
    LD A, (HL)          ; Load base value
    LD (DE), A          ; Store in working table
    INC HL              ; Next source
    INC DE              ; Next destination
    DJNZ init_achievement_table_loop

    ; Initialize mastery calculation parameters
    LD HL, mastery_calculation_data
    LD (mastery_data_pointer), HL

    ; Set achievement thresholds
    LD A, $05           ; 5 prisms for first mastery
    LD (prism_mastery_threshold), A
    LD A, $03           ; 3 interactions for spectral mastery
    LD (spectral_mastery_threshold), A
    LD A, $0A           ; 10 rescues for navigator mastery
    LD (navigator_mastery_threshold), A

    RET

; Initialize achievement display system for visual feedback
init_achievement_display_system:
    ; Set up achievement display area
    LD A, $01           ; Achievement display row
    LD (achievement_display_row), A
    LD A, $14           ; Achievement display column (20)
    LD (achievement_display_column), A

    ; Initialize achievement colors using attribute system
    LD A, $47           ; White on blue for achievements
    LD (achievement_text_attribute), A
    LD A, $43           ; Yellow on blue for highlights
    LD (achievement_highlight_attribute), A
    LD A, $45           ; Magenta on blue for mastery
    LD (mastery_attribute), A

    ; Clear achievement display area
    CALL clear_achievement_display_area

    ; Draw achievement labels
    CALL draw_achievement_labels

    RET

; Create achievement dimensional world with visual enhancements
create_achievement_dimensional_world:
    ; Draw enhanced dimensional maze with achievement zones
    CALL draw_achievement_dimensional_maze

    ; Create achievement prism locations
    CALL create_achievement_prism_locations

    ; Set up mastery checkpoint areas
    CALL setup_mastery_checkpoint_areas

    ; Initialize achievement particle effect areas
    CALL init_achievement_particle_areas

    RET

; Draw achievement dimensional maze with special zones
draw_achievement_dimensional_maze:
    ; Create base dimensional maze
    LD HL, $4000        ; Screen memory start
    LD DE, achievement_maze_pattern
    LD BC, 768          ; Copy maze pattern (24 × 32 characters)
    LDIR                ; Copy maze to screen

    ; Set up achievement zone attributes
    LD HL, $5800        ; Attribute memory start
    LD B, 24            ; Number of lines

create_achievement_attributes_loop:
    PUSH BC             ; Save line counter
    LD C, 32            ; Characters per line

create_achievement_line_attributes:
    ; Calculate achievement zone color based on position
    LD A, B             ; Line number
    ADD A, C            ; Add column for variation
    AND $07             ; Mask to valid colors
    ADD A, $40          ; Bright colors for achievement visibility

    ; Special zones for achievements
    LD D, A             ; Save calculated color
    LD A, B             ; Check line number
    CP $06              ; Achievement zone 1 (lines 6-9)
    JR C, normal_attribute
    CP $0A
    JR NC, check_zone_2

    ; Achievement zone 1 - prism collection zone
    LD A, $46           ; Yellow on blue
    JR store_attribute

check_zone_2:
    CP $12              ; Achievement zone 2 (lines 18-21)
    JR C, normal_attribute
    CP $16
    JR NC, normal_attribute

    ; Achievement zone 2 - mastery zone
    LD A, $45           ; Magenta on blue
    JR store_attribute

normal_attribute:
    LD A, D             ; Restore calculated color

store_attribute:
    LD (HL), A          ; Set attribute
    INC HL              ; Next character
    DEC C               ; Decrement character counter
    JR NZ, create_achievement_line_attributes

    POP BC              ; Restore line counter
    DJNZ create_achievement_attributes_loop

    RET

; Process achievement events from dimensional interactions
process_achievement_events:
    ; Check for dimensional prism collection events
    CALL check_prism_collection_achievements

    ; Check for spectral interaction events
    CALL check_spectral_interaction_achievements

    ; Check for navigator rescue events
    CALL check_navigator_rescue_achievements

    ; Check for mastery achievement events
    CALL check_mastery_achievement_events

    RET

; Check prism collection achievements using arithmetic calculations
check_prism_collection_achievements:
    ; Check each prism location for collection
    LD B, 8             ; Check 8 prism locations
    LD HL, prism_collection_flags

check_prism_collection_loop:
    LD A, (HL)          ; Load collection flag
    CP $00              ; Not collected?
    JR Z, next_prism_check

    ; Prism was collected - calculate achievement value
    LD A, $00           ; Clear collection flag
    LD (HL), A

    ; Calculate base achievement value
    LD A, B             ; Prism index
    DEC A               ; Convert to 0-based
    LD C, A             ; Save index

    ; Get prism value from lookup table
    LD HL, prism_achievement_values
    ADD HL, BC          ; Point to prism value
    LD A, (HL)          ; Load base value
    LD (temp_achievement_value), A

    ; Apply achievement multiplier
    CALL apply_achievement_multiplier

    ; Add to prism collection count
    LD A, (dimensional_prisms_collected)
    INC A
    LD (dimensional_prisms_collected), A

    ; Update achievement streak
    CALL update_achievement_streak

    ; Add calculated value to total achievements
    CALL add_achievement_to_total

    ; Create visual achievement feedback
    CALL create_prism_collection_feedback

    ; Restore prism index
    LD B, C
    INC B               ; Restore to 1-based

next_prism_check:
    INC HL              ; Next collection flag
    DJNZ check_prism_collection_loop

    RET

; Apply achievement multiplier using mathematical operations
apply_achievement_multiplier:
    ; Multiply temp_achievement_value by current multiplier
    LD A, (achievement_multiplier)
    CP $01              ; No multiplication needed?
    JR Z, no_multiplier

    ; Save original value
    LD A, (temp_achievement_value)
    LD B, A             ; B = original value
    LD C, A             ; C = accumulator

    ; Get multiplier count
    LD A, (achievement_multiplier)
    DEC A               ; One less iteration (already have base)
    JR Z, store_multiplied_result

    ; Multiply using repeated addition
    LD D, A             ; D = remaining iterations
multiplier_loop:
    LD A, C             ; Current total
    ADD A, B            ; Add base value
    LD C, A             ; Store new total
    DEC D               ; Decrement iterations
    JR NZ, multiplier_loop

store_multiplied_result:
    LD A, C             ; Final result
    LD (temp_achievement_value), A

no_multiplier:
    RET

; Update achievement streak with timing calculations
update_achievement_streak:
    ; Check if streak timer is active
    LD A, (streak_timer)
    CP $00              ; No active streak?
    JR Z, start_new_streak

    ; Active streak - increment count
    LD A, (achievement_streak)
    INC A
    CP $10              ; Cap at 16 streak
    JR C, store_streak_count
    LD A, $0F           ; Maximum 15 streak

store_streak_count:
    LD (achievement_streak), A

    ; Calculate streak multiplier (streak / 4 + 1)
    SRL A               ; Divide by 2
    SRL A               ; Divide by 4
    INC A               ; Add 1
    CP $05              ; Cap at 5x multiplier
    JR C, store_streak_multiplier
    LD A, $04           ; Maximum 4x (base + 3)

store_streak_multiplier:
    LD (achievement_multiplier), A

    ; Reset streak timer
    LD A, $78           ; 120 frames = 2 seconds
    LD (streak_timer), A

    JR streak_updated

start_new_streak:
    ; Start new achievement streak
    LD A, $01           ; Initialize streak
    LD (achievement_streak), A
    LD A, $01           ; Base multiplier
    LD (achievement_multiplier), A
    LD A, $78           ; 120 frames timer
    LD (streak_timer), A

streak_updated:
    RET

; Update mastery tracking with complex calculations
update_mastery_tracking:
    ; Check dimensional prism mastery
    CALL check_dimensional_prism_mastery

    ; Check spectral interaction mastery
    CALL check_spectral_interaction_mastery

    ; Check navigator rescue mastery
    CALL check_navigator_rescue_mastery

    ; Update streak timer
    LD A, (streak_timer)
    CP $00              ; Already zero?
    JR Z, no_streak_timer_update

    DEC A               ; Decrement timer
    LD (streak_timer), A
    JR NZ, no_streak_timer_update

    ; Streak expired - reset
    LD A, $00
    LD (achievement_streak), A
    LD A, $01
    LD (achievement_multiplier), A

no_streak_timer_update:
    RET

; Check dimensional prism mastery using threshold calculations
check_dimensional_prism_mastery:
    ; Check if we've reached prism mastery threshold
    LD A, (dimensional_prisms_collected)
    LD B, A             ; B = current count
    LD A, (prism_mastery_threshold)
    CP B                ; Threshold reached?
    JR NC, no_prism_mastery ; Not yet

    ; Check if this is a new mastery level
    LD A, (prism_mastery_level)
    LD C, A             ; C = current level

    ; Calculate new mastery level (count / threshold)
    LD A, B             ; Current count
    LD D, $00           ; Clear level counter
    LD E, A             ; E = remaining count

calculate_prism_mastery_level:
    LD A, (prism_mastery_threshold)
    CP E                ; Threshold > remaining?
    JR NC, prism_mastery_calculated

    ; Subtract threshold from remaining
    LD A, E
    SUB A, (prism_mastery_threshold)
    LD E, A
    INC D               ; Increment level
    JR calculate_prism_mastery_level

prism_mastery_calculated:
    ; Check if this is a new level
    LD A, D             ; New level
    CP C                ; Compare with current
    JR Z, no_prism_mastery ; Same level

    ; New mastery level achieved
    LD (prism_mastery_level), A

    ; Award mastery achievement points
    CALL award_mastery_achievement_points

    ; Create mastery visual effect
    CALL create_prism_mastery_effect

no_prism_mastery:
    RET

; Add achievement to total using efficient arithmetic
add_achievement_to_total:
    ; Add temp_achievement_value to total achievement score
    LD A, (total_achievement_score_low)
    ADD A, (temp_achievement_value)
    LD (total_achievement_score_low), A

    ; Handle carry to high byte
    JR NC, no_achievement_carry
    LD A, (total_achievement_score_high)
    INC A
    LD (total_achievement_score_high), A

no_achievement_carry:
    ; Check for achievement milestones
    CALL check_achievement_milestones

    RET

; Update achievement display system with visual feedback
update_achievement_display_system:
    ; Update prism collection display
    CALL update_prism_collection_display

    ; Update mastery level display
    CALL update_mastery_level_display

    ; Update achievement streak display
    CALL update_achievement_streak_display

    ; Update total achievement score display
    CALL update_total_achievement_display

    RET

; Update prism collection display using optimized routines
update_prism_collection_display:
    ; Position for prism display
    LD A, (achievement_display_row)
    LD (cursor_row), A
    LD A, (achievement_display_column)
    LD (cursor_column), A

    ; Display "PRISMS:" label
    LD HL, prism_label_text
    CALL display_achievement_text

    ; Display prism count
    LD A, (dimensional_prisms_collected)
    CALL convert_to_decimal_and_display

    ; Display separator
    LD A, '/'
    CALL display_achievement_character

    ; Display total available prisms
    LD A, (total_prisms_available)
    CALL convert_to_decimal_and_display

    RET

; Convert number to decimal and display
convert_to_decimal_and_display:
    ; A contains number to convert
    CP $0A              ; Two digits?
    JR C, single_digit_display

    ; Two digit number
    LD B, $00           ; Tens counter
two_digit_conversion_loop:
    CP $0A              ; >= 10?
    JR C, display_tens_digit
    SUB $0A             ; Subtract 10
    INC B               ; Increment tens
    JR two_digit_conversion_loop

display_tens_digit:
    PUSH AF             ; Save ones digit
    LD A, B             ; Tens digit
    ADD A, '0'          ; Convert to ASCII
    CALL display_achievement_character
    POP AF              ; Restore ones digit

single_digit_display:
    ADD A, '0'          ; Convert to ASCII
    CALL display_achievement_character

    RET

; Display achievement character with attribute effects
display_achievement_character:
    ; A contains character to display
    PUSH AF             ; Save character

    ; Calculate screen position
    LD A, (cursor_row)
    LD H, A
    LD A, (cursor_column)
    LD L, A
    CALL calculate_screen_address

    ; Display character
    POP AF              ; Restore character
    LD (HL), A

    ; Set attribute
    CALL calculate_attribute_address
    LD A, (achievement_text_attribute)
    LD (HL), A

    ; Advance cursor
    LD A, (cursor_column)
    INC A
    LD (cursor_column), A

    RET

; Update mastery level display with special effects
update_mastery_level_display:
    ; Position for mastery display
    LD A, (achievement_display_row)
    INC A               ; Next row
    LD (cursor_row), A
    LD A, (achievement_display_column)
    LD (cursor_column), A

    ; Display "MASTERY:" label
    LD HL, mastery_label_text
    CALL display_achievement_text

    ; Display prism mastery level
    LD A, (prism_mastery_level)
    CALL convert_to_decimal_and_display

    ; Display separator
    LD A, '-'
    CALL display_achievement_character

    ; Display spectral mastery level
    LD A, (spectral_mastery_level)
    CALL convert_to_decimal_and_display

    ; Display separator
    LD A, '-'
    CALL display_achievement_character

    ; Display navigator mastery level
    LD A, (navigator_mastery_level)
    CALL convert_to_decimal_and_display

    RET

; Create prism collection visual feedback
create_prism_collection_feedback:
    ; Create attribute flash effect at collection point
    CALL create_prism_flash_effect

    ; Create floating achievement text
    CALL create_floating_achievement_text

    ; Update achievement notification area
    CALL update_achievement_notification

    RET

; Create prism flash effect using attribute manipulation
create_prism_flash_effect:
    ; Find prism collection position
    LD A, (prism_collection_x)
    LD B, A             ; B = X position
    LD A, (prism_collection_y)
    LD C, A             ; C = Y position

    ; Calculate attribute address
    LD H, C             ; H = Y
    LD L, B             ; L = X
    CALL calculate_attribute_address

    ; Flash through bright colors
    LD A, (flash_color_counter)
    INC A
    AND $07             ; Keep in 0-7 range
    LD (flash_color_counter), A
    ADD A, $40          ; Make bright
    LD (HL), A          ; Set flashing attribute

    ; Set flash timer
    LD A, $10           ; 16 frames
    LD (prism_flash_timer), A

    RET

; Achievement calculation tables and data
base_achievement_table:
    DEFB $05, $0A, $0F, $14    ; Prism values
    DEFB $19, $1E, $23, $28    ; More prism values
    DEFB $02, $04, $06, $08    ; Spectral interaction values
    DEFB $0A, $0C, $0E, $10    ; Navigator rescue values

prism_label_text:
    DEFB "PRISMS:", $00

mastery_label_text:
    DEFB "MASTERY:", $00

; Achievement storage variables
dimensional_prisms_collected:   DEFB 0
spectral_interactions:          DEFB 0
navigator_rescues:              DEFB 0
mastery_achievements:           DEFB 0

; Achievement calculation variables
temp_achievement_value:         DEFB 0
last_achievement_value:         DEFB 0
achievement_multiplier:         DEFB 1
achievement_streak:             DEFB 0
streak_timer:                   DEFB 0

; Mastery tracking variables
prism_mastery_level:            DEFB 0
spectral_mastery_level:         DEFB 0
navigator_mastery_level:        DEFB 0
prism_mastery_threshold:        DEFB 5
spectral_mastery_threshold:     DEFB 3
navigator_mastery_threshold:    DEFB 10

; Total achievement tracking
total_achievement_score_low:    DEFB 0
total_achievement_score_high:   DEFB 0

; Display variables
achievement_display_row:        DEFB 1
achievement_display_column:     DEFB 20
achievement_text_attribute:     DEFB $47
achievement_highlight_attribute: DEFB $43
mastery_attribute:              DEFB $45
cursor_row:                     DEFB 0
cursor_column:                  DEFB 0

; Collection tracking
prism_collection_flags:         DEFS 8
prism_collection_x:             DEFB 0
prism_collection_y:             DEFB 0
prism_achievement_values:       DEFS 8
total_prisms_available:         DEFB 8

; Visual effect variables
flash_color_counter:            DEFB 0
prism_flash_timer:              DEFB 0

; Data pointers
mastery_data_pointer:           DEFW 0
achievement_point_values:       DEFS 16
mastery_calculation_data:       DEFS 32
achievement_maze_pattern:       DEFS 768
```
