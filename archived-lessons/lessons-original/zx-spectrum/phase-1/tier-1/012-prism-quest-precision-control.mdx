perform_optimized_keyboard_scan:
    ; Scan keyboard matrix with optimization
    LD HL, keyboard_current_state
    LD BC, $FEFE        ; Start with row 0
    LD D, $00           ; Row counter

keyboard_scan_loop:
    ; Read keyboard row
    IN A, (C)           ; Read keyboard row
    CPL                 ; Invert for positive logic
    LD (HL), A          ; Store row state

    ; Move to next row
    INC HL              ; Next storage location
    RLC B               ; Rotate to next row
    INC D               ; Increment row counter

    ; Check if all rows scanned

    CP $08              ; 8 keyboard rows
    JR NZ, keyboard_scan_loop

    RET

    AND A               ; Check if any keys pressed
    JR Z, next_simultaneous_row

    ; Count number of keys pressed in this row

    SRL C               ; Shift right
    JR NC, skip_key_count
    INC A               ; Increment key counter
skip_key_count:
    LD A, C             ; Check if more bits
    AND A
    JR NZ, count_keys_in_row

    ; Store simultaneous key count
    LD (DE), A

next_simultaneous_row:
    INC HL              ; Next keyboard row
    INC DE              ; Next buffer position
    DJNZ simultaneous_key_check_loop

    RET

apply_precision_input_filtering:
    ; Compare current and previous keyboard states

    LD DE, keyboard_previous_state
    LD BC, keyboard_change_mask
    LD A, $08           ; 8 rows to compare

input_filtering_loop:
    LD A, (HL)          ; Current state
    LD D, A             ; Save current
    LD A, (DE)          ; Previous state
    XOR D               ; Calculate changes
    LD (BC), A          ; Store change mask

    ; Apply debouncing filter
    PUSH HL
    PUSH DE
    PUSH BC
    CALL apply_debouncing_filter
    POP BC
    POP DE
    POP HL

    ; Move to next row
    INC HL
    INC DE
    INC BC
    DEC A
    JR NZ, input_filtering_loop

    ; Copy current state to previous for next frame
    LD HL, keyboard_current_state
    LD DE, keyboard_previous_state
    LD BC, $08
    LDIR

    RET

; Apply debouncing filter for precision input
apply_debouncing_filter:
    ; Check debounce counter for current input
    LD A, (input_debounce_counter)
    AND A               ; Check if debouncing
    JR Z, no_debouncing_active

    ; Debouncing active - decrement counter
    DEC A
    LD (input_debounce_counter), A

    ; Return previous stable state during debounce
    RET

    JR Z, input_stable

    ; Input changed - start debounce timer

input_stable:
    RET

; Queue precision command with timestamp and priority
queue_precision_command:
    ; A = command type, B = direction, C = distance
    ; Check if command queue has space
    LD HL, (command_queue_count)
    LD A, H             ; Get count
    CP (command_queue_size)
    RET NC              ; Queue full

    ; Add command to queue
    LD HL, (command_queue_write_pointer)

    ; Store command type
    LD (HL), A
    INC HL

    ; Store direction
    LD (HL), B
    INC HL

    ; Store distance
    LD (HL), C
    INC HL

    ; Store timestamp
    LD A, (precision_frame_counter)
    LD (HL), A
    INC HL

    ; Update write pointer
    LD (command_queue_write_pointer), HL

    ; Increment queue count
    LD A, (command_queue_count)
    INC A
    LD (command_queue_count), A

    RET

    LD A, (command_queue_count)
