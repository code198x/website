## Your Maze Challenges Materialize

Let's start with the magic - adding interactive puzzle elements to your dimensional maze:

```nasm
; Prism Quest - Maze Challenges Creation
    ORG $8000

start:
    ; Set up enhanced dimensional maze from Lessons 1-3
    CALL clear_screen
    CALL draw_dimensional_maze

    ; Initialize dimensional navigator from previous lessons
    LD BC, $0205        ; Navigator position: B=X(2), C=Y(5)
    LD (navigator_x), BC ; Store in memory

    ; Initialize dimensional shift system from Lesson 3
    CALL init_color_dimensions

    ; Create maze challenge system
    CALL init_puzzle_system

    ; Create interactive puzzle elements
    CALL create_maze_obstacles

    ; Set up puzzle objectives
    CALL create_puzzle_objectives

    ; Main dimensional puzzle loop
main_loop:
    CALL read_navigator_input
    CALL update_navigator_position
    CALL check_obstacle_collisions
    CALL check_puzzle_completion
    CALL update_puzzle_feedback
    JR main_loop

; Initialize puzzle challenge system
init_puzzle_system:
    ; Set up puzzle state variables
    LD A, $00           ; No puzzles solved initially
    LD (puzzles_solved), A ; Store puzzle progress

    ; Initialize obstacle states
    LD A, $01           ; Obstacle 1 active
    LD (obstacle_1_state), A
    LD A, $01           ; Obstacle 2 active
    LD (obstacle_2_state), A
    LD A, $01           ; Obstacle 3 active
    LD (obstacle_3_state), A

    ; Set puzzle timer for dynamic challenges
    LD A, $FF           ; Maximum timer value
    LD (puzzle_timer), A

    RET

; Create interactive maze obstacles
create_maze_obstacles:
    ; Create Color Gate Obstacle (requires red dimension)
    LD HL, $4000 + (10*32) + 12  ; Obstacle position
    LD A, $DB           ; Gate character (solid block)
    LD (HL), A          ; Place obstacle

    ; Set obstacle attribute (red gate)
    LD HL, $5800 + (10*32) + 12  ; Attribute position
    LD A, $42           ; Red on black (gate color)
    LD (HL), A          ; Set gate attribute

    ; Create Pressure Plate Puzzle
    LD HL, $4000 + (8*32) + 18   ; Pressure plate position
    LD A, $7E           ; Pressure plate character
    LD (HL), A          ; Place pressure plate

    ; Set pressure plate attribute
    LD HL, $5800 + (8*32) + 18   ; Attribute position
    LD A, $46           ; Yellow on black (interactive)
    LD (HL), A          ; Set pressure plate attribute

    ; Create Dimensional Lock (requires specific color sequence)
    LD HL, $4000 + (14*32) + 25  ; Lock position
    LD A, $23           ; Lock character
    LD (HL), A          ; Place lock

    ; Set lock attribute (changes based on state)
    LD HL, $5800 + (14*32) + 25  ; Attribute position
    LD A, $41           ; Blue on black (locked state)
    LD (HL), A          ; Set lock attribute

    RET

; Create puzzle objectives and goals
create_puzzle_objectives:
    ; Create dimensional keys scattered in maze
    LD HL, $4000 + (6*32) + 8    ; Key 1 position
    LD A, $2B           ; Key character
    LD (HL), A          ; Place key 1

    ; Set key 1 attribute
    LD HL, $5800 + (6*32) + 8    ; Attribute position
    LD A, $43           ; Green on black
    LD (HL), A          ; Set key attribute

    ; Create key 2
    LD HL, $4000 + (12*32) + 30  ; Key 2 position
    LD A, $2B           ; Key character
    LD (HL), A          ; Place key 2

    ; Set key 2 attribute
    LD HL, $5800 + (12*32) + 30  ; Attribute position
    LD A, $44           ; Cyan on black
    LD (HL), A          ; Set key attribute

    ; Create goal portal
    LD HL, $4000 + (16*32) + 20  ; Goal position
    LD A, $0F           ; Portal character (bright)
    LD (HL), A          ; Place goal portal

    ; Set goal attribute
    LD HL, $5800 + (16*32) + 20  ; Attribute position
    LD A, $47           ; White on black (goal)
    LD (HL), A          ; Set goal attribute

    RET

; Check collisions with maze obstacles
check_obstacle_collisions:
    ; Calculate navigator screen position for collision checking
    LD A, (navigator_y) ; Load Y position
    ADD A, A            ; Multiply by 2
    ADD A, A            ; Multiply by 4
    ADD A, A            ; Multiply by 8
    ADD A, A            ; Multiply by 16
    ADD A, A            ; Multiply by 32
    LD H, A             ; High byte of offset
    LD A, (navigator_x) ; Load X position
    LD L, A             ; Low byte of offset
    LD BC, $4000        ; Character memory base
    ADD HL, BC          ; Calculate navigator character position

    ; Check what character navigator is on
    LD A, (HL)          ; Read character at navigator position

    ; Check for Color Gate collision
    CP $DB              ; Compare with gate character
    JR Z, handle_gate_collision ; Jump if Zero flag set (equal)

    ; Check for Pressure Plate collision
    CP $7E              ; Compare with pressure plate character
    JR Z, handle_plate_collision ; Jump if Zero flag set

    ; Check for Key collision
    CP $2B              ; Compare with key character
    JR Z, handle_key_collision ; Jump if Zero flag set

    ; Check for Lock collision
    CP $23              ; Compare with lock character
    JR Z, handle_lock_collision ; Jump if Zero flag set

    ; Check for Goal collision
    CP $0F              ; Compare with goal character
    JR Z, handle_goal_collision ; Jump if Zero flag set

    ; No special collision
    RET

; Handle color gate collision (requires specific dimension)
handle_gate_collision:
    ; Check if in correct dimension
    LD A, (current_dimension) ; Load current dimension
    CP $01              ; Compare with red dimension
    JR NZ, gate_blocked ; Jump if Not Zero (wrong dimension)

    ; Correct dimension - open gate
    LD A, $00           ; Clear gate
    LD (obstacle_1_state), A ; Mark gate as disabled

    ; Clear gate character
    LD A, $20           ; Space character
    LD (HL), A          ; Clear gate

    ; Flash success feedback
    CALL flash_success_feedback

    RET

gate_blocked:
    ; Wrong dimension - block movement
    CALL flash_blocked_feedback

    ; Move navigator back one position
    LD A, (navigator_x) ; Load current X
    DEC A               ; Move back
    LD (navigator_x), A ; Store corrected position

    RET

; Handle pressure plate activation
handle_plate_collision:
    ; Check if plate already activated
    LD A, (obstacle_2_state) ; Load plate state
    CP $00              ; Compare with activated state
    JR Z, plate_done    ; Jump if Zero (already activated)

    ; Activate pressure plate
    LD A, $00           ; Mark as activated
    LD (obstacle_2_state), A ; Store new state

    ; Change plate appearance
    LD BC, $5800        ; Attribute base
    ADD HL, BC          ; Point to attribute memory
    SUB HL, BC          ; Convert back to character address
    LD BC, $1800        ; Offset to attribute memory
    ADD HL, BC          ; Point to attribute

    LD A, $45           ; Bright green (activated)
    LD (HL), A          ; Set activated attribute

    ; Trigger plate effect (remove distant obstacle)
    CALL activate_plate_effect

plate_done:
    RET

; Handle key collection
handle_key_collision:
    ; Determine which key was collected by checking position
    LD A, (navigator_x) ; Load navigator X
    CP $08              ; Compare with key 1 X position
    JR Z, collect_key_1 ; Jump if Zero (at key 1)

    CP $30              ; Compare with key 2 X position
    JR Z, collect_key_2 ; Jump if Zero (at key 2)

    RET                 ; Unknown key position

collect_key_1:
    ; Collect first key
    LD A, (keys_collected) ; Load current key count
    OR $01              ; Set bit 0 for key 1
    LD (keys_collected), A ; Store updated count

    ; Remove key from maze
    LD A, $20           ; Space character
    LD (HL), A          ; Clear key

    CALL flash_collection_feedback
    RET

collect_key_2:
    ; Collect second key
    LD A, (keys_collected) ; Load current key count
    OR $02              ; Set bit 1 for key 2
    LD (keys_collected), A ; Store updated count

    ; Remove key from maze
    LD A, $20           ; Space character
    LD (HL), A          ; Clear key

    CALL flash_collection_feedback
    RET

; Handle dimensional lock interaction
handle_lock_collision:
    ; Check if all keys collected
    LD A, (keys_collected) ; Load collected keys
    CP $03              ; Compare with both keys collected
    JR NZ, lock_blocked ; Jump if Not Zero (missing keys)

    ; Check if in correct dimensional sequence
    CALL check_dimensional_sequence
    CP $01              ; Compare result with success
    JR NZ, lock_blocked ; Jump if Not Zero (wrong sequence)

    ; Unlock successful
    LD A, $00           ; Mark lock as disabled
    LD (obstacle_3_state), A

    ; Clear lock
    LD A, $20           ; Space character
    LD (HL), A          ; Clear lock

    CALL flash_unlock_feedback
    RET

lock_blocked:
    ; Lock remains blocked
    CALL flash_blocked_feedback

    ; Move navigator back
    LD A, (navigator_x) ; Load current X
    DEC A               ; Move back
    LD (navigator_x), A ; Store corrected position

    RET

; Handle goal portal collision
handle_goal_collision:
    ; Check if all puzzles solved
    LD A, (obstacle_1_state) ; Check gate
    CP $00              ; Compare with disabled
    JR NZ, goal_blocked ; Jump if Not Zero (gate still active)

    LD A, (obstacle_2_state) ; Check pressure plate
    CP $00              ; Compare with activated
    JR NZ, goal_blocked ; Jump if Not Zero (plate not activated)

    LD A, (obstacle_3_state) ; Check lock
    CP $00              ; Compare with unlocked
    JR NZ, goal_blocked ; Jump if Not Zero (lock still active)

    ; All puzzles solved - goal reached!
    CALL handle_puzzle_completion
    RET

goal_blocked:
    ; Not all puzzles solved
    CALL flash_incomplete_feedback
    RET

; Check dimensional sequence for advanced puzzles
check_dimensional_sequence:
    ; Read last 3 dimensional changes
    LD A, (dimension_history) ; Load history
    AND $07             ; Mask to last 3 changes
    CP $05              ; Compare with required sequence (red->blue->green)
    JR Z, sequence_correct ; Jump if Zero (correct sequence)

    ; Wrong sequence
    LD A, $00           ; Return failure
    RET

sequence_correct:
    LD A, $01           ; Return success
    RET

; Update puzzle feedback and visual effects
update_puzzle_feedback:
    ; Update puzzle timer for dynamic effects
    LD A, (puzzle_timer) ; Load timer
    DEC A               ; Decrement timer
    LD (puzzle_timer), A ; Store updated timer
    CP $00              ; Compare with zero
    JR NZ, timer_continue ; Jump if Not Zero (timer active)

    ; Reset timer
    LD A, $FF           ; Reset to maximum
    LD (puzzle_timer), A

    ; Create pulsing effect on active puzzles
    CALL pulse_active_puzzles

timer_continue:
    RET

; Create pulsing visual effects on unsolved puzzles
pulse_active_puzzles:
    ; Pulse gate if still active
    LD A, (obstacle_1_state) ; Load gate state
    CP $01              ; Compare with active
    JR NZ, pulse_plate  ; Jump if Not Zero (gate disabled)

    ; Pulse gate attribute
    LD HL, $5800 + (10*32) + 12 ; Gate attribute position
    LD A, (HL)          ; Read current attribute
    XOR $08             ; Toggle bright bit
    LD (HL), A          ; Write pulsing attribute

pulse_plate:
    ; Pulse pressure plate if not activated
    LD A, (obstacle_2_state) ; Load plate state
    CP $01              ; Compare with inactive
    JR NZ, pulse_lock   ; Jump if Not Zero (plate activated)

    ; Pulse plate attribute
    LD HL, $5800 + (8*32) + 18  ; Plate attribute position
    LD A, (HL)          ; Read current attribute
    XOR $08             ; Toggle bright bit
    LD (HL), A          ; Write pulsing attribute

pulse_lock:
    ; Pulse lock if still active
    LD A, (obstacle_3_state) ; Load lock state
    CP $01              ; Compare with locked
    JR NZ, pulse_done   ; Jump if Not Zero (lock opened)

    ; Pulse lock attribute
    LD HL, $5800 + (14*32) + 25 ; Lock attribute position
    LD A, (HL)          ; Read current attribute
    XOR $08             ; Toggle bright bit
    LD (HL), A          ; Write pulsing attribute

pulse_done:
    RET

; Flash success feedback
flash_success_feedback:
    LD A, $04           ; Green border
    OUT ($FE), A        ; Flash border green

    ; Brief delay
    LD BC, $0200        ; Delay counter
success_delay:
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, success_delay ; Continue until zero

    ; Restore border
    LD A, $00           ; Black border
    OUT ($FE), A        ; Restore border

    RET

; Flash blocked feedback
flash_blocked_feedback:
    LD A, $02           ; Red border
    OUT ($FE), A        ; Flash border red

    ; Brief delay
    LD BC, $0100        ; Delay counter
blocked_delay:
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, blocked_delay ; Continue until zero

    ; Restore border
    LD A, $00           ; Black border
    OUT ($FE), A        ; Restore border

    RET

; Flash collection feedback
flash_collection_feedback:
    LD A, $06           ; Yellow border
    OUT ($FE), A        ; Flash border yellow

    ; Brief delay
    LD BC, $0150        ; Delay counter
collect_delay:
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, collect_delay ; Continue until zero

    ; Restore border
    LD A, $00           ; Black border
    OUT ($FE), A        ; Restore border

    RET

; Flash unlock feedback
flash_unlock_feedback:
    LD A, $03           ; Cyan border
    OUT ($FE), A        ; Flash border cyan

    ; Brief delay
    LD BC, $0300        ; Delay counter
unlock_delay:
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, unlock_delay ; Continue until zero

    ; Restore border
    LD A, $00           ; Black border
    OUT ($FE), A        ; Restore border

    RET

; Flash incomplete feedback
flash_incomplete_feedback:
    LD A, $05           ; Magenta border
    OUT ($FE), A        ; Flash border magenta

    ; Brief delay
    LD BC, $0180        ; Delay counter
incomplete_delay:
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, incomplete_delay ; Continue until zero

    ; Restore border
    LD A, $00           ; Black border
    OUT ($FE), A        ; Restore border

    RET

; Handle complete puzzle solution
handle_puzzle_completion:
    ; Flash victory sequence
    LD B, $05           ; Flash 5 times
victory_flash:
    PUSH BC             ; Save counter

    LD A, $07           ; White border
    OUT ($FE), A        ; Flash white

    LD BC, $0200        ; Delay
victory_delay_1:
    DEC BC
    LD A, B
    OR C
    JR NZ, victory_delay_1

    LD A, $00           ; Black border
    OUT ($FE), A        ; Flash black

    LD BC, $0200        ; Delay
victory_delay_2:
    DEC BC
    LD A, B
    OR C
    JR NZ, victory_delay_2

    POP BC              ; Restore counter
    DJNZ victory_flash  ; Continue flashing

    ; Mark puzzle as completed
    LD A, $01           ; Puzzle completed
    LD (puzzles_solved), A

    RET

; Activate pressure plate effect
activate_plate_effect:
    ; Remove a distant wall obstacle
    LD HL, $4000 + (6*32) + 25  ; Wall position
    LD A, $20           ; Space character
    LD (HL), A          ; Remove wall

    ; Clear wall attribute
    LD HL, $5800 + (6*32) + 25  ; Wall attribute position
    LD A, $47           ; Normal attribute
    LD (HL), A          ; Set normal attribute

    RET

; Check puzzle completion status
check_puzzle_completion:
    ; Check if victory condition met
    LD A, (puzzles_solved) ; Load completion status
    CP $01              ; Compare with completed
    JR Z, puzzle_complete ; Jump if Zero (completed)

    RET                 ; Not completed yet

puzzle_complete:
    ; Puzzle is complete - could transition to next level
    ; For now, just maintain victory state
    RET

; Enhanced screen clearing (from previous lessons)
clear_screen:
    ; Clear character memory
    LD HL, $4000        ; Character memory start
    LD BC, $1800        ; 6144 bytes
    LD A, $00           ; Clear character

clear_char_loop:
    LD (HL), A          ; Clear character
    INC HL              ; Next position
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, clear_char_loop

    ; Clear attribute memory
    LD HL, $5800        ; Attribute memory start
    LD BC, $0300        ; 768 bytes
    LD A, $47           ; White on black

clear_attr_loop:
    LD (HL), A          ; Clear attribute
    INC HL              ; Next position
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, clear_attr_loop

    RET

; Enhanced maze drawing (from previous lessons)
draw_dimensional_maze:
    ; Draw basic maze structure
    LD HL, $4000 + (4*32) + 8   ; Wall position
    LD A, $FF           ; Solid wall character
    LD B, 16            ; Wall length

draw_wall_loop:
    LD (HL), A          ; Place wall character
    INC HL              ; Next position
    DJNZ draw_wall_loop ; Continue until B=0

    RET

; Enhanced navigator input (from previous lessons)
read_navigator_input:
    ; Read keyboard input
    LD A, $FD           ; Select keyboard row
    IN A, ($FE)         ; Read keyboard port

    ; Check for movement keys (simplified)
    BIT 0, A            ; Check key state
    JR NZ, no_input     ; No key pressed

    ; Process movement
    CALL move_navigator_right

no_input:
    RET

; Enhanced navigator movement (from previous lessons)
move_navigator_right:
    LD A, (navigator_x) ; Load current X
    INC A               ; Move right
    CP 31               ; Check boundary
    JR NC, no_move      ; Don't move if at edge
    LD (navigator_x), A ; Store new position
no_move:
    RET

; Enhanced navigator position update (from previous lessons)
update_navigator_position:
    ; Load navigator position
    LD BC, (navigator_x) ; Load X,Y position

    ; Calculate screen position
    CALL calc_navigator_screen_position

    ; Place navigator character
    LD A, $81           ; Navigator character
    LD (HL), A          ; Place navigator

    RET

; Enhanced screen position calculation (from previous lessons)
calc_navigator_screen_position:
    LD A, C             ; Load Y position
    ADD A, A            ; Multiply by 2
    ADD A, A            ; Multiply by 4
    ADD A, A            ; Multiply by 8
    ADD A, A            ; Multiply by 16
    ADD A, A            ; Multiply by 32
    LD H, A             ; High byte
    LD L, B             ; X position as low byte
    LD BC, $4000        ; Screen base
    ADD HL, BC          ; Final screen position
    RET

; Enhanced color dimension initialization (from Lesson 3)
init_color_dimensions:
    LD A, $01           ; Dimension 1
    LD (current_dimension), A ; Store current dimension

    ; Initialize dimension history for sequence puzzles
    LD A, $00           ; Clear history
    LD (dimension_history), A

    RET

; Memory variables for puzzle system
navigator_x:        DB 2
navigator_y:        DB 5
current_dimension:  DB 1
dimension_history:  DB 0
puzzles_solved:     DB 0
obstacle_1_state:   DB 1    ; Gate state
obstacle_2_state:   DB 1    ; Pressure plate state
obstacle_3_state:   DB 1    ; Lock state
keys_collected:     DB 0    ; Bit field for collected keys
puzzle_timer:       DB $FF  ; Dynamic effects timer
```

## What Just Happened? The Magic Behind Maze Challenge Logic

Now that you've experienced your maze challenges, let's understand how **Z80 status flags** orchestrated this interactive puzzle system.

### Z80 Status Flags: Your Puzzle Logic Controller

The most important thing to understand: **status flags automatically provide information for puzzle decisions**. In your maze challenge code:

**Zero Flag (Z)**: Testing equality and puzzle states

```nasm
CP $DB              ; Compare with gate character
JR Z, handle_gate_collision ; Jump if Zero flag set (equal)
CP $01              ; Compare obstacle state with active
JR Z, gate_blocked  ; Jump if Zero flag set (obstacle active)
```

**Carry Flag (C)**: Boundary checking and range validation

```nasm
CP $08              ; Compare navigator X with key position
JR Z, collect_key_1 ; Jump if Zero (at correct position)
DEC BC              ; Decrement delay counter
LD A, B             ; Check if BC reached zero
OR C                ; Combine B and C (sets Zero if both zero)
JR NZ, continue_delay ; Jump if Not Zero (counter still active)
```

**Overflow and Sign Flags**: Complex puzzle state validation

```nasm
LD A, (keys_collected) ; Load collected keys bit field
OR $01              ; Set bit 0 for key 1 (affects flags)
CP $03              ; Compare with both keys collected
JR NZ, lock_blocked ; Jump if Not Zero (not all keys)
```

**Z80 Status Flag Examples in Maze Challenge Logic:**

| Flag               | Example                      | Use Case                                        |
| ------------------ | ---------------------------- | ----------------------------------------------- |
| **Zero Flag**      | `CP $DB / JR Z, handle_gate` | Test collision with specific puzzle elements    |
| **Carry Flag**     | `DEC BC / JR NZ, continue`   | Manage timing delays and animation counters     |
| **Combined Flags** | `OR $01 / CP $03`            | Track puzzle completion states and requirements |

### Collision Detection Using Status Flags

**Character-Based Collision System:**

```nasm
LD A, (HL)          ; Read character at navigator position
CP $DB              ; Compare with gate character (sets Zero if equal)
JR Z, handle_gate   ; Jump if Zero flag set (collision detected)
CP $7E              ; Compare with pressure plate (sets Zero if equal)
JR Z, handle_plate  ; Jump if Zero flag set (on pressure plate)
```

**State Management with Flags:**

```nasm
LD A, (obstacle_state) ; Load obstacle status
CP $00              ; Compare with disabled state (sets Zero if disabled)
JR Z, obstacle_clear ; Jump if Zero (obstacle removed)
CP $01              ; Compare with active state (sets Zero if active)
JR Z, obstacle_blocks ; Jump if Zero (obstacle blocking)
```

### Why Status Flags Matter for Puzzle Games

**Automatic State Testing**: Flags are set automatically by comparisons

```nasm
LD A, (keys_collected) ; Load key status
CP $03              ; Check if both keys collected (sets flags automatically)
JR NZ, incomplete   ; Branch based on automatic flag setting
```

**Efficient Puzzle Logic**: No manual state calculations needed

```nasm
LD A, (dimension_history) ; Load dimensional sequence
AND $07             ; Mask to relevant bits (affects Zero flag)
CP $05              ; Check required sequence (sets flags)
JR Z, sequence_correct ; Decision made based on automatic comparison
```

**Interactive Feedback**: Perfect for responsive puzzle elements

```nasm
DEC A               ; Decrement timer (sets Zero when reaches 0)
JR NZ, timer_active ; Continue if Not Zero (timer still running)
CALL pulse_puzzles  ; Timer expired - create visual effect
```

## Practice: Create Advanced Puzzle Mechanics

Let's enhance your maze system with complex multi-step puzzles:

```nasm
    ; Create sequence-based puzzle
    CALL init_sequence_puzzle

init_sequence_puzzle:
    ; Set up color sequence requirement
    LD A, $00           ; Clear sequence progress
    LD (sequence_progress), A

    ; Define required sequence: Red -> Blue -> Green -> Red
    LD HL, required_sequence
    LD A, $01           ; Red
    LD (HL), A
    INC HL
    LD A, $02           ; Blue
    LD (HL), A
    INC HL
    LD A, $04           ; Green
    LD (HL), A
    INC HL
    LD A, $01           ; Red again
    LD (HL), A

    RET

check_sequence_puzzle:
    ; Check if navigator entered correct dimensional sequence
    LD A, (current_dimension) ; Load current dimension
    LD HL, sequence_progress   ; Point to progress counter
    LD B, (HL)          ; Load current progress position

    ; Check if current dimension matches required step
    LD HL, required_sequence ; Point to sequence
    LD C, $00           ; Clear offset
    LD A, B             ; Copy progress to A
    ADD A, C            ; Add offset (A = B + C)
    LD C, A             ; Use as index
    ADD HL, BC          ; Point to required step

    LD A, (HL)          ; Load required dimension
    LD HL, current_dimension ; Point to current dimension
    CP (HL)             ; Compare required with current (sets flags)
    JR NZ, sequence_reset ; Jump if Not Zero (wrong dimension)

    ; Correct step - advance sequence
    LD HL, sequence_progress
    INC (HL)            ; Advance progress
    LD A, (HL)          ; Load new progress
    CP $04              ; Compare with sequence length (sets flags)
    JR Z, sequence_complete ; Jump if Zero (sequence complete)

    ; Sequence continues
    CALL flash_sequence_progress
    RET

sequence_reset:
    ; Wrong dimension - reset sequence
    LD HL, sequence_progress
    LD (HL), $00        ; Reset to beginning
    CALL flash_sequence_fail
    RET

sequence_complete:
    ; Sequence complete - unlock special area
    CALL unlock_secret_area
    RET

; Memory for advanced puzzles
sequence_progress:   DB 0
required_sequence:   DB 0, 0, 0, 0  ; Space for 4-step sequence
```
