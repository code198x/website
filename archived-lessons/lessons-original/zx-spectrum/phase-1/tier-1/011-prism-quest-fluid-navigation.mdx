; Initialize timing loop system for consistent performance
init_timing_loop_system:
    ; Set up interrupt-based timing
    CALL init_interrupt_timing
    CALL init_frame_synchronization
    CALL init_performance_monitoring
    CALL init_timing_optimization

    DI                  ; Disable interrupts

    LD A, $C0           ; High byte of interrupt table
    LD I, A             ; Set interrupt vector high byte
    LD HL, $C0C0        ; Interrupt vector address
    LD (HL), $00        ; Low byte of interrupt routine
    INC HL
    LD (HL), $C1        ; High byte of interrupt routine

    ; Create interrupt routine at $C100
    LD HL, timing_interrupt
    LD DE, $C100
    LD BC, timing_interrupt_end - timing_interrupt

check_navigation_zone_2:

store_navigation_attribute:

    LD (frame_timing_enabled), A

; Start frame timing for performance monitoring
start_frame_timing:
    ; Record frame start using system timing

    LD (frame_start_time), A

    ; Increment frame counter
    INC A
    LD (frame_timing_counter), A

    RET

; Read fluid input with smooth response

    LD BC, $FEFE        ; Keyboard row 1-5
    IN A, (C)           ; Read input
    AND $1F             ; Mask relevant bits
    LD (current_fluid_input), A

    ; Compare with previous input for smooth transitions
    LD B, A             ; Save current input
    LD A, (previous_input)
    XOR B               ; Check for changes
    LD (input_change_mask), A

    ; Store current input as previous for next frame
    LD A, B
    LD (previous_input), A

    ; Process input changes for fluid response
    CALL process_fluid_input_changes

    RET

; Process fluid input changes for smooth navigation

    JR Z, no_navigation_change

no_navigation_change:

calculate_fluid_navigator_velocity:

    LD A, (current_fluid_input)

    JR Z, check_right_fluid
    LD A, (acceleration_factor)

    LD (desired_velocity_x), A
    JR check_vertical_fluid_input

check_right_fluid:
    BIT 1, A            ; Check '2' key (right)
    JR Z, no_horizontal_fluid_input
    LD A, (acceleration_factor)
    LD (desired_velocity_x), A
    JR check_vertical_fluid_input

no_horizontal_fluid_input:

    LD A, (navigator_velocity_x)
    CALL apply_fluid_damping
    LD (desired_velocity_x), A

check_vertical_fluid_input:
    ; Calculate desired Y velocity similarly
    LD A, (current_fluid_input)
    BIT 2, A            ; Check '3' key (up)
    JR Z, check_down_fluid
    LD A, (acceleration_factor)
    NEG                 ; Make negative for up movement
    LD (desired_velocity_y), A
    JR fluid_velocity_calculated

check_down_fluid:
    BIT 3, A            ; Check '4' key (down)
    JR Z, no_vertical_fluid_input
    LD A, (acceleration_factor)
    LD (desired_velocity_y), A
    JR fluid_velocity_calculated

no_vertical_fluid_input:
    ; No vertical input - apply fluid damping
    LD A, (navigator_velocity_y)
    CALL apply_fluid_damping
    LD (desired_velocity_y), A

fluid_velocity_calculated:
    RET

; Apply fluid damping for smooth deceleration
apply_fluid_damping:
    ; A contains current velocity
    ; Apply damping factor for smooth deceleration
    CP $00              ; Check velocity direction
    JR Z, damping_complete ; Already zero
    JP M, negative_velocity_damping

    ; Positive velocity - reduce toward zero
    LD B, A             ; Save velocity
    LD A, (velocity_damping_factor)
    LD C, A             ; Load damping factor
    LD A, B             ; Restore velocity
    SUB C               ; Apply damping
    JP P, damping_complete ; Still positive
    LD A, $00           ; Clamp to zero
    JR damping_complete

negative_velocity_damping:
    ; Negative velocity - increase toward zero
    LD B, A             ; Save velocity
    LD A, (velocity_damping_factor)
    LD C, A             ; Load damping factor
    LD A, B             ; Restore velocity
    ADD A, C            ; Apply damping (add to negative)
    JP M, damping_complete ; Still negative
    LD A, $00           ; Clamp to zero

damping_complete:
    RET

; Apply fluid velocity smoothing for professional motion
apply_fluid_velocity_smoothing:
    ; Smooth X velocity toward desired velocity
    LD A, (desired_velocity_x)
    LD B, A             ; B = desired velocity
    LD A, (navigator_velocity_x)
    LD C, A             ; C = current velocity

    ; Calculate velocity difference
    LD A, B             ; Desired velocity
    SUB C               ; Subtract current velocity
    JR Z, smooth_y_velocity_fluid ; No change needed

    ; Apply smoothing factor
    LD B, A             ; Save difference
    LD A, (navigation_smoothing_factor)
    LD D, A             ; Load smoothing factor

    ; Limit velocity change rate
    LD A, B             ; Restore difference
    CP $00              ; Check sign
    JP M, negative_x_diff_fluid

    ; Positive difference - limit change rate
    CP D                ; Compare with limit
    JR C, apply_x_change_fluid
    LD A, D             ; Use limit
    JR apply_x_change_fluid

negative_x_diff_fluid:
    ; Negative difference - limit change rate
    NEG                 ; Make positive for comparison
    CP D                ; Compare with limit
    JR C, restore_negative_x
    LD A, D             ; Use limit

restore_negative_x:
    NEG                 ; Make negative again

apply_x_change_fluid:
    ADD A, C            ; Add to current velocity
    LD (navigator_velocity_x), A

smooth_y_velocity_fluid:
    ; Similar smoothing for Y velocity
    LD A, (desired_velocity_y)
    LD B, A
    LD A, (navigator_velocity_y)
    LD C, A

    LD A, B
    SUB C
    JR Z, velocity_smoothing_fluid_complete

    LD B, A
    LD A, (navigation_smoothing_factor)
    LD D, A

    LD A, B
    CP $00
    JP M, negative_y_diff_fluid

    CP D
    JR C, apply_y_change_fluid
    LD A, D
    JR apply_y_change_fluid

negative_y_diff_fluid:
    NEG
    CP D
    JR C, restore_negative_y
    LD A, D

restore_negative_y:
    NEG

apply_y_change_fluid:
    ADD A, C
    LD (navigator_velocity_y), A

velocity_smoothing_fluid_complete:
    RET

; Update navigator position with fluid sub-pixel precision
update_navigator_position_fluid:
    ; Update X position with fractional precision
    LD A, (navigator_smooth_x_fraction)
    ADD A, (navigator_velocity_x)
    LD (navigator_smooth_x_fraction), A

    ; Check for pixel boundary crossing
    JP NC, no_x_pixel_change_fluid

    ; Crossed pixel boundary - update position
    LD BC, (navigator_x)        ; Load current position
    INC B                       ; Increment X
    LD (navigator_x), BC        ; Store new position

no_x_pixel_change_fluid:
    ; Update Y position with fractional precision
    LD A, (navigator_smooth_y_fraction)
    ADD A, (navigator_velocity_y)
    LD (navigator_smooth_y_fraction), A

    ; Check for pixel boundary crossing
    JP NC, no_y_pixel_change_fluid

    ; Crossed pixel boundary - update position
    LD BC, (navigator_x)        ; Load current position
    INC C                       ; Increment Y
    LD (navigator_x), BC        ; Store new position

no_y_pixel_change_fluid:
    RET

; Update optimized attribute display for fluid visual performance
update_optimized_attribute_display:
    ; Check memory update interval for efficiency
    LD A, (memory_update_counter)
    INC A
    LD (memory_update_counter), A

    LD B, A             ; Save counter
    LD A, (memory_update_interval)
    CP B                ; Time for attribute update?
    JR NZ, skip_attribute_update

    ; Reset counter and update attributes
    LD A, $00
    LD (memory_update_counter), A

    ; Update attribute display efficiently
    CALL update_navigator_attributes_optimized
    CALL update_entity_attributes_optimized
    CALL update_achievement_attributes_optimized

skip_attribute_update:
    RET

; Complete frame timing for consistent 25fps
complete_frame_timing:
    ; Calculate frame time used
    LD A, (frame_timing_counter)
    LD B, A             ; Current time
    LD A, (frame_start_time)
    LD C, A             ; Start time
    LD A, B             ; Current time
    SUB C               ; Calculate time used
    LD (frame_time_used), A

    ; Check if we need to wait
    LD A, (target_frame_time)
    LD B, A             ; Target time
    LD A, (frame_time_used)
    CP B                ; Compare with target
    JR NC, frame_complete ; Already used enough time

    ; Wait for remaining time
    LD A, B             ; Target time
    SUB A, (frame_time_used) ; Calculate remaining time
    LD B, A             ; Remaining time

wait_remaining_time:
    ; Simple wait loop for remaining time
    DJNZ wait_remaining_time

frame_complete:
    RET

; Timing interrupt routine for frame synchronization
timing_interrupt:
    ; Save registers
    PUSH AF
    PUSH BC
    PUSH HL

    ; Update frame timing counter
    LD A, (frame_timing_counter)
    INC A
    LD (frame_timing_counter), A

    ; Check for frame update trigger
    LD B, A             ; Current counter
    LD A, (target_frame_interval)
    LD C, A             ; Target interval
    LD A, B             ; Current counter
    AND C               ; Check interval
    JR NZ, timing_interrupt_complete

    ; Set frame update flag
    LD A, $01
    LD (frame_update_flag), A
    JR timing_interrupt_done

timing_interrupt_complete:
    ; Clear frame update flag
    LD A, $00
    LD (frame_update_flag), A

timing_interrupt_done:
    ; Restore registers
    POP HL
    POP BC
    POP AF

    ; Return from interrupt
    EI
    RET
timing_interrupt_end:

; Fluid navigation variables and data
navigator_smooth_x_fraction:    DEFB 0  ; Sub-pixel X position
navigator_smooth_y_fraction:    DEFB 0  ; Sub-pixel Y position
navigator_velocity_x:           DEFB 0  ; Current X velocity
navigator_velocity_y:           DEFB 0  ; Current Y velocity
desired_velocity_x:             DEFB 0  ; Target X velocity
desired_velocity_y:             DEFB 0  ; Target Y velocity

; Navigation smoothing parameters
navigation_smoothing_factor:    DEFB 3  ; Navigation smoothing amount
velocity_damping_factor:        DEFB 2  ; Velocity damping rate
acceleration_factor:            DEFB 6  ; Acceleration amount
navigation_interpolation_factor: DEFB 0 ; Interpolation factor

; Frame timing variables
frame_counter:                  DEFB 0  ; Current frame count
target_frame_interval:          DEFB 3  ; Frames between updates (25fps)
frame_update_flag:              DEFB 0  ; Should update this frame
frame_timing_counter:           DEFB 0  ; Interrupt frame counter
frame_timing_accumulator:       DEFB 0  ; Timing accumulator
frame_timing_enabled:           DEFB 1  ; Timing system enabled

; Frame performance monitoring
frame_start_time:               DEFB 0  ; Frame start time
frame_time_used:                DEFB 0  ; Time used this frame
target_frame_time:              DEFB 50 ; Target frame time (25fps)
last_frame_time:                DEFB 0  ; Previous frame timing

; Input processing variables
current_fluid_input:            DEFB 0  ; Current frame input
previous_input:                 DEFB 0  ; Previous frame input
input_change_mask:              DEFB 0  ; Input change detection

; Memory optimization variables
memory_update_counter:          DEFB 0  ; Memory update frame counter
memory_update_interval:         DEFB 4  ; Update every 4 frames
navigation_pool_pointer:        DEFW 0  ; Navigation memory pool pointer
attribute_base_pointer:         DEFW 0  ; Attribute memory base pointer
navigator_data_pointer:         DEFW 0  ; Navigator data pointer

; Data structures
efficient_navigator_data:       DEFS 16 ; Efficient navigator data
navigation_memory_pool:         DEFS 64 ; Navigation memory pool
optimized_maze_pattern:         DEFS 768 ; Optimized maze data
