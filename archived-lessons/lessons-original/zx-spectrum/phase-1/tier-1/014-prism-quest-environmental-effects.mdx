# Lesson 14: Prism Quest - Environmental Effects

**Your dimensional world needs atmospheric effects!** In your advanced entity behaviors from Lesson 13, you created intelligent spectral AI opponents. Today, you'll add dynamic environmental effects with particle systems, atmospheric hazards, and immersive visual effects that transform your static dimensional maze into a living, breathing spectral world. You'll master graphics programming through creative ZX Spectrum techniques that create engaging environmental storytelling!

create_environmental_attributes_loop:

create_environmental_line_attributes:
    ; Calculate environmental zone colors

    ; Special environmental zones for effects

    ; Environmental zone 2 - hazard zone

    JR store_environmental_attribute

normal_environmental_attribute:
    LD A, D             ; Restore calculated color

store_environmental_attribute:

    INC HL              ; Next character
    DEC C               ; Decrement character counter
    JR NZ, create_environmental_line_attributes

    POP BC              ; Restore line counter
    DJNZ create_environmental_attributes_loop

    RET

; Update particle systems for dynamic environmental effects
update_particle_systems:
    ; Check if it's time to update particles
    LD A, (environmental_update_counter)

    LD (environmental_update_counter), A

    LD A, (environmental_update_frequency)

    JR NZ, skip_particle_update

    LD (environmental_update_counter), A

    ; Spawn new particles
    CALL spawn_dimensional_particles

    ; Update existing particles
    CALL update_existing_particles

    ; Remove expired particles
    CALL remove_expired_particles

skip_particle_update:
    RET

; Spawn dimensional particles based on game state
spawn_dimensional_particles:
    ; Check if we can spawn more particles
    LD A, (current_particle_count)
    LD B, A
    LD A, (max_active_particles)
    CP B                ; Compare with max
    JR C, no_particle_spawn ; Skip if at maximum

    ; Determine particle type based on dimensional state
    CALL determine_dimensional_particle_spawn_type

    ; Spawn atmospheric particles
    CALL spawn_atmospheric_particles

    ; Spawn effect particles from game events
    CALL spawn_dimensional_effect_particles

    ; Spawn hazard particles
    CALL spawn_hazard_particles

no_particle_spawn:
    RET

; Spawn atmospheric particles for ambient environment
spawn_atmospheric_particles:
    ; Check atmospheric particle spawn rate
    LD A, (atmospheric_particle_spawn_counter)
    INC A
    LD (atmospheric_particle_spawn_counter), A
    AND $0F             ; Spawn every 16 frames (slower for Spectrum)
    JR NZ, atmospheric_spawn_complete

    ; Find available particle slot
    CALL find_available_particle_slot
    CP $FF              ; Check if slot found
    JR Z, atmospheric_spawn_complete

    ; Create atmospheric particle
    LD C, A             ; Save slot index
    LD HL, particle_active_array
    ADD A, L
    LD L, A
    JR NC, no_active_carry
    INC H
no_active_carry:
    LD (HL), $01        ; Particle active

    ; Set particle type
    LD HL, particle_type_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_type_carry
    INC H
no_type_carry:
    LD (HL), $01        ; Atmospheric particle type

    ; Random spawn position at edges of screen
    CALL generate_random_number
    AND $1F             ; Limit to screen width
    LD B, A             ; Save X position

    ; Set X position
    LD HL, particle_x_position_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_x_carry
    INC H
no_x_carry:
    LD (HL), B          ; Store X position

    ; Set Y position at top
    LD HL, particle_y_position_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_y_carry
    INC H
no_y_carry:
    LD (HL), $01        ; Top of screen

    ; Set atmospheric particle velocity
    LD HL, particle_x_velocity_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_x_vel_carry
    INC H
no_x_vel_carry:
    LD (HL), $00        ; No horizontal movement

    LD HL, particle_y_velocity_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_y_vel_carry
    INC H
no_y_vel_carry:
    LD (HL), $01        ; Slow downward drift

    ; Set particle life
    LD HL, particle_life_timer_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_life_carry
    INC H
no_life_carry:
    LD (HL), $FF        ; Long life (255 frames)

    ; Increment particle count
    LD A, (current_particle_count)
    INC A
    LD (current_particle_count), A

atmospheric_spawn_complete:
    RET

; Check for spectral interaction events and spawn particles
check_spectral_interaction_events:
    ; Check spectral interaction flags
    LD B, $06           ; Check all 6 spectrals
    LD HL, spectral_interaction_flags

check_interaction_loop:
    LD A, (HL)
    AND A               ; Check if interaction occurred
    JR Z, next_interaction_check

    ; Spectral interaction occurred - create effect particles
    LD (HL), $00        ; Clear interaction flag

    ; Spawn interaction particle burst
    LD D, $03           ; Spawn 3 interaction particles
spawn_interaction_particles_loop:
    PUSH BC             ; Save spectral counter
    PUSH DE             ; Save particle counter
    PUSH HL             ; Save flag pointer

    CALL find_available_particle_slot
    CP $FF              ; Check if slot found
    JR Z, interaction_particle_complete

    ; Create interaction particle
    LD C, A             ; Save slot index
    LD HL, particle_active_array
    ADD A, L
    LD L, A
    JR NC, no_int_active_carry
    INC H
no_int_active_carry:
    LD (HL), $01        ; Particle active

    ; Set interaction particle type
    LD HL, particle_type_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_int_type_carry
    INC H
no_int_type_carry:
    LD (HL), $02        ; Interaction particle type

    ; Set interaction position from spectral
    LD A, (spectral_interaction_x)
    LD HL, particle_x_position_array
    LD B, C
    LD C, A             ; Save X position
    LD A, B
    ADD A, L
    LD L, A
    JR NC, no_int_x_carry
    INC H
no_int_x_carry:
    LD (HL), C          ; Store X position

    LD A, (spectral_interaction_y)
    LD HL, particle_y_position_array
    LD B, C
    LD C, A             ; Save Y position
    LD A, B
    ADD A, L
    LD L, A
    JR NC, no_int_y_carry

no_int_y_carry:
    LD (HL), C          ; Store Y position

    ; Random interaction velocity
    CALL generate_random_number

    SUB $04             ; Center around 0 (-4 to +3)
    LD HL, particle_x_velocity_array
    LD B, C
    LD C, A             ; Save X velocity
    LD A, B
    ADD A, L
    LD L, A
    JR NC, no_int_x_vel_carry
    INC H
no_int_x_vel_carry:
    LD (HL), C

    CALL generate_random_number
    AND $07             ; Limit velocity range
    SUB $04             ; Center around 0 (-4 to +3)
    LD HL, particle_y_velocity_array

    LD C, A             ; Save Y velocity

    ADD A, L

    LD A, B             ; Restore slot index
    ADD A, L
    LD L, A
    JR NC, no_int_life_carry
    INC H
no_int_life_carry:

    ; Increment particle count
    LD A, (current_particle_count)

    LD (current_particle_count), A

interaction_particle_complete:
    POP HL              ; Restore flag pointer
    POP DE              ; Restore particle counter
    POP BC              ; Restore spectral counter
    DEC D
    JR NZ, spawn_interaction_particles_loop

next_interaction_check:

    DJNZ check_interaction_loop

; Update existing particles with physics simulation
update_existing_particles:
    ; Process all active particles
    LD B, $10           ; Check all 16 particles
    LD C, $00           ; Particle index

update_particles_loop:

    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_check_carry
    INC H
no_check_carry:

    JR Z, next_particle_update

    CALL update_particle_physics

    ; Update particle life
    CALL update_particle_life

next_particle_update:

    DJNZ update_particles_loop

; Update particle physics for realistic movement
update_particle_physics:
    ; C contains particle index
    ; Update X position
    LD HL, particle_x_position_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_phys_x_carry
    INC H
no_phys_x_carry:
    LD D, (HL)          ; Load X position

    LD HL, particle_x_velocity_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_phys_x_vel_carry
    INC H
no_phys_x_vel_carry:

    ADD A, (HL)         ; Add velocity
    LD D, A             ; Save new X position

    ; Check X boundaries
    CP $20              ; Right boundary
    JR C, check_particle_y_physics

    ; Outside boundaries - mark for removal

    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_phys_life_carry
    INC H

    LD (HL), $01        ; 1 frame life (will expire next update)

check_particle_y_physics:

    LD HL, particle_x_position_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_phys_store_x_carry
    INC H
no_phys_store_x_carry:
    LD (HL), D          ; Store new X position

    ; Update Y position
    LD HL, particle_y_position_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_phys_y_carry
    INC H
no_phys_y_carry:

    LD HL, particle_y_velocity_array

    ADD A, L
    LD L, A
    JR NC, no_phys_y_vel_carry
    INC H

    ADD A, (HL)         ; Add velocity
    LD D, A             ; Save new Y position

    ; Check Y boundaries
    CP $18              ; Bottom boundary
    JR C, store_particle_y_position

    ; Below screen - mark for removal
    LD HL, particle_life_timer_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_phys_y_life_carry
    INC H
no_phys_y_life_carry:
    LD (HL), $01        ; 1 frame life (will expire next update)

store_particle_y_position:
    ; Store Y position
    LD HL, particle_y_position_array
    LD A, C
    ADD A, L
    LD L, A
    JR NC, no_phys_store_y_carry
    INC H
no_phys_store_y_carry:
    LD (HL), D          ; Store new Y position

    RET

; Update atmospheric effects for environmental immersion
update_atmospheric_effects:
    ; Update atmospheric color cycling
    CALL update_atmospheric_color_cycling

    ; Update atmospheric density
    CALL update_atmospheric_density

    ; Update environmental mood
    CALL update_environmental_mood

    RET

; Update atmospheric color cycling for dynamic environment

    ; Increment color cycle counter
    LD A, (color_cycle_counter)
    INC A
    LD (color_cycle_counter), A

    LD B, A             ; Save counter
    LD A, (color_cycle_speed)

    JR NZ, color_cycling_complete

    ; Reset counter and cycle color
    LD A, $00
    LD (color_cycle_counter), A

    ; Cycle through atmospheric colors
    LD A, (current_atmospheric_color)
    INC A
    CP $08              ; Wrap at 8 colors
    JR C, store_atmospheric_color
    LD A, $01           ; Reset to start color (not black)

    LD (current_atmospheric_color), A

    ; Update border color for atmospheric effect
    OUT ($FE), A        ; Set border color

color_cycling_complete:
    RET

; Update dimensional hazards for dynamic challenges

    ; Update hazard spawn timing
    CALL update_hazard_spawn_timing

    ; Update existing hazards
    CALL update_existing_hazards

    ; Check hazard interactions
    CALL check_hazard_interactions

    RET

    LD A, (hazard_spawn_timer)
    AND A               ; Check if zero
    JR Z, reset_hazard_timer
    DEC A
    LD (hazard_spawn_timer), A
    JR hazard_spawn_timing_complete

    ; Reset timer and spawn hazard
    LD A, (hazard_spawn_interval)
    LD (hazard_spawn_timer), A

    ; Check if we can spawn more hazards
    LD A, (active_hazard_count)
    LD B, A
    LD A, (max_dimensional_hazards)
    CP B                ; Compare with max
    JR C, hazard_spawn_timing_complete ; Skip if at maximum

    ; Spawn new dimensional hazard
    CALL spawn_dimensional_hazard

    RET

; Spawn dimensional hazard for gameplay challenge

    CALL find_available_hazard_slot
    CP $FF              ; Check if slot found
    JR Z, hazard_spawn_complete

    LD HL, hazard_active_array

    LD L, A
    JR NC, no_haz_active_carry
    INC H

    LD (HL), $01        ; Hazard active

    ; Determine hazard type based on current dimensional state
    LD A, (global_environmental_state)
    AND $03             ; 4 hazard types
    LD HL, hazard_type_array
    LD B, C
    LD C, A             ; Save hazard type
    LD A, B
    ADD A, L
    LD L, A
    JR NC, no_haz_type_carry
    INC H
no_haz_type_carry:
    LD (HL), C

    ; Set hazard position
    CALL generate_random_number
    AND $1F             ; Limit to screen area
    ADD A, $02          ; Offset from edges
    LD HL, hazard_x_position_array
    LD B, C
    LD C, A             ; Save X position
    LD A, B
    ADD A, L
    LD L, A
    JR NC, no_haz_x_carry
