```nasm
; Circuit Surge - Energy Blasts Creation
.segment "HEADER"
.byte "NES", $1A
.byte $02, $01
.byte $00, $00

.segment "CODE"
RESET:
    ; Initialize NES for precise electrical combat control
    SEI
    CLD
    LDX #$FF
    TXS

    ; Set up enhanced digital circuit world from Lessons 1-4
    LDA #$3F            ; Point to palette
    STA $2006           ; PPU address high
    LDA #$00            ; Point to palette start
    STA $2006           ; PPU address low

    ; Enhanced circuit colors with combat elements
    LDA #$0F            ; Black background
    STA $2007           ; Write palette
    LDA #$30            ; White conductors
    STA $2007           ; Write palette
    LDA #$16            ; Red energy blasts
    STA $2007           ; Write palette
    LDA #$27            ; Yellow electrical discharges
    STA $2007           ; Write palette

    ; Initialize current conductor from previous lessons
    LDA #$05            ; Starting X position
    STA $80             ; Store in zero page
    LDA #$03            ; Starting Y position
    STA $81             ; Store in zero page

    ; Initialize circuit challenges from Lesson 4
    JSR InitChallengeSystem
    JSR CreateCircuitObstacles

    ; Initialize energy combat system
    JSR InitCombatSystem

    ; Create electrical weapon systems
    JSR InitWeaponSystems

    ; Initialize energy management
    JSR InitEnergyManagement

    ; Main electrical combat loop
MainLoop:
    JSR ReadCombatInput
    JSR UpdateConductorCombat
    JSR UpdateWeaponSystems
    JSR UpdateCircuitObstacles
    JSR CheckElectricalCollisions
    JSR UpdateEnergySystems
    JSR UpdateCombatEffects
    JMP MainLoop

; Initialize electrical combat system
InitCombatSystem:
    ; Set up combat state variables in zero page
    LDA #$64            ; Conductor voltage (100)
    STA $A0             ; Store conductor voltage

    LDA #$32            ; Conductor charge (50)
    STA $A1             ; Store conductor charge

    LDA #$20            ; Weapon capacitance (32 shots)
    STA $A2             ; Store weapon capacitance

    ; Initialize combat timing
    LDA #$00            ; Combat timer
    STA $A3             ; Store timer

    ; Set electrical combat flags
    LDA #$00            ; No weapon firing initially
    STA $A4             ; Weapon discharge flag
    STA $A5             ; Hit spark flag
    STA $A6             ; Circuit destruction flag

    RTS

; Initialize electrical weapon systems
InitWeaponSystems:
    ; Set up energy pulse array (8 pulses maximum)
    LDX #$07            ; Pulse index counter
ClearEnergyPulses:
    LDA #$00            ; Inactive pulse
    STA $B0,X           ; Clear pulse active flag
    STA $B8,X           ; Clear pulse X position
    STA $C0,X           ; Clear pulse Y position
    STA $C8,X           ; Clear pulse direction
    DEX                 ; Decrement index
    BPL ClearEnergyPulses ; Continue until all cleared

    ; Set up weapon parameters
    LDA #$08            ; Weapon voltage
    STA $D0             ; Store weapon damage
    LDA #$04            ; Pulse speed
    STA $D1             ; Store pulse speed
    LDA #$10            ; Weapon charge cost
    STA $D2             ; Store charge per shot

    RTS

; Initialize energy management systems
InitEnergyManagement:
    ; Set up charge regeneration
    LDA #$3C            ; Charge regen rate (60 frames)
    STA $E0             ; Store regen timer

    LDA #$02            ; Charge regen amount
    STA $E1             ; Store regen per cycle

    ; Set up voltage systems
    LDA #$FF            ; Voltage regen rate (255 frames)
    STA $E2             ; Store voltage timer

    LDA #$01            ; Voltage regen amount
    STA $E3             ; Store voltage per cycle

    RTS

; Read electrical combat input from player
ReadCombatInput:
    ; Read controller for combat controls
    LDA $4016           ; Read controller port 1

    ; Check for energy blast (A button)
    AND #%00000001      ; Check A button bit
    BEQ NoWeaponDischarge ; Branch if not pressed

    ; Energy blast requested
    LDA #$01            ; Set discharge flag
    STA $A4             ; Store discharge request
    JMP InputDone

NoWeaponDischarge:
    LDA #$00            ; Clear discharge flag
    STA $A4             ; Store no discharge

InputDone:
    RTS

; Update conductor electrical combat systems
UpdateConductorCombat:
    ; Check if energy blast requested
    LDA $A4             ; Load discharge flag
    BEQ NoDischarge     ; Branch if no discharge requested

    ; Check if conductor has charge for weapon
    LDA $A1             ; Load conductor charge
    CMP $D2             ; Compare with weapon cost
    BCC NoDischarge     ; Branch if insufficient charge

    ; Check weapon cooldown
    LDA $A3             ; Load combat timer
    BNE NoDischarge     ; Branch if weapon on cooldown

    ; Fire electrical weapon
    JSR FireEnergyPulse

    ; Consume charge
    LDA $A1             ; Load current charge
    SEC                 ; Set carry for subtraction
    SBC $D2             ; Subtract weapon cost
    STA $A1             ; Store new charge

    ; Set weapon cooldown
    LDA #$08            ; Cooldown time
    STA $A3             ; Store cooldown timer

NoDischarge:
    ; Update combat timer
    LDA $A3             ; Load combat timer
    BEQ TimerDone       ; Skip if already zero
    DEC $A3             ; Decrement timer

TimerDone:
    RTS

; Fire electrical energy pulse
FireEnergyPulse:
    ; Find empty pulse slot
    LDX #$07            ; Start with last pulse
FindEmptySlot:
    LDA $B0,X           ; Load pulse active flag
    BEQ FoundSlot       ; Branch if empty slot found
    DEX                 ; Try previous slot
    BPL FindEmptySlot   ; Continue if more slots
    RTS                 ; No empty slots available

FoundSlot:
    ; Activate energy pulse
    LDA #$01            ; Set active flag
    STA $B0,X           ; Store in pulse array

    ; Set pulse position to conductor position
    LDA $80             ; Load conductor X
    CLC                 ; Clear carry
    ADC #$01            ; Offset slightly right
    STA $B8,X           ; Store pulse X

    LDA $81             ; Load conductor Y
    STA $C0,X           ; Store pulse Y

    ; Set pulse direction (moving right)
    LDA #$01            ; Direction right
    STA $C8,X           ; Store direction

    ; Play electrical discharge sound
    JSR PlayDischargeSound

    RTS

; Update all electrical weapon pulses
UpdateWeaponSystems:
    ; Loop through all energy pulses
    LDX #$07            ; Start with last pulse
UpdatePulseLoop:
    ; Check if pulse is active
    LDA $B0,X           ; Load active flag
    BEQ SkipPulse       ; Skip if inactive

    ; Update pulse position
    LDA $C8,X           ; Load direction
    CMP #$01            ; Check if moving right
    BEQ MoveRight

MoveLeft:
    ; Move pulse left
    LDA $B8,X           ; Load X position
    SEC                 ; Set carry for subtraction
    SBC $D1             ; Subtract speed
    BCC DeactivatePulse ; Deactivate if off screen
    STA $B8,X           ; Store new position
    JMP CheckPulseCollision

MoveRight:
    ; Move pulse right
    LDA $B8,X           ; Load X position
    CLC                 ; Clear carry
    ADC $D1             ; Add speed
    CMP #$1E            ; Check screen boundary
    BCS DeactivatePulse ; Deactivate if off screen
    STA $B8,X           ; Store new position

CheckPulseCollision:
    ; Check collision with circuit obstacles
    JSR CheckPulseObstacleCollision
    JMP ContinuePulseUpdate

DeactivatePulse:
    ; Deactivate off-screen pulse
    LDA #$00            ; Clear active flag
    STA $B0,X           ; Store in array

ContinuePulseUpdate:
SkipPulse:
    DEX                 ; Previous pulse
    BPL UpdatePulseLoop ; Continue until all processed

    RTS

; Check energy pulse collision with circuit obstacles
CheckPulseObstacleCollision:
    ; Save pulse index
    STX $F0             ; Temporary storage

    ; Check collision with Obstacle 1
    LDA $91             ; Load Obstacle 1 status
    BEQ CheckObstacle2  ; Skip if inactive

    ; Compare positions
    LDA $B8,X           ; Load pulse X
    CMP $95             ; Compare with Obstacle 1 X
    BNE CheckObstacle2  ; Different X positions

    LDA $C0,X           ; Load pulse Y
    CMP $96             ; Compare with Obstacle 1 Y
    BNE CheckObstacle2  ; Different Y positions

    ; Electrical hit detected!
    JSR HandleObstacleHit

    ; Deactivate pulse
    LDX $F0             ; Restore pulse index
    LDA #$00            ; Clear active flag
    STA $B0,X           ; Deactivate pulse

    ; Destroy Obstacle 1
    LDA #$00            ; Destroy obstacle
    STA $91             ; Mark Obstacle 1 as destroyed

    JMP CollisionCheckDone

CheckObstacle2:
    ; Similar collision check for Obstacle 2
    LDA $92             ; Load Obstacle 2 status
    BEQ CheckObstacle3  ; Skip if inactive

    LDX $F0             ; Restore pulse index
    LDA $B8,X           ; Load pulse X
    CMP $98             ; Compare with Obstacle 2 X
    BNE CheckObstacle3  ; Different X positions

    LDA $C0,X           ; Load pulse Y
    CMP $99             ; Compare with Obstacle 2 Y
    BNE CheckObstacle3  ; Different Y positions

    ; Electrical hit detected!
    JSR HandleObstacleHit

    ; Deactivate pulse and obstacle
    LDX $F0             ; Restore pulse index
    LDA #$00            ; Clear flags
    STA $B0,X           ; Deactivate pulse
    STA $92             ; Mark Obstacle 2 as destroyed

    JMP CollisionCheckDone

CheckObstacle3:
    ; Similar collision check for Obstacle 3
    LDA $93             ; Load Obstacle 3 status
    BEQ CollisionCheckDone ; Skip if inactive

    LDX $F0             ; Restore pulse index
    LDA $B8,X           ; Load pulse X
    CMP $9B             ; Compare with Obstacle 3 X
    BNE CollisionCheckDone ; Different X positions

    LDA $C0,X           ; Load pulse Y
    CMP $9C             ; Compare with Obstacle 3 Y
    BNE CollisionCheckDone ; Different Y positions

    ; Electrical hit detected!
    JSR HandleObstacleHit

    ; Deactivate pulse and obstacle
    LDX $F0             ; Restore pulse index
    LDA #$00            ; Clear flags
    STA $B0,X           ; Deactivate pulse
    STA $93             ; Mark Obstacle 3 as destroyed

CollisionCheckDone:
    LDX $F0             ; Restore pulse index
    RTS

; Handle electrical obstacle hit effects
HandleObstacleHit:
    ; Flash screen for electrical hit effect
    LDA #$16            ; Red electrical flash
    STA $2001           ; PPU control

    ; Play electrical hit sound
    JSR PlayElectricalHitSound

    ; Award voltage/charge
    LDA $A1             ; Load current charge
    CLC                 ; Clear carry
    ADC #$05            ; Add charge reward
    CMP #$64            ; Check maximum
    BCC StoreCharge     ; Store if within limit
    LDA #$64            ; Clamp to maximum

StoreCharge:
    STA $A1             ; Store updated charge

    ; Set destruction effect flag
    LDA #$10            ; Effect duration
    STA $A6             ; Store destruction timer

    RTS

; Check electrical collisions (obstacle damage to conductor)
CheckElectricalCollisions:
    ; Check collision with Obstacle 1
    LDA $91             ; Load Obstacle 1 status
    BEQ CheckConductorObstacle2 ; Skip if inactive

    ; Compare conductor and Obstacle 1 positions
    LDA $80             ; Load conductor X
    CMP $95             ; Compare with Obstacle 1 X
    BNE CheckConductorObstacle2 ; Different X positions

    LDA $81             ; Load conductor Y
    CMP $96             ; Compare with Obstacle 1 Y
    BNE CheckConductorObstacle2 ; Different Y positions

    ; Electrical collision detected! Handle conductor damage
    JSR HandleConductorDamage

CheckConductorObstacle2:
    ; Similar collision check for Obstacle 2
    LDA $92             ; Load Obstacle 2 status
    BEQ CheckConductorObstacle3 ; Skip if inactive

    LDA $80             ; Load conductor X
    CMP $98             ; Compare with Obstacle 2 X
    BNE CheckConductorObstacle3 ; Different X positions

    LDA $81             ; Load conductor Y
    CMP $99             ; Compare with Obstacle 2 Y
    BNE CheckConductorObstacle3 ; Different Y positions

    ; Electrical collision detected!
    JSR HandleConductorDamage

CheckConductorObstacle3:
    ; Similar collision check for Obstacle 3
    LDA $93             ; Load Obstacle 3 status
    BEQ ConductorCollisionDone ; Skip if inactive

    LDA $80             ; Load conductor X
    CMP $9B             ; Compare with Obstacle 3 X
    BNE ConductorCollisionDone ; Different X positions

    LDA $81             ; Load conductor Y
    CMP $9C             ; Compare with Obstacle 3 Y
    BNE ConductorCollisionDone ; Different Y positions

    ; Electrical collision detected!
    JSR HandleConductorDamage

ConductorCollisionDone:
    RTS

; Handle conductor taking electrical damage
HandleConductorDamage:
    ; Reduce conductor voltage
    LDA $A0             ; Load current voltage
    SEC                 ; Set carry for subtraction
    SBC #$0A            ; Subtract damage (10 voltage)
    BCS StoreVoltage    ; Store if positive
    LDA #$00            ; Clamp to zero

StoreVoltage:
    STA $A0             ; Store updated voltage

    ; Flash screen red for electrical damage
    LDA #$16            ; Red electrical color
    STA $2001           ; PPU control

    ; Set hit spark timer
    LDA #$08            ; Spark duration
    STA $A5             ; Store spark timer

    ; Play electrical damage sound
    JSR PlayDamageSound

    ; Check for circuit failure
    LDA $A0             ; Load voltage
    BNE DamageDone      ; Continue if voltage remaining

    ; Circuit failure sequence
    JSR HandleCircuitFailure

DamageDone:
    RTS

; Update electrical energy management systems
UpdateEnergySystem:
    ; Update charge regeneration
    DEC $E0             ; Decrement charge timer
    BNE CheckVoltageRegen ; Skip if timer not zero

    ; Regenerate charge
    LDA $A1             ; Load current charge
    CLC                 ; Clear carry
    ADC $E1             ; Add regen amount
    CMP #$64            ; Check maximum charge
    BCC StoreRegenCharge ; Store if within limit
    LDA #$64            ; Clamp to maximum

StoreRegenCharge:
    STA $A1             ; Store updated charge

    ; Reset charge timer
    LDA #$3C            ; Reset timer
    STA $E0             ; Store reset value

CheckVoltageRegen:
    ; Update voltage regeneration
    DEC $E2             ; Decrement voltage timer
    BNE EnergyUpdateDone ; Skip if timer not zero

    ; Check if voltage is below maximum
    LDA $A0             ; Load current voltage
    CMP #$64            ; Compare with maximum
    BCS ResetVoltageTimer ; Skip if at maximum

    ; Regenerate voltage
    CLC                 ; Clear carry
    ADC $E3             ; Add regen amount
    CMP #$64            ; Check maximum
    BCC StoreRegenVoltage ; Store if within limit
    LDA #$64            ; Clamp to maximum

StoreRegenVoltage:
    STA $A0             ; Store updated voltage

ResetVoltageTimer:
    ; Reset voltage timer
    LDA #$FF            ; Reset timer
    STA $E2             ; Store reset value

EnergyUpdateDone:
    RTS

; Update electrical combat visual and audio effects
UpdateCombatEffects:
    ; Update electrical spark effect
    LDA $A5             ; Load spark timer
    BEQ CheckDestructionEffect ; Skip if no spark

    DEC $A5             ; Decrement spark timer
    BNE SparkContinue   ; Continue sparking

    ; Spark finished - restore normal display
    LDA #$1E            ; Normal PPU control
    STA $2001           ; PPU control

SparkContinue:
CheckDestructionEffect:
    ; Update destruction effect
    LDA $A6             ; Load destruction timer
    BEQ DrawCombatHUD   ; Skip if no effect

    DEC $A6             ; Decrement effect timer

    ; Create electrical destruction visual effect
    LDA $A6             ; Load timer
    AND #$03            ; Mask for color cycling
    CLC                 ; Clear carry
    ADC #$16            ; Offset electrical color
    STA $2001           ; PPU electrical flash effect

DrawCombatHUD:
    ; Draw electrical heads-up display
    JSR DrawVoltageBar
    JSR DrawChargeBar
    JSR DrawEnergyPulses

    RTS

; Draw conductor voltage bar (simplified for space)
DrawVoltageBar:

    LDA #$20            ; Nametable address high
    STA $2006           ; PPU address high
    LDA #$40            ; Voltage bar position
    STA $2006           ; PPU address low

    ; Display voltage value
    LDA $A0             ; Load voltage
    JSR DisplayElectricalValue

    RTS

; Draw conductor charge bar (simplified for space)
DrawChargeBar:
    ; Position and draw charge display
    LDA #$20            ; Nametable address high
    STA $2006           ; PPU address high
    LDA #$60            ; Charge bar position
    STA $2006           ; PPU address low

    ; Display charge value

    JSR DisplayElectricalValue

    RTS

; Draw active energy pulses (simplified for space)
DrawEnergyPulses:
    ; Loop through all energy pulses
    LDX #$07            ; Start with last pulse
DrawPulseLoop:

    LDA $B0,X           ; Load active flag
    BEQ SkipDrawPulse   ; Skip if inactive

    ; Calculate pulse nametable position
    LDA $C0,X           ; Load pulse Y
    ASL                 ; Multiply by 2
    ASL                 ; Multiply by 4
    ASL                 ; Multiply by 8
    ASL                 ; Multiply by 16
    ASL                 ; Multiply by 32
    CLC                 ; Clear carry
    ADC $B8,X           ; Add pulse X
    TAY                 ; Transfer to Y for indexing

    ; Draw pulse character
    LDA #$2E            ; Energy pulse character
    STA $2007           ; PPU data

SkipDrawPulse:
    DEX                 ; Previous pulse
    BPL DrawPulseLoop   ; Continue until all processed

    RTS

; Play electrical discharge sound
PlayDischargeSound:
    ; Use pulse channel for electrical discharge sound
    LDA #$81            ; Duty cycle, constant volume
    STA $4000           ; APU pulse 1 control

    LDA #$F0            ; High frequency for electrical discharge
    STA $4002           ; APU pulse 1 frequency low
    LDA #$04            ; High frequency
    STA $4003           ; APU pulse 1 frequency high

    RTS

; Play electrical hit sound
PlayElectricalHitSound:
    ; Use noise channel for electrical hit sound
    LDA #$30            ; Volume
    STA $400C           ; APU noise control

    LDA #$08            ; Electrical hit frequency
    STA $400E           ; APU noise frequency

    RTS

; Play electrical damage sound
PlayDamageSound:
    ; Use triangle channel for electrical damage sound
    LDA #$81            ; Linear counter
    STA $4008           ; APU triangle control

    LDA #$80            ; Electrical damage frequency
    STA $400A           ; APU triangle frequency low
    LDA #$02            ; Electrical damage frequency high
    STA $400B           ; APU triangle frequency high

    RTS

; Handle circuit failure sequence
HandleCircuitFailure:
    ; Flash screen multiple times with electrical pattern
    LDX #$08            ; Flash count
CircuitFailureFlash:
    LDA #$16            ; Red electrical color
    STA $2001           ; PPU control

    ; Electrical delay
    LDY #$FF            ; Delay counter
CircuitFailureDelay1:
    DEY                 ; Decrement counter
    BNE CircuitFailureDelay1 ; Continue until zero

    LDA #$0F            ; Black color
    STA $2001           ; PPU control

    ; Electrical delay
    LDY #$FF            ; Delay counter
CircuitFailureDelay2:
    DEY                 ; Decrement counter
    BNE CircuitFailureDelay2 ; Continue until zero

    DEX                 ; Decrement flash counter
    BNE CircuitFailureFlash ; Continue flashing

    ; Reset circuit
    JMP RESET

; Display electrical value (simplified)
DisplayElectricalValue:
    ; Convert and display electrical value (simplified)
    LSR                 ; Shift right 4 times
    LSR
    LSR
    LSR
    CLC                 ; Clear carry
    ADC #$30            ; Convert to ASCII-like
    STA $2007           ; PPU data
    RTS

; Enhanced circuit challenge system from Lesson 4 (simplified)
InitChallengeSystem:
    LDA #$03            ; Number of circuit obstacles
    STA $90             ; Store obstacle count

    ; Initialize obstacle states
    LDA #$01            ; All obstacles active
    STA $91             ; Obstacle 1 status
    STA $92             ; Obstacle 2 status
    STA $93             ; Obstacle 3 status

    RTS

CreateCircuitObstacles:
    ; Create Obstacle 1
    LDA #$0A            ; Obstacle 1 X position
    STA $95             ; Store X
    LDA #$05            ; Obstacle 1 Y position
    STA $96             ; Store Y

    ; Create Obstacle 2
    LDA #$15            ; Obstacle 2 X position
    STA $98             ; Store X
    LDA #$08            ; Obstacle 2 Y position
    STA $99             ; Store Y

    ; Create Obstacle 3
    LDA #$20            ; Obstacle 3 X position
    STA $9B             ; Store X
    LDA #$0C            ; Obstacle 3 Y position
    STA $9C             ; Store Y

    RTS

UpdateCircuitObstacles:
    ; Simplified obstacle behavior (from Lesson 4)
    ; Obstacle 1 horizontal patrol
    LDA $91             ; Check if active
    BEQ SkipObstacle1   ; Skip if destroyed

    LDA $95             ; Load Obstacle 1 X
    CLC                 ; Clear carry
    ADC #$01            ; Move right
    CMP #$1C            ; Check boundary
    BCC StoreObstacle1X ; Store if within bounds
    LDA #$05            ; Reset to start

StoreObstacle1X:
    STA $95             ; Store new position

SkipObstacle1:
    ; Similar updates for Obstacle 2 and 3 (simplified)
    RTS

.segment "VECTORS"
.word RESET, RESET, RESET
```

```nasm

    JSR InitAdvancedElectricalWeapons

InitAdvancedElectricalWeapons:

    LDA #$01            ; Weapon type 1 (rapid pulse)
    STA ElectricalWeaponTypes
    LDA #$02            ; Weapon type 2 (spread discharge)
    STA ElectricalWeaponTypes+1

    ; Initialize electrical weapon stats arrays
    LDX #$02            ; Three electrical weapon types
InitElectricalWeaponStats:
    LDA ElectricalWeaponVoltage,X ; Load voltage value
    STA CurrentVoltage,X ; Store current stats
    LDA ElectricalWeaponSpeed,X ; Load speed value
    STA CurrentSpeed,X ; Store current stats
    DEX                 ; Previous weapon
    BPL InitElectricalWeaponStats ; Continue until all initialized

    RTS

FireAdvancedElectricalWeapon:
    ; Load current electrical weapon type
    LDA CurrentElectricalWeapon ; Load selected weapon
    TAX                 ; Use as array index

    ; Fire based on electrical weapon type
    LDA ElectricalWeaponTypes,X ; Load weapon type
    CMP #$01            ; Compare with rapid pulse
    BEQ FireRapidPulse
    CMP #$02            ; Compare with spread discharge
    BEQ FireSpreadDischarge
    CMP #$03            ; Compare with heavy voltage
    BEQ FireHeavyVoltage

FireRapidPulse:
    ; Fire multiple rapid electrical pulses
    LDX #$03            ; Fire 3 pulses
RapidPulseLoop:
    JSR CreateEnergyPulse ; Create pulse
    DEX                 ; Decrement counter
    BNE RapidPulseLoop  ; Continue until all fired
    RTS

FireSpreadDischarge:
    ; Fire electrical spread pattern
```
