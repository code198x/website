; Generate circuit progress code from current electrical state
GenerateCircuitProgressCode:
    ; Clear circuit progress code buffer
    LDX #$0B
clear_circuit_progress_buffer_loop:
    LDA #$00
    STA circuit_progress_code_buffer,X
    DEX
    BPL clear_circuit_progress_buffer_loop

    ; Encode conductor state (compact representation for NES)
    LDA conductor_x_position
    LSR                 ; Compress X position to 7 bits
    AND #$7F
    STA temp_circuit_encoding_buffer

    LDA conductor_y_position
    LSR                 ; Compress Y position to 7 bits
    AND #$7F
    STA temp_circuit_encoding_buffer+1

    ; Encode electrical achievement score (8 bits total)
    LDA conductor_achievement_score_high
    AND #$0F            ; Take lower 4 bits of high score
    ASL                 ; Shift left 4 positions
    ASL
    ASL
    ASL
    TAY                 ; Save in Y

    LDA conductor_achievement_score_low
    LSR                 ; Shift right 4 positions
    LSR
    LSR
    LSR
    ORA conductor_achievement_score_high ; This should be TYA, ORA temp storage
    STA temp_circuit_encoding_buffer+2

    ; Encode electrical level and circuit achievements (8 bits)
    LDA current_electrical_level
    AND #$0F            ; 4 bits for electrical level
    ASL                 ; Shift left 4 positions
    ASL
    ASL
    ASL
    TAY                 ; Save in Y

    LDA circuit_achievements_unlocked
    AND #$0F            ; 4 bits for circuit achievements
    TYA                 ; Get level bits back
    ORA circuit_achievements_unlocked ; Combine with level bits
    STA temp_circuit_encoding_buffer+3

    ; Calculate circuit checksum
    JSR CalculateCircuitProgressCodeChecksum
    LDA circuit_progress_code_checksum
    STA temp_circuit_encoding_buffer+4

    ; Convert binary data to circuit character code
    JSR ConvertBinaryToCircuitCharacterCode

    RTS

; Convert binary data to circuit character code
ConvertBinaryToCircuitCharacterCode:
    ; Convert 5 bytes of binary data to 8-character code
    LDX #$00            ; Source byte index
    LDY #$00            ; Destination character index

convert_circuit_character_loop:
    ; Take 5 bits from current byte (simplified bit packing)
    LDA temp_circuit_encoding_buffer,X
    AND #$1F            ; 5 bits = 32 values (0-31)

    ; Convert to character using circuit encoding table
    TAX
    LDA circuit_character_encoding_table,X
    STA circuit_progress_code_buffer,Y

    ; Move to next position
    LDX temp_source_index ; Need to restore X from temp storage
    INX                 ; Next source byte
    STX temp_source_index
    INY                 ; Next destination character
    CPY #$08            ; 8 characters
    BNE convert_circuit_character_loop

    RTS

; Calculate circuit progress code checksum
CalculateCircuitProgressCodeChecksum:
    ; Simple checksum of circuit encoded data
    LDA #$00
    STA circuit_progress_code_checksum

    LDX #$03            ; 4 bytes of data (excluding checksum)
circuit_checksum_loop:
    LDA circuit_progress_code_checksum
    CLC
    ADC temp_circuit_encoding_buffer,X
    STA circuit_progress_code_checksum
    DEX
    BPL circuit_checksum_loop

    ; Apply circuit polynomial transformation
    LDA circuit_progress_code_checksum
    TAY                 ; Save original
    LDA checksum_polynomial
    EOR circuit_progress_code_checksum ; XOR with polynomial
    STA circuit_progress_code_checksum

    RTS

; Generate electrical access password for current circuit stage
GenerateElectricalAccessPassword:
    ; Simple electrical encoding using level number and circuit checksum
    LDA current_electrical_level
    STA electrical_access_password_buffer

    ; Add electrical-specific modifier
    CLC
    ADC #$2B            ; Offset to avoid simple patterns
    STA electrical_access_password_buffer+1

    ; Add circuit achievement-based modifier
    LDA circuit_achievements_unlocked
    CLC
    ADC current_electrical_level
    STA electrical_access_password_buffer+2

    ; Calculate electrical checksum
    LDA electrical_access_password_buffer
    CLC
    ADC electrical_access_password_buffer+1
    CLC
    ADC electrical_access_password_buffer+2
    EOR #$99            ; XOR with electrical pattern
    STA electrical_access_password_buffer+3

    ; Convert to circuit characters
    LDX #$00
convert_electrical_password_loop:
    LDA electrical_access_password_buffer,X
    AND #$1F            ; 5 bits
    TAY
    LDA circuit_character_encoding_table,Y
    STA electrical_code_display_buffer,X
    INX
    CPX #$04            ; 4 characters
    BNE convert_electrical_password_loop

    RTS

; Validate entered electrical password
ValidateEnteredElectricalPassword:
    ; Check password type
    LDA current_password_type
    CMP #$01            ; Circuit progress code
    BEQ validate_circuit_progress_code
    CMP #$02            ; Electrical access code
    BEQ validate_electrical_access_code

    ; Unknown password type
    LDA #$00
    STA password_validation_result
    RTS

validate_circuit_progress_code:
    ; Convert entered characters back to binary
    JSR ConvertCircuitCharacterCodeToBinary

    ; Recalculate circuit checksum
    JSR CalculateCircuitProgressCodeChecksum

    ; Compare with embedded checksum
    LDA circuit_progress_code_checksum
    CMP temp_circuit_encoding_buffer+4
    BNE circuit_progress_validation_failed

    ; Checksum valid - restore electrical game state
    JSR RestoreElectricalGameStateFromCode
    LDA #$01
    STA password_validation_result
    RTS

circuit_progress_validation_failed:
    LDA #$00
    STA password_validation_result
    RTS

validate_electrical_access_code:
    ; Validate electrical access code checksum
    LDA password_entry_buffer
    CLC
    ADC password_entry_buffer+1
    CLC
    ADC password_entry_buffer+2
    EOR #$99            ; Apply electrical pattern
    CMP password_entry_buffer+3
    BNE electrical_access_validation_failed

    ; Valid electrical access code - unlock circuit
    LDA password_entry_buffer
    SEC
    SBC #$2B            ; Remove offset
    STA unlocked_electrical_level_number

    ; Set circuit unlock flag
    TAX
    LDA #$01
    STA circuit_unlock_table,X

    LDA #$01
    STA password_validation_result
    RTS

electrical_access_validation_failed:
    LDA #$00
    STA password_validation_result
    RTS

; Process electrical password entry input
ProcessElectricalPasswordEntryInput:
    ; Read controller input for NES
    JSR ReadControllerByte
    LDA controller_input
    AND #$80            ; Check A button
    BEQ check_electrical_password_other_buttons

    ; A button pressed - validate password
    JMP electrical_password_entry_complete

check_electrical_password_other_buttons:
    LDA controller_input
    AND #$40            ; Check B button
    BEQ check_electrical_password_dpad

    ; B button pressed - delete character
    JMP electrical_password_entry_delete

check_electrical_password_dpad:
    ; Handle D-pad for character selection (simplified)
    JSR ProcessElectricalPasswordCharacterSelection

no_electrical_password_input:
    RTS

electrical_password_entry_complete:
    ; Validate entered electrical password
    JSR ValidateEnteredElectricalPassword

    ; Check validation result
    LDA password_validation_result
    BEQ electrical_password_invalid

    ; Electrical password valid
    JSR DisplayElectricalPasswordSuccessMessage
    LDA #$00
    STA password_entry_mode
    RTS

electrical_password_invalid:
    JSR DisplayElectricalPasswordErrorMessage
    JSR ClearElectricalPasswordEntryBuffer
    RTS

electrical_password_entry_delete:
    ; Remove last character
    LDA password_entry_position
    BEQ no_electrical_password_input
    DEC password_entry_position

    ; Clear character from buffer
    LDX password_entry_position
    LDA #$00
    STA password_entry_buffer,X

    ; Update display
    JSR UpdateElectricalPasswordEntryDisplay
    RTS

; Display electrical password generation interface
DisplayElectricalPasswordGenerationInterface:
    ; Clear screen area for electrical password display
