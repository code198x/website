; Perform optimized controller read for precision input
PerformOptimizedControllerRead:
    ; Strobe controller for fresh read
    LDA #$01            ; Enable strobe
    STA $4016           ; Controller 1 strobe
    LDA #$00            ; Disable strobe to latch
    STA $4016           ; Controller 1 strobe

    ; Read all 8 controller buttons
    LDX #$08            ; 8 buttons to read
    LDA #$00            ; Clear input accumulator
    STA TempControllerInput

ControllerReadLoop:
    LDA $4016           ; Read controller button
    AND #$01            ; Mask single bit
    CLC
    ROL TempControllerInput ; Rotate into input accumulator
    ORA TempControllerInput ; Combine with accumulator
    STA TempControllerInput
    DEX                 ; Decrement button counter
    BNE ControllerReadLoop

    ; Store complete controller state
    LDA TempControllerInput
    STA ControllerCurrentState

    RTS

; Process button combination detection for precision control
ProcessButtonCombinationDetection:
    ; Check for simultaneous button presses
    LDA ControllerCurrentState
    STA TempButtonCheck

    ; Count number of buttons pressed simultaneously
    LDX #$00            ; Clear button count
    LDY #$08            ; Check all 8 buttons

CountButtonsLoop:
    LSR TempButtonCheck ; Shift button into carry
    BCC NoButtonPressed
    INX                 ; Increment button count
NoButtonPressed:
    DEY                 ; Decrement button counter
    BNE CountButtonsLoop

    ; Store simultaneous button count
    STX SimultaneousButtonCount

    ; Check for multi-button combinations
    CPX MultiButtonThreshold
    BCC NoMultiButtonCombination

    ; Multi-button combination detected
    LDA #$01
    STA MultiButtonCombinationFlag
    JMP ButtonCombinationComplete

NoMultiButtonCombination:
    LDA #$00
    STA MultiButtonCombinationFlag

ButtonCombinationComplete:
    RTS

; Apply precision input filtering for clean control
ApplyPrecisionInputFiltering:
    ; Compare current and previous controller states
    LDA ControllerCurrentState
    EOR ControllerPreviousState
    STA ControllerChangeState

    ; Apply debouncing filter
    LDA InputDebounceCounter
    BEQ NoDebounceActive

    ; Debouncing active - decrement counter
    DEC InputDebounceCounter

    ; Return previous stable state during debounce
    LDA ControllerPreviousState
    STA ControllerCurrentState
    RTS

NoDebounceActive:
    ; Check if input has changed
    LDA ControllerChangeState
    BEQ InputStable

    ; Input changed - start debounce timer
    LDA InputDebounceFrames
    STA InputDebounceCounter

    ; Process button press/release detection
    JSR ProcessButtonStateChanges

    ; Copy current state to previous for next frame
    LDA ControllerCurrentState
    STA ControllerPreviousState

    LDA ControllerChangeState
    AND ControllerPreviousState
    STA ButtonReleaseFlags

    ; Update button hold states
    LDA ControllerCurrentState
    STA ButtonHoldFlags

    RTS

; Process input polling optimization for responsive control

    LDA PollingCounter

    RTS

    STA PollingCounter

SkipInputPolling:
    RTS

    AND #$08
    BEQ CheckDownPress

    BEQ CheckLeftPress
    JSR QueuePrecisionDownMovement

    JSR QueuePrecisionLeftMovement

    RTS

; Queue precision command with timestamp and priority
QueuePrecisionCommand:
    ; Check if command queue has space
    LDA CommandQueueCount
    CMP CommandQueueSize
    BCS PrecisionQueueFull

    ; Add command to queue
    LDY #$00
    LDA TempCommandType
    STA (CommandQueueWritePointerLow),Y

    INY
    LDA TempCommandDirection
    STA (CommandQueueWritePointerLow),Y

    INY
    LDA TempCommandDistance
    STA (CommandQueueWritePointerLow),Y

    INY
    LDA PrecisionFrameCounter
    STA (CommandQueueWritePointerLow),Y

    ; Update queue count and pointers
    INC CommandQueueCount
    JSR AdvanceCommandQueueWritePointer

PrecisionQueueFull:
    RTS

; Execute precision commands with exact timing
ExecutePrecisionCommands:
    ; Check if commands available
    LDA CommandQueueCount
    BEQ NoCommandsToExecute

    JSR ProcessNextPrecisionCommand

    JSR UpdatePrecisionCommandPointers

    RTS

ProcessNextPrecisionCommand:
    ; Get command from queue
    LDY #$00
    LDA (CommandQueueReadPointerLow),Y
    STA CurrentCommandType

    INY
    LDA (CommandQueueReadPointerLow),Y
    STA CurrentCommandDirection

    INY
    LDA (CommandQueueReadPointerLow),Y
    STA CurrentCommandDistance

    INY
    LDA (CommandQueueReadPointerLow),Y
    STA CurrentCommandTimestamp

    ; Execute command based on type
    LDA CurrentCommandType
    CMP #$01            ; Movement command?
    BEQ ExecutePrecisionMovementCommand
    CMP #$02            ; Action command?
    BEQ ExecutePrecisionActionCommand

    RTS

    RTS

    LDA CurrentCommandDistance
    STA PrecisionMovementDistance

    CMP #$02            ; Down?
    BEQ MoveConductorDownPrecision
    CMP #$04            ; Left?

    CLC
    ADC PrecisionMovementDistance
    CMP #$E0            ; Check boundary (224 pixels)
    BCC StoreNewYDown
    LDA #$DF            ; Clamp to boundary
StoreNewYDown:
    STA ConductorYPosition
    RTS

MoveConductorLeftPrecision:
    LDA ConductorXPosition
    SEC
    SBC PrecisionMovementDistance
    BCS StoreNewXLeft
    LDA #$00            ; Clamp to boundary
StoreNewXLeft:
    STA ConductorXPosition
    RTS
