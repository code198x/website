CalculateSmoothConductorVelocity:

    STA ElectricalBoostInput

    AND #$01            ; Check B button (electrical brake)
    STA ElectricalBrakeInput
    
    LDA $4016           ; Read next bit
    AND #$01            ; Check Select button
    STA SelectInput
    
    LDA $4016           ; Read next bit
    AND #$01            ; Check Start button

    ; Read directional input for electrical conductor movement

    STA UpInput
    
    LDA $4016           ; Read next bit
    AND #$01            ; Check Down
    STA DownInput
    
    LDA $4016           ; Read next bit
    AND #$01            ; Check Left
    STA LeftInput
    
    LDA $4016           ; Read next bit
    AND #$01            ; Check Right
    STA RightInput

    BNE CalculateRightVelocity

CalculateRightVelocity:
    LDA AccelerationFactor
    STA DesiredVelocityX
    JMP CalculateYVelocity

CalculateLeftVelocity:
    LDA AccelerationFactor
    EOR #$FF            ; Make negative (two's complement)
    CLC
    ADC #$01
    STA DesiredVelocityX
    JMP CalculateYVelocity

ApplyXDamping:
    ; No horizontal input - apply electrical resistance damping
    LDA ConductorVelocityX
    BEQ CalculateYVelocity
    BMI DampNegativeX
    
    ; Positive velocity - reduce toward zero
    SEC
    SBC VelocityDampingFactor
    BCS StoreXDamping
    LDA #$00            ; Clamp to zero
    JMP StoreXDamping
    
DampNegativeX:
    ; Negative velocity - increase toward zero
    CLC
    ADC VelocityDampingFactor
    BMI StoreXDamping
    LDA #$00            ; Clamp to zero

StoreXDamping:
    STA DesiredVelocityX

CalculateYVelocity:
    ; Calculate desired Y velocity for electrical movement
    LDA DownInput       ; Check down input
    BNE CalculateDownVelocity
    LDA UpInput         ; Check up input
    BNE CalculateUpVelocity
    JMP ApplyYDamping   ; No vertical input - apply damping

CalculateDownVelocity:
    LDA AccelerationFactor
    STA DesiredVelocityY
    JMP ElectricalVelocityCalculated

CalculateUpVelocity:
    LDA AccelerationFactor
    EOR #$FF            ; Make negative
    CLC
    ADC #$01
    STA DesiredVelocityY
    JMP ElectricalVelocityCalculated

ApplyYDamping:
    ; No vertical input - apply electrical resistance damping
    LDA ConductorVelocityY
    BEQ ElectricalVelocityCalculated
    BMI DampNegativeY
    
    ; Positive velocity - reduce toward zero
    SEC
    SBC VelocityDampingFactor
    BCS StoreYDamping
    LDA #$00            ; Clamp to zero
    JMP StoreYDamping
    
DampNegativeY:
    ; Negative velocity - increase toward zero

    ADC VelocityDampingFactor
    BMI StoreYDamping
    LDA #$00            ; Clamp to zero

StoreYDamping:
    STA DesiredVelocityY

ElectricalVelocityCalculated:
    RTS

; Apply electrical velocity smoothing for professional motion feel
ApplyElectricalVelocitySmoothing:
    ; Smooth X velocity toward desired electrical velocity
    LDA DesiredVelocityX
    SEC
    SBC ConductorVelocityX  ; Calculate velocity difference
    BEQ SmoothYVelocity     ; No change needed
    
    ; Apply electrical smoothing factor
    BMI NegativeXDiff
    
    ; Positive difference - limit change rate for electrical systems
    CMP ElectricalSmoothingLimit
    BCC ApplyXChange
    LDA ElectricalSmoothingLimit
    JMP ApplyXChange

    ; Negative difference - limit change rate
    EOR #$FF
    CLC
    ADC #$01            ; Make positive
    CMP ElectricalSmoothingLimit
    BCC RestoreNegativeX
    LDA ElectricalSmoothingLimit
    
RestoreNegativeX:
    EOR #$FF
    CLC
    ADC #$01            ; Make negative again
    
ApplyXChange:
    CLC
    ADC ConductorVelocityX
    STA ConductorVelocityX

    LDA DesiredVelocityY
    SEC
    SBC ConductorVelocityY

    CMP ElectricalSmoothingLimit

    LDA ElectricalSmoothingLimit
    JMP ApplyYChange

    EOR #$FF
    CLC
    ADC #$01
    CMP ElectricalSmoothingLimit
    BCC RestoreNegativeY
    LDA ElectricalSmoothingLimit
    
RestoreNegativeY:
    EOR #$FF
    CLC
    ADC #$01
    
ApplyYChange:
    CLC
    ADC ConductorVelocityY
    STA ConductorVelocityY
