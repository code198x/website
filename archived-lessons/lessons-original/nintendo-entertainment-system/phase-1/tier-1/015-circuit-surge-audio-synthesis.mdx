# Lesson 15: Circuit Surge - Audio Synthesis

**Your electrical world needs dynamic sound!** In your environmental effects from Lesson 14, you created immersive visual atmospheres with particle systems. Today, you'll add dynamic audio synthesis with procedural sound effects, musical patterns, and responsive audio feedback that transforms your visual effects into a complete audio-visual experience. You'll master sound programming through authentic APU techniques that create engaging, electrical musical gameplay!

## Your Audio Synthesis Energizes

Let's create dynamic audio synthesis using APU programming and procedural electrical sound generation:

UpdateAudioSynthesisSystems:

; Update electrical musical patterns for dynamic music
UpdateElectricalMusicalPatterns:
    ; Update musical beat timer
    INC MusicalBeatTimer

    BCC MusicalPatternsContinue

    ; Reset beat timer and advance beat
    LDA #$00
    STA MusicalBeatTimer

    LDA CurrentMusicalBeat
    CMP BeatsPerPattern
    BCC NoPatternChange

    LDA #$00

    LDA MusicalPatternTimer
    CMP PatternChangeInterval
    BCC NoPatternSetChange

    ; Reset pattern timer and advance pattern set
    LDA #$00
    STA MusicalPatternTimer
    JSR AdvanceElectricalMusicalPatternSet

NoPatternSetChange:
NoPatternChange:
MusicalPatternsContinue:
    ; Generate current musical frame
    JSR GenerateElectricalMusicalFrame

    RTS

; Advance electrical musical pattern set based on game state
AdvanceElectricalMusicalPatternSet:
    ; Check conductor action intensity to determine patterns
    LDA ConductorActionIntensity
    CMP #$0C            ; High intensity threshold
    BCS HighIntensityElectricalPatterns
    CMP #$06            ; Medium intensity threshold
    BCS MediumIntensityElectricalPatterns

    ; Low intensity - ambient electrical patterns
    LDA CurrentMelodyPattern
    CLC
    ADC #$01
    AND #$03            ; 4 ambient patterns
    STA CurrentMelodyPattern
    JMP ElectricalPatternSetComplete

MediumIntensityElectricalPatterns:
    ; Medium intensity - active electrical patterns
    LDA CurrentMelodyPattern
    CLC
    ADC #$01
    AND #$03            ; 4 medium patterns
    CLC
    ADC #$04            ; Offset to medium patterns
    STA CurrentMelodyPattern
    JMP ElectricalPatternSetComplete

HighIntensityElectricalPatterns:
    ; High intensity - dramatic electrical patterns
    LDA CurrentMelodyPattern
    CLC
    ADC #$01
    AND #$03            ; 4 high patterns
    CLC
    ADC #$08            ; Offset to high patterns
    STA CurrentMelodyPattern

ElectricalPatternSetComplete:
    ; Update harmony, bass, and percussion patterns
    JSR UpdateHarmonyPatternFromMelody
    JSR UpdateBassPatternFromMelody
    JSR UpdatePercussionPatternFromIntensity

; Generate electrical musical frame for current patterns
GenerateElectricalMusicalFrame:
    ; Generate melody on pulse channel 1
    JSR GenerateElectricalMelodyChannel

    ; Generate harmony on pulse channel 2
    JSR GenerateElectricalHarmonyChannel

    ; Generate bass on triangle channel
    JSR GenerateElectricalBassChannel

    ; Generate percussion on noise channel
    JSR GenerateElectricalPercussionChannel

    RTS

; Generate electrical melody channel using APU Pulse 1
GenerateElectricalMelodyChannel:
    ; Get current melody pattern and beat
    LDA CurrentMelodyPattern
    ASL                 ; Pattern * 2
    ASL                 ; Pattern * 4
    ASL                 ; Pattern * 8
    ASL                 ; Pattern * 16 (beats per pattern)
    CLC
    ADC CurrentMusicalBeat
    TAY

    ; Get note from pattern table
    LDA ElectricalMelodyPatternTable,Y
    STA CurrentMelodyNote

    ; Convert note to APU frequency
    JSR ConvertNoteToAPUFrequency

    ; Set pulse channel 1 frequency
    LDA FrequencyLow
    STA $4002           ; Frequency low
    LDA FrequencyHigh
    STA $4003           ; Frequency high and length

    ; Set pulse channel 1 duty cycle and volume
    LDA #$BF            ; 50% duty cycle, constant volume, max volume
    STA $4000

    ; No sweep for melody
    LDA #$08            ; Disable sweep
    STA $4001

    RTS

; Generate electrical harmony channel using APU Pulse 2
GenerateElectricalHarmonyChannel:
    ; Get harmony offset based on current chord
    LDA CurrentHarmonyPattern
    TAY
    LDA ElectricalHarmonyOffsetTable,Y
    CLC
    ADC CurrentMelodyNote

    ; Convert harmony note to APU frequency
    JSR ConvertNoteToAPUFrequency

    ; Set pulse channel 2 frequency
    LDA FrequencyLow
    STA $4006           ; Frequency low
    LDA FrequencyHigh
    STA $4007           ; Frequency high and length

    ; Set pulse channel 2 duty cycle and volume (softer than melody)
    LDA #$B8            ; 50% duty cycle, constant volume, 75% volume
    STA $4004

    ; No sweep for harmony
    LDA #$08            ; Disable sweep
    STA $4005

    RTS

; Generate electrical bass channel using APU Triangle
GenerateElectricalBassChannel:
    ; Check if this is a bass beat
    LDA CurrentMusicalBeat
    AND #$03            ; Check beat position in 4/4 time
    TAY
    LDA ElectricalBassPatternTable,Y
    BEQ BassChannelComplete

    ; Generate bass note (one octave below melody)
    LDA CurrentMelodyNote
    SEC
    SBC #$0C            ; Down one octave

    ; Convert bass note to APU frequency
    JSR ConvertNoteToAPUFrequency

    ; Set triangle channel frequency
    LDA FrequencyLow
    STA $400A           ; Frequency low
    LDA FrequencyHigh
    STA $400B           ; Frequency high and length

    ; Set triangle channel linear counter
    LDA #$81            ; Linear counter load, length counter halt
    STA $4008

BassChannelComplete:
    RTS

; Generate electrical percussion channel using APU Noise
GenerateElectricalPercussionChannel:
    ; Check if this is a percussion beat
    LDA CurrentMusicalBeat
    AND #$07            ; Check beat position in 8-beat pattern
    TAY
    LDA CurrentPercussionPattern
    ASL                 ; Pattern * 2
    ASL                 ; Pattern * 4
    ASL                 ; Pattern * 8
    CLC
    ADC CurrentMusicalBeat
    AND #$07            ; Wrap to 8 beats
    TAY

    LDA ElectricalPercussionPatternTable,Y
    BEQ PercussionChannelComplete

    ; Generate electrical percussion sound
    LDA #$08            ; Short period for electrical sound
    STA $400E           ; Period and loop

    ; Set noise channel volume
    LDA #$9A            ; Constant volume, 75% volume
    STA $400C

    ; Set length counter
    LDA #$08            ; Short length
    STA $400F

PercussionChannelComplete:
    RTS

; Convert musical note to APU frequency
ConvertNoteToAPUFrequency:
    ; A contains note number (0-95, covering 8 octaves)
    ; Returns frequency in FrequencyLow and FrequencyHigh

    ; Clamp note to APU range
    CMP #$60            ; Maximum note (96)
    BCC NoteInAPURange
    LDA #$5F            ; Clamp to maximum

NoteInAPURange:
    ; Save original note
    STA TempNote

    ; Get base frequency from table
    AND #$0F            ; Get note within octave (0-15 for chromatic)
    TAY
    LDA APUBaseFrequencyTableLow,Y
    STA FrequencyLow
    LDA APUBaseFrequencyTableHigh,Y
    STA FrequencyHigh

    ; Calculate octave shift
    LDA TempNote
    LSR                 ; Divide by 2
    LSR                 ; Divide by 4
    LSR                 ; Divide by 8
    LSR                 ; Divide by 16 to get octave
    BEQ APUFrequencyComplete

    ; Shift frequency down by octaves (higher frequency for APU)
    TAY                 ; Octave count
APUOctaveShiftLoop:
    LSR FrequencyHigh   ; Shift high byte right
    ROR FrequencyLow    ; Rotate low byte right
    DEY
    BNE APUOctaveShiftLoop

APUFrequencyComplete:
    RTS

; Create electrical sound effect with specified type and priority
CreateElectricalSoundEffect:
    ; A = sound type, Y = priority
    ; Find available sound effect slot
    JSR FindAvailableElectricalSoundEffectSlot
    CPX #$FF            ; Check if slot found
    BEQ ElectricalSoundEffectCreationComplete

    LDA #$01            ; Sound effect active
    STA SoundEffectActiveArray,X

    STA SoundEffectPriorityArray,X

    ; Get sound effect parameters from type

    STA SoundEffectTypeArray,X
    TAY                 ; Use as table index

    LDA ElectricalSoundEffectFrequencyTable,Y
    STA SoundEffectFrequencyArray,X
    LDA ElectricalSoundEffectVolumeTable,Y
    STA SoundEffectVolumeArray,X
    LDA ElectricalSoundEffectDurationTable,Y
    STA SoundEffectDurationArray,X
    LDA ElectricalSoundEffectDutyCycleTable,Y
