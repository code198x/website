.segment "HEADER"
.byte "NES", $1A

RESET:

    SEI
    CLD
    LDX #$FF
    TXS

    ; Set up enhanced electrical world for network management
    JSR InitCompleteElectricalState

    ; Initialize new electrical network systems
    JSR InitElectricalNetworkSystem
    JSR InitConductorArraySystem

    JSR InitNetworkCoordination

    JSR CreateElectricalNetworkWorld

ElectricalNetworkLoop:

    JSR UpdatePrimaryConductor

    JSR UpdateCurrentFlowNetworks

InitElectricalNetworkSystem:

    JSR InitConductorNetworkArrays
    JSR InitCurrentFlowArrays
    JSR InitElectricalNodeArrays
    JSR InitNetworkManagementSystem

    LDA #$06            ; Maximum 6 conductors (memory efficient)
    STA MaxConductorCount

    STA MaxCurrentFlowCount
    LDA #$04            ; Maximum 4 electrical nodes
    STA MaxElectricalNodeCount

; Initialize conductor network arrays using memory-efficient design
InitConductorNetworkArrays:
    ; Conductor position arrays (6 conductors max for NES constraints)
    LDX #$05            ; Conductor index (0-5)
ClearConductorPositions:
    LDA #$00            ; Inactive conductor marker
    STA ConductorXArray,X       ; Clear X position
    STA ConductorYArray,X       ; Clear Y position
    STA ConductorActiveArray,X  ; Clear active flag
    DEX
    BPL ClearConductorPositions

    ; Conductor electrical state arrays
    LDX #$05            ; Conductor index (0-5)
ClearConductorStates:
    LDA #$00            ; Default electrical state
    STA ConductorTypeArray,X        ; Clear conductor type
    STA ConductorVoltageArray,X     ; Clear voltage level
    STA ConductorCurrentArray,X     ; Clear current flow
    STA ConductorVelocityXArray,X   ; Clear X velocity
    STA ConductorVelocityYArray,X   ; Clear Y velocity
    DEX
    BPL ClearConductorStates

    ; Conductor network connection arrays
    LDX #$05            ; Conductor index (0-5)
ClearConductorConnections:
    LDA #$FF            ; No connection marker
    STA ConductorConnectionArray,X  ; Clear connection to other conductors
    LDA #$00            ; Default network state
    STA ConductorNetworkIDArray,X   ; Clear network ID
    DEX
    BPL ClearConductorConnections

    RTS

; Initialize current flow arrays for electrical network management
InitCurrentFlowArrays:
    ; Current flow position arrays (8 flows max)
    LDX #$07            ; Current flow index (0-7)
ClearCurrentFlowPositions:
    LDA #$00            ; Inactive flow marker
    STA CurrentFlowXArray,X     ; Clear X position
    STA CurrentFlowYArray,X     ; Clear Y position
    STA CurrentFlowActiveArray,X ; Clear active flag
    DEX
    BPL ClearCurrentFlowPositions

    ; Current flow electrical properties
    LDX #$07            ; Current flow index (0-7)
ClearCurrentFlowProperties:
    LDA #$00            ; Default properties
    STA CurrentFlowTypeArray,X      ; Clear flow type (AC/DC/pulse)
    STA CurrentFlowIntensityArray,X ; Clear intensity level
    STA CurrentFlowDirectionArray,X ; Clear flow direction
    STA CurrentFlowVelocityXArray,X ; Clear X velocity
    STA CurrentFlowVelocityYArray,X ; Clear Y velocity
    DEX
    BPL ClearCurrentFlowProperties

    ; Current flow network data
    LDX #$07            ; Current flow index (0-7)
ClearCurrentFlowNetwork:
    LDA #$FF            ; No source marker
    STA CurrentFlowSourceArray,X    ; Clear source conductor
    STA CurrentFlowTargetArray,X    ; Clear target conductor
    LDA #$00            ; Default network data
    STA CurrentFlowNetworkArray,X   ; Clear network membership
    DEX
    BPL ClearCurrentFlowNetwork

    RTS

; Initialize electrical node arrays for network topology
InitElectricalNodeArrays:

    LDX #$03            ; Node index (0-3)
ClearNodePositions:

    STA ElectricalNodeYArray,X      ; Clear Y position

    DEX
    BPL ClearNodePositions

    ; Node electrical properties
    LDX #$03            ; Node index (0-3)
ClearNodeProperties:

    STA ElectricalNodeTypeArray,X   ; Set node type
    LDA #$00            ; Default electrical state
    STA ElectricalNodeVoltageArray,X ; Clear voltage
    STA ElectricalNodeCurrentArray,X ; Clear current
    STA ElectricalNodeCapacityArray,X ; Clear capacity
    DEX
    BPL ClearNodeProperties

    ; Node connection matrix (simplified for NES)
    LDX #$0F            ; 4 nodes × 4 connections each
ClearNodeConnections:
    LDA #$00            ; No connection
    STA NodeConnectionMatrix,X      ; Clear connection matrix
    DEX
    BPL ClearNodeConnections

    RTS

; Create electrical network world using enhanced circuit design
CreateElectricalNetworkWorld:

    JSR SetupNetworkPPU

    ; Create electrical network topology
    JSR CreateElectricalNetworkTopology

    ; Set up network coordination zones
    JSR SetupNetworkCoordinationZones

    ; Initialize electrical network visual elements
    JSR InitElectricalNetworkVisuals

    RTS

; Setup PPU for electrical network visualization
SetupNetworkPPU:
    ; Wait for PPU ready
    BIT $2002
    BPL SetupNetworkPPU

    ; Set up enhanced electrical network palette
    LDA #$3F            ; Point to palette
    STA $2006           ; PPU address high
    LDA #$00            ; Point to palette start
    STA $2006           ; PPU address low

    ; Network electrical color palette
    LDA #$0F            ; Black background
    STA $2007           ; Write palette

    STA $2007           ; Write palette
    LDA #$16            ; Red high voltage
    STA $2007           ; Write palette
    LDA #$27            ; Yellow medium voltage
    STA $2007           ; Write palette
    LDA #$11            ; Blue low voltage
    STA $2007           ; Write palette
    LDA #$25            ; Green current flows
    STA $2007           ; Write palette
    LDA #$35            ; Purple electrical nodes
    STA $2007           ; Write palette
    LDA #$38            ; Orange network effects
    STA $2007           ; Write palette

    ; Additional network visualization colors
    LDA #$2A            ; Cyan network connections
    STA $2007           ; Write palette
    LDA #$1A            ; Magenta network pulses
    STA $2007           ; Write palette
    LDA #$0A            ; Gray network infrastructure
    STA $2007           ; Write palette
    LDA #$20            ; Light gray network base
    STA $2007           ; Write palette

    ; Enable PPU for network rendering
    LDA #$1E            ; Enable background and sprites
    STA $2001           ; PPU control

    RTS

; Update conductor networks using efficient array processing
UpdateConductorNetworks:
    ; Process all active conductors in network
    LDX #$00            ; Start with conductor 0

UpdateConductorLoop:
    ; Check if this conductor is active
    LDA ConductorActiveArray,X
    BEQ NextConductor   ; Skip if inactive

    ; Update conductor electrical properties
    JSR UpdateConductorElectricalProperties

    ; Update conductor movement within network
    JSR UpdateConductorNetworkMovement

    ; Update conductor network connections
    JSR UpdateConductorNetworkConnections

    ; Update conductor network coordination
    JSR UpdateConductorNetworkCoordination

NextConductor:
    INX                 ; Next conductor index
    CPX #$06            ; Check all 6 conductor slots
    BNE UpdateConductorLoop

    RTS

; Update conductor electrical properties based on network state
UpdateConductorElectricalProperties:
    ; X contains conductor index
    ; Update voltage based on network connections
    LDA ConductorConnectionArray,X
    CMP #$FF            ; No connection?
    BEQ ConductorNoConnection

    ; Connected to another conductor - balance voltage
    TAY                 ; Y = connected conductor index
    LDA ConductorVoltageArray,Y     ; Load connected voltage
    CLC
    ADC ConductorVoltageArray,X     ; Add current voltage
    LSR                 ; Divide by 2 (average)
    STA ConductorVoltageArray,X     ; Store balanced voltage
    STA ConductorVoltageArray,Y     ; Balance both conductors

    JMP ConductorElectricalComplete

ConductorNoConnection:
    ; Isolated conductor - voltage decay
    LDA ConductorVoltageArray,X
    BEQ ConductorElectricalComplete ; Already zero
    DEC ConductorVoltageArray,X     ; Voltage decay

ConductorElectricalComplete:
    ; Update current flow based on voltage
    LDA ConductorVoltageArray,X
    LSR                 ; Current = voltage / 2 (simplified)
    STA ConductorCurrentArray,X

    RTS

; Update conductor movement within network constraints
UpdateConductorNetworkMovement:
    ; X contains conductor index
    ; Update X position
    LDA ConductorXArray,X           ; Load current X
    CLC
    ADC ConductorVelocityXArray,X   ; Add X velocity
    STA ConductorXArray,X           ; Store new X

    ; Update Y position
    LDA ConductorYArray,X           ; Load current Y
    CLC
    ADC ConductorVelocityYArray,X   ; Add Y velocity
    STA ConductorYArray,X           ; Store new Y

    ; Check network boundaries and connections
    JSR CheckConductorNetworkBoundaries

    ; Apply network forces (attraction to nodes/other conductors)
    JSR ApplyNetworkForces

    RTS

; Apply network forces for realistic electrical behavior
ApplyNetworkForces:
    ; X contains conductor index
    ; Check attraction to electrical nodes
    LDY #$00            ; Node index

CheckNodeAttractionLoop:
    ; Check if node is active
    LDA ElectricalNodeActiveArray,Y
    BEQ NextNodeAttraction

    ; Calculate distance to node
    LDA ElectricalNodeXArray,Y
    SEC
    SBC ConductorXArray,X
    BPL NodeDistanceXPositive
    EOR #$FF            ; Make positive (simple abs)
    CLC
    ADC #$01

NodeDistanceXPositive:
    CMP #$10            ; Attraction range (16 pixels)
    BCS NextNodeAttraction

    ; Apply attraction force
    LDA ElectricalNodeXArray,Y
    CMP ConductorXArray,X
    BCC AttractLeft

    ; Attract right
    INC ConductorVelocityXArray,X
    JMP CheckYAttraction

AttractLeft:
    DEC ConductorVelocityXArray,X

CheckYAttraction:
    LDA ElectricalNodeYArray,Y
    CMP ConductorYArray,X
    BCC AttractUp

    ; Attract down
    INC ConductorVelocityYArray,X
    JMP NextNodeAttraction

AttractUp:
    DEC ConductorVelocityYArray,X

NextNodeAttraction:
    INY                 ; Next node
    CPY #$04            ; Check all 4 nodes
    BNE CheckNodeAttractionLoop

    RTS

; Update current flow networks using synchronized movement
UpdateCurrentFlowNetworks:
    ; Process all active current flows
    LDX #$00            ; Start with flow 0

UpdateCurrentFlowLoop:
    ; Check if this current flow is active
    LDA CurrentFlowActiveArray,X
    BEQ NextCurrentFlow ; Skip if inactive

    ; Update current flow movement
    JSR UpdateCurrentFlowMovement

    ; Update current flow electrical properties
    JSR UpdateCurrentFlowElectricalProperties

    ; Update current flow network routing
    JSR UpdateCurrentFlowNetworkRouting

NextCurrentFlow:
    INX                 ; Next current flow index
    CPX #$08            ; Check all 8 current flow slots
    BNE UpdateCurrentFlowLoop

    RTS

; Update current flow movement along network paths
UpdateCurrentFlowMovement:
    ; X contains current flow index
    ; Check if flow has source and target
    LDA CurrentFlowSourceArray,X
    CMP #$FF            ; No source?
    BEQ CurrentFlowFreeMovement

    ; Move toward target conductor
    LDA CurrentFlowTargetArray,X
    CMP #$FF            ; No target?
    BEQ CurrentFlowFreeMovement

    ; Calculate movement toward target
    TAY                 ; Y = target conductor index
    LDA ConductorXArray,Y           ; Target X
    CMP CurrentFlowXArray,X         ; Compare with current X
    BCC FlowMoveLeft

    ; Move right toward target
    INC CurrentFlowXArray,X
    JMP FlowCheckY

FlowMoveLeft:
    DEC CurrentFlowXArray,X

FlowCheckY:
    LDA ConductorYArray,Y           ; Target Y
    CMP CurrentFlowYArray,X         ; Compare with current Y
    BCC FlowMoveUp

    ; Move down toward target
    INC CurrentFlowYArray,X
    JMP CurrentFlowMovementComplete

FlowMoveUp:
    DEC CurrentFlowYArray,X
    JMP CurrentFlowMovementComplete

CurrentFlowFreeMovement:
    ; Free movement using velocity
    LDA CurrentFlowXArray,X
    CLC
    ADC CurrentFlowVelocityXArray,X
    STA CurrentFlowXArray,X

    LDA CurrentFlowYArray,X
    CLC
    ADC CurrentFlowVelocityYArray,X
    STA CurrentFlowYArray,X

CurrentFlowMovementComplete:
    RTS

; Process network coordination for synchronized behavior
ProcessNetworkCoordination:
    ; Coordinate all network elements for harmonious electrical behavior
    JSR CoordinateNetworkVoltages
    JSR CoordinateNetworkCurrents
    JSR CoordinateNetworkTiming
    JSR CoordinateNetworkEffects

    RTS

; Coordinate network voltages for realistic electrical behavior
CoordinateNetworkVoltages:
    ; Balance voltages across connected conductors
    LDX #$00            ; Conductor index

VoltageBalanceLoop:
    ; Check if conductor is active
    LDA ConductorActiveArray,X
    BEQ NextVoltageBalance

    ; Check if conductor has connections
    LDA ConductorConnectionArray,X
    CMP #$FF            ; No connection?
    BEQ NextVoltageBalance

    ; Balance voltage with connected conductor
    TAY                 ; Y = connected conductor
    LDA ConductorVoltageArray,X     ; This conductor voltage
    CLC
    ADC ConductorVoltageArray,Y     ; Add connected voltage
    LSR                 ; Divide by 2
    STA TempVoltage     ; Store temporary result

    ; Apply balanced voltage to both conductors
    LDA TempVoltage
    STA ConductorVoltageArray,X
    STA ConductorVoltageArray,Y

NextVoltageBalance:
    INX                 ; Next conductor
    CPX #$06            ; Check all conductors
    BNE VoltageBalanceLoop

    RTS

; Process network collisions and interactions
ProcessNetworkCollisions:
    ; Check conductor vs conductor interactions
    JSR CheckConductorNetworkInteractions

    ; Check current flow vs conductor interactions
    JSR CheckCurrentFlowConductorInteractions

    ; Check conductor vs electrical node interactions
    JSR CheckConductorNodeInteractions

    RTS

; Check conductor network interactions
CheckConductorNetworkInteractions:
    ; Check all conductor pairs for network interactions
    LDX #$00            ; Outer conductor index

ConductorInteractionOuterLoop:
    ; Check if outer conductor is active
    LDA ConductorActiveArray,X
    BEQ NextOuterConductor

    ; Inner loop: check against other conductors
    LDY #$00            ; Inner conductor index

ConductorInteractionInnerLoop:
    ; Don't check conductor against itself
    CPX Y               ; Same conductor?
    BEQ NextInnerConductor

    ; Check if inner conductor is active
    LDA ConductorActiveArray,Y
    BEQ NextInnerConductor

    ; Check distance between conductors
    LDA ConductorXArray,X
    SEC
    SBC ConductorXArray,Y
    BPL ConductorDistanceXPositive
    EOR #$FF            ; Make positive
    CLC
    ADC #$01

ConductorDistanceXPositive:
    CMP #$04            ; Interaction distance (4 pixels)
    BCS NextInnerConductor

    ; Check Y distance
    LDA ConductorYArray,X
    SEC
    SBC ConductorYArray,Y
    BPL ConductorDistanceYPositive
    EOR #$FF            ; Make positive
    CLC
    ADC #$01

ConductorDistanceYPositive:
    CMP #$04            ; Interaction distance
    BCS NextInnerConductor

    ; Conductors are close - create network connection
    JSR CreateNetworkConnection

NextInnerConductor:
    INY                 ; Next inner conductor
    CPY #$06            ; Check all conductors
    BNE ConductorInteractionInnerLoop

NextOuterConductor:
    INX                 ; Next outer conductor
    CPX #$06            ; Check all conductors
    BNE ConductorInteractionOuterLoop

    RTS

; Create network connection between conductors
CreateNetworkConnection:
    ; X = first conductor, Y = second conductor
    ; Connect X to Y
    TYA                 ; A = Y (second conductor)
    STA ConductorConnectionArray,X  ; Connect first to second

    ; Connect Y to X
    TXA                 ; A = X (first conductor)
    STA ConductorConnectionArray,Y  ; Connect second to first

    ; Set same network ID for both
    LDA ConductorNetworkIDArray,X
    BNE ConnectionHasNetwork

    ; Create new network ID
    INC NetworkIDCounter
    LDA NetworkIDCounter
    STA ConductorNetworkIDArray,X
    STA ConductorNetworkIDArray,Y
    JMP ConnectionComplete

ConnectionHasNetwork:
    ; Use existing network ID
    STA ConductorNetworkIDArray,Y

ConnectionComplete:
    ; Create visual connection effect
    JSR CreateConnectionEffect

    ; Play connection audio
    JSR PlayConnectionSound

    RTS

; Update network visual effects
UpdateNetworkVisualEffects:
    ; Display all network conductors
    JSR DisplayNetworkConductors

    ; Display all current flows
    JSR DisplayNetworkCurrentFlows

    ; Display electrical nodes
    JSR DisplayElectricalNodes

    ; Display network connections
    JSR DisplayNetworkConnections

    RTS

; Display network conductors with electrical state visualization
DisplayNetworkConductors:
    LDX #$00            ; Conductor index

DisplayConductorLoop:
    ; Check if conductor is active
    LDA ConductorActiveArray,X
    BEQ NextConductorDisplay

    ; Get conductor position
    LDA ConductorXArray,X
    STA TempX
    LDA ConductorYArray,X
    STA TempY

    ; Get conductor character based on voltage
    LDA ConductorVoltageArray,X
    CMP #$80            ; High voltage?
    BCS HighVoltageConductor
    CMP #$40            ; Medium voltage?
    BCS MediumVoltageConductor

    ; Low voltage conductor
    LDA #$43            ; Character 'C'
    JMP StoreConductorChar

MediumVoltageConductor:
    LDA #$44            ; Character 'D'
    JMP StoreConductorChar

HighVoltageConductor:
    LDA #$45            ; Character 'E'

StoreConductorChar:
    STA TempChar

    ; Display conductor
    JSR DisplayCharacterAtPosition

NextConductorDisplay:
    INX                 ; Next conductor
    CPX #$06            ; Check all conductors
    BNE DisplayConductorLoop

    RTS

; Data arrays for electrical network management
ConductorXArray:            .res 6      ; Conductor X positions
ConductorYArray:            .res 6      ; Conductor Y positions
ConductorActiveArray:       .res 6      ; Conductor active flags
ConductorTypeArray:         .res 6      ; Conductor type IDs

CurrentFlowDirectionArray:  .res 8      ; Current flow directions
CurrentFlowVelocityXArray:  .res 8      ; Current flow X velocities
CurrentFlowVelocityYArray:  .res 8      ; Current flow Y velocities

CurrentFlowNetworkArray:    .res 8      ; Current flow network IDs

ElectricalNodeXArray:       .res 4      ; Electrical node X positions

; Network management variables
ActiveConductorCount:       .res 1      ; Current active conductor count
