    ; Read B button

    AND #$01            ; Mask B button
    STA $C1             ; Store B button state

    ; Read Select button

    AND #$01            ; Mask Select button
    STA $C2             ; Store Select button state

    ; Read Start button
    LDA $4016           ; Read controller
    AND #$01            ; Mask Start button
    STA $C3             ; Store Start button state

    ; Read directional buttons
    LDA $4016           ; Read Up
    AND #$01
    STA $C4             ; Store Up state
    LDA $4016           ; Read Down
    AND #$01
    STA $C5             ; Store Down state
    LDA $4016           ; Read Left
    AND #$01
    STA $C6             ; Store Left state
    LDA $4016           ; Read Right
    AND #$01
    STA $C7             ; Store Right state
