    cmp #$01        ; Check player aggression

    cmp #$02        ; Check health status

    lda decision_left_branches,y
    sta current_decision_node
    jmp continue_traversal

    lda decision_right_branches,y

    ; Check if we've reached a leaf node
    ldy current_decision_node
    lda decision_node_types,y
