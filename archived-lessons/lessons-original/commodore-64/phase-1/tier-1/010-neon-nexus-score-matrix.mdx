## Your Score Matrix Activates

```nasm
; Neon Nexus - Score Matrix System
* = $0801
        .byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

start:
    ; Initialize complete C64 environment
    lda #$93        ; Clear screen
    jsr $ffd2       ; CHROUT
    
    ; Set up enhanced neon visual environment for scoring
    lda #$00        ; Black background
    sta $d020       ; Border color
    sta $d021       ; Background color
    
    ; Initialize comprehensive game state from lessons 1-9
    jsr init_complete_game_state
    jsr init_entity_swarm_system
    
    ; Initialize new score matrix systems
    jsr init_score_matrix_system
    jsr init_multiplier_system
    jsr init_combo_system
    jsr init_achievement_system
    
    ; Set up score-ready arena
    jsr create_score_matrix_arena
    
    ; Score Matrix main loop
score_matrix_loop:
    ; Process all input
    jsr read_complete_input
    
    ; Update player and entity systems
    jsr update_player_movement
    jsr update_player_combat
    jsr update_enemy_swarm
    jsr update_projectile_swarm
    
    ; Process scoring systems
    jsr process_score_events
    jsr update_score_calculations
    jsr update_multiplier_system
    jsr update_combo_system
    
    ; Process all interactions with scoring
    jsr process_swarm_collisions_with_scoring
    jsr update_achievement_tracking
    
    ; Update visual and audio systems
    jsr update_score_visual_effects
    jsr update_score_display_system
    jsr update_audio_systems
    
    ; Check for score-based victory conditions
    jsr check_score_victory_conditions
    
    jmp score_matrix_loop

; Initialize score matrix system using arithmetic operations
init_score_matrix_system:
    ; Set up score calculation systems
    jsr init_score_calculation_system
    jsr init_score_display_system
    jsr init_score_arithmetic_system
    jsr init_score_storage_system
    
    ; Initialize score values using BCD format
    lda #$00        ; Clear all score digits
    sta score_millions
    sta score_hundred_thousands
    sta score_ten_thousands
    sta score_thousands
    sta score_hundreds
    sta score_tens
    sta score_ones
    
    ; Initialize score calculation variables
    sta temp_score_low
    sta temp_score_high
    sta last_score_low
    sta last_score_high
    
    ; Set initial scoring parameters
    lda #$01        ; Base score multiplier
    sta score_multiplier
    lda #$00        ; No combo initially
    sta combo_count
    sta combo_timer
    
    rts

; Initialize score calculation system for arithmetic operations
init_score_calculation_system:
    ; Set up score calculation lookup tables
    ldx #$0f        ; Initialize 16 score values
init_score_table_loop:
    lda base_score_table,x
    sta enemy_score_values,x
    dex
    bpl init_score_table_loop
    
    ; Initialize multiplier calculation table
    ldx #$07        ; Initialize 8 multiplier values
init_multiplier_table_loop:
    lda base_multiplier_table,x
    sta multiplier_values,x
    dex
    bpl init_multiplier_table_loop
    
    ; Initialize combo calculation parameters
    lda #$05        ; Combo time limit (5 seconds)
    sta combo_time_limit
    lda #$02        ; Minimum enemies for combo
    sta combo_minimum
    
    rts

; Initialize score display system for visual feedback
init_score_display_system:
    ; Set up score display area at top of screen
    lda #$04        ; Score display row
    sta score_display_row
    lda #$02        ; Score display column
    sta score_display_column
    
    ; Initialize score display colors
    lda #$01        ; White text
    sta score_text_color
    lda #$07        ; Yellow highlights
    sta score_highlight_color
    lda #$03        ; Cyan multiplier
    sta multiplier_color
    
    ; Clear score display area
    jsr clear_score_display_area
    
    ; Draw score labels
    jsr draw_score_labels
    
    rts

; Create score matrix arena with visual enhancements
create_score_matrix_arena:
    ; Draw enhanced arena with scoring zones
    jsr draw_score_arena_boundaries
    
    ; Create score multiplier zones
    jsr create_multiplier_zones
    
    ; Set up combo notification areas
    jsr setup_combo_notification_areas
    
    ; Initialize score particle effect areas
    jsr init_score_particle_areas
    
    rts

; Process score events from game interactions
process_score_events:
    ; Check for enemy destruction events
    jsr check_enemy_destruction_scoring
    
    ; Check for projectile hit events
    jsr check_projectile_hit_scoring
    
    ; Check for combo events
    jsr check_combo_scoring_events
    
    ; Check for special achievement events
    jsr check_achievement_scoring_events
    
    rts

; Check enemy destruction scoring using arithmetic calculations
check_enemy_destruction_scoring:
    ; Check each enemy slot for recent destruction
    ldx #$07        ; Check all 8 enemy slots
check_destruction_loop:
    lda enemy_destruction_flags,x
    beq next_destruction_check
    
    ; Enemy was destroyed - calculate score
    lda #$00        ; Clear destruction flag
    sta enemy_destruction_flags,x
    
    ; Get enemy type for score calculation
    lda enemy_type_array,x
    tax             ; X = enemy type
    
    ; Calculate base score using lookup table
    jsr calculate_enemy_base_score
    
    ; Apply current multiplier
    jsr apply_score_multiplier
    
    ; Add to combo if within time limit
    jsr update_combo_from_destruction
    
    ; Add calculated score to total
    jsr add_score_to_total
    
    ; Create visual score feedback
    jsr create_score_visual_feedback
    
    ; Restore enemy index
    ldx enemy_destruction_index

next_destruction_check:
    dex
    bpl check_destruction_loop
    
    rts

; Calculate enemy base score using arithmetic operations
calculate_enemy_base_score:
    ; X contains enemy type (0-7)
    ; Load base score from lookup table
    lda enemy_score_values,x
    sta temp_score_low
    lda enemy_score_values+8,x  ; High byte
    sta temp_score_high
    
    ; Apply difficulty bonus (arithmetic calculation)
    lda current_difficulty_level
    beq no_difficulty_bonus     ; No bonus at level 0
    
    ; Multiply base score by difficulty level
    lda temp_score_low
    ldx current_difficulty_level
difficulty_multiply_loop:
    clc
    adc temp_score_low          ; Add base score
    bcc no_carry_difficulty
    inc temp_score_high         ; Handle carry
no_carry_difficulty:
    dex
    bne difficulty_multiply_loop
    
    sta temp_score_low

no_difficulty_bonus:
    rts

; Apply score multiplier using BCD arithmetic
apply_score_multiplier:
    ; Multiply temp_score by current multiplier using BCD
    lda score_multiplier
    cmp #$01
    beq no_multiplier           ; No multiplication needed
    
    ; Save original score
    lda temp_score_low
    sta original_score_low
    lda temp_score_high
    sta original_score_high
    
    ; Clear result
    lda #$00
    sta temp_score_low
    sta temp_score_high
    
    ; Multiply using repeated addition (BCD safe)
    ldx score_multiplier
multiplier_loop:
    ; Add original score to result
    sed                         ; Set decimal mode for BCD
    lda temp_score_low
    clc
    adc original_score_low
    sta temp_score_low
    
    lda temp_score_high
    adc original_score_high
    sta temp_score_high
    cld                         ; Clear decimal mode
    
    dex
    bne multiplier_loop

no_multiplier:
    rts

; Update combo system with timing and calculations
update_combo_system:
    ; Check if combo timer is active
    lda combo_timer
    beq no_active_combo
    
    ; Decrement combo timer
    dec combo_timer
    bne combo_still_active
    
    ; Combo expired - reset combo count
    lda #$00
    sta combo_count
    jmp no_active_combo

combo_still_active:
    ; Check for combo multiplier bonus
    lda combo_count
    cmp combo_minimum
    bcc no_active_combo         ; Not enough for combo
    
    ; Calculate combo multiplier (count / 2 + 1)
    lsr                         ; Divide by 2
    clc
    adc #$01                    ; Add 1
    cmp #$09                    ; Cap at 9x multiplier
    bcc store_combo_multiplier
    lda #$09                    ; Maximum 9x
    
store_combo_multiplier:
    sta combo_multiplier

no_active_combo:
    rts

; Update combo from enemy destruction
update_combo_from_destruction:
    ; Increment combo count
    inc combo_count
    
    ; Reset combo timer
    lda #$3c        ; 60 frames = 1 second
    sta combo_timer
    
    ; Update score multiplier based on combo
    lda combo_count
    cmp #$02        ; 2+ enemies
    bcc no_combo_bonus
    cmp #$05        ; 5+ enemies
    bcc small_combo_bonus
    cmp #$0a        ; 10+ enemies
    bcc medium_combo_bonus
    
    ; Large combo bonus
    lda #$05        ; 5x multiplier
    sta score_multiplier
    jmp combo_bonus_applied
    
medium_combo_bonus:
    lda #$03        ; 3x multiplier
    sta score_multiplier
    jmp combo_bonus_applied
    
small_combo_bonus:
    lda #$02        ; 2x multiplier
    sta score_multiplier
    jmp combo_bonus_applied

no_combo_bonus:
    lda #$01        ; 1x multiplier
    sta score_multiplier

combo_bonus_applied:
    rts

; Add calculated score to total using BCD arithmetic
add_score_to_total:
    ; Add temp_score to main score using BCD
    sed                         ; Set decimal mode
    
    ; Add ones place
    lda score_ones
    clc
    adc temp_score_low
    sta score_ones
    
    ; Handle carries through all digits
    lda score_tens
    adc #$00                    ; Add carry
    cmp #$0a
    bcc tens_no_carry
    sec
    sbc #$0a
    sta score_tens
    
    lda score_hundreds
    adc #$01                    ; Carry from tens
    cmp #$0a
    bcc hundreds_no_carry
    sec
    sbc #$0a
    sta score_hundreds
    
    lda score_thousands
    adc #$01                    ; Carry from hundreds
    cmp #$0a
    bcc thousands_no_carry
    sec
    sbc #$0a
    sta score_thousands
    
    lda score_ten_thousands
    adc #$01                    ; Carry from thousands
    cmp #$0a
    bcc ten_thousands_no_carry
    sec
    sbc #$0a
    sta score_ten_thousands
    
    lda score_hundred_thousands
    adc #$01                    ; Carry from ten thousands
    cmp #$0a
    bcc hundred_thousands_no_carry
    sec
    sbc #$0a
    sta score_hundred_thousands
    
    lda score_millions
    adc #$01                    ; Carry from hundred thousands
    sta score_millions
    jmp score_addition_complete

tens_no_carry:
    sta score_tens
    jmp score_addition_complete

hundreds_no_carry:
    sta score_hundreds
    jmp score_addition_complete

thousands_no_carry:
    sta score_thousands
    jmp score_addition_complete

ten_thousands_no_carry:
    sta score_ten_thousands
    jmp score_addition_complete

hundred_thousands_no_carry:
    sta score_hundred_thousands

score_addition_complete:
    cld                         ; Clear decimal mode
    rts

; Update score display system with visual feedback
update_score_display_system:
    ; Update main score display
    jsr update_main_score_display
    
    ; Update multiplier display
    jsr update_multiplier_display
    
    ; Update combo display
    jsr update_combo_display
    
    ; Update achievement display
    jsr update_achievement_display
    
    rts

; Update main score display using optimized routines
update_main_score_display:
    ; Position cursor for score display
    lda score_display_row
    sta cursor_row
    lda score_display_column
    sta cursor_column
    
    ; Convert and display millions digit
    lda score_millions
    beq skip_millions           ; Don't display leading zeros
    clc
    adc #$30                    ; Convert to ASCII
    jsr display_score_digit
    
skip_millions:
    ; Convert and display hundred thousands
    lda score_hundred_thousands
    bne display_hundred_thousands
    lda score_millions          ; Check if we need leading zero
    beq skip_hundred_thousands
    
display_hundred_thousands:
    lda score_hundred_thousands
    clc
    adc #$30                    ; Convert to ASCII
    jsr display_score_digit
    
skip_hundred_thousands:
    ; Convert and display ten thousands
    lda score_ten_thousands
    bne display_ten_thousands
    lda score_hundred_thousands ; Check for leading zeros
    bne display_ten_thousands
    lda score_millions
    beq skip_ten_thousands
    
display_ten_thousands:
    lda score_ten_thousands
    clc
    adc #$30                    ; Convert to ASCII
    jsr display_score_digit
    
skip_ten_thousands:
    ; Always display thousands, hundreds, tens, ones
    lda score_thousands
    clc
    adc #$30                    ; Convert to ASCII
    jsr display_score_digit
    
    lda score_hundreds
    clc
    adc #$30                    ; Convert to ASCII
    jsr display_score_digit
    
    lda score_tens
    clc
    adc #$30                    ; Convert to ASCII
    jsr display_score_digit
    
    lda score_ones
    clc
    adc #$30                    ; Convert to ASCII
    jsr display_score_digit
    
    rts

; Display score digit with visual effects
display_score_digit:
    ; A contains ASCII digit to display
    pha                         ; Save digit
    
    ; Calculate screen position
    lda cursor_row
    asl                         ; × 2
    asl                         ; × 4
    asl                         ; × 8
    asl                         ; × 16
    asl                         ; × 32 (40 chars per line, close enough)
    clc
    adc cursor_column
    clc
    adc #<$0400                 ; Add screen base address
    sta screen_pointer_low
    
    lda #>$0400
    adc #$00                    ; Add carry
    sta screen_pointer_high
    
    ; Display digit
    pla                         ; Restore digit
    ldy #$00
    sta (screen_pointer_low),y
    
    ; Set color attribute
    lda screen_pointer_low
    clc
    adc #<($d800-$0400)         ; Convert to color RAM
    sta color_pointer_low
    lda screen_pointer_high
    adc #>($d800-$0400)
    sta color_pointer_high
    
    lda score_text_color
    sta (color_pointer_low),y
    
    ; Advance cursor
    inc cursor_column
    
    rts

; Update multiplier display with dynamic effects
update_multiplier_display:
    ; Position cursor for multiplier display
    lda #$04        ; Row 4
    sta cursor_row
    lda #$15        ; Column 21
    sta cursor_column
    
    ; Display "×" symbol
    lda #$d7        ; Multiplication symbol
    jsr display_score_digit
    
    ; Display multiplier value
    lda score_multiplier
    clc
    adc #$30        ; Convert to ASCII
    jsr display_score_digit
    
    ; Add visual effect for high multipliers
    lda score_multiplier
    cmp #$03
    bcc no_multiplier_effect
    
    ; Flash multiplier for high values
    jsr create_multiplier_flash_effect

no_multiplier_effect:
    rts

; Update combo display with counter and timer
update_combo_display:
    ; Position cursor for combo display
    lda #$05        ; Row 5
    sta cursor_row
    lda #$02        ; Column 2
    sta cursor_column
    
    ; Check if combo is active
    lda combo_count
    cmp combo_minimum
    bcc no_combo_display
    
    ; Display "COMBO:" label
    ldx #$00
combo_label_loop:
    lda combo_label_text,x
    beq combo_label_complete
    jsr display_score_digit
    inx
    jmp combo_label_loop

combo_label_complete:
    ; Display combo count
    lda combo_count
    cmp #$0a        ; Two digits?
    bcc single_digit_combo
    
    ; Two digit combo
    ldx #$00        ; Tens digit
two_digit_combo_loop:
    lda combo_count
    sec
    sbc #$0a
    bcc display_combo_tens
    sta combo_count
    inx
    jmp two_digit_combo_loop
    
display_combo_tens:
    txa
    clc
    adc #$30        ; Convert to ASCII
    jsr display_score_digit
    
single_digit_combo:
    lda combo_count
    clc
    adc #$30        ; Convert to ASCII
    jsr display_score_digit

no_combo_display:
    rts

; Create score visual feedback for enemy destruction
create_score_visual_feedback:
    ; Create floating score text at enemy position
    jsr create_floating_score_text
    
    ; Create particle effect for score gain
    jsr create_score_particle_effect
    
    ; Flash score display for large scores
    lda temp_score_high
    bne large_score_flash
    lda temp_score_low
    cmp #$50        ; Score > 80?
    bcc no_score_flash
    
large_score_flash:
    jsr create_score_flash_effect

no_score_flash:
    rts

; Create floating score text effect
create_floating_score_text:
    ; Find available floating text slot
    ldx #$03        ; Check 4 floating text slots
find_text_slot:
    lda floating_text_active,x
    beq found_text_slot
    dex
    bpl find_text_slot
    rts             ; No available slots

found_text_slot:
    ; Activate floating text
    lda #$01
    sta floating_text_active,x
    
    ; Set position to destroyed enemy location
    lda enemy_destruction_x
    sta floating_text_x,x
    lda enemy_destruction_y
    sta floating_text_y,x
    
    ; Set score value to display
    lda temp_score_low
    sta floating_text_score_low,x
    lda temp_score_high
    sta floating_text_score_high,x
    
    ; Set animation timer
    lda #$3c        ; 60 frames = 1 second
    sta floating_text_timer,x
    
    ; Set upward velocity
    lda #$ff        ; -1 (upward movement)
    sta floating_text_velocity,x
    
    rts

; Score calculation tables and data
base_score_table:
    .byte $10, $20, $30, $50    ; Enemy type scores (low byte)
    .byte $80, $c0, $00, $50    ; More enemy type scores
    .byte $00, $00, $00, $00    ; Enemy type scores (high byte)
    .byte $00, $00, $01, $01    ; More enemy type scores (high)

base_multiplier_table:
    .byte $01, $02, $03, $04    ; Multiplier values
    .byte $05, $06, $08, $0a    ; Higher multiplier values

combo_label_text:
    .byte "COMBO:", $00

; Score storage (BCD format)
score_millions:             .res 1
score_hundred_thousands:    .res 1
score_ten_thousands:        .res 1
score_thousands:            .res 1
score_hundreds:             .res 1
score_tens:                 .res 1
score_ones:                 .res 1

; Score calculation variables
temp_score_low:             .res 1
temp_score_high:            .res 1
original_score_low:         .res 1
original_score_high:        .res 1
last_score_low:             .res 1
last_score_high:            .res 1

; Multiplier and combo variables
score_multiplier:           .res 1
combo_count:                .res 1
combo_timer:                .res 1
combo_multiplier:           .res 1
combo_time_limit:           .res 1
combo_minimum:              .res 1

; Display variables
score_display_row:          .res 1
score_display_column:       .res 1
score_text_color:           .res 1
score_highlight_color:      .res 1
multiplier_color:           .res 1
cursor_row:                 .res 1
cursor_column:              .res 1
screen_pointer_low:         .res 1
screen_pointer_high:        .res 1
color_pointer_low:          .res 1
color_pointer_high:         .res 1

; Enemy destruction tracking
enemy_destruction_flags:    .res 8
enemy_destruction_index:    .res 1
enemy_destruction_x:        .res 1
enemy_destruction_y:        .res 1
enemy_score_values:         .res 16
current_difficulty_level:   .res 1

; Floating text system
floating_text_active:       .res 4
floating_text_x:            .res 4
floating_text_y:            .res 4
floating_text_score_low:    .res 4
floating_text_score_high:   .res 4
floating_text_timer:        .res 4
floating_text_velocity:     .res 4

; Multiplier values table
multiplier_values:          .res 8
```
