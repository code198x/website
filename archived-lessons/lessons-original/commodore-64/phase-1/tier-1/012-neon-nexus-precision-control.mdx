    ; A contains raw input
    ; Apply noise filtering and debouncing

    lda previous_raw_input
    sta temp_input_compare
    pla             ; Restore raw input

    ; Check for input stability (debouncing)
    cmp temp_input_compare
    beq input_is_stable

    ; Input changed - start debounce timer

    lda #$00
    sta input_stability_counter

    rts

    inc input_stability_counter
    lda input_stability_counter
    cmp input_debounce_time
    bcc input_still_debouncing

    lda raw_joystick_input
    sta current_stable_input
    sta previous_raw_input

    lda current_stable_input
    rts

    ; Process fire button with precision timing

    and #$10        ; Fire button mask

    ; Check for button press (0 to 1 transition)
    lda previous_fire_button_state
    eor current_fire_button_state
    and current_fire_button_state
    sta fire_button_press_flag

    ; Check for button release (1 to 0 transition)
    lda previous_fire_button_state
    eor current_fire_button_state
    and previous_fire_button_state
    sta fire_button_release_flag

    ; Update button hold timing
    lda current_fire_button_state
    beq fire_button_not_held

    ; Fire button is held - increment hold counter
    inc fire_button_hold_counter
    lda fire_button_hold_counter
    cmp button_hold_threshold

    sta fire_button_hold_flag
    jmp fire_button_hold_check_complete

    sta fire_button_hold_flag
