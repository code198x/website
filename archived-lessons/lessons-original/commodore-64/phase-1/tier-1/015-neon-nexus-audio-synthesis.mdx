    sta $d402,x     ; Clear pulse width low
    sta $d403,x     ; Clear pulse width high
    sta $d405,x     ; Clear attack/decay
    sta $d406,x     ; Clear sustain/release

; Advance melody pattern based on game state
advance_melody_pattern:
    ; Check game intensity to determine pattern progression
    lda player_action_intensity
    cmp #$08        ; High intensity threshold
    bcs high_intensity_melody
    cmp #$04        ; Medium intensity threshold
    bcs medium_intensity_melody
    
    ; Low intensity - calm melody
    lda current_melody_pattern
    clc
    adc #$01
    and #$03        ; 4 calm patterns
    sta current_melody_pattern
    rts
    
medium_intensity_melody:
    ; Medium intensity - more active melody
    lda current_melody_pattern
    clc
    adc #$01
    and #$03        ; 4 medium patterns
    clc
    adc #$04        ; Offset to medium patterns
    sta current_melody_pattern
    rts
    
high_intensity_melody:
    ; High intensity - dramatic melody
    lda current_melody_pattern
    clc
    adc #$01
    and #$03        ; 4 high patterns
    clc
    adc #$08        ; Offset to high patterns
    sta current_melody_pattern
    rts

; Generate musical frame for current patterns
generate_musical_frame:
    ; Generate melody on voice 0
    jsr generate_melody_voice
    
    ; Generate harmony on voice 1
    jsr generate_harmony_voice
    
    ; Generate bass/rhythm on voice 2
    jsr generate_bass_voice

; Generate melody voice based on current pattern
generate_melody_voice:
    ; Get current melody pattern
    ldy current_melody_pattern
    
    ; Get current beat position
    lda musical_pattern_timer
    lsr             ; Divide by musical tempo
    lsr
    and #$07        ; 8 beats per pattern
    tax
    
    ; Get note from pattern table
    lda melody_pattern_table,y
    clc
    adc melody_beat_offset_table,x
    
    ; Convert note to frequency
    jsr convert_note_to_frequency
    
    ; Set voice 0 frequency
    sta $d400       ; Frequency low
    txa
    sta $d401       ; Frequency high
    
    ; Set voice 0 envelope and waveform
    lda #$09        ; Attack=0, Decay=9
    sta $d405
    lda #$00        ; Sustain=0, Release=0
    sta $d406
    lda #$11        ; Triangle wave, gate on
    sta $d404
    
    rts

; Generate harmony voice based on melody
generate_harmony_voice:
    ; Get harmony offset based on current chord
    ldy current_harmony_pattern
    lda harmony_offset_table,y
    
    ; Add to melody note for harmony
    clc
    adc current_melody_note
    
    ; Convert to frequency
    jsr convert_note_to_frequency
    
    ; Set voice 1 frequency
    sta $d407       ; Frequency low
    txa
    sta $d408       ; Frequency high
    
    ; Set voice 1 envelope (softer than melody)
    lda #$05        ; Attack=0, Decay=5
    sta $d40c
    lda #$60        ; Sustain=6, Release=0
    sta $d40d
    lda #$21        ; Sawtooth wave, gate on
    sta $d40b
    
    rts

; Generate bass voice for rhythm and foundation
generate_bass_voice:
    ; Check if this is a bass beat
    lda musical_pattern_timer
    and #$07        ; Check beat position
    tax
    lda bass_pattern_table,x
    beq bass_voice_complete
    
    ; Generate bass note (octave below melody)
    lda current_melody_note
    sec
    sbc #$0c        ; Down one octave
    
    ; Convert to frequency
    jsr convert_note_to_frequency
    
    ; Set voice 2 frequency
    sta $d40e       ; Frequency low
    txa
    sta $d40f       ; Frequency high
    
    ; Set voice 2 envelope (punchy bass)
    lda #$02        ; Attack=0, Decay=2
    sta $d412
    lda #$00        ; Sustain=0, Release=0
    sta $d413
    lda #$41        ; Pulse wave, gate on
    sta $d411
    
bass_voice_complete:
    rts

    bpl check_particle_sound_loop

check_player_movement_sounds:

    beq player_movement_sound_complete

    lda player_velocity_x
    clc
    adc player_velocity_y
    cmp #$04        ; Movement threshold
    bcc player_movement_sound_complete
    
    ; Create movement sound effect
    lda #$03        ; Movement sound type
    ldy #$01        ; Low priority
    jsr create_sound_effect
    
player_movement_sound_complete:
    rts

; Create sound effect with specified type and priority
create_sound_effect:
    ; A = sound type, Y = priority
    ; Find available sound effect slot
    jsr find_available_sound_effect_slot
    cpx #$ff        ; Check if slot found
    beq sound_effect_creation_complete
    
    ; Create sound effect
    lda #$01        ; Sound effect active
    sta sound_effect_active_array,x
    
    ; Set sound effect properties
    tya             ; Priority
    sta sound_effect_priority_array,x
    
    ; Get sound effect parameters from type
    lda sound_effect_type
    sta sound_effect_type_array,x
    tax             ; Use as table index
    
    lda sound_effect_frequency_table,x
    sta sound_effect_frequency_array,y
    lda sound_effect_waveform_table,x
    sta sound_effect_waveform_array,y
    lda sound_effect_duration_table,x
    sta sound_effect_duration_array,y
    lda sound_effect_envelope_table,x
    sta sound_effect_envelope_array,y
    
    ; Assign to appropriate voice
    jsr assign_sound_effect_voice
    
    ; Increment sound effect count
    inc current_sound_effect_count
    
sound_effect_creation_complete:
    rts

; Process active sound effects
process_active_sound_effects:
    ; Process all active sound effects
    ldx #$00        ; Start with effect 0
process_sound_effects_loop:
    ; Check if sound effect is active
    lda sound_effect_active_array,x
    beq next_sound_effect_process
    
    ; Update sound effect
    jsr update_individual_sound_effect
    
next_sound_effect_process:
    inx
    cpx #$10        ; Check all 16 sound effects
    bne process_sound_effects_loop
    
    rts

; Update individual sound effect
update_individual_sound_effect:
    ; X contains sound effect index
    ; Decrement duration
    dec sound_effect_duration_array,x
    bne sound_effect_still_active
    
    ; Sound effect expired
    lda #$00
    sta sound_effect_active_array,x
    dec current_sound_effect_count
    
    ; Turn off voice if this was the only effect using it
    jsr check_voice_usage
    
sound_effect_still_active:
    ; Update sound effect parameters
    jsr update_sound_effect_parameters
    
    rts

; Convert musical note to SID frequency
convert_note_to_frequency:
    ; A contains note number (0-95, covering 8 octaves)
    ; Returns frequency in A (low) and X (high)
    tay             ; Save note number
    
    ; Get base frequency from table
    lda note_number
    and #$0f        ; Get note within octave (0-11)
    tax
    lda base_frequency_table_low,x
    sta temp_freq_low
    lda base_frequency_table_high,x
    sta temp_freq_high
    
    ; Calculate octave shift
    tya             ; Restore note number
    lsr
    lsr
    lsr
    lsr             ; Divide by 16 to get octave
    beq frequency_conversion_complete
    
    ; Shift frequency up by octaves
    tax             ; Octave count
octave_shift_loop:
    asl temp_freq_low
    rol temp_freq_high
    dex
    bne octave_shift_loop
    
frequency_conversion_complete:
    lda temp_freq_low
    ldx temp_freq_high
    rts

process_audio_visual_synchronization:
