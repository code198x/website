init_atmospheric_effects:

; Create environmental effects arena with atmospheric enhancement
create_environmental_effects_arena:
    ; Draw enhanced arena with environmental zones
    jsr draw_environmental_arena_boundaries

    ; Create atmospheric effect zones
    jsr create_atmospheric_effect_zones

    jsr setup_environmental_hazard_areas

; Update particle systems for dynamic environmental effects
update_particle_systems:
    ; Check if it's time to update particles
    inc environmental_update_counter
    lda environmental_update_counter
    cmp environmental_update_frequency
    bcc skip_particle_update

    ; Reset counter and update particles
    lda #$00
    sta environmental_update_counter

    ; Spawn new particles
    jsr spawn_environmental_particles

    ; Update existing particles
    jsr update_existing_particles

    ; Remove expired particles
    jsr remove_expired_particles

skip_particle_update:
    rts

; Spawn environmental particles based on game state
spawn_environmental_particles:
    ; Check if we can spawn more particles
    lda current_particle_count
    cmp max_active_particles
    bcs no_particle_spawn

    ; Determine particle type based on environmental state
    jsr determine_particle_spawn_type

    ; Spawn atmospheric particles
    jsr spawn_atmospheric_particles

    ; Spawn effect particles from game events
    jsr spawn_effect_particles

    ; Spawn hazard particles
    jsr spawn_hazard_particles

no_particle_spawn:
    rts

; Spawn atmospheric particles for ambient environment
spawn_atmospheric_particles:
    ; Check atmospheric particle spawn rate
    lda atmospheric_particle_spawn_counter
    inc atmospheric_particle_spawn_counter
    and #$07        ; Spawn every 8 frames
    bne atmospheric_spawn_complete

    ; Find available particle slot
    jsr find_available_particle_slot
    cpx #$ff        ; Check if slot found
    beq atmospheric_spawn_complete

    ; Create atmospheric particle
    lda #$01        ; Particle active
    sta particle_active_array,x
    lda #$01        ; Atmospheric particle type
    sta particle_type_array,x

    ; Random spawn position at top of screen
    jsr generate_random_number
    and #$3f        ; Limit to screen width
    clc
    adc #$02        ; Offset from edge
    sta particle_x_position_array,x
    lda #$02        ; Top of screen
    sta particle_y_position_array,x

    ; Set atmospheric particle velocity
    lda #$00        ; No horizontal movement
    sta particle_x_velocity_array,x
    lda #$01        ; Slow downward drift
    sta particle_y_velocity_array,x

    ; Set particle properties
    lda #$ff        ; Long life (255 frames)
    sta particle_life_timer_array,x
    lda current_atmospheric_color
    sta particle_color_array,x

    ; Increment particle count
    inc current_particle_count

atmospheric_spawn_complete:
    rts

; Spawn effect particles from game events
spawn_effect_particles:
    ; Check for explosion events
    jsr check_explosion_events

    ; Check for impact events
    jsr check_impact_events

    ; Check for powerup events
    jsr check_powerup_events

    rts

; Check for explosion events and spawn particles
check_explosion_events:
    ; Check entity destruction flags
    ldx #$07        ; Check all 8 entities
check_explosion_loop:
    lda entity_destruction_flags,x
    beq next_explosion_check

    ; Entity destroyed - create explosion particles
    lda #$00        ; Clear destruction flag
    sta entity_destruction_flags,x

    ; Spawn explosion particle burst
    ldy #$06        ; Spawn 6 explosion particles
spawn_explosion_particles_loop:
    phy             ; Save Y counter
    jsr find_available_particle_slot
    cpx #$ff        ; Check if slot found
    beq explosion_particle_complete

    ; Create explosion particle
    lda #$01        ; Particle active
    sta particle_active_array,x
    lda #$02        ; Explosion particle type
    sta particle_type_array,x

    ; Set explosion position from entity
    lda entity_destruction_x
    sta particle_x_position_array,x
    lda entity_destruction_y
    sta particle_y_position_array,x

    ; Random explosion velocity
    jsr generate_random_number
    and #$0f        ; Limit velocity range
    sec
    sbc #$08        ; Center around 0 (-8 to +7)
    sta particle_x_velocity_array,x

    jsr generate_random_number
    and #$0f        ; Limit velocity range
    sec
    sbc #$08        ; Center around 0 (-8 to +7)
    sta particle_y_velocity_array,x

    ; Set explosion particle properties
    lda #$1e        ; Short life (30 frames)
    sta particle_life_timer_array,x
    lda #$02        ; Red explosion color
    sta particle_color_array,x

    ; Increment particle count
    inc current_particle_count

explosion_particle_complete:
    ply             ; Restore Y counter
    dey
    bne spawn_explosion_particles_loop

next_explosion_check:
    dex
    bpl check_explosion_loop

    rts

; Update existing particles with physics simulation
update_existing_particles:
    ; Process all active particles
    ldx #$00        ; Start with particle 0
update_particles_loop:
    ; Check if particle is active
    lda particle_active_array,x
    beq next_particle_update

    ; Update particle physics
    jsr update_particle_physics

    ; Update particle life
    jsr update_particle_life

    ; Update particle visual properties
    jsr update_particle_visual_properties

next_particle_update:
    inx
    cpx #$20        ; Check all 32 particles
    bne update_particles_loop

    rts

; Update particle physics for realistic movement
update_particle_physics:
    ; X contains particle index
    ; Update X position
    lda particle_x_position_array,x
    clc
    adc particle_x_velocity_array,x
    sta particle_x_position_array,x

    ; Check X boundaries
    cmp #$28        ; Right boundary
    bcc check_particle_y_physics
    cmp #$d8        ; Left boundary (negative wrap)
    bcs check_particle_y_physics

    ; Outside boundaries - mark for removal
    lda #$01        ; 1 frame life (will expire next update)
    sta particle_life_timer_array,x

check_particle_y_physics:
    ; Update Y position
    lda particle_y_position_array,x
    clc
    adc particle_y_velocity_array,x
    sta particle_y_position_array,x

    ; Check Y boundaries
    cmp #$18        ; Bottom boundary
    bcc particle_physics_complete

    ; Below screen - mark for removal
    lda #$01        ; 1 frame life (will expire next update)
    sta particle_life_timer_array,x

particle_physics_complete:
    ; Apply environmental forces based on particle type
    lda particle_type_array,x
    cmp #$01        ; Atmospheric particle
    beq apply_atmospheric_forces
    cmp #$02        ; Explosion particle
    beq apply_explosion_forces

    rts

apply_atmospheric_forces:
    ; Apply atmospheric drift and gravity
    lda particle_y_velocity_array,x
    cmp #$03        ; Maximum fall speed
    bcs atmospheric_forces_complete
    inc particle_y_velocity_array,x ; Slight gravity

atmospheric_forces_complete:
    rts

apply_explosion_forces:
    ; Apply explosion particle deceleration
    lda particle_x_velocity_array,x
    beq check_explosion_y_decel
    bmi explosion_x_negative

    ; Positive X velocity - decelerate
    dec particle_x_velocity_array,x
    jmp check_explosion_y_decel

explosion_x_negative:
    ; Negative X velocity - decelerate
    inc particle_x_velocity_array,x

check_explosion_y_decel:
    lda particle_y_velocity_array,x
    beq explosion_forces_complete
    bmi explosion_y_negative

    ; Positive Y velocity - decelerate
    dec particle_y_velocity_array,x
    jmp explosion_forces_complete

explosion_y_negative:
    ; Negative Y velocity - decelerate (add gravity)
    inc particle_y_velocity_array,x

explosion_forces_complete:
    rts

; Update atmospheric effects for environmental immersion
update_atmospheric_effects:
    ; Update atmospheric color cycling
    jsr update_atmospheric_color_cycling

    ; Update atmospheric density
    jsr update_atmospheric_density

    ; Update environmental mood
    jsr update_environmental_mood

    rts

; Update atmospheric color cycling for dynamic environment
update_atmospheric_color_cycling:
    ; Increment color cycle counter
    inc color_cycle_counter
    lda color_cycle_counter
    cmp color_cycle_speed
    bcc color_cycling_complete

    ; Reset counter and cycle color
    lda #$00
    sta color_cycle_counter

    ; Cycle through atmospheric colors
    inc current_atmospheric_color
    lda current_atmospheric_color
    cmp #$08        ; Wrap at 8 colors
    bcc store_atmospheric_color
    lda #$02        ; Reset to start color

store_atmospheric_color:
    sta current_atmospheric_color

    ; Update border color for atmospheric effect
    sta $d020       ; Border color

color_cycling_complete:
    rts

; Update environmental hazards for dynamic challenges
update_environmental_hazards:
    ; Update hazard spawn timing
    jsr update_hazard_spawn_timing

    ; Update existing hazards
    jsr update_existing_hazards

    ; Check hazard interactions
    jsr check_hazard_interactions

    rts

; Update hazard spawn timing for dynamic challenge
update_hazard_spawn_timing:
    ; Decrement spawn timer
    dec hazard_spawn_timer
    bne hazard_spawn_timing_complete

    ; Reset timer and spawn hazard
    lda hazard_spawn_interval
    sta hazard_spawn_timer

    ; Check if we can spawn more hazards
    lda active_hazard_count
    cmp max_environmental_hazards
    bcs hazard_spawn_timing_complete

    ; Spawn new environmental hazard
    jsr spawn_environmental_hazard

hazard_spawn_timing_complete:
    rts

; Spawn environmental hazard for gameplay challenge
spawn_environmental_hazard:
    ; Find available hazard slot
    jsr find_available_hazard_slot
    cpx #$ff        ; Check if slot found
    beq hazard_spawn_complete

    ; Create environmental hazard
    lda #$01        ; Hazard active
    sta hazard_active_array,x

    ; Determine hazard type based on current game state
    lda global_environmental_state
    and #$03        ; 4 hazard types
    sta hazard_type_array,x

    ; Set hazard position
    jsr generate_random_number
    and #$3f        ; Limit to screen area
    clc
    adc #$04        ; Offset from edges
    sta hazard_x_position_array,x

    jsr generate_random_number
    and #$1f        ; Limit Y range
    clc
    adc #$04        ; Offset from top
    sta hazard_y_position_array,x

    ; Set hazard properties based on type
    ldy hazard_type_array,x
    lda hazard_size_table,y
    sta hazard_size_array,x
    lda hazard_intensity_table,y
    sta hazard_intensity_array,x
    lda hazard_duration_table,y
    sta hazard_timer_array,x

    ; Increment active hazard count
    inc active_hazard_count

hazard_spawn_complete:
    rts

; Render particle systems for visual effects
render_particle_systems:
    ; Render all active particles
    ldx #$00        ; Start with particle 0
render_particles_loop:
    ; Check if particle is active
    lda particle_active_array,x
    beq next_particle_render

    ; Render individual particle
    jsr render_individual_particle

next_particle_render:
    inx
    cpx #$20        ; Check all 32 particles
    bne render_particles_loop

    rts

; Render individual particle with VIC-II optimization
render_individual_particle:
    ; X contains particle index
    ; Get particle screen position
    lda particle_x_position_array,x
    sta temp_particle_screen_x
    lda particle_y_position_array,x
    sta temp_particle_screen_y

    ; Calculate screen memory address
    lda temp_particle_screen_y
    asl
    asl
    asl
    asl
    asl             ; Multiply by 32 (approximate for 40 chars)
    clc
    adc temp_particle_screen_x
    clc
    adc #<$0400     ; Add screen base
    sta particle_screen_pointer_low

    lda #>$0400
    adc #$00
    sta particle_screen_pointer_high

    ; Get particle character based on type
    ldy particle_type_array,x
    lda particle_character_table,y

    ; Display particle character
    ldy #$00
    sta (particle_screen_pointer_low),y

    ; Set particle color
    lda particle_screen_pointer_low
    clc
    adc #<($d800-$0400)  ; Convert to color RAM
    sta particle_color_pointer_low
    lda particle_screen_pointer_high
    adc #>($d800-$0400)
    sta particle_color_pointer_high

    lda particle_color_array,x
    sta (particle_color_pointer_low),y

    rts

particle_color_pointer_high:        .res 1  ; Particle color pointer high
