; Update score system with conditional branching
update_score_system:
    ; Update combo timer
    lda $ec         ; Load combo timer
    beq reset_combo ; Branch if timer expired
    dec $ec         ; Decrement combo timer
    jmp score_update_done

reset_combo:
    ; Reset combo multiplier
    lda #$01        ; Reset multiplier to base
    sta $ea         ; Store base multiplier
    lda #$00        ; Clear combo count
    sta $eb         ; Store cleared combo

score_update_done:
    rts

; Award score points with multiplier logic
award_score_points:
    ; A register contains base points to award
    ; Apply combo multiplier
    ldx $ea         ; Load current multiplier
    cpx #$01        ; Check if base multiplier
    beq add_base_score ; Branch if no multiplier

    ; Apply multiplier (simplified: double for combo)
    asl             ; Double the score

add_base_score:
    ; Add to total score (16-bit addition)
    clc             ; Clear carry
    adc $e8         ; Add to score low byte
    sta $e8         ; Store updated low byte
    bcc score_award_done ; Branch if no carry

    ; Handle carry to high byte
    inc $e9         ; Increment score high byte

score_award_done:
    ; Increment combo counter
    inc $eb         ; Increment combo count

    ; Reset combo timer
    lda #$3c        ; Reset combo timer (60 frames)
    sta $ec         ; Store reset timer

    ; Check for combo threshold
    lda $eb         ; Load combo count
    cmp #$05        ; Check for 5-hit combo
    bcc combo_check_done ; Branch if below threshold

    ; Combo threshold reached - increase multiplier
    lda $ea         ; Load current multiplier
    cmp #$04        ; Check maximum multiplier
    bcs combo_check_done ; Branch if at maximum

    inc $ea         ; Increase multiplier

    ; Play combo sound
    jsr play_combo_sound

combo_check_done:
    rts

; Check level progression conditions
check_level_progression:
    ; Check if level complete flag is set
    lda $f1         ; Load level complete flag
    beq progression_check_done ; Branch if level not complete

    ; Check if victory sequence finished
    lda $f5         ; Load victory timer
    bne progression_check_done ; Branch if victory sequence still running

    ; Level complete - check for game completion

progression_check_done:

advance_to_next_level:

    ; Scale difficulty based on level

    ; High level scaling
    inc $da         ; Increase enemy count
    lda $db         ; Load enemy health base
    clc             ; Clear carry
    adc #$10        ; Add health increase
    sta $db         ; Store increased health
    jmp reset_level_state

normal_scaling:
    ; Normal scaling for early levels
    lda $db         ; Load enemy health base
    clc             ; Clear carry

    sta $db         ; Store increased health

reset_level_state:
    ; Reset level-specific variables
    lda #$00        ; Clear enemies defeated
    sta $f4         ; Reset defeat counter
    lda #$00        ; Clear victory flags
    sta $f0         ; Clear victory flag
    sta $f1         ; Clear level complete flag

    ; Restore player resources for new level
    lda #$64        ; Full health for new level
    sta $a0         ; Restore player health
    lda #$32        ; Partial energy for new level
    sta $a1         ; Restore player energy

    ; Award level advancement bonus

    jsr award_score_points

    ; Reinitialize enemies for new level
    jsr create_enemy_entities

    ; Play level advancement sound
    jsr play_level_advance_sound

    rts

; Update victory sequence with conditional effects
update_victory_sequence:
    ; Check if victory sequence active
    lda $f5         ; Load victory timer
    beq victory_sequence_done ; Branch if sequence finished

    ; Update victory timer
    dec $f5         ; Decrement victory timer

    ; Create victory visual effects based on timer
    lda $f5         ; Load current timer
    cmp #$30        ; Check if in first half of sequence
    bcc victory_second_half ; Branch if second half

    ; First half - bright flashing effects
    lda $f5         ; Load timer
    and #$04        ; Mask for flashing
    beq victory_bright ; Branch for bright frame

    ; Dim frame
    lda #$01        ; Dim colors
    sta $d020       ; Border color
    sta $d021       ; Background color
    jmp victory_effect_done

victory_bright:
    ; Bright frame
    lda #$07        ; Bright colors
    sta $d020       ; Border color
    sta $d021       ; Background color
    jmp victory_effect_done

victory_second_half:
    ; Second half - color cycling
    lda $f5         ; Load timer
    and #$07        ; Mask for color cycle
    clc             ; Clear carry
    adc #$02        ; Offset colors
    sta $d020       ; Border color cycling

victory_effect_done:
    rts

victory_sequence_done:
    ; Victory sequence finished - restore normal colors
    lda #$00        ; Black border
    sta $d020       ; Border color
    lda #$06        ; Blue background
    sta $d021       ; Background color

    rts

; Award various victory rewards using conditional logic
award_victory_rewards:

store_health_reward:

    lda $a1         ; Load current energy
    clc             ; Clear carry
    adc $f8         ; Add victory energy bonus

    lda #$64        ; Clamp to maximum

store_energy_reward:

    ; Award base score bonus
    lda $ef         ; Load victory score bonus
    jsr award_score_points

; Award perfect level bonus
award_perfect_bonus:
    ; Double score bonus for perfect level
    lda $ef         ; Load base bonus
    asl             ; Double it
    jsr award_score_points

    ; Award achievement energy bonus
    lda $a1         ; Load current energy
    clc             ; Clear carry
    adc $ce         ; Add achievement energy bonus
    cmp #$64        ; Check maximum
    bcc store_perfect_energy ; Store if within limit
    lda #$64        ; Clamp to maximum

store_perfect_energy:
    sta $a1         ; Store updated energy

    rts

; Award speed completion bonus
award_speed_bonus:
    ; Triple score bonus for speed completion
    lda $ef         ; Load base bonus
    asl             ; Double it
    clc             ; Clear carry
    adc $ef         ; Add base again (triple total)
    jsr award_score_points

    ; Award extra weapon ammo
    lda $a2         ; Load current ammo
    clc             ; Clear carry
    adc #$10        ; Add speed bonus ammo
    cmp #$32        ; Check maximum ammo
    bcc store_speed_ammo ; Store if within limit

store_speed_ammo:

; Award standard victory rewards

### Conditional Branching: Your Victory Management System

The most important thing to understand: **conditional branching enables sophisticated victory logic and game state management**. In your victory progression code:

**Victory Condition Checking**: Multiple victory paths with branching logic

```nasm
check_victory_conditions:
    lda $f4         ; Load enemies defeated
    cmp $f3         ; Compare with required count
    bcc check_other_conditions ; Branch if not enough enemies defeated

    ; Primary victory achieved!
    jsr trigger_level_victory
```

**Achievement Logic**: Conditional rewards based on performance

```nasm
check_speed_bonus:
    lda $a3         ; Load completion time
    cmp $cd         ; Compare with speed threshold
    bcs normal_victory ; Branch if too slow

    ; Speed completion achieved!
    lda #$01        ; Set speed achievement
    sta $cb         ; Store speed flag
```

**Victory System Examples Using Conditional Branching:**

| Technique             | Example                 | Use Case                          |
| --------------------- | ----------------------- | --------------------------------- |
| **Victory Checking**  | `CMP / BCC victory`     | Check multiple victory conditions |
| **Achievement Logic** | `CMP / BCS achievement` | Award conditional bonuses         |
| **State Management**  | `BEQ / BNE state`       | Control game progression flow     |
