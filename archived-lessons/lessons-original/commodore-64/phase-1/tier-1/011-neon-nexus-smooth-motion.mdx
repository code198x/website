start:

    ; Smooth Motion main loop
smooth_motion_loop:

    ; Process all input with smooth response

    jsr update_smooth_player_movement

    jsr check_victory_conditions_fast

    jsr wait_for_frame_completion

; Initialize smooth motion system using timing optimization
init_smooth_motion_system:
    ; Set up motion smoothing parameters
    jsr init_motion_smoothing_parameters
    jsr init_velocity_interpolation
    jsr init_acceleration_systems
    jsr init_timing_control

    ; Initialize smooth motion variables
    lda #$00        ; Clear all motion states
    sta player_smooth_x_fraction
    sta player_smooth_y_fraction
    sta motion_interpolation_factor
    sta frame_timing_accumulator

    ; Set motion smoothing parameters
    lda #$04        ; Motion smoothing factor (4 = smooth)
    sta motion_smoothing_factor
    lda #$02        ; Velocity damping factor
    sta velocity_damping_factor
    lda #$08        ; Acceleration factor
    sta acceleration_factor

    ; Initialize target frame rate control
    lda #$02        ; Target: 30fps (every 2 vblanks)
    sta target_frame_interval
    lda #$00        ; Frame counter
    sta frame_counter

    rts

; Initialize raster timing system for consistent performance
init_raster_timing_system:
    ; Set up raster interrupt timing
    lda #$7f        ; Disable CIA interrupts
    sta $dc0d       ; CIA 1 interrupt control
    sta $dd0d       ; CIA 2 interrupt control

    ; Clear interrupt pending flags
    lda $dc0d       ; CIA 1 interrupt status
    lda $dd0d       ; CIA 2 interrupt status

    ; Set up raster interrupt
    lda #$01        ; Enable raster interrupt
    sta $d01a       ; VIC interrupt mask

    ; Set raster line for interrupt (bottom of screen)
    lda #$fa        ; Raster line 250 (bottom)
    sta $d012       ; Raster line register
    lda $d011       ; VIC control register
    and #$7f        ; Clear bit 8 of raster line
    sta $d011       ; Store back

    ; Set interrupt vector
    lda #<raster_interrupt
    sta $0314       ; IRQ vector low
    lda #>raster_interrupt
    sta $0315       ; IRQ vector high

    ; Enable interrupts
    cli             ; Clear interrupt flag

    rts

; Initialize sprite optimization system for smooth entity display
init_sprite_optimization_system:
    ; Set up sprite multiplexing for smooth entity display
    jsr init_sprite_multiplexing
    jsr init_sprite_priority_system
    jsr init_sprite_timing_optimization
    jsr init_sprite_memory_management

    ; Configure sprites for optimal performance
    lda #$ff        ; Enable all 8 sprites
    sta $d015       ; Sprite enable register

    ; Set sprite priorities for consistent display
    lda #$00        ; Sprites behind background
    sta $d01b       ; Sprite background priority

    ; Configure sprite multicolor for efficiency
    lda #$00        ; Single color sprites (faster)
    sta $d01c       ; Sprite multicolor register

    ; Set sprite expand for performance
    lda #$00        ; No expansion (better performance)
    sta $d017       ; Sprite Y expand
    sta $d01d       ; Sprite X expand

    rts

; Setup raster interrupt for frame timing
setup_raster_interrupt:
    ; Initialize raster interrupt system
    lda #$00
    sta frame_timing_counter
    sta frame_timing_accumulator
    sta last_frame_time

    ; Set initial timing state
    lda #$01
    sta frame_timing_enabled

    rts

; Wait for VBlank for consistent timing
wait_for_vblank:
    ; Wait for VBlank start (raster line 0)
wait_vblank_loop:
    lda $d012       ; Read current raster line
    bne wait_vblank_loop ; Wait until line 0

    ; VBlank started - increment frame counter
    inc frame_counter

    ; Check if this is our target frame
    lda frame_counter
    cmp target_frame_interval
    bcc frame_skip      ; Skip this frame

    ; Reset frame counter for next cycle
    lda #$00
    sta frame_counter

    ; This is our update frame
    lda #$01
    sta frame_update_flag
    rts

frame_skip:
    ; Skip this frame for consistent timing
    lda #$00
    sta frame_update_flag
    rts

; Start frame timer for performance monitoring
start_frame_timer:
    ; Record frame start time using CIA timer
    lda $dc04       ; CIA 1 timer A low
    sta frame_start_time_low
    lda $dc05       ; CIA 1 timer A high
    sta frame_start_time_high

    rts

; Update smooth player movement using interpolation
update_smooth_player_movement:
    ; Check if we should update this frame
    lda frame_update_flag
    beq skip_player_update

    ; Read input and calculate desired velocity
    jsr calculate_smooth_player_velocity

    ; Apply velocity smoothing
    jsr apply_velocity_smoothing

    ; Update position with sub-pixel precision
    jsr update_player_position_smooth

    ; Apply boundary constraints smoothly
    jsr apply_smooth_boundary_constraints

skip_player_update:
    rts

; Calculate smooth player velocity from input
calculate_smooth_player_velocity:
    ; Read joystick input
    lda $dc00       ; Read joystick port 2
    and #$1f        ; Mask relevant bits
    eor #$1f        ; Invert logic

    ; Calculate desired X velocity
    lsr             ; Check right
    bcc check_left_smooth
    lda acceleration_factor
    sta desired_velocity_x
    jmp check_y_input_smooth

check_left_smooth:
    lsr             ; Check left
    bcc check_y_input_smooth
    lda acceleration_factor
    eor #$ff        ; Make negative (two's complement)
    clc
    adc #$01
    sta desired_velocity_x
    jmp check_y_input_smooth

check_y_input_smooth:
    ; No horizontal input - apply damping
    lda desired_velocity_x
    bne calculate_y_velocity_smooth

    ; Apply velocity damping to X
    lda player_velocity_x
    beq calculate_y_velocity_smooth
    bpl damp_positive_x

    ; Negative velocity - move toward zero
    clc
    adc velocity_damping_factor
    bpl zero_x_velocity
    sta player_velocity_x
    jmp calculate_y_velocity_smooth

damp_positive_x:
    ; Positive velocity - move toward zero
    sec
    sbc velocity_damping_factor
    bcc zero_x_velocity
    sta player_velocity_x
    jmp calculate_y_velocity_smooth

zero_x_velocity:
    lda #$00
    sta player_velocity_x

calculate_y_velocity_smooth:
    ; Similar Y velocity calculation
    lda $dc00       ; Re-read input for Y
    and #$1f        ; Mask relevant bits
    eor #$1f        ; Invert logic

    ; Skip to Y bits (already shifted by X processing)
    lsr             ; Skip right
    lsr             ; Skip left
    lsr             ; Check down
    bcc check_up_smooth
    lda acceleration_factor
    sta desired_velocity_y
    jmp apply_velocity_changes

check_up_smooth:
    lsr             ; Check up
    bcc apply_damping_y
    lda acceleration_factor
    eor #$ff        ; Make negative
    clc
    adc #$01
    sta desired_velocity_y
    jmp apply_velocity_changes

apply_damping_y:
    ; No vertical input - apply damping
    lda player_velocity_y
    beq apply_velocity_changes
    bpl damp_positive_y

    ; Negative velocity
    clc
    adc velocity_damping_factor
    bpl zero_y_velocity
    sta player_velocity_y
    jmp apply_velocity_changes

damp_positive_y:
    ; Positive velocity
    sec
    sbc velocity_damping_factor
    bcc zero_y_velocity
    sta player_velocity_y
    jmp apply_velocity_changes

zero_y_velocity:
    lda #$00
    sta player_velocity_y

apply_velocity_changes:
    rts

; Apply velocity smoothing for professional motion feel
apply_velocity_smoothing:
    ; Smooth X velocity toward desired velocity
    lda desired_velocity_x
    sec
    sbc player_velocity_x   ; Calculate velocity difference
    beq smooth_y_velocity   ; No change needed

    ; Apply smoothing factor
    bpl positive_x_diff

    ; Negative difference - limit change rate
    cmp velocity_smoothing_limit
    bcs apply_x_change
    lda velocity_smoothing_limit
    eor #$ff
    clc
    adc #$01            ; Make negative limit
    jmp apply_x_change

positive_x_diff:
    ; Positive difference - limit change rate
    cmp velocity_smoothing_limit
    bcc apply_x_change
    lda velocity_smoothing_limit

apply_x_change:
    clc
    adc player_velocity_x
    sta player_velocity_x

smooth_y_velocity:
    ; Similar smoothing for Y velocity
    lda desired_velocity_y
    sec
    sbc player_velocity_y
    beq velocity_smoothing_complete

    bpl positive_y_diff

    cmp velocity_smoothing_limit
    bcs apply_y_change
    lda velocity_smoothing_limit
    eor #$ff
    clc
    adc #$01
    jmp apply_y_change

positive_y_diff:
    cmp velocity_smoothing_limit
    bcc apply_y_change
    lda velocity_smoothing_limit

apply_y_change:
    clc
    adc player_velocity_y
    sta player_velocity_y

velocity_smoothing_complete:
    rts

; Update player position with sub-pixel precision
update_player_position_smooth:
    ; Update X position with fractional precision
    lda player_smooth_x_fraction
    clc
    adc player_velocity_x
    sta player_smooth_x_fraction

    ; Check for pixel boundary crossing
    bcc no_x_pixel_change

    ; Crossed pixel boundary - update position
    inc $80             ; Player X position

no_x_pixel_change:
    ; Update Y position with fractional precision
    lda player_smooth_y_fraction
    clc
    adc player_velocity_y
    sta player_smooth_y_fraction

    ; Check for pixel boundary crossing
    bcc no_y_pixel_change

    ; Crossed pixel boundary - update position
    inc $81             ; Player Y position

no_y_pixel_change:
    rts

; Update smooth entity swarm with optimized timing
update_smooth_entity_swarm:
    ; Check if we should update entities this frame
    lda frame_update_flag
    beq skip_entity_update

    ; Update entities with smooth motion
    ldx #$00            ; Start with entity 0

update_smooth_entity_loop:
    ; Check if entity is active
    lda enemy_active_array,x
    beq next_smooth_entity

    ; Update entity smooth movement
    jsr update_entity_smooth_movement

    ; Update entity AI with timing consideration
    jsr update_entity_ai_timed

    ; Update entity visual interpolation
    jsr update_entity_visual_interpolation

next_smooth_entity:
    inx
    cpx #$08            ; Check all 8 entities
    bne update_smooth_entity_loop

skip_entity_update:
    rts

; Update entity smooth movement using interpolation
update_entity_smooth_movement:
    ; X contains entity index
    ; Apply smooth movement to entity X velocity
    lda enemy_velocity_x_array,x
    clc
    adc enemy_smooth_x_fraction,x
    sta enemy_smooth_x_fraction,x
    bcc check_entity_y_movement

    ; Update entity X position
    lda enemy_x_array,x
    clc
    adc #$01
    sta enemy_x_array,x

check_entity_y_movement:
    ; Apply smooth movement to entity Y velocity
    lda enemy_velocity_y_array,x
    clc
    adc enemy_smooth_y_fraction,x
    sta enemy_smooth_y_fraction,x
    bcc entity_movement_complete

    ; Update entity Y position
    lda enemy_y_array,x
    clc
    adc #$01
    sta enemy_y_array,x

entity_movement_complete:
    rts

; Update optimized sprite display for smooth visual performance
update_optimized_sprite_display:
    ; Use sprite multiplexing for smooth entity display
    jsr setup_sprite_multiplexing

    ; Update sprite positions with smooth interpolation
    jsr update_sprite_positions_smooth

    jsr optimize_sprite_rendering_order

    ; Apply sprite visual effects efficiently

; Setup sprite multiplexing for multiple entities
setup_sprite_multiplexing:
    ; Count active entities for sprite allocation
    lda #$00
    sta active_entity_count
    ldx #$00

count_active_entities:
    lda enemy_active_array,x
    beq next_entity_count
    inc active_entity_count

next_entity_count:
    inx
    cpx #$08
    bne count_active_entities

    ; Allocate sprites based on active entities
    lda active_entity_count
    cmp #$08            ; More than 8 entities?
    bcc allocate_sprites

    ; Need sprite multiplexing
    jsr setup_advanced_sprite_multiplexing
    rts

allocate_sprites:
    ; Simple sprite allocation (8 or fewer entities)
    jsr setup_simple_sprite_allocation
    rts

; Wait for frame completion to maintain timing
wait_for_frame_completion:
    ; Calculate frame time used
    lda $dc04           ; Current timer low
    sec
    sbc frame_start_time_low
    sta frame_time_used_low

    lda $dc05           ; Current timer high
    sbc frame_start_time_high
    sta frame_time_used_high

    ; Check if we have time remaining
    lda frame_time_used_high
    cmp target_frame_time_high
    bcc frame_time_remaining
    bne frame_overrun

    lda frame_time_used_low
    cmp target_frame_time_low
    bcc frame_time_remaining

frame_overrun:
    ; Frame took too long - skip waiting
    rts

frame_time_remaining:
    ; Wait for remaining frame time
wait_frame_time_loop:
    lda $dc04           ; Check current time
    sec
    sbc frame_start_time_low
    lda $dc05
    sbc frame_start_time_high

    cmp target_frame_time_high
    bcc wait_frame_time_loop
    bne frame_wait_complete

    lda $dc04
    sec
    sbc frame_start_time_low
    cmp target_frame_time_low
    bcc wait_frame_time_loop

frame_wait_complete:
    rts

; Raster interrupt routine for timing control
raster_interrupt:
    ; Save registers
    pha
    txa
    pha
    tya
    pha

    ; Acknowledge interrupt
    lda $d019           ; VIC interrupt status
    sta $d019           ; Clear interrupt

    ; Update frame timing
    inc frame_timing_counter

    ; Restore registers
    pla
    tay
    pla
    tax
    pla

    ; Return from interrupt
    rti

; Smooth motion variables and data
player_smooth_x_fraction:       .res 1  ; Sub-pixel X position
player_smooth_y_fraction:       .res 1  ; Sub-pixel Y position
player_velocity_x:              .res 1  ; Current X velocity
player_velocity_y:              .res 1  ; Current Y velocity
desired_velocity_x:             .res 1  ; Target X velocity
desired_velocity_y:             .res 1  ; Target Y velocity

; Motion smoothing parameters
motion_smoothing_factor:        .res 1  ; Motion smoothing amount
velocity_damping_factor:        .res 1  ; Velocity damping rate
acceleration_factor:            .res 1  ; Acceleration amount
velocity_smoothing_limit:       .res 1  ; Maximum velocity change per frame
motion_interpolation_factor:    .res 1  ; Interpolation factor

; Frame timing variables
frame_counter:                  .res 1  ; Current frame count
target_frame_interval:          .res 1  ; Frames between updates
frame_update_flag:              .res 1  ; Should update this frame
frame_timing_counter:           .res 1  ; Interrupt frame counter
frame_timing_accumulator:       .res 1  ; Timing accumulator
frame_timing_enabled:           .res 1  ; Timing system enabled

; Frame performance monitoring
frame_start_time_low:           .res 1  ; Frame start time (low)
frame_start_time_high:          .res 1  ; Frame start time (high)
frame_time_used_low:            .res 1  ; Time used this frame (low)
frame_time_used_high:           .res 1  ; Time used this frame (high)
target_frame_time_low:          .res 1  ; Target frame time (low)
target_frame_time_high:         .res 1  ; Target frame time (high)
last_frame_time:                .res 1  ; Previous frame timing

; Entity smooth motion arrays
enemy_smooth_x_fraction:        .res 8  ; Entity sub-pixel X positions
enemy_smooth_y_fraction:        .res 8  ; Entity sub-pixel Y positions
entity_interpolation_factors:   .res 8  ; Per-entity interpolation factors

; Sprite optimization variables
active_entity_count:            .res 1  ; Number of active entities
sprite_allocation_table:        .res 8  ; Sprite allocation mapping
sprite_priority_table:          .res 8  ; Sprite rendering priority

; Timing constants
target_fps:                     .byte 30 ; Target 30 FPS
