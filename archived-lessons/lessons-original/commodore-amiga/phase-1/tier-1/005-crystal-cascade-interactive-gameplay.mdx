# Lesson 5: Crystal Cascade - Interactive Gameplay

**Your parallax world needs engaging interactive gameplay!** In the last lesson, you added professional smooth scrolling to your multi-layered world. Today, you'll create interactive gameplay systems - collision detection, entity interactions, and engaging mechanics that transform your visual showcase into a compelling game experience!

## Your Interactive Gameplay Engages

Let's start with the magic - adding interactive gameplay systems to your scrolling parallax world:

```m68k
; Crystal Cascade - Interactive Gameplay Creation
    SECTION code,CODE

start:
    ; Set up enhanced parallax world from Lessons 1-4
    MOVE.L  #$00040000, A0      ; Background bitplane 0
    MOVE.L  #$00042000, A1      ; Middle layer bitplane 1
    MOVE.L  #$00044000, A2      ; Foreground bitplane 2
    
    ; Initialize scrolling system from Lesson 4
    JSR     InitScrollingSystem
    JSR     ConfigureScrollRegisters
    JSR     CreateScrollBuffers
    
    ; Initialize interactive gameplay system
    JSR     InitGameplaySystem
    
    ; Create interactive entities
    JSR     CreateGameplayEntities
    
    ; Set up collision detection
    JSR     InitCollisionSystem
    
    ; Initialize game objectives
    JSR     InitGameObjectives
    
    ; Professional interactive game loop
main_loop:
    JSR     ProcessPlayerInput      ; From Lesson 4
    JSR     UpdateScrollPosition    ; From Lesson 4
    JSR     UpdateGameplayEntities  ; New gameplay updates
    JSR     ProcessCollisionDetection ; New collision system
    JSR     UpdateGameObjectives    ; New objective tracking
    JSR     UpdateParallaxLayers    ; From Lesson 4
    JSR     UpdateInteractiveEffects ; New interactive effects
    BRA.S   main_loop

; Initialize interactive gameplay system
InitGameplaySystem:
    ; Set up game state variables
    MOVE.W  #0, GameScore       ; Player score
    MOVE.W  #100, PlayerHealth  ; Player health
    MOVE.W  #50, PlayerEnergy   ; Player energy
    MOVE.W  #0, CrystalsCollected ; Crystals found
    MOVE.W  #20, CrystalsTotal  ; Total crystals in level
    
    ; Initialize gameplay flags
    MOVE.B  #0, GamePaused      ; Game not paused
    MOVE.B  #0, LevelComplete   ; Level not complete
    MOVE.B  #0, GameOver        ; Game not over
    MOVE.B  #1, PlayerActive    ; Player is active
    
    ; Set up interaction parameters
    MOVE.W  #8, InteractionRange ; Interaction distance
    MOVE.W  #30, InvulnerabilityTime ; Damage immunity frames
    MOVE.W  #0, InvulnerabilityCounter ; Current immunity
    
    RTS

; Create interactive gameplay entities
CreateGameplayEntities:
    ; Enhance crystal entity from Lesson 2 with gameplay
    MOVE.L  #CrystalEntity, A3  ; Point to entity data
    MOVE.W  #160, ENTITY_X(A3)  ; Entity X position
    MOVE.W  #100, ENTITY_Y(A3)  ; Entity Y position
    MOVE.B  #1, ENTITY_ACTIVE(A3) ; Mark as active
    MOVE.B  #1, ENTITY_TYPE(A3) ; Type: Collectible
    
    ; Create gameplay collectibles
    JSR     CreateCollectibleCrystals
    
    ; Create interactive hazards
    JSR     CreateHazardElements
    
    ; Create power-up entities
    JSR     CreatePowerUpEntities
    
    ; Create interactive platforms
    JSR     CreateInteractivePlatforms
    
    RTS

; Create collectible crystal entities
CreateCollectibleCrystals:
    ; Create Crystal 1
    MOVE.L  #Crystal1Entity, A4 ; Point to crystal data
    MOVE.W  #80, ENTITY_X(A4)   ; Position X
    MOVE.W  #120, ENTITY_Y(A4)  ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #2, ENTITY_TYPE(A4) ; Type: Crystal
    MOVE.W  #10, ENTITY_VALUE(A4) ; Point value
    
    ; Create Crystal 2
    MOVE.L  #Crystal2Entity, A4 ; Point to crystal data
    MOVE.W  #240, ENTITY_X(A4)  ; Position X
    MOVE.W  #80, ENTITY_Y(A4)   ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #2, ENTITY_TYPE(A4) ; Type: Crystal
    MOVE.W  #15, ENTITY_VALUE(A4) ; Higher value
    
    ; Create Crystal 3
    MOVE.L  #Crystal3Entity, A4 ; Point to crystal data
    MOVE.W  #320, ENTITY_X(A4)  ; Position X
    MOVE.W  #140, ENTITY_Y(A4)  ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #2, ENTITY_TYPE(A4) ; Type: Crystal
    MOVE.W  #20, ENTITY_VALUE(A4) ; Highest value
    
    RTS

; Create hazard elements
CreateHazardElements:
    ; Create Energy Drain Hazard
    MOVE.L  #Hazard1Entity, A4  ; Point to hazard data
    MOVE.W  #200, ENTITY_X(A4)  ; Position X
    MOVE.W  #160, ENTITY_Y(A4)  ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #3, ENTITY_TYPE(A4) ; Type: Hazard
    MOVE.W  #-5, ENTITY_VALUE(A4) ; Energy drain amount
    
    ; Create Moving Hazard
    MOVE.L  #Hazard2Entity, A4  ; Point to hazard data
    MOVE.W  #300, ENTITY_X(A4)  ; Position X
    MOVE.W  #100, ENTITY_Y(A4)  ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #3, ENTITY_TYPE(A4) ; Type: Hazard
    MOVE.W  #-10, ENTITY_VALUE(A4) ; Health damage
    MOVE.W  #2, ENTITY_VELOCITY_X(A4) ; Movement speed
    
    RTS

; Create power-up entities
CreatePowerUpEntities:
    ; Create Health Power-up
    MOVE.L  #PowerUp1Entity, A4 ; Point to power-up data
    MOVE.W  #400, ENTITY_X(A4)  ; Position X
    MOVE.W  #90, ENTITY_Y(A4)   ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #4, ENTITY_TYPE(A4) ; Type: Power-up
    MOVE.W  #25, ENTITY_VALUE(A4) ; Health restore amount
    
    ; Create Energy Power-up
    MOVE.L  #PowerUp2Entity, A4 ; Point to power-up data
    MOVE.W  #500, ENTITY_X(A4)  ; Position X
    MOVE.W  #130, ENTITY_Y(A4)  ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #4, ENTITY_TYPE(A4) ; Type: Power-up
    MOVE.W  #30, ENTITY_VALUE(A4) ; Energy restore amount
    
    RTS

; Create interactive platforms
CreateInteractivePlatforms:
    ; Create Moving Platform 1
    MOVE.L  #Platform1Entity, A4 ; Point to platform data
    MOVE.W  #150, ENTITY_X(A4)   ; Position X
    MOVE.W  #180, ENTITY_Y(A4)   ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #5, ENTITY_TYPE(A4)  ; Type: Platform
    MOVE.W  #1, ENTITY_VELOCITY_X(A4) ; Horizontal movement
    MOVE.W  #0, ENTITY_VELOCITY_Y(A4) ; No vertical movement
    
    ; Create Elevator Platform
    MOVE.L  #Platform2Entity, A4 ; Point to platform data
    MOVE.W  #350, ENTITY_X(A4)   ; Position X
    MOVE.W  #200, ENTITY_Y(A4)   ; Position Y
    MOVE.B  #1, ENTITY_ACTIVE(A4) ; Active
    MOVE.B  #5, ENTITY_TYPE(A4)  ; Type: Platform
    MOVE.W  #0, ENTITY_VELOCITY_X(A4) ; No horizontal movement
    MOVE.W  #1, ENTITY_VELOCITY_Y(A4) ; Vertical movement
    
    RTS

; Initialize collision detection system
InitCollisionSystem:
    ; Set up collision detection parameters
    MOVE.W  #16, CollisionTolerance ; Collision boundary size
    MOVE.W  #0, CollisionCount   ; Number of collisions this frame
    
    ; Initialize collision response table
    MOVE.L  #CollisionResponseTable, A5
    MOVE.L  A5, CollisionTablePtr
    
    ; Set up entity interaction matrix
    JSR     SetupInteractionMatrix
    
    RTS

; Set up entity interaction matrix
SetupInteractionMatrix:
    ; Define which entity types can interact
    ; Matrix format: [Type1][Type2] = Interaction allowed (1=yes, 0=no)
    
    MOVE.L  #InteractionMatrix, A5
    
    ; Player (Type 1) interactions
    MOVE.B  #1, 0(A5,1*8+2)    ; Player can interact with Crystals (Type 2)
    MOVE.B  #1, 0(A5,1*8+3)    ; Player can interact with Hazards (Type 3)
    MOVE.B  #1, 0(A5,1*8+4)    ; Player can interact with Power-ups (Type 4)
    MOVE.B  #1, 0(A5,1*8+5)    ; Player can interact with Platforms (Type 5)
    
    ; Set symmetric interactions (if needed)
    MOVE.B  #1, 0(A5,2*8+1)    ; Crystals can interact with Player
    MOVE.B  #1, 0(A5,3*8+1)    ; Hazards can interact with Player
    MOVE.B  #1, 0(A5,4*8+1)    ; Power-ups can interact with Player
    MOVE.B  #1, 0(A5,5*8+1)    ; Platforms can interact with Player
    
    RTS

; Update all gameplay entities
UpdateGameplayEntities:
    ; Update crystal entity from Lesson 2
    JSR     UpdateCrystalEntity
    
    ; Update collectible crystals
    JSR     UpdateCollectibleCrystals
    
    ; Update hazard elements
    JSR     UpdateHazardElements
    
    ; Update power-up entities
    JSR     UpdatePowerUpEntities
    
    ; Update interactive platforms
    JSR     UpdateInteractivePlatforms
    
    RTS

; Update collectible crystals
UpdateCollectibleCrystals:
    ; Update Crystal 1
    MOVE.L  #Crystal1Entity, A4
    JSR     UpdateCrystalAnimation
    
    ; Update Crystal 2
    MOVE.L  #Crystal2Entity, A4
    JSR     UpdateCrystalAnimation
    
    ; Update Crystal 3
    MOVE.L  #Crystal3Entity, A4
    JSR     UpdateCrystalAnimation
    
    RTS

; Update crystal animation
UpdateCrystalAnimation:
    ; Check if crystal is active
    TST.B   ENTITY_ACTIVE(A4)
    BEQ.S   crystal_anim_done
    
    ; Animate crystal rotation
    MOVE.W  ENTITY_FRAME(A4), D0
    ADDQ.W  #1, D0
    AND.W   #$000F, D0          ; Wrap to 16 frames
    MOVE.W  D0, ENTITY_FRAME(A4)
    
    ; Apply floating motion
    MOVE.W  ENTITY_Y(A4), D1
    MOVE.W  D0, D2
    LSR.W   #2, D2              ; Slow down floating
    BTST    #2, D2              ; Test floating direction
    BEQ.S   float_up
    
    ADDQ.W  #1, D1              ; Float down
    BRA.S   store_float_y
    
float_up:
    SUBQ.W  #1, D1              ; Float up
    
store_float_y:
    MOVE.W  D1, ENTITY_Y(A4)
    
crystal_anim_done:
    RTS

; Update hazard elements
UpdateHazardElements:
    ; Update Hazard 1 (stationary energy drain)
    MOVE.L  #Hazard1Entity, A4
    JSR     UpdateEnergyDrain
    
    ; Update Hazard 2 (moving hazard)
    MOVE.L  #Hazard2Entity, A4
    JSR     UpdateMovingHazard
    
    RTS

; Update energy drain hazard
UpdateEnergyDrain:
    ; Check if hazard is active
    TST.B   ENTITY_ACTIVE(A4)
    BEQ.S   energy_drain_done
    
    ; Animate energy drain effect
    MOVE.W  ENTITY_FRAME(A4), D0
    ADDQ.W  #2, D0              ; Faster animation
    AND.W   #$001F, D0          ; Wrap to 32 frames
    MOVE.W  D0, ENTITY_FRAME(A4)
    
energy_drain_done:
    RTS

; Update moving hazard
UpdateMovingHazard:
    ; Check if hazard is active
    TST.B   ENTITY_ACTIVE(A4)
    BEQ.S   moving_hazard_done
    
    ; Update position based on velocity
    MOVE.W  ENTITY_X(A4), D0
    ADD.W   ENTITY_VELOCITY_X(A4), D0
    
    ; Check boundaries and reverse if needed
    CMP.W   #50, D0
    BLT.S   reverse_horizontal
    CMP.W   #600, D0
    BGT.S   reverse_horizontal
    BRA.S   store_hazard_x
    
reverse_horizontal:
    NEG.W   ENTITY_VELOCITY_X(A4)  ; Reverse direction
    
store_hazard_x:
    MOVE.W  D0, ENTITY_X(A4)
    
    ; Animate hazard
    MOVE.W  ENTITY_FRAME(A4), D0
    ADDQ.W  #1, D0
    AND.W   #$0007, D0          ; Wrap to 8 frames
    MOVE.W  D0, ENTITY_FRAME(A4)
    
moving_hazard_done:
    RTS

; Update power-up entities
UpdatePowerUpEntities:
    ; Update Health Power-up
    MOVE.L  #PowerUp1Entity, A4
    JSR     UpdatePowerUpGlow
    
    ; Update Energy Power-up
    MOVE.L  #PowerUp2Entity, A4
    JSR     UpdatePowerUpGlow
    
    RTS

; Update power-up glow effect
UpdatePowerUpGlow:
    ; Check if power-up is active
    TST.B   ENTITY_ACTIVE(A4)
    BEQ.S   powerup_glow_done
    
    ; Animate glowing effect
    MOVE.W  ENTITY_FRAME(A4), D0
    ADDQ.W  #1, D0
    AND.W   #$003F, D0          ; Wrap to 64 frames for smooth glow
    MOVE.W  D0, ENTITY_FRAME(A4)
    
    ; Apply subtle bobbing motion
    MOVE.W  ENTITY_Y(A4), D1
    MOVE.W  D0, D2
    LSR.W   #3, D2              ; Slow bobbing
    BTST    #1, D2              ; Test bobbing direction
    BEQ.S   bob_up
    
    ADDQ.W  #1, D1              ; Bob down
    BRA.S   store_bob_y
    
bob_up:
    SUBQ.W  #1, D1              ; Bob up
    
store_bob_y:
    MOVE.W  D1, ENTITY_Y(A4)
    
powerup_glow_done:
    RTS

; Update interactive platforms
UpdateInteractivePlatforms:
    ; Update Moving Platform 1
    MOVE.L  #Platform1Entity, A4
    JSR     UpdateMovingPlatform
    
    ; Update Elevator Platform
    MOVE.L  #Platform2Entity, A4
    JSR     UpdateElevatorPlatform
    
    RTS

; Update moving platform
UpdateMovingPlatform:
    ; Check if platform is active
    TST.B   ENTITY_ACTIVE(A4)
    BEQ.S   moving_platform_done
    
    ; Update horizontal position
    MOVE.W  ENTITY_X(A4), D0
    ADD.W   ENTITY_VELOCITY_X(A4), D0
    
    ; Check platform boundaries
    CMP.W   #100, D0
    BLT.S   reverse_platform_h
    CMP.W   #300, D0
    BGT.S   reverse_platform_h
    BRA.S   store_platform_x
    
reverse_platform_h:
    NEG.W   ENTITY_VELOCITY_X(A4)  ; Reverse direction
    
store_platform_x:
    MOVE.W  D0, ENTITY_X(A4)
    
moving_platform_done:
    RTS

; Update elevator platform
UpdateElevatorPlatform:
    ; Check if platform is active
    TST.B   ENTITY_ACTIVE(A4)
    BEQ.S   elevator_platform_done
    
    ; Update vertical position
    MOVE.W  ENTITY_Y(A4), D0
    ADD.W   ENTITY_VELOCITY_Y(A4), D0
    
    ; Check elevator boundaries
    CMP.W   #80, D0
    BLT.S   reverse_platform_v
    CMP.W   #220, D0
    BGT.S   reverse_platform_v
    BRA.S   store_platform_y
    
reverse_platform_v:
    NEG.W   ENTITY_VELOCITY_Y(A4)  ; Reverse direction
    
store_platform_y:
    MOVE.W  D0, ENTITY_Y(A4)
    
elevator_platform_done:
    RTS

; Process collision detection
ProcessCollisionDetection:
    ; Check player collisions with all entities
    JSR     CheckPlayerCrystalCollisions
    JSR     CheckPlayerHazardCollisions
    JSR     CheckPlayerPowerUpCollisions
    JSR     CheckPlayerPlatformCollisions
    
    ; Update invulnerability timer
    JSR     UpdateInvulnerability
    
    RTS

; Check player collisions with crystals
CheckPlayerCrystalCollisions:
    ; Check collision with Crystal 1
    MOVE.L  #CrystalEntity, A3   ; Player entity
    MOVE.L  #Crystal1Entity, A4  ; Crystal entity
    JSR     CheckEntityCollision
    TST.W   D7                   ; Check collision result
    BEQ.S   check_crystal_2
    JSR     HandleCrystalCollection
    
check_crystal_2:
    ; Check collision with Crystal 2
    MOVE.L  #Crystal2Entity, A4
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   check_crystal_3
    JSR     HandleCrystalCollection
    
check_crystal_3:
    ; Check collision with Crystal 3
    MOVE.L  #Crystal3Entity, A4
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   crystal_collisions_done
    JSR     HandleCrystalCollection
    
crystal_collisions_done:
    RTS

; Check entity collision between A3 and A4
CheckEntityCollision:
    ; Load entity positions
    MOVE.W  ENTITY_X(A3), D0     ; Player X
    MOVE.W  ENTITY_Y(A3), D1     ; Player Y
    MOVE.W  ENTITY_X(A4), D2     ; Other entity X
    MOVE.W  ENTITY_Y(A4), D3     ; Other entity Y
    
    ; Check if other entity is active
    TST.B   ENTITY_ACTIVE(A4)
    BEQ.S   no_collision
    
    ; Calculate distance
    SUB.W   D0, D2               ; Delta X
    BPL.S   positive_dx
    NEG.W   D2                   ; Make positive
positive_dx:
    
    SUB.W   D1, D3               ; Delta Y
    BPL.S   positive_dy
    NEG.W   D3                   ; Make positive
positive_dy:
    
    ; Check collision boundaries
    CMP.W   CollisionTolerance, D2
    BGT.S   no_collision
    CMP.W   CollisionTolerance, D3
    BGT.S   no_collision
    
    ; Collision detected
    MOVE.W  #1, D7
    RTS
    
no_collision:
    MOVE.W  #0, D7
    RTS

; Handle crystal collection
HandleCrystalCollection:
    ; Deactivate collected crystal
    MOVE.B  #0, ENTITY_ACTIVE(A4)
    
    ; Add to score
    MOVE.W  GameScore, D0
    ADD.W   ENTITY_VALUE(A4), D0
    MOVE.W  D0, GameScore
    
    ; Increment crystals collected
    ADDQ.W  #1, CrystalsCollected
    
    ; Restore some energy
    MOVE.W  PlayerEnergy, D0
    ADDQ.W  #5, D0
    CMP.W   #100, D0
    BLE.S   store_energy
    MOVE.W  #100, D0
store_energy:
    MOVE.W  D0, PlayerEnergy
    
    ; Play collection effect
    JSR     PlayCrystalSound
    JSR     CreateCollectionEffect
    
    RTS

; Check player collisions with hazards
CheckPlayerHazardCollisions:
    ; Skip if player is invulnerable
    TST.W   InvulnerabilityCounter
    BNE.S   hazard_collisions_done
    
    ; Check collision with Hazard 1
    MOVE.L  #CrystalEntity, A3   ; Player entity
    MOVE.L  #Hazard1Entity, A4   ; Hazard entity
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   check_hazard_2
    JSR     HandleHazardDamage
    
check_hazard_2:
    ; Check collision with Hazard 2
    MOVE.L  #Hazard2Entity, A4
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   hazard_collisions_done
    JSR     HandleHazardDamage
    
hazard_collisions_done:
    RTS

; Handle hazard damage
HandleHazardDamage:
    ; Apply damage based on hazard type
    MOVE.W  ENTITY_VALUE(A4), D0  ; Damage amount (negative)
    
    ; Apply to health or energy based on hazard
    CMP.W   #-10, D0
    BLT.S   health_damage
    
energy_damage:
    ; Energy damage
    MOVE.W  PlayerEnergy, D1
    ADD.W   D0, D1               ; Add negative damage
    BPL.S   store_energy_damage
    MOVE.W  #0, D1               ; Clamp to zero
store_energy_damage:
    MOVE.W  D1, PlayerEnergy
    BRA.S   set_invulnerability
    
health_damage:
    ; Health damage
    MOVE.W  PlayerHealth, D1
    ADD.W   D0, D1               ; Add negative damage
    BPL.S   store_health_damage
    MOVE.W  #0, D1               ; Clamp to zero
store_health_damage:
    MOVE.W  D1, PlayerHealth
    
set_invulnerability:
    ; Set invulnerability period
    MOVE.W  InvulnerabilityTime, InvulnerabilityCounter
    
    ; Play damage effect
    JSR     PlayDamageSound
    JSR     CreateDamageEffect
    
    RTS

; Check player collisions with power-ups
CheckPlayerPowerUpCollisions:
    ; Check collision with Power-up 1
    MOVE.L  #CrystalEntity, A3   ; Player entity
    MOVE.L  #PowerUp1Entity, A4  ; Power-up entity
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   check_powerup_2
    JSR     HandlePowerUpCollection
    
check_powerup_2:
    ; Check collision with Power-up 2
    MOVE.L  #PowerUp2Entity, A4
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   powerup_collisions_done
    JSR     HandlePowerUpCollection
    
powerup_collisions_done:
    RTS

; Handle power-up collection
HandlePowerUpCollection:
    ; Deactivate collected power-up
    MOVE.B  #0, ENTITY_ACTIVE(A4)
    
    ; Apply power-up effect
    MOVE.W  ENTITY_VALUE(A4), D0  ; Restoration amount
    
    ; Determine if health or energy power-up
    CMP.W  #30, D0
    BGT.S  health_powerup
    
energy_powerup:
    ; Energy power-up
    MOVE.W  PlayerEnergy, D1
    ADD.W   D0, D1
    CMP.W   #100, D1
    BLE.S   store_energy_powerup
    MOVE.W  #100, D1
store_energy_powerup:
    MOVE.W  D1, PlayerEnergy
    BRA.S   powerup_effect_done
    
health_powerup:
    ; Health power-up
    MOVE.W  PlayerHealth, D1
    ADD.W   D0, D1
    CMP.W   #100, D1
    BLE.S   store_health_powerup
    MOVE.W  #100, D1
store_health_powerup:
    MOVE.W  D1, PlayerHealth
    
powerup_effect_done:
    ; Play power-up effect
    JSR     PlayPowerUpSound
    JSR     CreatePowerUpEffect
    
    RTS

; Check player collisions with platforms
CheckPlayerPlatformCollisions:
    ; Check collision with Platform 1
    MOVE.L  #CrystalEntity, A3   ; Player entity
    MOVE.L  #Platform1Entity, A4 ; Platform entity
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   check_platform_2
    JSR     HandlePlatformInteraction
    
check_platform_2:
    ; Check collision with Platform 2
    MOVE.L  #Platform2Entity, A4
    JSR     CheckEntityCollision
    TST.W   D7
    BEQ.S   platform_collisions_done
    JSR     HandlePlatformInteraction
    
platform_collisions_done:
    RTS

; Handle platform interaction
HandlePlatformInteraction:
    ; Move player with platform
    MOVE.W  ENTITY_VELOCITY_X(A4), D0
    MOVE.W  ENTITY_VELOCITY_Y(A4), D1
    
    ; Apply platform movement to player
    MOVE.W  ENTITY_X(A3), D2
    ADD.W   D0, D2
    MOVE.W  D2, ENTITY_X(A3)
    
    MOVE.W  ENTITY_Y(A3), D2
    ADD.W   D1, D2
    MOVE.W  D2, ENTITY_Y(A3)
    
    RTS

; Update invulnerability system
UpdateInvulnerability:
    ; Decrement invulnerability counter
    TST.W   InvulnerabilityCounter
```

```m68k
    MOVE.L  #CrystalEntity, A3      ; Player entity

    MOVE.W  EntityCount, D6         ; Load entity count
    
interaction_loop:
    MOVE.L  (A5)+, A4               ; Load next entity pointer
    JSR     CheckEntityCollision    ; Generic collision check
    TST.W   D7                      ; Check collision result
    BEQ.S   next_interaction        ; No collision
    
    ; Collision detected - call appropriate handler
    MOVE.B  ENTITY_TYPE(A4), D0     ; Load entity type
    LSL.W   #2, D0                  ; Multiply by 4 for long pointer
    LEA     InteractionCallbacks, A6 ; Point to callback table
    MOVE.L  0(A6,D0.W), A6          ; Load callback address
    JSR     (A6)                    ; Call interaction handler
    
next_interaction:
    SUBQ.W  #1, D6                  ; Decrement entity count
    BNE.S   interaction_loop        ; Continue until all processed
    
    RTS

    ; Enhanced crystal collection with combo system

    BLT.S   normal_crystal
```
