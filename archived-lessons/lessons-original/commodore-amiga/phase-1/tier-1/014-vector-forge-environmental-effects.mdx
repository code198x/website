# Lesson 14: Vector Forge - Environmental Effects

**Your vector world needs atmospheric effects!** In your advanced entity behaviors from Lesson 13, you created intelligent vector-based AI opponents. Today, you'll add dynamic environmental effects with particle systems, atmospheric hazards, and immersive visual effects that transform your static vector elements into a living, breathing forge world. You'll master advanced graphics programming through authentic Amiga techniques that create engaging environmental storytelling!

        ; Random spawn position across screen width
        bsr     generate_random_number
        and.l   #$1ff,d1        ; Limit to screen width (512 pixels)
        lea     particle_x_position_array,a0
        lsl.l   #1,d0           ; Convert to word offset
        move.w  d1,(a0,d0.l)    ; Store X position

        ; Set Y position at top of screen
        lea     particle_y_position_array,a0
        move.w  #0,(a0,d0.l)    ; Top of screen

        lea     particle_x_velocity_array,a0
        move.w  #0,(a0,d0.l)    ; No horizontal movement

        lea     particle_y_velocity_array,a0
        move.w  #$40,(a0,d0.l)  ; Slow downward drift (fixed point)

        lea     particle_life_timer_array,a0

spawn_manipulation_particles_loop:

        lea     particle_type_array,a1
        move.b  #2,(a1,d0.l)    ; Manipulation particle type

        move.l  vector_manipulation_x,d1

        move.l  d0,d4
        lsl.l   #1,d4           ; Convert to word offset

; Update particle physics for realistic movement
update_particle_physics:
        ; d3 contains particle index
        ; Update X position
        lea     particle_x_position_array,a0
        move.l  d3,d0
        lsl.l   #1,d0           ; Convert to word offset
        move.w  (a0,d0.l),d1    ; Load X position

        lea     particle_x_velocity_array,a0
        add.w   (a0,d0.l),d1    ; Add velocity

        ; Check X boundaries
        cmp.w   #512,d1         ; Right boundary
        blt.s   check_particle_y_physics
        cmp.w   #-256,d1        ; Left boundary (allow negative for wrapping)
        bgt.s   check_particle_y_physics

        ; Outside boundaries - mark for removal
        lea     particle_life_timer_array,a0
        move.w  #1,(a0,d0.l)    ; 1 frame life (will expire next update)

check_particle_y_physics:
        ; Store X position
        lea     particle_x_position_array,a0
        move.w  d1,(a0,d0.l)

        ; Update Y position
        lea     particle_y_position_array,a0
        move.w  (a0,d0.l),d1    ; Load Y position

        lea     particle_y_velocity_array,a0
        add.w   (a0,d0.l),d1    ; Add velocity

        ; Check Y boundaries
        cmp.w   #256,d1         ; Bottom boundary
        blt.s   store_particle_y_position

        ; Below screen - mark for removal
        lea     particle_life_timer_array,a0
        move.w  #1,(a0,d0.l)    ; 1 frame life (will expire next update)

        ; Store Y position
        lea     particle_y_position_array,a0
        move.w  d1,(a0,d0.l)

        ; Apply environmental forces based on particle type
        lea     particle_type_array,a0
        move.l  d3,d0
        move.b  (a0,d0.l),d1    ; Load particle type

        cmp.b   #1,d1           ; Atmospheric particle
        beq.s   apply_atmospheric_forces
        cmp.b   #2,d1           ; Manipulation particle
        beq.s   apply_manipulation_forces

        rts

apply_atmospheric_forces:
        ; Apply atmospheric drift and gravity
        lea     particle_y_velocity_array,a0
        move.l  d3,d0
        lsl.l   #1,d0           ; Convert to word offset
        move.w  (a0,d0.l),d1
        cmp.w   #$80,d1         ; Maximum fall speed (fixed point)
        bge.s   atmospheric_forces_complete
        addq.w  #2,d1           ; Slight gravity acceleration
        move.w  d1,(a0,d0.l)

atmospheric_forces_complete:
        rts

apply_manipulation_forces:
        ; Apply manipulation particle deceleration
        lea     particle_x_velocity_array,a0
        move.l  d3,d0
        lsl.l   #1,d0           ; Convert to word offset
        move.w  (a0,d0.l),d1
        beq.s   check_manipulation_y_decel
        bpl.s   manipulation_x_positive

        ; Negative X velocity - decelerate
        addq.w  #8,d1           ; Deceleration amount
        bpl.s   manipulation_x_zero
        move.w  d1,(a0,d0.l)
        bra.s   check_manipulation_y_decel
manipulation_x_zero:
        clr.w   (a0,d0.l)
        bra.s   check_manipulation_y_decel

manipulation_x_positive:
        ; Positive X velocity - decelerate
        subq.w  #8,d1           ; Deceleration amount
        bpl.s   manipulation_x_store
        clr.w   d1
manipulation_x_store:
        move.w  d1,(a0,d0.l)

check_manipulation_y_decel:
        lea     particle_y_velocity_array,a0
        move.w  (a0,d0.l),d1
        beq.s   manipulation_forces_complete
        bpl.s   manipulation_y_positive

        ; Negative Y velocity - decelerate (add gravity)
        addq.w  #8,d1           ; Deceleration amount
        bpl.s   manipulation_y_zero
        move.w  d1,(a0,d0.l)
        bra.s   manipulation_forces_complete
manipulation_y_zero:
        clr.w   (a0,d0.l)
        bra.s   manipulation_forces_complete

manipulation_y_positive:
        ; Positive Y velocity - decelerate
        subq.w  #8,d1           ; Deceleration amount
        bpl.s   manipulation_y_store
        clr.w   d1
manipulation_y_store:
        move.w  d1,(a0,d0.l)

; Update atmospheric color cycling for dynamic environment
update_atmospheric_color_cycling:
        ; Increment color cycle counter
        addq.l  #1,color_cycle_counter
        move.l  color_cycle_counter,d0
        move.l  color_cycle_speed,d1
        cmp.l   d1,d0
        blt.s   color_cycling_complete

        ; Reset counter and cycle color
        clr.l   color_cycle_counter

        ; Cycle through atmospheric colors
        move.l  current_atmospheric_color,d0
        addq.l  #$111,d0        ; Increment RGB components
        cmp.l   #$fff,d0        ; Check for overflow
