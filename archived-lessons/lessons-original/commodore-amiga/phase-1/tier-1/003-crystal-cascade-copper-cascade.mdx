```nasm
; Crystal Cascade - Copper Cascade Creation
    SECTION code,CODE

start:
    ; Set up multi-layered parallax world from Lessons 1-2
    MOVE.L  #$00040000, A0      ; Background bitplane 0
    MOVE.W  #$0123, D0          ; Mountain pattern
    MOVE.W  D0, (A0)            ; Draw mountains

    MOVE.L  #$00042000, A1      ; Middle bitplane 1
    MOVE.W  #$4567, D1          ; Crystal formation pattern
    MOVE.W  D1, 20(A1)          ; Offset crystals

    ; Initialize floating entities from Lesson 2
    MOVE.L  #CrystalEntity, A3  ; Point to entity data
    MOVE.W  #160, ENTITY_X(A3)  ; Entity X position
    MOVE.W  #100, ENTITY_Y(A3)  ; Entity Y position

    ; Initialize copper cascade system
    JSR     InitCopperCascade

    ; Create dynamic color cycling system
    JSR     CreateColorCycles

    ; Set up copper list for automated effects
    JSR     ActivateCopperList

    ; Professional game loop with copper effects
loop:
    JSR     UpdateCrystalEntity     ; Update entity from Lesson 2
    JSR     UpdateColorCycles       ; Update dynamic colors
    JSR     SyncCopperEffects       ; Synchronize copper timing
    BRA.S   loop

; Initialize copper cascade effects system
InitCopperCascade:
    ; Set up copper list in chip memory
    MOVE.L  #CopperList, A0     ; Point to copper list (absolute addressing)
    MOVE.L  A0, $DFF080         ; Set copper list pointer (absolute addressing)

    ; Initialize color registers with base palette
    MOVE.L  #$00DFF180, A1      ; Color register base (absolute addressing)
    MOVE.W  #$000F, (A1)        ; Color 0: Deep space blue (register indirect)
    MOVE.W  #$0F4F, 2(A1)       ; Color 1: Crystal cyan (register indirect with offset)
    MOVE.W  #$0F8F, 4(A1)       ; Color 2: Bright crystal (register indirect with offset)
    MOVE.W  #$0FFF, 6(A1)       ; Color 3: Pure white highlights (register indirect with offset)

    ; Set up gradient base for copper effects
    MOVE.L  #ColorGradientData, A2 ; Point to gradient data (absolute addressing)
    MOVE.L  A2, GradientPointer    ; Store pointer for updates (absolute addressing)

    RTS

; Create dynamic color cycling system
CreateColorCycles:
    ; Initialize cycle counters using different addressing modes
    MOVE.B  #$00, CycleCounter     ; Start cycle at 0 (immediate to absolute)
    MOVE.W  #$0001, CycleDirection ; Cycle direction (immediate to absolute)
    MOVE.L  #$00000040, CycleSpeed ; Cycle timing (immediate to absolute)

    ; Set up color interpolation system
    MOVE.L  #$00DFF180, A4      ; Color register base (absolute addressing)

    ; Create color cycle pattern for crystal effects
    MOVE.W  #$001F, D0          ; Start color: Dark blue (immediate addressing)
    MOVE.W  #$0F8F, D1          ; End color: Bright cyan (immediate addressing)
    MOVE.L  #32, D2             ; Cycle steps (immediate addressing)

    ; Store cycle parameters
    MOVE.W  D0, CycleStart      ; Store start color (absolute addressing)
    MOVE.W  D1, CycleEnd        ; Store end color (absolute addressing)
    MOVE.L  D2, CycleSteps      ; Store step count (absolute addressing)

    RTS

; Update dynamic color cycling effects
UpdateColorCycles:
    ; Load cycle parameters using various addressing modes
    MOVE.B  CycleCounter, D0    ; Load current position (absolute addressing)
    MOVE.W  CycleDirection, D1  ; Load direction (absolute addressing)
    MOVE.L  CycleSpeed, D2      ; Load timing (absolute addressing)

    ; Update cycle position
    ADD.B   D1, D0              ; Advance cycle position
    CMP.B   #32, D0             ; Check upper bound (immediate addressing)
    BLE.S   check_lower_bound   ; Within bounds
    MOVE.W  #-1, CycleDirection ; Reverse direction (immediate to absolute)
    MOVE.B  #31, D0             ; Clamp to maximum (immediate addressing)

check_lower_bound:
    TST.B   D0                  ; Check if below zero
    BGE.S   store_position      ; Positive, continue
    MOVE.W  #1, CycleDirection  ; Forward direction (immediate to absolute)
    MOVE.B  #0, D0              ; Clamp to minimum (immediate addressing)

store_position:
    MOVE.B  D0, CycleCounter    ; Store updated position (absolute addressing)

    ; Calculate interpolated color
    JSR     InterpolateColor

    ; Apply to multiple color registers for gradient effect
    MOVE.L  #$00DFF180, A1      ; Color register base (absolute addressing)
    MOVE.W  D3, 2(A1)           ; Set color 1 (register indirect with displacement)

    ; Create shifted variants for depth
    MOVE.W  D3, D4              ; Copy color (register to register)
    LSR.W   #1, D4              ; Darken for background (immediate addressing)
    AND.W   #$0777, D4          ; Mask to valid color range (immediate addressing)
    MOVE.W  D4, 4(A1)           ; Set color 2 (register indirect with displacement)

    RTS

; Interpolate color between cycle start and end
InterpolateColor:
    ; Load cycle parameters
    MOVE.W  CycleStart, D4      ; Load start color (absolute addressing)
    MOVE.W  CycleEnd, D5        ; Load end color (absolute addressing)
    MOVE.B  CycleCounter, D0    ; Load position (absolute addressing)

    ; Extract RGB components from start color
    MOVE.W  D4, D6              ; Copy start color
    AND.W   #$0F00, D6          ; Extract red component (immediate addressing)
    LSR.W   #8, D6              ; Shift to low bits

    MOVE.W  D4, D7              ; Copy start color
    AND.W   #$00F0, D7          ; Extract green component (immediate addressing)
    LSR.W   #4, D7              ; Shift to low bits

    MOVE.W  D4, D0              ; Copy start color
    AND.W   #$000F, D0          ; Extract blue component (immediate addressing)

    ; Extract RGB components from end color
    MOVE.W  D5, A5              ; Copy end color to address register
    MOVE.L  A5, D1              ; Move to data register for manipulation
    AND.W   #$0F00, D1          ; Extract red component (immediate addressing)
    LSR.W   #8, D1              ; Shift to low bits

    ; Simplified interpolation for demonstration
    ADD.W   D1, D6              ; Combine red components
    LSR.W   #1, D6              ; Average (immediate addressing)
    LSL.W   #8, D6              ; Shift back to red position

    ; Combine interpolated color
    MOVE.W  D6, D3              ; Start with red component
    OR.W    #$0088, D3          ; Add fixed green/blue (immediate addressing)

    RTS

; Activate copper list for automated effects
ActivateCopperList:
    ; Enable copper coprocessor
    MOVE.W  #$8080, $DFF096     ; Enable copper DMA (immediate to absolute)

    ; Start copper list execution
    MOVE.L  #CopperList, $DFF080 ; Set copper list pointer (absolute addressing)

    RTS

; Synchronize copper effects with frame timing
SyncCopperEffects:
    ; Wait for vertical blank for smooth animation
    MOVE.L  #$00DFF004, A0      ; VPOSR register (absolute addressing)

wait_vblank:
    MOVE.W  (A0), D0            ; Read vertical position (register indirect)
    AND.W   #$0100, D0          ; Check vertical blank bit (immediate addressing)
    BEQ.S   wait_vblank         ; Wait until VBlank

    ; Update copper color registers during VBlank
    MOVE.L  #$00DFF180, A1      ; Color register base (absolute addressing)
    MOVE.B  CycleCounter, D0    ; Load cycle position (absolute addressing)

    ; Create dynamic background gradient
    LSL.W   #2, D0              ; Multiply by 4 for color index
    MOVE.L  #ColorGradientData, A2 ; Point to gradient table (absolute addressing)
    MOVE.W  0(A2,D0.W), 8(A1)   ; Set background color (indexed with offset)

    RTS

; Enhanced entity placement with copper synchronization
PlaceCrystalEntity:
    ; Load entity position (from Lesson 2, enhanced)
    MOVE.L  #CrystalEntity, A5  ; Entity data pointer (absolute addressing)
    MOVE.W  ENTITY_X(A5), D3    ; Load X coordinate (register indirect with displacement)
    MOVE.W  ENTITY_Y(A5), D4    ; Load Y coordinate (register indirect with displacement)

    ; Calculate precise screen address using multiple addressing modes
    MOVE.L  #$00046000, A6      ; Entity bitplane address (absolute addressing)

    ; Y offset calculation using efficient addressing
    MOVE.W  D4, D5              ; Copy Y coordinate (register to register)
    MULU.W  #40, D5             ; Multiply by bytes per row (immediate addressing)
    ADD.L   D5, A6              ; Add Y offset (register indirect with index)

    ; X offset calculation
    MOVE.W  D3, D6              ; Copy X coordinate (register to register)
    LSR.W   #3, D6              ; Divide by 8 pixels per byte (immediate addressing)
    ADD.W   D6, A6              ; Add X offset (register indirect with displacement)

    ; Place entity with color cycling effect
    MOVE.B  CycleCounter, D0    ; Load current cycle (absolute addressing)
    AND.B   #$07, D0            ; Mask to 8 patterns (immediate addressing)
    MOVE.L  #EntityPatterns, A0 ; Point to pattern table (absolute addressing)
    MOVE.B  0(A0,D0.W), (A6)    ; Place animated pattern (indexed addressing)

    RTS

; Copper list for automated visual effects
CopperList:
    ; Wait for specific scanlines and change colors
    DC.W $2C01, $FFFE          ; Wait for line 44
    DC.W $0180, $001F          ; Set color 0 to dark blue
    DC.W $3001, $FFFE          ; Wait for line 48
    DC.W $0180, $002F          ; Set color 0 to medium blue
    DC.W $3401, $FFFE          ; Wait for line 52
    DC.W $0180, $004F          ; Set color 0 to bright blue
    DC.W $FFFF, $FFFE          ; End copper list

; Professional entity data structure (from Lesson 2)
CrystalEntity:
ENTITY_X        DC.W    160     ; X position (16-bit precision)
ENTITY_Y        DC.W    100     ; Y position (16-bit precision)
ENTITY_VELOCITY_X DC.B  1       ; X velocity (8-bit efficiency)
ENTITY_VELOCITY_Y DC.B  2       ; Y velocity (8-bit efficiency)
ENTITY_ID       DC.L    $12345678 ; Unique entity ID (32-bit)
ENTITY_COLOR    DC.W    $0F8F   ; Entity color data (16-bit)

; Color cycling variables
CycleCounter:    DC.B    0      ; Current cycle position
CycleDirection:  DC.W    1      ; Cycle direction (1 or -1)
CycleSpeed:      DC.L    64     ; Cycle timing
CycleStart:      DC.W    $001F  ; Start color
CycleEnd:        DC.W    $0F8F  ; End color
CycleSteps:      DC.L    32     ; Number of interpolation steps
GradientPointer: DC.L    0      ; Pointer to gradient data

; Color gradient data for copper effects
ColorGradientData:
    DC.W $000F, $001F, $002F, $003F, $004F, $005F, $006F, $007F
    DC.W $008F, $009F, $00AF, $00BF, $00CF, $00DF, $00EF, $00FF
    DC.W $01FF, $02FF, $03FF, $04FF, $05FF, $06FF, $07FF, $08FF
    DC.W $09FF, $0AFF, $0BFF, $0CFF, $0DFF, $0EFF, $0FFF, $0FFF

; Entity animation patterns
EntityPatterns:
    DC.B $2A, $2B, $2C, $2D, $2E, $2F, $30, $31 ; 8 animation frames
```

### 68000 Addressing Modes: Your Professional Graphics Toolkit

The most important thing to understand: **68000 addressing modes enable efficient custom chip programming**. In your copper cascade code:

**Immediate Addressing (#value)**: Loading graphics constants

```nasm
MOVE.W  #$0F8F, D1          ; Load color value (immediate addressing)
MOVE.L  #$00DFF180, A1      ; Load color register address (immediate addressing)
MOVE.W  #$8080, $DFF096     ; Enable copper DMA (immediate addressing)
```

**Absolute Addressing (address)**: Accessing custom chip registers

```nasm
MOVE.W  D0, CycleCounter    ; Store to cycle variable (absolute addressing)
MOVE.L  A0, $DFF080         ; Set copper list pointer (absolute addressing)
MOVE.W  (A0), D0            ; Read from VPOSR register (absolute addressing)
```

**Register Indirect with Displacement (offset(An))**: Efficient array and structure access

```nasm
MOVE.W  D0, 2(A1)           ; Set color register 1 (register indirect with displacement)
MOVE.W  ENTITY_X(A5), D3    ; Load entity X position (register indirect with displacement)
MOVE.B  0(A0,D0.W), (A6)    ; Indexed pattern access (register indirect with index)
```

**68000 Addressing Mode Examples in Professional Graphics Control:**

| Mode                  | Example              | Use Case                                            |
| --------------------- | -------------------- | --------------------------------------------------- |
| **Immediate**         | `MOVE.W #$0F8F, D1`  | Load color values and custom chip constants         |
| **Absolute**          | `MOVE.L A0, $DFF080` | Control custom chip registers and hardware          |
| **Register Indirect** | `MOVE.W D0, 2(A1)`   | Efficient color register and structure manipulation |

### Quick Reference: Today's Professional Graphics Instructions
