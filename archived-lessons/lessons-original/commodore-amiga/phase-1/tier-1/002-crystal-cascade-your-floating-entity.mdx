```nasm
; Crystal Cascade - Floating Entity Creation
    SECTION code,CODE

start:
    ; Set up multi-layered parallax world from Lesson 1
    MOVE.L  #$00040000, A0      ; Background bitplane 0
    MOVE.W  #$0123, D0          ; Mountain pattern
    MOVE.W  D0, (A0)            ; Draw mountains
    MOVE.W  D0, 40(A0)          ; Continue pattern

    MOVE.L  #$00042000, A1      ; Middle bitplane 1
    MOVE.W  #$4567, D1          ; Crystal formation pattern
    MOVE.W  D1, 20(A1)          ; Offset crystals

    MOVE.L  #$00044000, A2      ; Foreground bitplane 2
    MOVE.W  #$FEDC, D2          ; Metal structure pattern
    MOVE.W  D2, 10(A2)          ; Foreground elements

    ; Create floating crystal entity with complex data
    MOVE.L  #CrystalEntity, A3  ; Point to entity data structure

    ; Initialize entity attributes using different data sizes
    MOVE.W  #160, ENTITY_X(A3)      ; X position (16-bit precision)
    MOVE.W  #100, ENTITY_Y(A3)      ; Y position (16-bit precision)
    MOVE.B  #1, ENTITY_VELOCITY_X(A3) ; X velocity (8-bit speed)
    MOVE.B  #2, ENTITY_VELOCITY_Y(A3) ; Y velocity (8-bit speed)
    MOVE.L  #$12345678, ENTITY_ID(A3) ; Unique ID (32-bit)
    MOVE.W  #$0F8F, ENTITY_COLOR(A3) ; Entity color (16-bit RGB)

    ; Calculate entity screen position using 68000 precision
    JSR     PlaceCrystalEntity

    ; Set stunning parallax colors
    MOVE.L  #$00DFF180, A4      ; Color palette
    MOVE.W  #$000F, (A4)        ; Deep blue mountains
    MOVE.W  #$0F0F, 2(A4)       ; Bright cyan crystals
    MOVE.W  #$FF0F, 4(A4)       ; Metallic gold structures
    MOVE.W  #$0FFF, 6(A4)       ; Bright white entity

    ; Professional game loop
loop:
    JSR     UpdateCrystalEntity  ; Update entity position
    JSR     PlaceCrystalEntity   ; Redraw entity
    BRA.S   loop

; Professional entity placement using 68000 precision
PlaceCrystalEntity:
    ; Load entity position with professional precision
    MOVE.L  #CrystalEntity, A5  ; Entity data pointer
    MOVE.W  ENTITY_X(A5), D3    ; Load X coordinate (16-bit)
    MOVE.W  ENTITY_Y(A5), D4    ; Load Y coordinate (16-bit)

    ; Calculate precise screen address
    MOVE.L  #$00046000, A6      ; Entity bitplane (layer 3)

    ; Calculate Y offset (320 bytes per row in high-res)
    MOVE.W  D4, D5              ; Copy Y coordinate
    MULU.W  #40, D5             ; Multiply by bytes per row
    ADD.L   D5, A6              ; Add Y offset

    ; Calculate X offset (byte position)
    MOVE.W  D3, D6              ; Copy X coordinate
    LSR.W   #3, D6              ; Divide by 8 (pixels per byte)
    ADD.W   D6, A6              ; Add X offset

    ; Place crystal entity character
    MOVE.B  #$2A, (A6)          ; Crystal entity pattern

    RTS

; Professional entity movement update
UpdateCrystalEntity:
    MOVE.L  #CrystalEntity, A5  ; Entity data pointer

    ; Update X position with professional precision
    MOVE.W  ENTITY_X(A5), D0    ; Load current X
    MOVE.B  ENTITY_VELOCITY_X(A5), D1 ; Load X velocity
    EXT.W   D1                  ; Extend byte to word
    ADD.W   D1, D0              ; Add velocity to position

    ; Boundary checking with professional precision
    CMP.W   #320, D0            ; Check right boundary
    BLE.S   x_boundary_ok       ; Branch if within bounds
    NEG.B   ENTITY_VELOCITY_X(A5) ; Reverse X velocity
    MOVE.W  #319, D0            ; Clamp to boundary
x_boundary_ok:
    MOVE.W  D0, ENTITY_X(A5)    ; Store updated X

    ; Update Y position with same precision
    MOVE.W  ENTITY_Y(A5), D0    ; Load current Y
    MOVE.B  ENTITY_VELOCITY_Y(A5), D1 ; Load Y velocity
    EXT.W   D1                  ; Extend byte to word
    ADD.W   D1, D0              ; Add velocity to position

    ; Y boundary checking
    CMP.W   #256, D0            ; Check bottom boundary
    BLE.S   y_boundary_ok       ; Branch if within bounds
    NEG.B   ENTITY_VELOCITY_Y(A5) ; Reverse Y velocity
    MOVE.W  #255, D0            ; Clamp to boundary
y_boundary_ok:
    MOVE.W  D0, ENTITY_Y(A5)    ; Store updated Y

    RTS

; Professional entity data structure
CrystalEntity:
ENTITY_X        DC.W    0       ; X position (16-bit precision)
ENTITY_Y        DC.W    0       ; Y position (16-bit precision)
ENTITY_VELOCITY_X DC.B  0       ; X velocity (8-bit efficiency)
ENTITY_VELOCITY_Y DC.B  0       ; Y velocity (8-bit efficiency)
ENTITY_ID       DC.L    0       ; Unique entity ID (32-bit)
ENTITY_COLOR    DC.W    0       ; Entity color data (16-bit)
ENTITY_LAYER    DC.B    0       ; Parallax layer assignment
ENTITY_ACTIVE   DC.B    0       ; Active flag
```

### Data and Address Registers: Your Professional Toolkit

The most important thing to understand: **the 68000's rich register set enables professional entity management**. In your crystal entity code:

**Data Registers (D0-D7)**: Your calculation engine

- **D0**: Entity X coordinate calculations
- **D1**: Entity velocity processing
- **D2**: Foreground layer data
- **D3**: Precise X position tracking
- **D4**: Precise Y position tracking
- **D5-D7**: Mathematical calculations and temporary values

**Address Registers (A0-A7)**: Your memory management system

- **A0**: Background layer memory pointer
- **A1**: Middle layer memory pointer
- **A2**: Foreground layer memory pointer
- **A3**: Entity data structure pointer
- **A4**: Color palette memory pointer
- **A5-A6**: Entity processing and screen calculation pointers

### Professional Data Size Management

The 68000's multiple data sizes enable optimal entity storage:

```nasm
MOVE.W  #160, ENTITY_X(A3)      ; 16-bit precision for coordinates
MOVE.B  #1, ENTITY_VELOCITY_X(A3) ; 8-bit efficiency for velocities
MOVE.L  #$12345678, ENTITY_ID(A3) ; 32-bit capacity for unique IDs
```

**Why this matters for professional games**:

- **16-bit coordinates**: Smooth, precise positioning across large game worlds
- **8-bit velocities**: Memory-efficient speed values with sufficient range
- **32-bit IDs**: Support for thousands of unique game entities

### The MOVE Instruction Family: Professional Data Management

You now have the most powerful data movement system available:

```nasm
MOVE.B  source, destination     ; 8-bit moves (flags, small values)
```

**Professional advantage**: The 68000 can move data between any combination of registers, memory, and immediate values with any data size.

## Practice: Create Multiple Professional Entities

Let's expand your parallax world with a complete entity system:

```nasm
    ; Create floating crystal swarm (multiple entities)
    MOVE.L  #EntityArray, A0    ; Entity array pointer
    MOVE.L  #4, D7              ; Create 5 entities (0-4)

create_entity_loop:
    ; Calculate entity data pointer
    MOVE.L  #ENTITY_SIZE, D0    ; Size of each entity structure
    MULU.W  D7, D0              ; Multiply by entity index
    MOVE.L  A0, A1              ; Copy base pointer
    ADD.L   D0, A1              ; Add offset for this entity

    ; Initialize entity with unique attributes
    MOVE.W  D7, D1              ; Use loop counter for variation
    MULU.W  #50, D1             ; Spread entities across screen
    ADD.W   #50, D1             ; Add base offset
    MOVE.W  D1, ENTITY_X(A1)    ; Set unique X position

    MOVE.W  D7, D2              ; Use counter for Y variation
    MULU.W  #30, D2             ; Different Y spacing
    ADD.W   #80, D2             ; Add base Y offset
    MOVE.W  D2, ENTITY_Y(A1)    ; Set unique Y position

    ; Vary velocities for interesting movement
    MOVE.B  D7, D3              ; Base velocity on index
    ADD.B   #1, D3              ; Ensure non-zero velocity
    MOVE.B  D3, ENTITY_VELOCITY_X(A1)

    MOVE.B  #2, ENTITY_VELOCITY_Y(A1) ; Standard Y velocity

    ; Create unique entity ID
    MOVE.L  D7, D4              ; Base ID on index
    LSL.L   #8, D4              ; Shift for uniqueness
    OR.L    #$C0DE0000, D4      ; Add entity type identifier
    MOVE.L  D4, ENTITY_ID(A1)   ; Store unique ID

    DBRA    D7, create_entity_loop ; Continue for all entities

; Professional entity structure definition
ENTITY_SIZE     EQU     16      ; Total size per entity
EntityArray:
    DS.B    80                  ; Space for 5 entities × 16 bytes each
```

### Quick Reference: Today's Professional Programming Instructions

- **MOVE.W source, dest**: Move 16-bit data (coordinates, precise values)
- **MOVE.B source, dest**: Move 8-bit data (flags, small efficient values)
- **MOVE.L source, dest**: Move 32-bit data (addresses, large values)
- **MULU.W factor, reg**: Multiply for professional calculations
- **JSR subroutine**: Call professional subroutines for complex operations
