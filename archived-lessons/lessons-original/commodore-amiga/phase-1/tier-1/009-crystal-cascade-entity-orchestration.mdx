    JSR     InitCrystalEntityVisuals

; Initialize crystal entity arrays using professional organization
InitCrystalEntityArrays:
    ; Primary crystal entities (12 entities for rich experience)
    MOVE.W  #11, D0                         ; Entity index (0-11)
    LEA     CrystalEntityArray, A0          ; Point to entity array
    
InitCrystalEntityLoop:
    ; Clear entity position data
    MOVE.L  #$00000000, (A0)+              ; Clear X,Y position (long)
    MOVE.L  #$00000000, (A0)+              ; Clear velocity data (long)
    
    ; Clear entity state data
    MOVE.W  #$0000, (A0)+                  ; Clear entity type
    MOVE.W  #$0000, (A0)+                  ; Clear entity state
    MOVE.W  #$0000, (A0)+                  ; Clear entity health
    MOVE.W  #$0000, (A0)+                  ; Clear entity behavior
    
    ; Clear entity hierarchy data
    MOVE.W  #$FFFF, (A0)+                  ; Clear parent ID (-1 = no parent)
    MOVE.W  #$0000, (A0)+                  ; Clear formation ID
    MOVE.W  #$0000, (A0)+                  ; Clear coordination flags
    MOVE.W  #$0000, (A0)+                  ; Clear orchestration state
    
    DBRA    D0, InitCrystalEntityLoop       ; Continue until all initialized

InitFormationCrystalEntities:
    ; Formation entities move in coordinated patterns

InitFormationLoop:

    MOVE.W  #$0000, (A0)+                  ; Formation member count
    MOVE.W  #$0000, (A0)+                  ; Formation sync state
    MOVE.W  #$0000, (A0)+                  ; Formation effect state

; Setup multi-entity bitplanes using hardware optimization
SetupMultiEntityBitplanes:
    ; Configure bitplanes for complex entity rendering
    MOVE.L  #$00040000, A0                  ; Bitplane 0 (background)
    MOVE.L  #$00042000, A1                  ; Bitplane 1 (crystal formations)
    MOVE.L  #$00044000, A2                  ; Bitplane 2 (floating entities)
    MOVE.L  #$00046000, A3                  ; Bitplane 3 (effects and UI)
    
    ; Set up custom chip registers for multi-entity support
    MOVE.W  #$4200, $DFF100                 ; BPLCON0: 4 bitplanes, color enable
    MOVE.W  #$0000, $DFF102                 ; BPLCON1: no horizontal scroll
    MOVE.W  #$0024, $DFF104                 ; BPLCON2: sprites over bitplanes
    
    ; Set bitplane pointers
    MOVE.L  A0, $DFF0E0                     ; BPL1PTH/L
    MOVE.L  A1, $DFF0E4                     ; BPL2PTH/L
    MOVE.L  A2, $DFF0E8                     ; BPL3PTH/L
    MOVE.L  A3, $DFF0EC                     ; BPL4PTH/L

; Update crystal formations using coordinated behavior
UpdateCrystalFormations:
    ; Process all active crystal formations

    RTS

; Update primary formations using professional organization
UpdatePrimaryFormations:
    ; Update main crystal formations (0-3)
    MOVE.W  #3, D7                          ; Formation counter
    LEA     FormationEntityArray, A6        ; Formation data pointer
    
UpdatePrimaryFormationLoop:
    ; Check if formation is active
    MOVE.W  8(A6), D0                       ; Load formation type
    BEQ.S   NextPrimaryFormation            ; Skip if inactive
    
    ; Update formation position
    JSR     UpdateFormationPosition
    
    ; Update formation behavior
    JSR     UpdateFormationBehavior
    
    ; Update formation member coordination
    JSR     UpdateFormationMemberCoordination
    
NextPrimaryFormation:
    LEA     32(A6), A6                      ; Next formation (32 bytes each)
    DBRA    D7, UpdatePrimaryFormationLoop  ; Continue for all primary formations
    
    RTS

; Update formation position using mathematical coordination
UpdateFormationPosition:
    ; A6 points to current formation data
    ; Update formation center based on movement pattern
    MOVE.W  8(A6), D0                       ; Load formation type
    CMP.W   #$0001, D0                      ; Crystal ring formation?
    BEQ.S   UpdateRingFormation
    CMP.W   #$0002, D0                      ; Crystal spiral formation?
    BEQ.S   UpdateSpiralFormation
    CMP.W   #$0003, D0                      ; Crystal wave formation?
    BEQ.S   UpdateWaveFormation
    
    ; Default formation (static)
    RTS

UpdateRingFormation:
    ; Circular formation that rotates
    MOVE.W  10(A6), D0                      ; Load formation phase
    ADDQ.W  #2, D0                          ; Increment phase
    AND.W   #$00FF, D0                      ; Keep in 0-255 range
    MOVE.W  D0, 10(A6)                      ; Store updated phase
    
    ; Calculate center position based on phase
    ; Use sine/cosine approximation for smooth circular motion
    JSR     CalculateCircularMotion
    
    RTS

UpdateSpiralFormation:
    ; Expanding/contracting spiral formation
    MOVE.W  10(A6), D0                      ; Load formation phase
    ADDQ.W  #3, D0                          ; Increment phase (faster)
    AND.W   #$01FF, D0                      ; Keep in 0-511 range
    MOVE.W  D0, 10(A6)                      ; Store updated phase
    
    ; Calculate spiral radius variation
    MOVE.W  12(A6), D1                      ; Load base radius
    MOVE.W  D0, D2                          ; Copy phase
    LSR.W   #4, D2                          ; Divide by 16
    ADD.W   D2, D1                          ; Add to radius
    
    ; Apply spiral calculation
    JSR     CalculateSpiralMotion
    
    RTS

UpdateWaveFormation:
    ; Sine wave formation
    MOVE.W  10(A6), D0                      ; Load formation phase
    ADDQ.W  #1, D0                          ; Increment phase (slower)
    AND.W   #$007F, D0                      ; Keep in 0-127 range
    MOVE.W  D0, 10(A6)                      ; Store updated phase
    
    ; Calculate wave motion
    JSR     CalculateWaveMotion
    
    RTS

; Update entity hierarchies using structured coordination
UpdateEntityHierarchies:
    ; Process parent-child relationships in crystal entities
    JSR     UpdateParentChildRelationships
    JSR     UpdateHierarchyBehaviors
    JSR     UpdateHierarchyEffects
    
    RTS

; Update parent-child relationships using efficient traversal
UpdateParentChildRelationships:
    ; Process all entities and update child positions based on parents
    MOVE.W  #11, D7                         ; Entity counter
    LEA     CrystalEntityArray, A5          ; Entity data pointer
    
UpdateHierarchyLoop:
    ; Check if entity has a parent
    MOVE.W  16(A5), D0                      ; Load parent ID
    CMP.W   #$FFFF, D0                      ; No parent?
    BEQ.S   NextHierarchyEntity             ; Skip if no parent
    
    ; Calculate parent entity address
    MULU.W  #32, D0                         ; Parent offset (32 bytes per entity)
    LEA     CrystalEntityArray, A4          ; Base entity array
    ADD.L   D0, A4                          ; Point to parent entity
    
    ; Update child position relative to parent
    JSR     UpdateChildPositionRelativeToParent
    
    ; Update child behavior based on parent state
    JSR     UpdateChildBehaviorFromParent
    
NextHierarchyEntity:
    LEA     32(A5), A5                      ; Next entity (32 bytes each)
    DBRA    D7, UpdateHierarchyLoop         ; Continue for all entities
    
    RTS

; Update child position relative to parent using vector mathematics
UpdateChildPositionRelativeToParent:
    ; A4 points to parent entity, A5 points to child entity
    ; Get parent position
    MOVE.L  (A4), D0                        ; Parent X,Y position
    
    ; Get child offset from formation data
    MOVE.W  18(A5), D1                      ; Formation ID
    BEQ.S   SimpleChildOffset               ; Use simple offset if no formation
    
    ; Calculate formation-based offset
    JSR     CalculateFormationOffset
    
SimpleChildOffset:
    ; Apply simple offset to child
    MOVE.L  (A5), D2                        ; Child current position
    ADD.L   D0, D2                          ; Add parent position
    MOVE.L  D2, (A5)                        ; Store child new position
    
    RTS

ProcessOrchestrationCoordination:

; Coordinate formation interactions using intelligent algorithms
CoordinateFormationInteractions:
    ; Check for formation-to-formation interactions
    MOVE.W  #3, D7                          ; Outer formation counter
    LEA     FormationEntityArray, A6        ; Outer formation pointer
    
FormationInteractionOuterLoop:
    ; Check if outer formation is active
    MOVE.W  8(A6), D0                       ; Load formation type
    BEQ.S   NextOuterFormation              ; Skip if inactive
    
    ; Check against other formations
    MOVE.W  #3, D6                          ; Inner formation counter
    LEA     FormationEntityArray, A5        ; Inner formation pointer
    
FormationInteractionInnerLoop:
    ; Don't check formation against itself
    CMP.L   A6, A5                          ; Same formation?
    BEQ.S   NextInnerFormation              ; Skip self
    
    ; Check if inner formation is active
    MOVE.W  8(A5), D1                       ; Load inner formation type
    BEQ.S   NextInnerFormation              ; Skip if inactive
    
    ; Calculate distance between formations
    JSR     CalculateFormationDistance
    
    ; Check for interaction based on distance
    CMP.W   #$0040, D0                      ; Interaction distance (64 pixels)
    BGT.S   NextInnerFormation              ; Too far apart
    
    ; Process formation interaction
    JSR     ProcessFormationInteraction
    
NextInnerFormation:
    LEA     32(A5), A5                      ; Next inner formation
    DBRA    D6, FormationInteractionInnerLoop
    
NextOuterFormation:
    LEA     32(A6), A6                      ; Next outer formation
    DBRA    D7, FormationInteractionOuterLoop
    
    RTS

; Update multi-entity visual effects using hardware acceleration
UpdateMultiEntityVisuals:

    JSR     RenderCrystalEntities

; Render crystal entities using optimized blitter operations
RenderCrystalEntities:
    ; Use blitter for fast entity rendering
    MOVE.W  #11, D7                         ; Entity counter
    LEA     CrystalEntityArray, A5          ; Entity data pointer
    
RenderCrystalEntityLoop:
    ; Check if entity is active
    MOVE.W  10(A5), D0                      ; Load entity state
    BEQ.S   NextCrystalEntityRender         ; Skip if inactive
    
    ; Get entity position
    MOVE.L  (A5), D0                        ; Entity X,Y position
    MOVE.W  D0, D2                          ; Y position (low word)
    SWAP    D0                              ; X position (high word)
    
    ; Get entity visual type
    MOVE.W  8(A5), D1                       ; Entity type
    
    ; Render entity using blitter
    JSR     BlitCrystalEntity
    
NextCrystalEntityRender:
    LEA     32(A5), A5                      ; Next entity
    DBRA    D7, RenderCrystalEntityLoop     ; Continue for all entities
    
    RTS
