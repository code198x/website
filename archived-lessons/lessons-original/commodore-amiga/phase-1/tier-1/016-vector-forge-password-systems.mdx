        or.b    d1,d0

; Convert binary data to vector character code
convert_binary_to_vector_character_code:
        ; Convert 8 bytes of binary data to 10-character code
        moveq   #0,d0                           ; Source byte index
        moveq   #0,d1                           ; Destination character index

        ; Take 5 bits from current byte (simplified bit packing)
        lea     temp_vector_encoding_buffer,a0
        move.b  (a0,d0.w),d2                    ; Load source byte
        andi.b  #$1F,d2                         ; 5 bits = 32 values (0-31)

        ; Convert to character using vector encoding table
        lea     vector_character_encoding_table,a0
        move.b  (a0,d2.w),d3                    ; Load character

        ; Store in vector progress code buffer

        ; Move to next position

        addq.w  #1,d1                           ; Next destination character
        cmp.w   #10,d1                          ; 10 characters
        blt     convert_vector_character_loop

; Calculate vector progress code checksum
calculate_vector_progress_code_checksum:

        move.b  vector_progress_code_checksum,d1
        add.b   (a0,d0.w),d1
        move.b  d1,vector_progress_code_checksum

        move.b  vector_progress_code_checksum,d0

        move.b  vector_achievements_unlocked,d0

        add.b   dimensional_access_password_buffer+1,d0
