## Complete Game Structure

```nasm
; ============================================
; NES PONG - COMPLETE GAME
; Tier 1 Final Deliverable
; Lessons 001-032 Synthesis
; ============================================

; iNES header
.segment "HEADER"
.byte "NES", $1A
.byte $02               ; 2 × 16KB PRG-ROM
.byte $01               ; 1 × 8KB CHR-ROM
.byte $01               ; Mapper 0, vertical mirroring
.byte $00

; ============================================
; ZERO PAGE VARIABLES
; ============================================
.segment "ZEROPAGE"
; System
nmi_ready: .res 1
frame_count: .res 1
temp: .res 4

; Game state
game_state: .res 1
pause_flag: .res 1

; Ball
ball_x: .res 1
ball_y: .res 1
ball_vx: .res 1
ball_vy: .res 1

; Paddles
paddle1_x: .res 1
paddle1_y: .res 1
paddle1_vy: .res 1
paddle2_x: .res 1
paddle2_y: .res 1
paddle2_vy: .res 1

; Controllers
buttons1: .res 1
buttons2: .res 1
buttons1_prev: .res 1
buttons2_prev: .res 1

; Scoring
score_p1: .res 1
score_p2: .res 1

; Effects
collision_cooldown: .res 1
flash_timer: .res 1

; ============================================
; MAIN ENTRY AND LOOP
; ============================================
.segment "CODE"
Reset:
    SEI
    CLD
    ; PPU warmup
    BIT $2002
:   BIT $2002
    BPL :-
:   BIT $2002
    BPL :-

    JSR InitAPU
    JSR InitPPU
    JSR InitGame

    ; Enable NMI
    LDA #%10000000
    STA $2000

    JMP MainLoop

NMI:
    PHA
    TXA
    PHA
    TYA
    PHA

    ; OAM DMA
    LDA #$00
    STA $2003
    LDA #>oam_buffer
    STA $4014

    ; Nametable updates
    ; [Score display update if flagged]

    PLA
    TAY
    PLA
    TAX
    PLA

    INC frame_count
    LDA #$01
    STA nmi_ready
    RTI

MainLoop:
    ; Wait for VBlank
:   LDA nmi_ready
    BEQ :-
    LDA #$00
    STA nmi_ready

    ; State machine
    LDA game_state
    CMP #STATE_TITLE
    BEQ TitleScreen
    CMP #STATE_PLAYING
    BEQ PlayingState
    CMP #STATE_PAUSED
    BEQ PausedState
    CMP #STATE_GAMEOVER
    BEQ GameOverState

    JMP MainLoop

; ============================================
; GAME STATES
; ============================================
TitleScreen:
    JSR ReadControllers
    ; [Detect Start button]
    ; [Transition to Playing]
    JMP MainLoop

PlayingState:
    JSR ReadControllers

    ; Check for pause
    ; [Start button detection]

    ; Game logic
    JSR UpdatePaddle1
    JSR UpdatePaddle2
    JSR CheckAllCollisions
    JSR UpdateBallPhysics
    JSR CheckBallBoundaries
    JSR UpdateFlashEffect

    ; Update sprites
    JSR UpdatePaddleSprites
    JSR UpdateBallSprite

    ; Update button history
    LDA buttons1
    STA buttons1_prev
    LDA buttons2
    STA buttons2_prev

    JMP MainLoop

PausedState:
    JSR ReadControllers
    ; [Unpause on Start]
    JMP MainLoop

GameOverState:
    JSR ReadControllers
    ; [Restart on Start]
    JMP MainLoop

; ============================================
; [INCLUDE ALL GAME LOGIC FROM L001-031]
; - ReadControllers (L13-16, L29)
; - UpdatePaddle1/2 (L13-16, L23)
; - Ball physics (L17-20)
; - Collision detection (L21-24)
; - Scoring logic (L25-28)
; - Sound effects (L30)
; - Visual effects (L31)
; ============================================

; ============================================
; CHR-ROM DATA
; ============================================
.segment "CHARS"
; [All tile data]
; - Paddle tiles ($00-$01)
; - Ball tile ($02)
; - Border/line tiles
; - Digit tiles ($30-$39)
; - Letter tiles (title/game over)
```

```

    ↓

    ↓

    ↓ (Start pressed)

    ↓ (Score >= 11)

```
