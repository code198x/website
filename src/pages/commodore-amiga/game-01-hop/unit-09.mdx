## What You're Building

Every arcade game needs stakes. Lives give the player a limited number of chances, and score gives them a reason to keep playing. This unit adds:

- **Lives system** starting with 3 lives
- **Score tracking** that increases when the frog advances
- **On-screen display** showing lives and score
- **Game over detection** when lives reach zero

By the end, death will actually cost something, and progress will be rewarded.

## The Lives System

Lives are simple: start with a number (traditionally 3), decrement on death, and end the game at zero.

```asm
STARTING_LIVES  equ 3

lives:          dc.w STARTING_LIVES
```

When the frog dies, we decrement lives in our `kill_frog` routine:

```asm
kill_frog:
            tst.w   frog_state
            bne.s   .done

            move.w  #STATE_DYING,frog_state
            move.w  #DEATH_FRAMES,death_timer

            ; Lose a life
            subq.w  #1,lives

.done:
            rts
```

In `update_death`, we check if lives have run out before respawning:

```asm
.respawn:
            ; Check for game over
            tst.w   lives
            beq.s   .game_over

            ; Respawn
            move.w  #FROG_START_X,frog_x
            move.w  #FROG_START_Y,frog_y
            move.w  #STATE_ALIVE,frog_state
            clr.w   on_log
            bra.s   .done

.game_over:
            move.w  #STATE_GAMEOVER,frog_state
```

We add a new state for game over:

```asm
STATE_ALIVE     equ 0
STATE_DYING     equ 1
STATE_DEAD      equ 2
STATE_GAMEOVER  equ 3
```

## The Scoring System

Score increases when the frog makes forward progress. The simplest approach: award points whenever the frog hops upward.

```asm
SCORE_HOP       equ 10          ; Points per forward hop

score:          dc.l 0          ; 32-bit for larger scores
```

We use a long word (32 bits) for score because it can grow quite large. In the movement handler, we award points for upward hops:

```asm
handle_movement:
            ; ... edge detection code ...

            btst    #3,d1               ; Up pressed?
            beq.s   .no_up
            move.w  frog_y,d0
            sub.w   #HOP_DISTANCE,d0
            cmp.w   #48,d0
            blt.s   .no_up
            move.w  d0,frog_y
            clr.w   on_log

            ; Award points for forward progress
            add.l   #SCORE_HOP,score

.no_up:
            ; ... rest of movement code ...
```

## Displaying Numbers

To show lives and score on screen, we need to draw numbers. This requires:
1. A font - bitmap data for digits 0-9
2. A drawing routine to blit characters to the screen

### The Digit Font

Each digit is 8 pixels wide and 8 pixels tall - one byte per row, 8 bytes per character:

```asm
; Simple 8x8 pixel font for digits 0-9
font_digits:
            ; 0
            dc.b    %00111100
            dc.b    %01100110
            dc.b    %01101110
            dc.b    %01110110
            dc.b    %01100110
            dc.b    %01100110
            dc.b    %00111100
            dc.b    %00000000

            ; 1
            dc.b    %00011000
            dc.b    %00111000
            dc.b    %00011000
            dc.b    %00011000
            dc.b    %00011000
            dc.b    %00011000
            dc.b    %01111110
            dc.b    %00000000

            ; ... and so on for 2-9
```

Each digit takes 8 bytes, so digit N starts at `font_digits + (N * 8)`.

### Drawing a Character

We use the Blitter to copy font data to the screen. Since our font is exactly 8 pixels wide (one byte), this is straightforward:

```asm
; Draw a single digit at screen position
; d0 = digit (0-9)
; d1 = X position (must be byte-aligned)
; d2 = Y position
draw_digit:
            lea     CUSTOM,a5

            ; Wait for blitter
.wait:
            btst    #14,DMACONR(a5)
            bne.s   .wait

            ; Calculate font source address
            lea     font_digits,a0
            lsl.w   #3,d0               ; d0 * 8
            add.w   d0,a0               ; a0 = font data for this digit

            ; Calculate screen destination
            lea     bitplane,a1
            move.w  d2,d3
            mulu.w  #SCREEN_WIDTH/8,d3  ; Y * bytes per row
            lsr.w   #3,d1               ; X / 8 for byte offset
            add.w   d1,d3
            add.w   d3,a1               ; a1 = screen destination

            ; Set up blitter
            move.w  #$ffff,BLTAFWM(a5)
            move.w  #$ffff,BLTALWM(a5)
            move.l  #$09f00000,BLTCON0(a5)  ; A->D, no shifts
            move.w  #0,BLTAMOD(a5)
            move.w  #(SCREEN_WIDTH/8)-1,BLTDMOD(a5)
            move.l  a0,BLTAPTH(a5)
            move.l  a1,BLTDPTH(a5)
            move.w  #(8<<6)|1,BLTSIZE(a5)   ; 8 rows, 1 word

            rts
```

### Drawing Multi-Digit Numbers

Score can be several digits. We need to extract each digit and draw them left to right:

```asm
; Draw score at position
; Score is in 'score' variable
draw_score:
            move.l  score,d0
            move.w  #SCORE_X,d4         ; Starting X position
            move.w  #SCORE_Y,d5         ; Y position

            ; We'll draw 6 digits (max 999999)
            move.w  #5,d7               ; Loop counter

            ; Divide by powers of 10, largest first
            lea     powers_of_10,a2

.digit_loop:
            move.l  (a2)+,d2            ; Get divisor
            moveq   #0,d1               ; Digit counter

.div_loop:
            cmp.l   d2,d0
            blt.s   .div_done
            sub.l   d2,d0
            addq.w  #1,d1
            bra.s   .div_loop

.div_done:
            ; d1 = digit value
            movem.l d0/d4-d7/a2,-(sp)
            move.w  d1,d0               ; Digit to draw
            move.w  d4,d1               ; X position
            move.w  d5,d2               ; Y position
            bsr     draw_digit
            movem.l (sp)+,d0/d4-d7/a2

            addq.w  #8,d4               ; Next X position
            dbf     d7,.digit_loop

            rts

powers_of_10:
            dc.l    100000,10000,1000,100,10,1
```

This approach:
1. Starts with the largest power of 10 (100000)
2. Counts how many times we can subtract it - that's our digit
3. Draws that digit
4. Moves to the next power of 10
5. Repeats for all 6 digits

### Drawing Lives

Lives are simpler - just one or two digits:

```asm
; Draw lives count
draw_lives:
            move.w  lives,d0
            move.w  #LIVES_X,d1
            move.w  #LIVES_Y,d2
            bsr     draw_digit
            rts
```

If you want to show lives as icons (little frogs), you could use sprites or blitter objects instead of numbers.

## Screen Layout

We need to reserve screen space for the HUD (Heads-Up Display):

```
+----------------------------------+
|  SCORE: 000000      LIVES: 3     |  <- HUD area (top 16 pixels)
+----------------------------------+
|                                  |
|           GAME AREA              |
|                                  |
+----------------------------------+
```

Adjust your zone boundaries to account for the HUD:

```asm
HUD_HEIGHT      equ 16

SCORE_X         equ 64              ; "SCORE: " label width
SCORE_Y         equ 4
LIVES_X         equ 280
LIVES_Y         equ 4

; Shift game zones down
WATER_TOP       equ 64+HUD_HEIGHT   ; Was 64
WATER_BOT       equ 108+HUD_HEIGHT  ; Was 108
ROAD_TOP        equ 120+HUD_HEIGHT  ; Was 120
ROAD_BOT        equ 200+HUD_HEIGHT  ; Was 200
FROG_START_Y    equ 220+HUD_HEIGHT  ; Was 220
```

## Drawing Labels

We also need to draw "SCORE:" and "LIVES:" labels. We can add letter definitions to our font, or use pre-rendered bitmap labels:

```asm
; Pre-rendered "SCORE:" label (48x8 pixels = 6 bytes wide, 8 tall)
label_score:
            dc.b    %01111100,%01111100,%01111110,%01111100,%01111110,%00000000
            dc.b    %11000000,%11000110,%11000110,%11000110,%11000000,%00011000
            dc.b    %11000000,%11000110,%11000110,%11001100,%11000000,%00011000
            dc.b    %01111100,%11000110,%11001100,%11111000,%11111100,%00011000
            dc.b    %00000110,%11000110,%11111100,%11001100,%11000000,%00000000
            dc.b    %00000110,%11000110,%11000110,%11000110,%11000000,%00011000
            dc.b    %11111100,%01111100,%11000110,%11000110,%11111110,%00011000
            dc.b    %00000000,%00000000,%00000000,%00000000,%00000000,%00000000
```

For simplicity, you could also just hard-code positions and draw the HUD once at startup rather than every frame.

## Clearing the HUD Area

Before drawing numbers, we need to clear the previous values. Use the Blitter to fill the number area with zeros:

```asm
; Clear the score number area
clear_score_area:
            lea     CUSTOM,a5

.wait:
            btst    #14,DMACONR(a5)
            bne.s   .wait

            lea     bitplane,a0
            add.w   #(SCORE_Y*(SCREEN_WIDTH/8))+(SCORE_X/8),a0

            move.w  #$ffff,BLTAFWM(a5)
            move.w  #$ffff,BLTALWM(a5)
            move.l  #$01000000,BLTCON0(a5)  ; D=0 (clear)
            move.l  a0,BLTDPTH(a5)
            move.w  #(SCREEN_WIDTH/8)-6,BLTDMOD(a5)
            move.w  #(8<<6)|3,BLTSIZE(a5)   ; 8 rows, 3 words (48 pixels)

            rts
```

## Updated Main Loop

Add HUD drawing to your main loop:

```asm
.draw:
            bsr     update_frog_sprite
            bsr     draw_objects

            ; Draw HUD
            bsr     clear_score_area
            bsr     draw_score
            bsr     draw_lives

            ; Check for exit
            btst    #6,CIAA_PRA
            bne     main_loop
```

## Game Over Handling

When `frog_state` is `STATE_GAMEOVER`, the main loop should stop normal gameplay:

```asm
main_loop:
            bsr     wait_vblank

            ; Check for game over
            cmp.w   #STATE_GAMEOVER,frog_state
            beq.s   .game_over_loop

            ; ... normal game code ...
            bra     main_loop

.game_over_loop:
            ; Game is over - just wait for exit
            btst    #6,CIAA_PRA
            bne.s   .game_over_loop

            ; Exit
            bra     exit
```

Later units will add a "GAME OVER" message and restart option.

## The Complete Code

Here's the full `hop.asm` with lives and scoring:

```asm
;=============================================================================
; HOP - Unit 09: Lives and Scoring
;=============================================================================
; Complete takeover Amiga game - no OS, direct hardware access
; Adds: lives counter, score tracking, HUD display
;=============================================================================

            include "hardware/custom.i"
            include "hardware/dmabits.i"

;=============================================================================
; Constants
;=============================================================================

CUSTOM          equ $dff000

; Display
SCREEN_WIDTH    equ 320
SCREEN_HEIGHT   equ 256
BITPLANE_SIZE   equ (SCREEN_WIDTH/8)*SCREEN_HEIGHT
HUD_HEIGHT      equ 16

; DMA and interrupts
DMAF_SETCLR     equ $8000
DMAF_COPPER     equ $0080
DMAF_RASTER     equ $0100
DMAF_BLITTER    equ $0040
DMAF_SPRITE     equ $0020

; Custom chip register offsets
DMACONR         equ $002
VPOSR           equ $004
VHPOSR          equ $006
DMACON          equ $096
INTENA          equ $09a
INTREQ          equ $09c
BPLCON0         equ $100
BPLCON1         equ $102
BPLCON2         equ $104
BPL1PTH         equ $0e0
BPL1PTL         equ $0e2
DIWSTRT         equ $08e
DIWSTOP         equ $090
DDFSTRT         equ $092
DDFSTOP         equ $094
BLTCON0         equ $040
BLTCON1         equ $042
BLTAFWM         equ $044
BLTALWM         equ $046
BLTAMOD         equ $064
BLTDMOD         equ $066
BLTAPTH         equ $050
BLTDPTH         equ $054
BLTSIZE         equ $058
SPR0PTH         equ $120
SPR0PTL         equ $122
COLOR00         equ $180
COP1LCH         equ $080
COP1LCL         equ $082
COPJMP1         equ $088

; CIA
CIAA_PRA        equ $bfe001
JOY1DAT         equ $00c

; Frog
FROG_START_X    equ 152
FROG_START_Y    equ 236             ; Adjusted for HUD
FROG_WIDTH      equ 16
FROG_HEIGHT     equ 16
HOP_DISTANCE    equ 16

; Scoring
SCORE_HOP       equ 10
SCORE_X         equ 8
SCORE_Y         equ 4
LIVES_X         equ 280
LIVES_Y         equ 4
STARTING_LIVES  equ 3

; Object system
OBJ_X           equ 0
OBJ_Y           equ 2
OBJ_SPEED       equ 4
OBJ_TYPE        equ 6
OBJ_SIZE        equ 8
NUM_OBJECTS     equ 8

; Object types
TYPE_NONE       equ 0
TYPE_CAR        equ 1
TYPE_LOG        equ 2

; Zone boundaries (adjusted for HUD)
WATER_TOP       equ 80
WATER_BOT       equ 124
ROAD_TOP        equ 136
ROAD_BOT        equ 216

; Object dimensions
OBJ_WIDTH       equ 32
OBJ_HEIGHT      equ 16

; Game states
STATE_ALIVE     equ 0
STATE_DYING     equ 1
STATE_DEAD      equ 2
STATE_GAMEOVER  equ 3
DEATH_FRAMES    equ 30

;=============================================================================
; Entry Point
;=============================================================================

            section code,code

start:
            ; Disable interrupts and DMA
            lea     CUSTOM,a5
            move.w  #$7fff,INTENA(a5)
            move.w  #$7fff,DMACON(a5)
            move.w  #$7fff,INTREQ(a5)

            ; Initialise game
            bsr     init_display
            bsr     init_sprites
            bsr     init_objects
            bsr     init_frog

            ; Enable required DMA
            move.w  #DMAF_SETCLR|DMAF_COPPER|DMAF_RASTER|DMAF_SPRITE|DMAF_BLITTER,DMACON(a5)

main_loop:
            bsr     wait_vblank

            ; Check for game over
            cmp.w   #STATE_GAMEOVER,frog_state
            beq.s   .game_over_loop

            ; Always update world
            bsr     update_objects

            ; Check game state
            cmp.w   #STATE_ALIVE,frog_state
            bne.s   .skip_input

            ; Process input and collisions when alive
            bsr     read_joystick
            bsr     handle_movement
            bsr     handle_platform
            bsr     check_bounds_death
            bsr     check_car_death
            bsr     check_water_death
            bra.s   .draw

.skip_input:
            bsr     update_death

.draw:
            bsr     update_frog_sprite
            bsr     draw_objects
            bsr     draw_hud

            ; Check for exit (left mouse button)
            btst    #6,CIAA_PRA
            bne     main_loop

            bra.s   exit

.game_over_loop:
            bsr     draw_hud
            btst    #6,CIAA_PRA
            bne.s   .game_over_loop

exit:
            ; Exit - restore system
            lea     CUSTOM,a5
            move.w  #$7fff,DMACON(a5)
            move.w  #$7fff,INTENA(a5)
            moveq   #0,d0
            rts

;=============================================================================
; Initialisation
;=============================================================================

init_display:
            lea     CUSTOM,a5

            ; Patch copper list with bitplane address
            lea     bitplane,a0
            move.l  a0,d0
            lea     copperlist,a1
            move.w  d0,bitplane_ptr_lo-copperlist+2(a1)
            swap    d0
            move.w  d0,bitplane_ptr_hi-copperlist+2(a1)

            ; Point copper to our list
            lea     copperlist,a0
            move.l  a0,COP1LCH(a5)
            move.w  COPJMP1(a5),d0

            rts

init_sprites:
            lea     CUSTOM,a5
            lea     frog_sprite,a0
            move.l  a0,d0
            move.w  d0,SPR0PTL(a5)
            swap    d0
            move.w  d0,SPR0PTH(a5)

            ; Disable other sprites
            lea     null_sprite,a0
            move.l  a0,d0
            move.w  #1,d1
.disable:
            move.w  d0,SPR0PTL+4(a5,d1.w)
            swap    d0
            move.w  d0,SPR0PTH+4(a5,d1.w)
            swap    d0
            addq.w  #4,d1
            cmp.w   #32,d1
            blt.s   .disable

            rts

init_objects:
            lea     objects,a0

            ; Car 1 - lane 1
            move.w  #0,OBJ_X(a0)
            move.w  #156,OBJ_Y(a0)
            move.w  #2,OBJ_SPEED(a0)
            move.w  #TYPE_CAR,OBJ_TYPE(a0)
            add.w   #OBJ_SIZE,a0

            ; Car 2 - lane 1
            move.w  #160,OBJ_X(a0)
            move.w  #156,OBJ_Y(a0)
            move.w  #2,OBJ_SPEED(a0)
            move.w  #TYPE_CAR,OBJ_TYPE(a0)
            add.w   #OBJ_SIZE,a0

            ; Car 3 - lane 2
            move.w  #280,OBJ_X(a0)
            move.w  #176,OBJ_Y(a0)
            move.w  #-3,OBJ_SPEED(a0)
            move.w  #TYPE_CAR,OBJ_TYPE(a0)
            add.w   #OBJ_SIZE,a0

            ; Car 4 - lane 2
            move.w  #120,OBJ_X(a0)
            move.w  #176,OBJ_Y(a0)
            move.w  #-3,OBJ_SPEED(a0)
            move.w  #TYPE_CAR,OBJ_TYPE(a0)
            add.w   #OBJ_SIZE,a0

            ; Car 5 - lane 3
            move.w  #60,OBJ_X(a0)
            move.w  #196,OBJ_Y(a0)
            move.w  #1,OBJ_SPEED(a0)
            move.w  #TYPE_CAR,OBJ_TYPE(a0)
            add.w   #OBJ_SIZE,a0

            ; Log 1 - water lane 1
            move.w  #20,OBJ_X(a0)
            move.w  #92,OBJ_Y(a0)
            move.w  #1,OBJ_SPEED(a0)
            move.w  #TYPE_LOG,OBJ_TYPE(a0)
            add.w   #OBJ_SIZE,a0

            ; Log 2 - water lane 1
            move.w  #180,OBJ_X(a0)
            move.w  #92,OBJ_Y(a0)
            move.w  #1,OBJ_SPEED(a0)
            move.w  #TYPE_LOG,OBJ_TYPE(a0)
            add.w   #OBJ_SIZE,a0

            ; Log 3 - water lane 2
            move.w  #100,OBJ_X(a0)
            move.w  #108,OBJ_Y(a0)
            move.w  #-2,OBJ_SPEED(a0)
            move.w  #TYPE_LOG,OBJ_TYPE(a0)

            rts

init_frog:
            move.w  #FROG_START_X,frog_x
            move.w  #FROG_START_Y,frog_y
            move.w  #STATE_ALIVE,frog_state
            clr.w   on_log
            clr.w   death_timer
            clr.w   joy_last
            move.w  #STARTING_LIVES,lives
            clr.l   score
            rts

;=============================================================================
; Input Handling
;=============================================================================

read_joystick:
            lea     CUSTOM,a5
            move.w  JOY1DAT(a5),d0

            moveq   #0,d1

            ; Right: bit 1 XOR bit 9
            move.w  d0,d2
            lsr.w   #1,d2
            eor.w   d0,d2
            btst    #0,d2
            beq.s   .no_right
            or.w    #1,d1
.no_right:

            ; Left: bit 9 XOR bit 8
            move.w  d0,d2
            lsr.w   #1,d2
            eor.w   d0,d2
            btst    #8,d2
            beq.s   .no_left
            or.w    #2,d1
.no_left:

            ; Down: bit 1
            btst    #1,d0
            beq.s   .no_down
            or.w    #4,d1
.no_down:

            ; Up: bit 9
            btst    #9,d0
            beq.s   .no_up
            or.w    #8,d1
.no_up:

            move.w  d1,joy_current
            rts

handle_movement:
            move.w  joy_current,d0
            move.w  joy_last,d1
            not.w   d1
            and.w   d0,d1
            move.w  d0,joy_last

            ; Up - award points for forward progress
            btst    #3,d1
            beq.s   .no_up
            move.w  frog_y,d0
            sub.w   #HOP_DISTANCE,d0
            cmp.w   #48,d0
            blt.s   .no_up
            move.w  d0,frog_y
            clr.w   on_log
            add.l   #SCORE_HOP,score
.no_up:

            btst    #2,d1
            beq.s   .no_down
            move.w  frog_y,d0
            add.w   #HOP_DISTANCE,d0
            cmp.w   #FROG_START_Y,d0
            bgt.s   .no_down
            move.w  d0,frog_y
            clr.w   on_log
.no_down:

            btst    #0,d1
            beq.s   .no_right
            move.w  frog_x,d0
            add.w   #HOP_DISTANCE,d0
            cmp.w   #304,d0
            bgt.s   .no_right
            move.w  d0,frog_x
.no_right:

            btst    #1,d1
            beq.s   .no_left
            move.w  frog_x,d0
            sub.w   #HOP_DISTANCE,d0
            bmi.s   .no_left
            move.w  d0,frog_x
.no_left:

            rts

;=============================================================================
; Platform and Collision
;=============================================================================

handle_platform:
            clr.w   on_log

            move.w  frog_y,d0
            cmp.w   #WATER_TOP,d0
            blt.s   .done
            cmp.w   #WATER_BOT,d0
            bge.s   .done

            lea     objects,a0
            move.w  #NUM_OBJECTS-1,d7

.loop:
            cmp.w   #TYPE_LOG,OBJ_TYPE(a0)
            bne.s   .next

            move.w  frog_y,d0
            move.w  OBJ_Y(a0),d1
            sub.w   d1,d0
            bmi.s   .next
            cmp.w   #OBJ_HEIGHT,d0
            bge.s   .next

            move.w  frog_x,d0
            move.w  OBJ_X(a0),d1
            sub.w   d1,d0
            add.w   #FROG_WIDTH,d0
            bmi.s   .next
            cmp.w   #OBJ_WIDTH+FROG_WIDTH,d0
            bge.s   .next

            move.w  #1,on_log
            move.w  OBJ_SPEED(a0),d0
            add.w   d0,frog_x
            bra.s   .done

.next:
            add.w   #OBJ_SIZE,a0
            dbf     d7,.loop

.done:
            rts

check_car_death:
            tst.w   frog_state
            bne.s   .done

            lea     objects,a0
            move.w  #NUM_OBJECTS-1,d7

.loop:
            cmp.w   #TYPE_CAR,OBJ_TYPE(a0)
            bne.s   .next

            move.w  frog_y,d0
            move.w  OBJ_Y(a0),d1
            sub.w   d1,d0
            bmi.s   .next
            cmp.w   #OBJ_HEIGHT,d0
            bge.s   .next

            move.w  frog_x,d0
            move.w  OBJ_X(a0),d1
            sub.w   d1,d0
            add.w   #FROG_WIDTH,d0
            bmi.s   .next
            cmp.w   #OBJ_WIDTH+FROG_WIDTH,d0
            bge.s   .next

            bsr     kill_frog
            bra.s   .done

.next:
            add.w   #OBJ_SIZE,a0
            dbf     d7,.loop
.done:
            rts

check_water_death:
            tst.w   frog_state
            bne.s   .done

            move.w  frog_y,d0
            cmp.w   #WATER_TOP,d0
            blt.s   .done
            cmp.w   #WATER_BOT,d0
            bge.s   .done

            tst.w   on_log
            bne.s   .done

            bsr     kill_frog
.done:
            rts

check_bounds_death:
            tst.w   frog_state
            bne.s   .done

            move.w  frog_x,d0
            bmi.s   .kill
            cmp.w   #SCREEN_WIDTH-FROG_WIDTH,d0
            bgt.s   .kill
.done:
            rts
.kill:
            bsr     kill_frog
            rts

kill_frog:
            tst.w   frog_state
            bne.s   .done

            move.w  #STATE_DYING,frog_state
            move.w  #DEATH_FRAMES,death_timer
            subq.w  #1,lives
.done:
            rts

update_death:
            cmp.w   #STATE_DYING,frog_state
            bne.s   .check_dead

            subq.w  #1,death_timer
            bne.s   .done

            move.w  #STATE_DEAD,frog_state
            bra.s   .respawn

.check_dead:
            cmp.w   #STATE_DEAD,frog_state
            bne.s   .done

.respawn:
            ; Check for game over
            tst.w   lives
            beq.s   .game_over

            move.w  #FROG_START_X,frog_x
            move.w  #FROG_START_Y,frog_y
            move.w  #STATE_ALIVE,frog_state
            clr.w   on_log
            bra.s   .done

.game_over:
            move.w  #STATE_GAMEOVER,frog_state

.done:
            rts

.loop:

            beq.s   .next

            add.w   d0,OBJ_X(a0)

            move.w  #-OBJ_WIDTH,OBJ_X(a0)
            bra.s   .next

            bgt.s   .next
            move.w  #SCREEN_WIDTH,OBJ_X(a0)

.next:
            add.w   #OBJ_SIZE,a0
            dbf     d7,.loop
            rts

            bne.s   .visible

            and.w   #1,d0
            bne.s   .hide

            beq.s   .hide

            add.w   #128,d0
            move.w  d0,d2
            lsr.w   #1,d0

            move.w  frog_y,d1
            add.w   #44,d1

            move.w  d1,d3
            lsl.w   #8,d3
            or.w    d0,d3
            move.w  d3,frog_sprite

            move.w  d1,d3
            add.w   #FROG_HEIGHT,d3
            lsl.w   #8,d3
            and.w   #1,d2
            or.w    d2,d3
            move.w  d3,frog_sprite+2
            bra.s   .done

.hide:
            clr.l   frog_sprite

.done:
            rts

draw_objects:
            lea     CUSTOM,a5
            lea     objects,a0
            lea     bitplane,a1
            move.w  #NUM_OBJECTS-1,d7

.loop:
            tst.w   OBJ_TYPE(a0)
            beq.s   .next

.wait:
            btst    #14,DMACONR(a5)
            bne.s   .wait

            move.w  OBJ_Y(a0),d0
            mulu.w  #SCREEN_WIDTH/8,d0
            move.w  OBJ_X(a0),d1
            bmi.s   .next
            lsr.w   #3,d1
            add.w   d1,d0
            lea     (a1,d0.w),a2

            move.w  #$ffff,BLTAFWM(a5)
            move.w  #$ffff,BLTALWM(a5)
            move.l  #$01f00000,BLTCON0(a5)
            move.l  a2,BLTDPTH(a5)
            move.w  #(SCREEN_WIDTH/8)-4,BLTDMOD(a5)
            move.w  #(OBJ_HEIGHT<<6)|2,BLTSIZE(a5)

.next:
            add.w   #OBJ_SIZE,a0
            dbf     d7,.loop
            rts

draw_score:

            move.l  score,d0
            move.w  #SCORE_X,d4
            move.w  #SCORE_Y,d5
            lea     powers_of_10,a2
            move.w  #5,d7

.digit_loop:
            move.l  (a2)+,d2
            moveq   #0,d1

.div_loop:
            cmp.l   d2,d0
            blt.s   .div_done
            sub.l   d2,d0
            addq.w  #1,d1
            bra.s   .div_loop

.div_done:
            movem.l d0/d4-d7/a2,-(sp)
            move.w  d1,d0
            move.w  d4,d1
            move.w  d5,d2
            bsr     draw_digit
            movem.l (sp)+,d0/d4-d7/a2

            addq.w  #8,d4
            dbf     d7,.digit_loop
            rts

draw_lives_count:
            move.w  lives,d0
            move.w  #LIVES_X,d1
            move.w  #LIVES_Y,d2
            bsr     draw_digit
            rts

; Draw single digit (0-9)
; d0 = digit, d1 = X, d2 = Y
draw_digit:
            lea     CUSTOM,a5
            movem.l d0-d2,-(sp)

.wait:
            btst    #14,DMACONR(a5)
            bne.s   .wait

            ; Calculate font source
            lea     font_digits,a0
            and.w   #$f,d0              ; Ensure 0-15
            lsl.w   #3,d0               ; * 8 bytes per digit
            add.w   d0,a0

            ; Calculate screen destination
            lea     bitplane,a1
            move.w  d2,d3
            mulu.w  #SCREEN_WIDTH/8,d3
            lsr.w   #3,d1
            add.w   d1,d3
            add.w   d3,a1

            ; Set up blitter - copy A to D
            move.w  #$ffff,BLTAFWM(a5)
            move.w  #$ffff,BLTALWM(a5)
            move.l  #$09f00000,BLTCON0(a5)
            move.w  #0,BLTAMOD(a5)
            move.w  #(SCREEN_WIDTH/8)-1,BLTDMOD(a5)
            move.l  a0,BLTAPTH(a5)
            move.l  a1,BLTDPTH(a5)
            move.w  #(8<<6)|1,BLTSIZE(a5)

            movem.l (sp)+,d0-d2
            rts

wait_vblank:

.wait:

            rts
```
