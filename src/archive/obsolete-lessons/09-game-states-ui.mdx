## Understanding Game State Management

### The State Machine Concept

Professional games aren't just continuous loops—they're state machines that transition between distinct modes:

<CodeBlock
  title="Classic Game State Flow"
  code={`INIT → TITLE → MENU → PLAYING → PAUSED
                 ↓         ↓
            HIGH_SCORE  GAME_OVER
                 ↑         ↓
                 ←─────────┘`}
/>

**Historical Implementation:**
Every classic C64 game used state machines:

- **"The Last Ninja"** - Complex states for intro, training, levels, death
- **"Bubble Bobble"** - Title → Demo → Playing → Bonus → Continue
- **"International Karate"** - Menu → Options → Fight → Victory → Tournament
- **"Paradroid"** - Title → Ship Select → Playing → Transfer → Destroyed

### State Variables and Control

<CodeBlock title="State Management System" language="asm" code={`; Game state definitions
STATE_INIT      = 0
STATE_TITLE     = 1
STATE_MENU      = 2
STATE_PLAYING   = 3
STATE_PAUSED    = 4
STATE_GAME_OVER = 5
STATE_HIGH_SCORE = 6

; Current state tracking
current_state: !byte STATE_INIT
previous_state: !byte STATE_INIT

; Main game loop with state management
main_loop:
; Wait for raster
lda #250

bne wait

        ; Handle current state
        lda current_state
        cmp #STATE_INIT

        beq handle_title
        cmp #STATE_MENU

jmp main_loop

<CodeBlock title="Interactive Menu Navigation" language="asm" code={`create_main_menu:
        jsr clear_screen
        
        ; Draw menu border
        jsr draw_menu_border
        
        ; Display menu title
        ldx #<menu_title
        ldy #>menu_title
        lda #5
        jsr print_centered_text

        ldx #0
display_options:
        lda menu_options_row,x
        tay
        lda menu_options_low,x
        sta text_ptr
        lda menu_options_high,x
        sta text_ptr+1
        
        ; Highlight selected option
        cpx current_menu_item
        bne normal_option
        
        ; Draw selection indicator
        lda #16                ; Column 16
        jsr calculate_position
        lda #$3e               ; '>' character
        sta (screen_ptr),y
        lda #$01               ; White
        sta (color_ptr),y

        ; Print option text
        lda #18                ; Column 18
        jsr print_text_at

        cpx #menu_item_count
        bne display_options
        rts

lda $dc00
sta joystick_state

        lsr
        bcs check_menu_down

        jsr play_menu_move_sound
        jsr update_menu_display

### Pause Functionality

<CodeBlock title="Professional Pause System" language="asm" code={`check_pause_input:
        ; Check for Run/Stop key or specific button
        lda $dc01              ; Keyboard matrix
        and #$80               ; Run/Stop key
        bne no_pause_request
        
        ; Debounce check
        lda pause_debounce
        bne no_pause_request
        
        ; Toggle pause state
        lda #STATE_PAUSED
        sta current_state
        lda #STATE_PLAYING
        sta previous_state
        
        ; Set debounce
        lda #30                ; Half second
        sta pause_debounce
        
        jsr play_pause_sound
        jsr display_pause_overlay
        
no_pause_request:
        ; Update debounce counter
        lda pause_debounce
        beq pause_check_done
        dec pause_debounce
pause_check_done:
        rts

display_pause_overlay:
; Draw semi-transparent pause overlay
; Save screen area
ldx #0
save_pause_area:
lda $0400+440,x ; Middle of screen
sta pause_buffer,x
lda $d800+440,x
sta pause_color_buffer,x
inx
cpx #120 ; 3 rows × 40 columns
bne save_pause_area

        ; Draw pause box
        ldx #0

pause_box_loop:
lda #$a0 ; Solid block
sta $0400+440,x
lda #$00 ; Black
sta $d800+440,x
inx
cpx #120
bne pause_box_loop

        ; Display "PAUSED" text
        ldx #<paused_text
        ldy #>paused_text
        lda #12                ; Center row
        jsr print_centered_text

        rts

restore_pause_area:
; Restore saved screen
ldx #0
restore_loop:
lda pause_buffer,x
sta $0400+440,x
lda pause_color_buffer,x
sta $d800+440,x
inx
cpx #120
bne restore_loop
rts

check_unpause_input:
; Check for unpause
lda $dc01
and #$80 ; Run/Stop again
bne no_unpause

        lda pause_debounce
        bne no_unpause

        ; Return to game
        jsr restore_pause_area
        lda previous_state
        sta current_state

        lda #30
        sta pause_debounce
        jsr play_unpause_sound

no_unpause:
rts

; Pause data
paused_text: !text " GAME PAUSED ",0
pause_debounce: !byte 0
pause_buffer: !fill 120,0
pause_color_buffer: !fill 120,0`} />

        ldy #>new_high_text

## Complete State Management Example

<CodeBlock title="Full Game with Complete State System" language="asm" code={`; Complete Game with Professional State Management
; Title → Menu → Playing → Game Over → High Score

        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

        ; State dispatcher
        lda current_state
