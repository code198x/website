5 REM *** ASTEROID DODGER - COMPLETE GAME ***
10 REM INITIALIZE COLORS

20 YELLOW$ = CHR$(158): CYAN$ = CHR$(159): OFF$ = CHR$(144)

50 REM TITLE SCREEN

70 SHIELD_TIME = 0: SPEED_BOOST = 0: MULTIPLIER = 1

205   END IF
210   REM CHECK COLLISION
215   IF SHIELD_TIME = 0 THEN
220     DX = ABS(SHIP_X - AST_X(I))

235   END IF
240 NEXT I
245 REM UPDATE POWER-UPS
250 FOR I = 1 TO 3
255   IF PWR_ACTIVE(I) = 0 AND RND(1) < 0.01 THEN
260     GOSUB 3500: REM SPAWN POWER-UP
265   END IF
270   IF PWR_ACTIVE(I) = 1 THEN
275     PWR_Y(I) = PWR_Y(I) + 0.5

285     REM CHECK COLLECTION
290     DX = ABS(SHIP_X - PWR_X(I))
295     DY = ABS(SHIP_Y - PWR_Y(I))
300     IF DX < 2 AND DY < 2 THEN GOSUB 4000: REM COLLECT!
305   END IF
310 NEXT I
315 REM UPDATE EFFECTS
320 IF SHIELD_TIME > 0 THEN SHIELD_TIME = SHIELD_TIME - 1
325 IF SPEED_BOOST > 0 THEN SPEED_BOOST = SPEED_BOOST - 1
330 REM INCREASE DIFFICULTY

340   LEVEL = LEVEL + 1: WAVE = 0

350 END IF

2045 NEXT I

2575 FOR I = 1 TO 3
2580   IF PWR_ACTIVE(I) = 1 THEN

2595     IF PWR_TYPE(I) = 1 THEN PRINT YELLOW$; "◆"; OFF$
2600     IF PWR_TYPE(I) = 2 THEN PRINT CYAN$; "●"; OFF$
2605     IF PWR_TYPE(I) = 3 THEN PRINT GREEN$; "★"; OFF$
2610   END IF
2615 NEXT I
2620 REM DRAW SHIP
2625 PRINT CHR$(19);: FOR Z = 1 TO SHIP_Y: PRINT CHR$(17);: NEXT Z
2630 PRINT TAB(SHIP_X);
2635 IF SHIELD_TIME > 0 AND (TI AND 2) = 0 THEN PRINT CYAN$;
2640 PRINT "▲"; OFF$
2645 REM DRAW HUD
2650 PRINT CHR$(19); CHR$(19); TAB(2);
2655 PRINT "SCORE:"; SCORE; " L:"; LEVEL; " x"; MULTIPLIER;

2670 RETURN
3000 REM COLLISION HANDLER
3005 GOSUB 6000: REM EXPLOSION EFFECT
3010 LIVES = LIVES - 1
3015 MULTIPLIER = 1
3020 IF LIVES = 0 THEN GAME_OVER = 1
3025 SHIP_X = 20: SHIP_Y = 20
3030 SHIELD_TIME = 60: REM MERCY INVINCIBILITY
