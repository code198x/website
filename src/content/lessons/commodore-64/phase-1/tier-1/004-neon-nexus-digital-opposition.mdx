update_enemy_1:

    bne enemy_1_done ; Skip if not active (Zero flag check)
    
    ; Load current position and direction
    lda $95         ; Load Enemy 1 X position

    adc #$01        ; Add 1 to X position
    cmp #$25        ; Compare with right boundary (sets flags)
    bcc store_x_1   ; Branch if carry clear (within bounds)

    lda #$02        ; Change to left direction
    sta $97         ; Store new direction
    lda #$24        ; Clamp X position
    jmp store_x_1

    sec             ; Set carry flag for subtraction
    sbc #$01        ; Subtract 1 from X position
    cmp #$05        ; Compare with left boundary (sets flags)
    bcs store_x_1   ; Branch if carry set (within bounds)
    
    ; Hit left boundary - change direction
    lda #$01        ; Change to right direction
    sta $97         ; Store new direction
    lda #$06        ; Clamp X position
    
store_x_1:
    sta $95         ; Store updated X position
    
enemy_1_done:
    rts

store_y_2:

enemy_2_done:

update_enemy_3:

    lda $93         ; Load Enemy 3 status

    bne enemy_3_done ; Branch if Zero flag clear

diagonal_down_right:

    adc #$01        ; Increment X

    txa             ; Transfer X (Y position) to A
    clc             ; Clear carry
    adc #$01        ; Increment Y
    cmp #$18        ; Check Y boundary (sets flags)
    bcs change_diagonal_1 ; Change direction if boundary hit

    lda $9b         ; Reload X position
    adc #$01        ; Increment X
    sta $9b         ; Store X
    stx $9c         ; Store Y

    jmp enemy_3_done
    
diagonal_up_left:

    txa             ; Transfer X (Y position) to A
    sec             ; Set carry
    sbc #$01        ; Decrement Y
    cmp #$0a        ; Check Y boundary (sets flags)
    bcc change_diagonal_2 ; Change direction if boundary hit

    stx $9c         ; Store Y

    asl             ; Multiply by 16

    clc             ; Clear carry
    adc $80         ; Add player X
    tay             ; Transfer to Y for indexing
    
    ; Draw player character
    lda #$2a        ; Player character (bright star)

    ; Set up Voice 1 for game sounds

## What Just Happened? The Magic Behind Digital Opposition

Now that you've seen your digital enemies in action, let's understand how **status flags** orchestrated this intelligent opposition system.

### Status Flags: Your AI Decision-Making System

The most important thing to understand: **status flags automatically provide information for AI decisions**. In your digital opposition code:

**Zero Flag (Z)**: Testing equality and status
```nasm
cmp #$01        ; Compare enemy status with active flag
bne enemy_1_done ; Branch if Not Equal (Zero flag clear)
lda $91         ; Load enemy status
cmp #$01        ; Compare with active (sets Zero flag if equal)
```

**Carry Flag (C)**: Boundary checking and range testing
```nasm
adc #$01        ; Add 1 to position
cmp #$25        ; Compare with boundary (sets Carry if ≥ boundary)
bcc store_x_1   ; Branch if Carry Clear (within bounds)
bcs change_direction ; Branch if Carry Set (hit boundary)
```

**Negative Flag (N)**: Direction and movement testing  
```nasm
sbc #$01        ; Subtract 1 from position
bmi reverse_direction ; Branch if Minus (result negative)
```

**Status Flag Examples in Digital Opposition AI Control:**

| Flag | Example | Use Case |
|------|---------|----------|
| **Zero Flag** | `cmp #$01 / bne skip_enemy` | Test if enemies are active and control AI execution |
| **Carry Flag** | `cmp #$25 / bcc within_bounds` | Check movement boundaries and trigger direction changes |
| **Negative Flag** | `sbc #$01 / bmi reverse` | Detect negative positions and handle boundary wrapping |

### The CMP Instruction: Your AI Brain

The **CMP (Compare)** instruction is essential for AI decision-making:

```nasm
CMP #value      ; Compare A register with value (sets flags but doesn't change A)
```

**How CMP Sets Flags:**
- **Zero Flag**: Set if A equals the compared value
- **Carry Flag**: Set if A is greater than or equal to compared value  
- **Negative Flag**: Set if result would be negative

**AI Decision Pattern:**
```nasm
lda enemy_direction    ; Load enemy direction
cmp #$01              ; Compare with "moving right"
beq move_right        ; Branch if equal (Zero flag set)
cmp #$02              ; Compare with "moving down"  
beq move_down         ; Branch if equal
```

### Why Status Flags Matter for Game AI

**Automatic Decision Making**: Flags are set automatically by arithmetic operations
```nasm
adc #$01        ; Move enemy right (automatically sets flags)
cmp #$25        ; Check boundary (automatically sets flags)
bcc continue    ; Decision made based on automatic flag
```

**Efficient Boundary Checking**: No manual calculations needed
```nasm
lda enemy_x     ; Load position
adc #$01        ; Move (sets Carry if overflow)
bcs hit_boundary ; Automatic boundary detection
```

**State Management**: Perfect for enemy status tracking
```nasm
lda enemy_status ; Load status
cmp #$01        ; Compare with active (sets Zero if match)
bne skip_enemy  ; Skip inactive enemies automatically
```

check_player_proximity:
    ; Check if enemy should chase player
    lda $95         ; Load Enemy 1 X position
    sec             ; Set carry for subtraction
    sbc $80         ; Subtract player X position
    bpl positive_diff ; Branch if positive difference
    
    ; Negative difference - enemy is right of player

    cmp #$05        ; Compare distance with chase range

    sta $97         ; Store as enemy direction
    
    ; Move toward player
    lda $95         ; Load enemy X

chase_right:

    jmp chase_done

chase_up:
    dec $96         ; Move enemy up toward player
    jmp chase_done
    
chase_down:
    inc $96         ; Move enemy down toward player
    
chase_done:
    rts
    
too_far:
    ; Resume normal patrol pattern
    lda #$01        ; Reset to patrol mode
    sta $97         ; Store direction
    rts
