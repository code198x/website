## Step 1: Simple Perceptron

Let's start with a basic perceptron for enemy decision making:

```asm6502
; Neural network data structures
; Perceptron with 4 inputs, 1 output
NN_INPUTS = 4

; Input weights (signed 8-bit: -128 to +127)
weights:    !byte 50, -30, 20, -10    ; Player distance X, Y, health, ammo

; Bias weight
bias:       !byte 10

; Learning rate (fixed point: 1.0 = 128)
learn_rate: !byte 16                  ; 0.125 learning rate

; Inputs (current frame)
inputs:     !byte 0, 0, 0, 0          ; Will be calculated each frame

; Previous decision and outcome
last_decision:  !byte 0               ; 0=retreat, 1=attack
last_outcome:   !byte 0               ; 0=bad, 1=good

perceptron_decide:
        ; Calculate inputs based on current game state
        jsr calculate_inputs
        
        ; Compute weighted sum
        lda #0
        sta nn_sum_low
        sta nn_sum_high
        
        ldx #0
weight_loop:
        ; Multiply input[x] * weight[x]
        lda inputs,x
        ldy weights,x
        jsr signed_multiply    ; Result in multiply_result
        
        ; Add to sum
        clc
        lda nn_sum_low
        adc multiply_result
        sta nn_sum_low
        lda nn_sum_high
        adc multiply_result+1
        sta nn_sum_high
        
        inx
        cpx #NN_INPUTS
        bne weight_loop
        
        ; Add bias
        clc
        lda nn_sum_low
        adc bias
        sta nn_sum_low
        bcc no_bias_carry
        inc nn_sum_high
no_bias_carry:
        
        ; Apply activation function (step function)
        lda nn_sum_high
        bpl positive_sum
        
        ; Negative sum = retreat (0)
        lda #0
        sta last_decision
        rts
        
positive_sum:
        ; Positive sum = attack (1)
        lda #1
        sta last_decision
        rts

calculate_inputs:
        ; Input 0: Player distance X (0-255)
        lda player_x
        sec
        sbc enemy_x
        bpl pos_x_diff
        eor #$ff        ; Absolute value
        inc
pos_x_diff:
        sta inputs+0
        
        ; Input 1: Player distance Y (0-255)
        lda player_y
        sec
        sbc enemy_y
        bpl pos_y_diff
        eor #$ff
        inc
pos_y_diff:
        sta inputs+1
        
        ; Input 2: Player "health" (based on recent hits taken)
        lda player_hit_recently
        sta inputs+2
        
        ; Input 3: "Ammo availability" (bullets on screen)
        jsr count_active_bullets
        sta inputs+3
        
        rts

count_active_bullets:
        ldx #0
        ldy #0
bullet_count_loop:
        lda bullet_active,x
        beq skip_count
        iny
skip_count:
        inx
        cpx #MAX_BULLETS
        bne bullet_count_loop
        
        tya     ; Count in A
        rts
```

## Step 2: Learning Algorithm

Implement basic reinforcement learning:

```asm6502
neural_learn:
        ; Only learn if we have a previous decision
        lda frame_counter
        cmp #1
        beq no_learning
        
        ; Evaluate the outcome of our last decision
        jsr evaluate_outcome
        
        ; If outcome was bad, adjust weights
        lda last_outcome
        bne no_learning         ; Good outcome = no change needed
        
        ; Bad outcome - adjust weights away from last decision
        jsr adjust_weights
        
no_learning:
        rts

evaluate_outcome:
        ; Determine if our last decision was good or bad
        ; Good: If we attacked and hit player, or retreated and avoided damage
        ; Bad: If we attacked and got hit, or retreated when we could have attacked
        
        lda last_decision
        bne eval_attack
        
eval_retreat:
        ; We retreated - was it good?
        lda enemy_hit_recently
        beq retreat_good        ; Didn't get hit = good retreat
        
        ; Got hit anyway = bad retreat
        lda #0
        sta last_outcome
        rts
        
retreat_good:
        lda #1
        sta last_outcome
        rts
        
eval_attack:
        ; We attacked - did we hit the player?
        lda player_hit_recently
        bne attack_good
        
        ; Didn't hit player = bad attack
        lda #0
        sta last_outcome
        rts
        
attack_good:
        lda #1
        sta last_outcome
        rts

adjust_weights:
        ; Adjust weights using simple delta rule
        ; If decision was wrong, move weights in opposite direction
        
        ldx #0
adjust_loop:
        ; Calculate weight adjustment
        lda inputs,x            ; Input value
        ldy learn_rate
        jsr signed_multiply     ; input * learn_rate
        
        ; If last decision was 1 (attack) and outcome was bad,
        ; subtract from weight. If decision was 0 (retreat) and bad,
        ; add to weight.
        lda last_decision
        bne subtract_weight
        
add_weight:
        ; Add adjustment to weight
        clc
        lda weights,x
        adc multiply_result
        sta weights,x
        jmp next_weight
        
subtract_weight:
        ; Subtract adjustment from weight
        sec
        lda weights,x
        sbc multiply_result
        sta weights,x
        
next_weight:
        inx
        cpx #NN_INPUTS
        bne adjust_loop
        
        rts
```

## Step 3: Pattern Recognition

Track and learn from player movement patterns:

```asm6502
; Pattern recognition system
PATTERN_LENGTH = 8
MAX_PATTERNS = 4

; Player movement history
movement_history:   !fill PATTERN_LENGTH, 0    ; 0=none, 1=up, 2=down, 3=left, 4=right
history_index:      !byte 0

; Known patterns and their frequencies
known_patterns:     !fill (MAX_PATTERNS * PATTERN_LENGTH), 0
pattern_frequency:  !fill MAX_PATTERNS, 0
pattern_count:      !byte 0

record_movement:
        ; Record player movement in history buffer
        lda player_moved
        beq no_movement
        
        ; Determine movement direction
        lda old_player_x
        cmp player_x
        beq check_y_movement
        bcc moved_right
        
        ; Moved left
        lda #3
        jmp store_movement
        
moved_right:
        lda #4
        jmp store_movement
        
check_y_movement:
        lda old_player_y
        cmp player_y
        beq no_movement
        bcc moved_down
        
        ; Moved up
        lda #1
        jmp store_movement
        
moved_down:
        lda #2
        
store_movement:
        ; Store in circular buffer
        ldx history_index
        sta movement_history,x
        
        ; Advance index
        inx
        cpx #PATTERN_LENGTH
        bne no_wrap
        ldx #0
no_wrap:
        stx history_index
        
        ; Try to recognize pattern
        jsr recognize_pattern
        
no_movement:
        rts

recognize_pattern:
        ; Check if current history matches known patterns
        ldx #0
pattern_check_loop:
        jsr compare_pattern     ; Compare pattern X with history
        lda pattern_match
        bne found_match
        
        inx
        cpx pattern_count
        bne pattern_check_loop
        
        ; No match found - learn new pattern if we have space
        ldx pattern_count
        cpx #MAX_PATTERNS
        beq no_more_patterns
        
        jsr learn_new_pattern
        inc pattern_count
        
found_match:
        ; Increment frequency of matched pattern
        ldx current_pattern
        inc pattern_frequency,x
        
        ; Use pattern to predict next move
        jsr predict_next_move
        
no_more_patterns:
        rts

compare_pattern:
        ; Compare pattern X with current movement history
        ; Store result in pattern_match (0=no match, 1=match)
        
        stx current_pattern
        
        ; Calculate start of pattern X in known_patterns array
        lda #0
        sta pattern_offset
        cpx #0
        beq start_compare
        
calc_offset:
        clc
        adc #PATTERN_LENGTH
        dex
        bne calc_offset
        sta pattern_offset
        
start_compare:
        ; Compare PATTERN_LENGTH bytes
        ldy #0
compare_loop:
        ldx pattern_offset
        lda known_patterns,x
        cmp movement_history,y
        bne no_match
        
        inc pattern_offset
        iny
        cpy #PATTERN_LENGTH
        bne compare_loop
        
        ; Full match!
        lda #1
        sta pattern_match
        rts
        
no_match:
        lda #0
        sta pattern_match
        rts

predict_next_move:
        ; Use most frequent pattern to predict player's next move
        ; Adjust enemy position accordingly
        
        ; Find most frequent pattern
        ldx #0
        lda #0
        sta max_frequency
        
frequency_loop:
        lda pattern_frequency,x
        cmp max_frequency
        bcc not_max
        
        sta max_frequency
        stx best_pattern
        
not_max:
        inx
        cpx pattern_count
        bne frequency_loop
        
        ; Get predicted next move from best pattern
        ; (This is simplified - would use pattern rotation)
        ldx best_pattern
        lda #PATTERN_LENGTH
        sta multiply_a
        txa
        sta multiply_b
        jsr multiply_8bit
        
        ; Get last move in pattern as prediction
        ldx multiply_result
        lda known_patterns + PATTERN_LENGTH - 1,x
        sta predicted_move
        
        ; Adjust enemy movement to counter prediction
        jsr counter_predicted_move
        
        rts
```

## Step 4: Adaptive Difficulty

Create AI that scales with player performance:

```asm6502
; Adaptive difficulty system
performance_window: !fill 16, 0        ; Last 16 game outcomes
window_index:       !byte 0
difficulty_level:   !byte 1            ; 1=easy, 2=normal, 3=hard

; Performance tracking
player_hits_taken:  !byte 0
player_hits_dealt:  !byte 0
games_completed:    !byte 0

update_difficulty:
        ; Calculate player performance over recent games
        jsr calculate_performance
        
        ; Adjust difficulty based on performance
        lda performance_score
        cmp #12         ; If player winning 75% of recent encounters
        bcs increase_difficulty
        
        cmp #4          ; If player winning less than 25%
        bcc decrease_difficulty
        
        ; Performance in acceptable range - no change
        rts
        
increase_difficulty:
        lda difficulty_level
        cmp #3
        beq max_difficulty
        inc difficulty_level
        jsr apply_difficulty
        
max_difficulty:
        rts
        
decrease_difficulty:
        lda difficulty_level
        cmp #1
        beq min_difficulty
        dec difficulty_level
        jsr apply_difficulty
        
min_difficulty:
        rts

apply_difficulty:
        ; Adjust AI parameters based on difficulty level
        lda difficulty_level
        cmp #1
        beq easy_mode
        cmp #2
        beq normal_mode
        
hard_mode:
        ; Aggressive AI
        lda #4          ; Fast enemy movement
        sta enemy_base_delay
        lda #32         ; High learning rate
        sta learn_rate
        lda #2          ; More enemy bullets
        sta enemy_fire_rate
        rts
        
normal_mode:
        ; Balanced AI
        lda #8
        sta enemy_base_delay
        lda #16
        sta learn_rate
        lda #1
        sta enemy_fire_rate
        rts
        
easy_mode:
        ; Gentle AI
        lda #12
        sta enemy_base_delay
        lda #8
        sta learn_rate
        lda #0
        sta enemy_fire_rate
        rts

calculate_performance:
        ; Count successes in performance window
        ldx #0
        lda #0
        sta performance_score
        
count_loop:
        lda performance_window,x
        beq skip_add
        inc performance_score
skip_add:
        inx
        cpx #16
        bne count_loop
        
        rts
```

## Step 5: Memory-Efficient Implementation

Optimize for C64 memory constraints:

```asm6502
; Compact neural network using 4-bit weights
; Each byte stores 2 weights (-8 to +7 range)

compact_weights:    !byte $32, $A1     ; Weights packed as nibbles

get_weight:
        ; Get weight X (0-3) from compact storage
        ; Returns signed weight in A register
        
        txa
        lsr                     ; Divide by 2 to get byte index
        tay
        lda compact_weights,y   ; Get packed byte
        
        txa
        and #1                  ; Check if even or odd weight
        beq even_weight
        
        ; Odd weight - use high nibble
        lsr
        lsr
        lsr
        lsr
        jmp sign_extend
        
even_weight:
        ; Even weight - use low nibble
        and #$0f
        
sign_extend:
        ; Convert 4-bit signed to 8-bit signed
        cmp #8
        bcc positive_weight
        
        ; Negative weight - extend sign
        ora #$f0
        
positive_weight:
        rts

set_weight:
        ; Set weight X to value in A register
        ; A contains new weight (-8 to +7)
        
        ; Clamp to 4-bit signed range
        cmp #8
        bcc check_negative
        lda #7          ; Max positive
        jmp do_set
        
check_negative:
        cmp #$f8
        bcs do_set
        lda #$f8        ; Max negative
        
do_set:
        sta temp_weight
        
        ; Calculate byte and nibble position
        txa
        lsr
        tay                     ; Byte index
        
        lda compact_weights,y   ; Get current packed byte
        
        txa
        and #1
        beq set_even
        
        ; Set odd weight (high nibble)
        and #$0f                ; Clear high nibble
        pha
        lda temp_weight
        and #$0f
        asl
        asl
        asl
        asl                     ; Shift to high nibble
        sta temp_weight
        pla
        ora temp_weight
        sta compact_weights,y
        rts
        
set_even:
        ; Set even weight (low nibble)
        and #$f0                ; Clear low nibble
        pha
        lda temp_weight
        and #$0f                ; Keep only low nibble
        sta temp_weight
        pla
        ora temp_weight
        sta compact_weights,y
        rts
```

## Step 6: Integration with Game Loop

Connect the neural network to enemy behavior:

```asm6502
intelligent_enemy_update:
        ; Record player movement for pattern learning
        jsr record_movement
        
        ; Use neural network to make decision
        jsr perceptron_decide
        
        ; Execute decision
        lda last_decision
        bne attack_behavior
        
retreat_behavior:
        ; Move away from player
        lda player_x
        cmp enemy_x
        beq retreat_y
        bcc retreat_right
        
        ; Retreat left
        dec enemy_x
        jmp retreat_done
        
retreat_right:
        inc enemy_x
        jmp retreat_done
        
retreat_y:
        lda player_y
        cmp enemy_y
        bcc retreat_down
        
        ; Retreat up
        dec enemy_y
        jmp retreat_done
        
retreat_down:
        inc enemy_y
        
retreat_done:
        ; Learn from this decision next frame
        jsr neural_learn
        rts
        
attack_behavior:
        ; Standard chase behavior but more aggressive
        jsr update_enemy        ; Use existing chase code
        
        ; Maybe fire if close enough
        lda player_x
        sec
        sbc enemy_x
        bpl check_distance
        eor #$ff
        adc #1
        
check_distance:
        cmp #3                  ; Within 3 characters
        bcs no_fire
        
        ; Fire at player (if enemy bullets implemented)
        jsr fire_enemy_bullet
        
no_fire:
        jsr neural_learn
        rts
```

## Understanding Neural Networks in Assembly

### Fixed-Point Arithmetic

Since the C64 lacks floating-point math, we use fixed-point:

```asm6502
; 8-bit signed weights: -128 to +127
; Learning rate: 1.0 = 128, 0.5 = 64, 0.125 = 16
```

### Memory Efficiency

- **Packed weights**: 2 weights per byte (4-bit each)
- **Circular buffers**: For movement history
- **Bit fields**: For pattern flags

### Simplified Algorithms

- **Perceptron**: Linear classifier with step activation
- **Delta rule**: Simple weight adjustment
- **Pattern matching**: Exact sequence comparison

## Neural Network Fundamentals

### Perceptron Formula

```
output = step(sum(input[i] * weight[i]) + bias)
```

### Learning Rule

```
weight[i] = weight[i] + learn_rate * input[i] * error
```

### Activation Functions

- **Step**: `output = (sum >= 0) ? 1 : 0`
- **Sign**: `output = (sum >= 0) ? 1 : -1`

## Performance Considerations

**Memory usage**: ~50 bytes for basic perceptron
**CPU cycles**: ~200 cycles per decision
**Learning speed**: Adjustable via learning rate
**Accuracy**: Improves over 10-20 games
