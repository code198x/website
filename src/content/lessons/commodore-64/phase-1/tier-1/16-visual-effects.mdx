# Visual Effects: Flash, Fade, and Game Feel
**Make your game feel alive with responsive visual feedback**

A game without visual effects feels dead. Compare playing "Super Mario Bros." with the star power-up (flashing rainbow colors, invincibility sparkles) versus just being told "you are invincible." The difference is game feel—that satisfying feedback that makes every action feel impactful. When "Defender" explodes an enemy, the screen flashes white. When "Pac-Man" eats a power pill, the screen pulses. When you lose a life in any game, something flashes red.

These effects aren't decoration—they're communication. A flash of red says "damage" faster than any text. A rainbow cycle says "power-up" instantly. A screen shake says "explosion" viscerally. Today you'll master these essential feedback techniques that separate amateur games from professional ones. These are the effects that make players feel every hit, every power-up, every victory.

This is where games gain that addictive, responsive feel that keeps players coming back.

## What You'll Learn

In this lesson, you'll discover:
- **Flash effects** - Screen and border flashes for instant feedback
- **Fade transitions** - Smooth color changes between states
- **Screen shake** - Impact effects through display manipulation
- **Color cycling** - Animated rainbow and pulse effects
- **Timing control** - Managing effect duration and speed
- **Visual communication** - Using effects to convey game information

## Flash Effects

### Damage Flash

The most essential feedback—showing the player they've been hit:

```asm
; Screen flash effects for game feedback
; Essential visual communication

        * = $0801
        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

        * = $080d
start:
        ; Initialize
        jsr setup_screen
        
        ; Display instructions
        jsr show_instructions
        
main_loop:
        ; Wait for input
        jsr $ffe4       ; GETIN
        beq main_loop
        
        ; Trigger effects
        cmp #'D'
        beq do_damage_flash
        cmp #'P'
        beq do_powerup_flash
        cmp #'C'
        beq do_collect_flash
        cmp #'W'
        beq do_warning_flash
        cmp #'Q'
        beq exit
        
        jmp main_loop
        
exit:
        rts

setup_screen:
        ; Clear and set colors
        lda #147
        jsr $ffd2
        
        lda #$00        ; Black border
        sta $d020
        lda #$00        ; Black background
        sta $d021
        
        rts

damage_flash:
        ; Classic red damage flash
        
        ; Save current colors
        lda $d020
        sta saved_border
        lda $d021
        sta saved_bg
        
        ; Flash intensity levels
        ldx #0
damage_loop:
        lda damage_pattern,x
        beq damage_done
        
        ; Set flash color
        sta $d020
        sta $d021
        
        ; Hold for timing
        lda damage_timing,x
        tay
damage_wait:
        lda #250
dwait:  cmp $d012
        bne dwait
        dey
        bne damage_wait
        
        inx
        jmp damage_loop
        
damage_done:
        ; Restore original colors
        lda saved_border
        sta $d020
        lda saved_bg
        sta $d021
        rts

do_damage_flash:
        jsr damage_flash
        jmp main_loop

powerup_flash:
        ; Rainbow power-up flash
        
        ldx #0
power_loop:
        lda rainbow_colors,x
        beq power_done
        
        sta $d020
        
        ; Quick cycle
        ldy #2
power_wait:
        lda #250
pwait:  cmp $d012
        bne pwait
        dey
        bne power_wait
        
        inx
        jmp power_loop
        
power_done:
        ; Return to black
        lda #$00
        sta $d020
        rts

do_powerup_flash:
        jsr powerup_flash
        jmp main_loop

collect_flash:
        ; Quick white flash for collection
        
        ; Bright flash
        lda #$01        ; White
        sta $d020
        sta $d021
        
        ; Very brief
        ldy #3
collect_wait:
        lda #250
cwait:  cmp $d012
        bne cwait
        dey
        bne collect_wait
        
        ; Return to normal
        lda #$00
        sta $d020
        sta $d021
        rts

do_collect_flash:
        jsr collect_flash
        jmp main_loop

warning_flash:
        ; Pulsing warning flash
        
        ldx #10         ; Number of pulses
warning_loop:
        ; Flash on
        lda #$02        ; Red
        sta $d020
        
        ldy #5
won:    lda #250
        cmp $d012
        bne won
        dey
        bne won
        
        ; Flash off
        lda #$00        ; Black
        sta $d020
        
        ldy #5
woff:   lda #250
        cmp $d012
        bne woff
        dey
        bne woff
        
        dex
        bne warning_loop
        rts

do_warning_flash:
        jsr warning_flash
        jmp main_loop

; Flash data
saved_border:   !byte 0
saved_bg:       !byte 0

damage_pattern:
        !byte $02,$0a,$02,$00,$02,$0a,$02,$00
        !byte 0         ; End marker

damage_timing:
        !byte 3,2,2,2,3,2,2,4
        !byte 0

rainbow_colors:
        !byte $02,$08,$07,$05,$03,$06,$04,$02
        !byte $08,$07,$05,$03,$06,$04,$00
        !byte 0
```

## Fade Effects

### Smooth Transitions

```asm
fade_to_black:
        ; Smooth fade from current to black
        
        lda $d021
        sta fade_start_color
        
        ; Determine fade path
        jsr calculate_fade_path
        
        ldx #0
fade_out_loop:
        lda fade_path,x
        cmp #$ff        ; End marker
        beq fade_out_done
        
        sta $d020
        sta $d021
        
        ; Fade speed
        ldy #4
fade_delay:
        lda #250
fwait:  cmp $d012
        bne fwait
        dey
        bne fade_delay
        
        inx
        jmp fade_out_loop
        
fade_out_done:
        rts

fade_from_black:
        ; Smooth fade from black to target
        
        ; Reverse the fade path
        ldx fade_steps
fade_in_loop:
        lda fade_path,x
        cmp #$ff
        beq fade_in_done
        
        sta $d020
        sta $d021
        
        ldy #4
fade_in_delay:
        lda #250
fiwait: cmp $d012
        bne fiwait
        dey
        bne fade_in_delay
        
        dex
        bpl fade_in_loop
        
fade_in_done:
        rts

calculate_fade_path:
        ; Build fade sequence based on start color
        lda fade_start_color
        
        ; Simple fade through grays
        cmp #$01        ; White
        beq fade_from_white
        cmp #$06        ; Blue
        beq fade_from_blue
        cmp #$05        ; Green
        beq fade_from_green
        
        ; Generic fade
        jmp generic_fade

fade_from_white:
        ldx #0
white_path:
        lda white_fade_table,x
        sta fade_path,x
        cmp #$ff
        beq path_done
        inx
        jmp white_path

fade_from_blue:
        ldx #0
blue_path:
        lda blue_fade_table,x
        sta fade_path,x
        cmp #$ff
        beq path_done
        inx
        jmp blue_path

fade_from_green:
        ldx #0
green_path:
        lda green_fade_table,x
        sta fade_path,x
        cmp #$ff
        beq path_done
        inx
        jmp green_path

generic_fade:
        ; Generic gray fade
        ldx #0
gen_path:
        lda generic_fade_table,x
        sta fade_path,x
        cmp #$ff
        beq path_done
        inx
        jmp gen_path

path_done:
        dex
        stx fade_steps
        rts

; Fade tables
white_fade_table:
        !byte $01,$0f,$0c,$0b,$00,$ff

blue_fade_table:
        !byte $06,$0e,$03,$0c,$0b,$00,$ff

green_fade_table:
        !byte $05,$0d,$03,$0c,$0b,$00,$ff

generic_fade_table:
        !byte $0f,$0c,$0b,$00,$ff

fade_start_color: !byte 0
fade_steps:       !byte 0
fade_path:        !byte 0,0,0,0,0,0,0,0,$ff
```

## Screen Shake

### Impact Effects

```asm
screen_shake:
        ; Shake screen for impact effect
        
        ; Save current scroll position
        lda $d016
        and #$f8        ; Clear scroll bits
        sta original_scroll
        
        ldx #0
shake_loop:
        lda shake_table,x
        cmp #$ff
        beq shake_done
        
        ; Apply shake offset
        ora original_scroll
        sta $d016
        
        ; Brief hold
        ldy shake_duration,x
shake_wait:
        lda #250
swait:  cmp $d012
        bne swait
        dey
        bne shake_wait
        
        inx
        jmp shake_loop
        
shake_done:
        ; Restore original position
        lda original_scroll
        sta $d016
        rts

explosion_shake:
        ; Big explosion shake
        
        ldx #3          ; Repeat count
big_shake:
        txa
        pha
        
        ; Shake sequence
        jsr screen_shake
        
        pla
        tax
        dex
        bne big_shake
        
        rts

subtle_shake:
        ; Small impact shake
        
        ; Quick left-right
        lda $d016
        and #$f8
        ora #$01        ; Shift right
        sta $d016
        
        ldy #2
        jsr wait_frames
        
        lda $d016
        and #$f8
        ora #$07        ; Shift left
        sta $d016
        
        ldy #2
        jsr wait_frames
        
        lda $d016
        and #$f8        ; Center
        sta $d016
        
        rts

wait_frames:
        ; Wait Y frames
wf_loop:
        lda #250
wf:     cmp $d012
        bne wf
        dey
        bne wf_loop
        rts

; Shake data
original_scroll: !byte 0

shake_table:
        !byte $02,$06,$01,$07,$03,$05,$00,$04
        !byte $02,$06,$00,$ff

shake_duration:
        !byte 1,1,1,1,1,1,2,1
        !byte 1,1,2,0
```

## Color Cycling

### Animated Color Effects

```asm
rainbow_cycle:
        ; Continuous rainbow effect
        
        lda cycle_position
        tax
        
rainbow_loop:
        lda rainbow_table,x
        sta $d020
        
        ; Speed control
        ldy cycle_speed
cycle_wait:
        lda #250
cyc:    cmp $d012
        bne cyc
        dey
        bne cycle_wait
        
        ; Next color
        inx
        txa
        and #$0f        ; Wrap at 16
        tax
        
        ; Check for completion
        cpx cycle_position
        bne rainbow_loop
        
        ; Update start position
        inc cycle_position
        lda cycle_position
        and #$0f
        sta cycle_position
        
        rts

pulse_effect:
        ; Pulsing glow effect
        
        ldx #0
pulse_loop:
        lda pulse_table,x
        cmp #$ff
        beq pulse_reverse
        
        sta $d020
        
        ldy #3
pulse_wait:
        lda #250
pls:    cmp $d012
        bne pls
        dey
        bne pulse_wait
        
        inx
        jmp pulse_loop
        
pulse_reverse:
        ; Pulse back
        dex
pulse_back:
        lda pulse_table,x
        sta $d020
        
        ldy #3
pulse_wait2:
        lda #250
pls2:   cmp $d012
        bne pls2
        dey
        bne pulse_wait2
        
        dex
        bpl pulse_back
        
        rts

fire_flicker:
        ; Flickering fire effect
        
        jsr get_random
        and #$03        ; 0-3
        tax
        lda fire_colors,x
        sta $d020
        
        ; Variable timing
        jsr get_random
        and #$07        ; 0-7
        tay
        iny             ; 1-8
flicker_wait:
        lda #250
flk:    cmp $d012
        bne flk
        dey
        bne flicker_wait
        
        rts

get_random:
        ; Simple random number
        lda random_seed
        asl
        asl
        adc random_seed
        adc #$37
        sta random_seed
        rts

; Cycling data
cycle_position: !byte 0
cycle_speed:    !byte 2

rainbow_table:
        !byte $02,$08,$07,$05,$0d,$03,$0e,$06
        !byte $04,$02,$08,$07,$05,$0d,$03,$0e

pulse_table:
        !byte $00,$0b,$0c,$0f,$01,$01,$0f,$0c
        !byte $0b,$00,$ff

fire_colors:
        !byte $02,$08,$0a,$07

random_seed:    !byte $42
```

## Character Effects

### Flashing Text and Sprites

```asm
flash_text:
        ; Flash text on screen
        
        lda flash_state
        eor #$01
        sta flash_state
        
        beq text_normal
        
text_inverse:
        ; Inverse video
        ldx #0
inv_loop:
        lda $0400,x
        ora #$80        ; Set inverse bit
        sta $0400,x
        inx
        cpx #40         ; One line
        bne inv_loop
        rts
        
text_normal:
        ; Normal video
        ldx #0
norm_loop:
        lda $0400,x
        and #$7f        ; Clear inverse bit
        sta $0400,x
        inx
        cpx #40
        bne norm_loop
        rts

flash_sprite:
        ; Flash sprite for invincibility
        
        lda sprite_flash_timer
        and #$04        ; Every 4 frames
        beq sprite_visible
        
sprite_hidden:
        lda $d015
        and #$fe        ; Hide sprite 0
        sta $d015
        jmp flash_sprite_done
        
sprite_visible:
        lda $d015
        ora #$01        ; Show sprite 0
        sta $d015
        
flash_sprite_done:
        inc sprite_flash_timer
        rts

color_flash_text:
        ; Flash text colors
        
        inc color_flash_pos
        lda color_flash_pos
        and #$07
        tax
        lda flash_colors,x
        
        ; Apply to text area
        ldx #0
color_text_loop:
        sta $d800,x
        inx
        cpx #40
        bne color_text_loop
        rts

; Flash data
flash_state:        !byte 0
sprite_flash_timer: !byte 0
color_flash_pos:    !byte 0

flash_colors:
        !byte $01,$07,$03,$05,$06,$04,$02,$08
```

## Combined Effects System

```asm
; Complete visual effects system
; All effects integrated

        * = $0801
        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

        * = $080d
main:
        jsr init_effects
        jsr show_menu
        
menu_loop:
        jsr $ffe4       ; Get key
        beq menu_loop
        
        ; Process selection
        cmp #'1'
        beq demo_flash
        cmp #'2'
        beq demo_fade
        cmp #'3'
        beq demo_shake
        cmp #'4'
        beq demo_cycle
        cmp #'5'
        beq demo_combined
        cmp #'Q'
        beq quit
        
        jmp menu_loop

quit:
        jsr cleanup
        rts

init_effects:
        ; Initialize system
        lda #147
        jsr $ffd2
        
        lda #$00
        sta $d020
        sta $d021
        
        ; Initialize sprite for demos
        lda #$01
        sta $d015       ; Enable sprite 0
        lda #100
        sta $d000       ; X position
        lda #100
        sta $d001       ; Y position
        lda #$07        ; Yellow
        sta $d027       ; Sprite color
        
        rts

demo_flash:
        ; Show all flash effects
        jsr damage_flash
        jsr wait_key
        jsr powerup_flash
        jsr wait_key
        jsr collect_flash
        jsr wait_key
        jsr warning_flash
        jmp main

demo_fade:
        ; Show fade effects
        lda #$06        ; Start blue
        sta $d021
        jsr fade_to_black
        jsr wait_key
        jsr fade_from_black
        jmp main

demo_shake:
        ; Show shake effects
        jsr screen_shake
        jsr wait_key
        jsr explosion_shake
        jsr wait_key
        jsr subtle_shake
        jmp main

demo_cycle:
        ; Show cycling effects
        ldx #50
cycle_demo:
        txa
        pha
        jsr rainbow_cycle
        pla
        tax
        dex
        bne cycle_demo
        
        jsr pulse_effect
        
        ldx #100
fire_demo:
        txa
        pha
        jsr fire_flicker
        pla
        tax
        dex
        bne fire_demo
        
        jmp main

demo_combined:
        ; Combined effect sequence
        
        ; Hit sequence
        jsr damage_flash
        jsr screen_shake
        
        ; Power-up sequence
        jsr powerup_flash
        ldx #30
powerup_cycle:
        txa
        pha
        jsr rainbow_cycle
        jsr flash_sprite
        pla
        tax
        dex
        bne powerup_cycle
        
        ; Victory sequence
        jsr pulse_effect
        jsr collect_flash
        
        jmp main

wait_key:
        jsr $ffe4
        beq wait_key
        rts

show_menu:
        ; Display menu
        ldx #5
        ldy #5
        jsr goto_xy
        
        ldx #0
menu_text:
        lda menu_string,x
        beq menu_done
        jsr $ffd2
        inx
        bne menu_text
menu_done:
        rts

goto_xy:
        ; Position cursor at X,Y
        lda #19         ; Home
        jsr $ffd2
        
        cpy #0
        beq gx
gy:     lda #17         ; Down
        jsr $ffd2
        dey
        bne gy
        
gx:     cpx #0
        beq gxy_done
        lda #29         ; Right
        jsr $ffd2
        dex
        bne gx
gxy_done:
        rts

cleanup:
        ; Restore defaults
        lda #$0e
        sta $d020
        lda #$06
        sta $d021
        lda $d016
        and #$f8
        sta $d016
        rts

; Menu text
menu_string:
        !text "VISUAL EFFECTS DEMO",13,13
        !text "1 - FLASH EFFECTS",13
        !text "2 - FADE EFFECTS",13
        !text "3 - SHAKE EFFECTS",13
        !text "4 - COLOR CYCLING",13
        !text "5 - COMBINED DEMO",13,13
        !text "Q - QUIT",13
        !byte 0
```

## Quick Reference

```text
Flash Effects:
- Damage: Red flash, 2-3 pulses
- Power-up: Rainbow cycle
- Collection: Single white flash
- Warning: Repeated red pulses

Fade Techniques:
- Through grays: Natural fade
- Direct to black: Quick transition
- Color-specific paths: Themed fades

Shake Patterns:
- Horizontal: Left-right motion
- Random: Chaotic explosion
- Diminishing: Start big, settle

Color Cycling:
- Rainbow: Full spectrum loop
- Pulse: Bright to dark and back
- Flicker: Random fire effect

Timing Guidelines:
- Flash: 2-5 frames per state
- Fade: 4-8 frames per step
- Shake: 1-2 frames per position
- Cycle: 2-4 frames per color

Performance Tips:
- Pre-calculate sequences
- Use lookup tables
- Minimize screen updates
- Sync with raster
```
