# Custom Character Sets: Creating Your Visual Identity
**Transform your complete game with unique graphics**

## Understanding Character Graphics

### The Memory-Efficient Revolution

In 1982, bitmap graphics required massive amounts of memory—8KB for a single screen. Character mode needed just 2KB for the same area, yet could display graphics just as detailed. How? By treating the screen as a grid of reusable 8×8 pixel tiles.

**The Brilliant Design:**
- **Screen memory** (1000 bytes) - Which character to display
- **Character definitions** (2048 bytes) - What each character looks like
- **Color memory** (1000 bytes) - Individual colors per position
- **Total: ~4KB** for unlimited visual variety

**Historical Impact:**
This efficiency enabled legendary games:
- **"Boulder Dash"** - Entire underground worlds in character mode
- **"Lode Runner"** - Complex level designs with minimal memory
- **"Paradroid"** - Detailed spaceship interiors
- **"Impossible Mission"** - Room after room of unique platforms

### Character Memory Organization

```text
Character Set Layout (2048 bytes total):
- 256 characters × 8 bytes each
- Each byte = one row of 8 pixels
- Bit 7 = leftmost pixel, Bit 0 = rightmost

Memory Locations:
ROM Charset: $D000-$D7FF (hidden under I/O)
RAM Options: $0800-$0FFF, $1000-$17FF, $1800-$1FFF, 
            $2000-$27FF, $2800-$2FFF, $3000-$37FF

VIC-II Control:
$D018 bits 1-3: Select character memory location
```

## Creating Custom Characters

### Copying ROM to RAM

The first step is copying the ROM character set to RAM where you can modify it:

```asm
init_custom_charset:
        ; BASIC launcher from previous lessons
        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

start:
        ; Initialize custom character set
        jsr copy_charset_to_ram
        jsr create_game_tiles
        jsr create_custom_font
        jsr init_animation_system
        
        ; Continue with game initialization
        jmp main_game_loop

copy_charset_to_ram:
        ; Tell VIC-II where our charset will be
        lda $d018
        and #$f0               ; Keep screen memory bits
        ora #$0c               ; Charset at $3000-$37FF
        sta $d018
        
        ; Critical: Disable interrupts during copy
        sei
        
        ; Switch out I/O to reveal character ROM
        lda $01
        pha                    ; Save memory configuration
        and #$fb               ; Clear bit 2 - reveal char ROM
        sta $01
        
        ; Fast unrolled copy (8 pages of 256 bytes)
        ldx #0
copy_loop:
        lda $d000,x            ; Character ROM page 0
        sta $3000,x            ; RAM destination
        lda $d100,x            ; Page 1
        sta $3100,x
        lda $d200,x            ; Page 2
        sta $3200,x
        lda $d300,x            ; Page 3
        sta $3300,x
        lda $d400,x            ; Page 4
        sta $3400,x
        lda $d500,x            ; Page 5
        sta $3500,x
        lda $d600,x            ; Page 6
        sta $3600,x
        lda $d700,x            ; Page 7
        sta $3700,x
        dex
        bne copy_loop
        
        ; Restore memory configuration
        pla
        sta $01
        cli                    ; Re-enable interrupts
        rts
```

## Animated Character Effects

### Dynamic Background Animation

```asm
init_animation_system:
        ; Set up animated tiles
        lda #0
        sta animation_frame
        sta animation_timer
        rts

animate_characters:
        ; Called every frame in main loop
        inc animation_timer
        lda animation_timer
        and #$07               ; Every 8 frames
        bne skip_animation
        
        ; Update animation frame
        inc animation_frame
        lda animation_frame
        and #$03               ; 4-frame animation
        sta animation_frame
        
        ; Animate data conduits (characters 140-143)
        jsr animate_conduits
        
        ; Animate network nodes (characters 144-147)
        jsr animate_network_nodes
        
        ; Animate data flow effects (character 148)
        jsr animate_data_flow
        
skip_animation:
        rts

animate_conduits:
        ; Create flowing data conduit effect
        ldx animation_frame
        lda conduit_frame_table_low,x
        sta source_ptr
        lda conduit_frame_table_high,x
        sta source_ptr+1
        
        ; Copy frame to character 140 (conduit tile)
        ldy #7
conduit_copy:
        lda (source_ptr),y
        sta $3000+(140*8),y
        dey
        bpl conduit_copy
        rts

animate_network_nodes:
        ; Pulsing network node effect
        ldx animation_frame
        
        ; Create pulsing using animation frame
        lda node_base_pattern,x
        
        ; Modify node character based on frame
        ldy #7
node_loop:
        lda node_base_pattern,y
        cpx #0
        beq store_node
        cpx #1
        beq node_frame_1
        cpx #2
        beq node_frame_2
        
        ; Frame 3 - maximum brightness
        ora #%00011000         ; Brighten center
        
store_node:
        sta $3000+(144*8),y
        dey
        bpl node_loop
        rts

node_frame_1:
        ora #%00001000         ; Small pulse
        jmp store_node
        
node_frame_2:
        ora #%00011000         ; Medium pulse
        jmp store_node

; Network animation data
conduit_frame_0:
        !byte %01010101        ; Data flowing right
        !byte %10101010
        !byte %01010101
        !byte %10101010
        !byte %01010101
        !byte %10101010
        !byte %01010101
        !byte %10101010

conduit_frame_1:
        !byte %10101010        ; Data flowing pattern shift
        !byte %01010101
        !byte %10101010
        !byte %01010101
        !byte %10101010
        !byte %01010101
        !byte %10101010
        !byte %01010101

        !byte %00000000
        !byte %00111100        ; Network node base
        !byte %01111110
        !byte %11100111
        !byte %11100111
        !byte %01111110
        !byte %00111100
        !byte %00000000

; Animation variables

animation_timer:  !byte 0
random_seed:      !byte $42
source_ptr:       !word 0

        !byte <conduit_frame_0, <conduit_frame_1
        !byte <conduit_frame_0, <conduit_frame_1
        
conduit_frame_table_high:
        !byte >conduit_frame_0, >conduit_frame_1
        !byte >conduit_frame_0, >conduit_frame_1
```

```asm
create_explosion_particles:
        ; Use characters 200-207 for particle animation
        ; Each frame shows explosion expanding
        
        ldx #0
particle_frame_loop:
        ; Frame 0: Small explosion
        lda explosion_frame_0,x
        sta $3000+(200*8),x
        
        ; Frame 1: Medium explosion
        lda explosion_frame_1,x
        sta $3000+(201*8),x
        
        ; Frame 2: Large explosion
        lda explosion_frame_2,x
        sta $3000+(202*8),x

        sta $3000+(203*8),x

trigger_explosion:
        ; Input: X,Y = character position
        stx explosion_x
        sty explosion_y
        lda #0
        sta explosion_frame
        sta explosion_timer
        lda #1
        sta explosion_active
        rts

update_explosion:
        lda explosion_active
        beq no_explosion
        
        ; Update timer
        inc explosion_timer
        lda explosion_timer
        and #$03               ; Every 4 frames
        bne same_explosion_frame
        
        ; Next frame
        inc explosion_frame
        lda explosion_frame
        cmp #4
        bne display_explosion
        
        ; Explosion finished
        lda #0
        sta explosion_active
        
        ; Clear explosion from screen

        ldy explosion_y

        rts

        clc
        adc #200               ; Explosion characters start at 200

        ldy explosion_y

no_explosion:
```

```asm
        ldy #8
        jsr plot_character

        ldy #12
        jsr plot_character

        jsr plot_character
        ldy #17
        jsr plot_character

        ; Input: A = character, X = column, Y = row

        asl
        asl                    ; × 8
        sta temp
        asl

        clc
        adc temp               ; × 40

        rts
```
