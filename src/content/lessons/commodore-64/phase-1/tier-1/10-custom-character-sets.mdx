## Understanding Character Graphics

### The Memory-Efficient Revolution

In 1982, bitmap graphics required massive amounts of memory—8KB for a single screen. Character mode needed just 2KB for the same area, yet could display graphics just as detailed. How? By treating the screen as a grid of reusable 8×8 pixel tiles.

**The Brilliant Design:**
- **Screen memory** (1000 bytes) - Which character to display
- **Character definitions** (2048 bytes) - What each character looks like
- **Color memory** (1000 bytes) - Individual colors per position
- **Total: ~4KB** for unlimited visual variety

**Historical Impact:**
This efficiency enabled legendary games:
- **"Boulder Dash"** - Entire underground worlds in character mode
- **"Lode Runner"** - Complex level designs with minimal memory
- **"Paradroid"** - Detailed spaceship interiors
- **"Impossible Mission"** - Room after room of unique platforms

## Creating Custom Characters

### Copying ROM to RAM

The first step is copying the ROM character set to RAM where you can modify it:

<CodeBlock title="Professional Character Set Initialization" language="asm" code={`init_custom_charset:
        ; BASIC launcher from previous lessons
        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

start:
        ; Initialize custom character set
        jsr copy_charset_to_ram
        jsr create_game_tiles
        jsr create_custom_font
        jsr init_animation_system
        
        ; Continue with game initialization
        jmp main_game_loop

copy_charset_to_ram:
        ; Tell VIC-II where our charset will be
        lda $d018
        and #$f0               ; Keep screen memory bits
        ora #$0c               ; Charset at $3000-$37FF
        sta $d018
        
        ; Critical: Disable interrupts during copy
        sei
        
        ; Switch out I/O to reveal character ROM
        lda $01
        pha                    ; Save memory configuration
        and #$fb               ; Clear bit 2 - reveal char ROM
        sta $01
        
        ; Fast unrolled copy (8 pages of 256 bytes)
        ldx #0
copy_loop:
        lda $d000,x            ; Character ROM page 0
        sta $3000,x            ; RAM destination
        lda $d100,x            ; Page 1
        sta $3100,x
        lda $d200,x            ; Page 2
        sta $3200,x
        lda $d300,x            ; Page 3
        sta $3300,x
        lda $d400,x            ; Page 4
        sta $3400,x
        lda $d500,x            ; Page 5
        sta $3500,x
        lda $d600,x            ; Page 6
        sta $3600,x
        lda $d700,x            ; Page 7
        sta $3700,x
        dex
        bne copy_loop
        
        ; Restore memory configuration
        pla
        sta $01
        cli                    ; Re-enable interrupts
        rts`} />

## Animated Character Effects

### Dynamic Background Animation

<CodeBlock title="Animated Character System" language="asm" code={`init_animation_system:
        ; Set up animated tiles
        lda #0
        sta animation_frame
        sta animation_timer
        rts

animate_characters:
        ; Called every frame in main loop
        inc animation_timer
        lda animation_timer
        and #$07               ; Every 8 frames
        bne skip_animation
        
        ; Update animation frame
        inc animation_frame
        lda animation_frame
        and #$03               ; 4-frame animation
        sta animation_frame
        
        ; Animate water tiles (characters 140-143)
        jsr animate_water
        
        ; Animate fire tiles (characters 144-147)
        jsr animate_fire
        
        ; Animate sparkle effects (character 148)
        jsr animate_sparkle
        
skip_animation:
        rts

animate_water:
        ; Create flowing water effect
        ldx animation_frame
        lda water_frame_table_low,x
        sta source_ptr
        lda water_frame_table_high,x
        sta source_ptr+1
        
        ; Copy frame to character 140
        ldy #7
water_copy:
        lda (source_ptr),y
        sta $3000+(140*8),y
        dey
        bpl water_copy
        rts

animate_fire:
        ; Flickering fire effect
        ldx animation_frame
        
        ; Randomize some pixels for flame flicker
        lda random_seed
        asl
        adc random_seed
        adc #$c5
        sta random_seed
        
        ; Modify fire character based on frame
        ldy #7
fire_loop:
        lda fire_base_pattern,y
        cpx #0
        beq store_fire
        cpx #1
        beq fire_frame_1
        cpx #2
        beq fire_frame_2
        
        ; Frame 3 - maximum flicker
        eor random_seed
        and #%00111100         ; Only middle pixels flicker
        ora fire_base_pattern,y
        
store_fire:
        sta $3000+(144*8),y
        dey
        bpl fire_loop
        rts

fire_frame_1:
        eor #%00011000         ; Small flicker
        jmp store_fire
        
fire_frame_2:
        eor #%00100100         ; Medium flicker
        jmp store_fire

; Animation data
water_frame_0:
        !byte %11001100
        !byte %00110011
        !byte %11001100
        !byte %00110011
        !byte %11001100
        !byte %00110011
        !byte %11001100
        !byte %00110011

water_frame_1:
        !byte %00110011
        !byte %11001100
        !byte %00110011
        !byte %11001100
        !byte %00110011
        !byte %11001100
        !byte %00110011
        !byte %11001100

fire_base_pattern:
        !byte %00011000
        !byte %00111100
        !byte %01111110
        !byte %01111110
        !byte %11111111
        !byte %11111111
        !byte %01111110
        !byte %00111100

; Animation variables
animation_frame:  !byte 0
animation_timer:  !byte 0
random_seed:      !byte $42
source_ptr:       !word 0

        !byte <water_frame_0, <water_frame_1
        !byte <water_frame_0, <water_frame_1
        
water_frame_table_high:
        !byte >water_frame_0, >water_frame_1
        !byte >water_frame_0, >water_frame_1`} />
