## Understanding C64 Character Graphics

### The 40×25 Grid Revolution

The C64's screen isn't just for text—it's a powerful graphics system. The 40×25 character grid (1000 positions) can display any of 256 different characters, each with individual colors. This system created the detailed worlds of countless classic games.

**Why Character Mode Was Brilliant:**
- **Memory efficient** - Entire detailed screen in just 2KB
- **Fast updates** - Change one byte to alter 8×8 pixels
- **Pattern reuse** - Same character appears multiple times
- **Color flexibility** - Each position has its own color
- **Hardware accelerated** - VIC-II handles all rendering

**Historical Impact:**
Games that defined the medium through character graphics:
- **"Lode Runner"** - Intricate mine shafts and ladders
- **"Boulder Dash"** - Dynamic cave systems
- **"Jumpman"** - Industrial platforms and machinery
- **"Impossible Mission"** - Detailed room interiors
- **"The Last Ninja"** - Atmospheric temple gardens

### Memory Architecture

```text
Screen Memory: $0400-$07E7 (1000 bytes)
- Each byte = which character to display (0-255)
- Arranged as 40 columns × 25 rows

Color Memory: $D800-$DBE7 (1000 bytes)  
- Each byte = color for that position (0-15)
- Same 40×25 arrangement

Character Set: $D000-$D7FF (default ROM)
- 256 characters × 8 bytes each = 2KB
- Can be redirected to RAM for custom graphics
```

## Designing Background Elements

### Creating Platform Graphics

Let's design classic platform game elements:

```asm
; Platform graphics patterns
; Each row represents one screen position

draw_platform:
        ; Input: X = starting column, Y = row
        ; Draw a solid platform 8 characters wide
        
        ; Calculate screen position: row * 40 + column
        tya
        asl                    ; × 2
        asl                    ; × 4
        asl                    ; × 8
        tay
        tya
        asl                    ; × 16
        asl                    ; × 32
        clc
        adc initial_y          ; × 40 (32 + 8)
        sta screen_pos_low
        lda #$04               ; High byte of $0400
        adc #0                 ; Add carry if needed
        sta screen_pos_high
        
        ; Draw platform pattern
        ldy x_position
draw_platform_loop:
        lda platform_pattern,x
        sta (screen_pos_low),y ; Draw character
        
        ; Set platform color
        lda #$0b               ; Dark gray
        sta $d800,y            ; Color memory
        
        inx
        iny
        cpx #8                 ; Platform width
        bne draw_platform_loop
        rts

platform_pattern:
        ; Character codes for solid platform look
        !byte $a0,$a0,$a0,$a0,$a0,$a0,$a0,$a0  ; Solid blocks
        
        ; Alternative patterns for variety:
brick_pattern:
        !byte $a6,$a7,$a6,$a7,$a6,$a7,$a6,$a7  ; Brick texture
        
metal_pattern:
        !byte $a1,$a2,$a1,$a2,$a1,$a2,$a1,$a2  ; Metal grating
```

### Building Complete Rooms

```asm
; Create network grid background for Neon Nexus
build_network_background:
        ; Clear screen first
        jsr clear_screen
        
        ; Draw network grid base pattern
        jsr draw_network_grid
        
        ; Add data flow channels
        jsr draw_data_channels
        
        ; Add network nodes at intersections
        jsr add_network_nodes
        
        ; Create animated circuit traces
        jsr initialize_circuit_traces
        
        ; Add background status indicators
        jsr add_network_status_display
        
        rts

draw_network_grid:
        ; Draw background network grid pattern
        ldy #24                ; Start from bottom row
grid_row_loop:
        ldx #39                ; 40 columns (0-39)
grid_col_loop:
        ; Calculate screen position
        tya
        jsr calc_row_offset
        
        ; Determine grid pattern based on position
        txa
        and #$07               ; Every 8 columns
        bne check_row_grid
        lda #$fa               ; Vertical grid line
        jmp store_grid
        
check_row_grid:
        tya  
        and #$07               ; Every 8 rows
        bne empty_grid
        lda #$fb               ; Horizontal grid line
        jmp store_grid
        
empty_grid:
        lda #$20               ; Space (transparent)
        
store_grid:
        sta $0400,x            ; Store character
        lda #$0b               ; Dark cyan grid color
        sta $d800,x            ; Store color
        
        dex
        bpl grid_col_loop
        dey
        bpl grid_row_loop
        rts

draw_data_channels:
        ; Draw main data flow channels
        ; Horizontal channel at row 12
        ldy #12
        ldx #39
horizontal_channel:
        lda #$fc               ; Data flow character
        sta $0400+(12*40),x    ; Row 12
        lda #$03               ; Cyan
        sta $d800+(12*40),x
        dex
        bpl horizontal_channel
        
        ; Vertical channel at column 20
        ldy #24
vertical_channel:
        tya
        jsr calc_row_offset
        lda #$fd               ; Vertical data flow
        ldx #20
        sta $0400,x
        lda #$03               ; Cyan
        sta $d800,x
        dey
        bpl vertical_channel
        rts

draw_left_wall:
        ; Draw vertical wall on left side
        ldy #24                ; 25 rows (0-24)
left_wall_loop:
        tya
        tax
        ; Calculate position: Y * 40
        jsr calculate_row_position
        lda #$e1               ; Vertical line character
        ldy #0                 ; Column 0
        sta (screen_pos_low),y
        lda #$0c               ; Gray
        sta (color_pos_low),y
        dey
        bpl left_wall_loop
        rts

draw_right_wall:
        ; Similar to left wall but at column 39
        ldy #24
right_wall_loop:
        tya
        tax
        jsr calculate_row_position
        lda #$e1               ; Vertical line
        ldy #39                ; Right column
        sta (screen_pos_low),y
        lda #$0c
        sta (color_pos_low),y
        dey
        bpl right_wall_loop
        rts
```

## Advanced Background Techniques

### Animated Background Elements

```asm
animate_background:
        ; Called every few frames to animate background elements
        
        ; Animate computer screens
        inc animation_frame
        lda animation_frame
        and #$0f               ; Every 16 frames
        bne skip_terminal_update
        
        jsr update_terminal_displays
        
skip_terminal_update:
        ; Animate water/lava (color cycling)
        lda animation_frame
        and #$07               ; Every 8 frames
        bne skip_water_update
        
        jsr cycle_water_colors
        
skip_water_update:
        ; Flashing danger indicators
        lda animation_frame
        and #$03               ; Every 4 frames
        bne skip_danger_flash
        
        jsr flash_danger_zones
        
skip_danger_flash:
        rts

update_terminal_displays:
        ; Change characters on computer screens
        ldx #0
terminal_update_loop:
        lda terminal_positions,x
        tay
        
        ; Generate random-looking data
        lda random_seed
        asl
        adc random_seed
        adc #$c5
        sta random_seed
        
        and #$0f               ; Limit to hex digits
        ora #$30               ; Convert to screen code
        sta $0400,y            ; Update screen
        
        inx
        cpx #4                 ; Number of terminals
        bne terminal_update_loop
        rts

cycle_water_colors:
        ; Create flowing water effect with color
        ldx #39
water_color_loop:
        ; Get current color
        lda $d800+400,x        ; Row 10 (water level)
        
        ; Cycle through blue shades
        cmp #$06               ; Dark blue?
        bne check_medium_blue
        lda #$0e               ; Go to light blue
        jmp set_water_color
        
check_medium_blue:
        cmp #$0e               ; Light blue?
        bne set_dark_blue
        lda #$03               ; Go to cyan
        jmp set_water_color
        
set_dark_blue:
        lda #$06               ; Back to dark blue
        
set_water_color:
        sta $d800+400,x
        dex
        bpl water_color_loop
        rts

flash_danger_zones:
        ; Flash dangerous areas between red and yellow
        lda danger_flash_state
        eor #$01
        sta danger_flash_state
        
        beq show_red
        lda #$07               ; Yellow
        jmp flash_danger
show_red:
        lda #$02               ; Red
        
flash_danger:
        sta $d800+600          ; Example danger position
        sta $d800+601
        sta $d800+602
        rts

; Animation variables
animation_frame:    !byte 0
random_seed:        !byte $42
danger_flash_state: !byte 0
ladder_height:      !byte 0
```

### Parallax Scrolling Effect

```asm
create_parallax_layers:
        ; Create depth with different scroll speeds
        ; Background layer (mountains/clouds) scrolls slowly
        ; Foreground layer (platforms) stays fixed
        
        ; Draw far background (mountains)
        ldx #39
mountain_loop:
        txa
        clc
        adc parallax_offset_slow
        and #$07               ; Repeat every 8 chars
        tay
        lda mountain_pattern,y
        sta $0400+80,x         ; Row 2
        lda #$0c               ; Gray
        sta $d800+80,x
        dex
        bpl mountain_loop
        
        ; Draw mid background (trees)
        ldx #39
tree_loop:
        txa
        clc
        adc parallax_offset_medium
        and #$0f               ; Repeat every 16
        tay
        lda tree_pattern,y
        sta $0400+200,x        ; Row 5
        lda #$05               ; Green
        sta $d800+200,x
        dex
        bpl tree_loop
        rts

update_parallax:
        ; Called each frame to scroll layers
        inc frame_counter
        lda frame_counter
        and #$03               ; Every 4 frames
        bne skip_slow_scroll
        
        inc parallax_offset_slow
        
skip_slow_scroll:
        lda frame_counter
        and #$01               ; Every 2 frames
        bne skip_medium_scroll
        
        inc parallax_offset_medium
        
skip_medium_scroll:
        rts

; Parallax patterns
mountain_pattern:
        !byte $20,$2f,$5c,$2f,$5c,$20,$20,$20
        
tree_pattern:
        !byte $20,$20,$54,$54,$54,$20,$20,$20
        !byte $20,$54,$54,$54,$20,$20,$20,$20

; Parallax variables
parallax_offset_slow:   !byte 0
parallax_offset_medium: !byte 0
frame_counter:         !byte 0
```

## Integrating Sprites with Backgrounds

### Platform Collision Detection

```asm
check_platform_collision:
        ; Check if sprite is standing on a platform
        ; Input: Sprite position in sprite_x_pos, sprite_y_pos
        
        ; Convert sprite position to character coordinates
        lda sprite_y_pos
        clc
        adc #21                ; Bottom of sprite
        lsr
        lsr
        lsr                    ; Divide by 8 for character row
        sta char_row
        
        lda sprite_x_pos
        lsr
        lsr
        lsr                    ; Divide by 8 for character column
        sta char_col
        
        ; Calculate screen memory position
        lda char_row
        asl                    ; × 2
        asl                    ; × 4
        asl                    ; × 8
        sta temp
        asl                    ; × 16
        asl                    ; × 32
        clc
        adc temp               ; × 40
        adc char_col           ; Add column
        tay
        
        ; Check character at position
        lda $0400,y
        cmp #$a0               ; Platform character?
        beq on_platform
        cmp #$e0               ; Floor character?
        beq on_platform
        
        ; Not on platform - apply gravity
        lda #0
        sta on_ground_flag
        rts
        
on_platform:
        ; On solid ground
        lda #1
        sta on_ground_flag
        
        ; Snap sprite to platform top
        lda char_row
        asl
        asl
        asl                    ; × 8
        sec
        sbc #21                ; Adjust for sprite height
        sta sprite_y_pos
        rts

; Collision variables
char_row:       !byte 0
char_col:       !byte 0
on_ground_flag: !byte 0
temp:           !byte 0
```

```asm
        jsr update_sprite_hardware
        jsr update_sound
        
        jmp main_loop

init_display:
        lda #$00               ; Black border
        sta $d020
        lda #$06               ; Blue background
        sta $d021
        rts

build_game_world:
        ; Create complete level layout
        jsr clear_screen
        
        ; Draw main structure
        jsr draw_room_walls
        
        ; Add multiple platform levels
        ldx #5
        ldy #20                ; Bottom platform
        lda #30                ; Width
        jsr draw_wide_platform
        
        ldx #10
        ldy #15                ; Middle left platform
        lda #10
        jsr draw_wide_platform
        
        ldx #20
        ldy #15                ; Middle right platform
        lda #10
        jsr draw_wide_platform
        
        ldx #12
        ldy #10                ; Top center platform
        lda #15
        jsr draw_wide_platform
        
        ; Add environment details
        jsr add_ladders
        jsr add_terminals
        jsr add_decorations
        
        ; Mark collectible positions
        jsr place_collectible_markers
        
        ; Add danger zones
        jsr create_danger_areas
        
        rts

draw_wide_platform:
        ; Input: X = start column, Y = row, A = width
        sta platform_width
        stx platform_x
        sty platform_y
        
platform_draw_loop:
        ; Calculate screen position
        lda platform_y
        jsr calculate_row_address
        
        ldy platform_x
        lda #$a0               ; Solid platform character
        sta (screen_pos_low),y
        
        ; Brick pattern for variety
        tya
        and #$01
        beq brick_light
        lda #$09               ; Brown
        jmp set_platform_color
brick_light:
        lda #$0b               ; Dark gray
        
set_platform_color:
        sta (color_pos_low),y
        
        inc platform_x
        dec platform_width
        bne platform_draw_loop
        rts

apply_physics:
        ; Apply gravity if not on platform
        lda on_ground_flag
        bne skip_gravity
        
        ; Fall
        inc sprite_y_pos
        inc sprite_y_pos
        
skip_gravity:
        ; Check for jump input
        lda joystick_state
        and #$01               ; Up = jump
        beq no_jump
        
        lda on_ground_flag
        beq no_jump            ; Can't jump in air
        
        ; Initiate jump
        lda #248               ; -8 velocity
        sta y_velocity
        lda #0
        sta on_ground_flag
        
no_jump:
        ; Apply velocity
        lda sprite_y_pos
        clc
        adc y_velocity
        sta sprite_y_pos
        
        ; Reduce velocity (gravity)
        inc y_velocity
        rts

; World building variables
platform_width: !byte 0
platform_x:     !byte 0
platform_y:     !byte 0
y_velocity:     !byte 0

; Screen calculation helpers
screen_pos_low: !byte 0
screen_pos_high: !byte 0
color_pos_low:  !byte 0
color_pos_high: !byte 0
```
