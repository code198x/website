# Screen Layout: Organizing Your Game Display
**Create professional game screens with proper UI organization**

Look at any classic C64 game screenshot. "Bubble Bobble" has its score at the top, lives at the bottom, and play area in the center. "Impossible Mission" splits the screen between the main view and the pocket computer interface. "Elite" divides its display into 3D view, scanner, and status panels. These aren't random choices—they follow design patterns that make games intuitive and playable.

The C64's 40×25 character display (plus borders) offers 1000 character positions to organize. How you arrange these determines whether your game feels professional or amateurish. Players have expectations: scores go at the top, lives in a corner, the action in the center. Break these conventions without good reason, and players feel lost.

Today you'll learn to create the same clean, organized layouts that made C64 games instantly understandable.

## What You'll Create

By lesson's end, you'll:
- Design multi-zone game screens

## Creating Status Bars

<CodeBlock title="Professional Status Bar Implementation" language="asm" code={`; Complete status bar system
; Professional game UI layout

        * = $0801
        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

        * = $080d
start:
        jsr clear_screen
        jsr create_top_bar
        jsr create_bottom_bar
        jsr create_play_area
        jsr initialize_ui
        
main_loop:
        ; Game loop with UI updates
        lda #250
wait:   cmp $d012
        bne wait
        
        jsr update_displays
        jmp main_loop

create_top_bar:
        ; Create professional top status bar
        
        ; Fill first two rows with background
        ldx #79         ; Two rows
        lda #160        ; Reverse space
fill_top:
        sta $0400,x
        lda #1          ; White
        sta $d800,x
        dex
        bpl fill_top
        
        ; Add labels
        ldx #0
score_label:
        lda score_text,x
        beq hi_label
        sta $0400,x
        inx
        bne score_label
        
hi_label:
        ldx #0
hi_loop:
        lda hi_text,x
        beq level_label
        sta $0410,x     ; Center position
        inx
        bne hi_loop
        
level_label:
        ldx #0
level_loop:
        lda level_text,x
        beq separator_line
        sta $0420,x     ; Right position
        inx
        bne level_loop
        
separator_line:
        ; Draw separator line
        ldx #39
        lda #64         ; Horizontal line
sep_loop:
        sta $0450,x     ; Row 2
        lda #12         ; Gray
        sta $d850,x
        dex
        bpl sep_loop
        
        rts

create_bottom_bar:
        ; Create bottom status area
        
        ; Separator line at row 22
        ldx #39
        lda #64         ; Horizontal line
bot_sep:
        sta $0400+(22*40),x
        lda #12         ; Gray
        sta $d800+(22*40),x
        dex
        bpl bot_sep
        
        ; Fill bottom two rows
        ldx #79
        lda #160        ; Reverse space
fill_bot:
        sta $0400+(23*40),x
        lda #1          ; White
        sta $d800+(23*40),x
        dex
        bpl fill_bot
        
        ; Add labels
        ldx #0
lives_label:
        lda lives_text,x
        beq msg_area
        sta $0400+(23*40),x
        inx
        bne lives_label
        
msg_area:
        ; Message area in center
        ldx #0
msg_loop:
        lda ready_text,x
        beq done_bottom
        sta $0400+(24*40)+15,x
        inx
        bne msg_loop
        
done_bottom:
        rts

create_play_area:
        ; Define and clear play area
        
        ; Draw side borders
        ldy #18         ; Rows 3-21
side_loop:
        ; Calculate row start
        tya
        clc
        adc #3          ; Start at row 3
        tax
        lda row_lo,x
        sta ptr
        lda row_hi,x
        sta ptr+1
        
        ; Left border
        lda #93         ; Vertical bar
        ldy #0
        sta (ptr),y
        
        ; Right border
        ldy #39
        sta (ptr),y
        
        ; Color borders
        lda ptr+1
        clc
        adc #$d4        ; Color RAM offset
        sta ptr+1
        
        lda #12         ; Gray
        ldy #0
        sta (ptr),y
        ldy #39
        sta (ptr),y
        
        dey
        bpl side_loop
        
        ; Clear play area interior
        ldx #3          ; Start row
clear_row:
        lda row_lo,x
        sta ptr
        lda row_hi,x
        sta ptr+1
        
        ldy #38         ; Columns 1-38
clear_col:
        lda #32         ; Space
        sta (ptr),y
        dey
        bne clear_col
        
        inx
        cpx #22         ; End row
        bne clear_row
        
        rts

; Text labels
score_text: !text "SCORE:000000"
            !byte 0
hi_text:    !text "HI:000000"
            !byte 0
level_text: !text "LEVEL:01"
            !byte 0
lives_text: !text "LIVES:"
            !byte 0
ready_text: !text "GET READY!"
            !byte 0

; Screen row address tables
row_lo:     !byte <$0400, <$0428, <$0450, <$0478
            !byte <$04a0, <$04c8, <$04f0, <$0518
            !byte <$0540, <$0568, <$0590, <$05b8
            !byte <$05e0, <$0608, <$0630, <$0658
            !byte <$0680, <$06a8, <$06d0, <$06f8
            !byte <$0720, <$0748, <$0770, <$0798
            !byte <$07c0

row_hi:     !byte >$0400, >$0428, >$0450, >$0478
            !byte >$04a0, >$04c8, >$04f0, >$0518
            !byte >$0540, >$0568, >$0590, >$05b8
            !byte >$05e0, >$0608, >$0630, >$0658
            !byte >$0680, >$06a8, >$06d0, >$06f8
            !byte >$0720, >$0748, >$0770, >$0798
            !byte >$07c0

ptr:        !word 0`} />

        ; Params after JSR: x1,y1,x2,y2

- Use lookup tables for speed
- Update only changed areas
- Buffer off-screen changes`} />
