# Enemy AI and Behaviors

Great enemies make great games. From simple patrol patterns to complex hunting behaviors, enemy AI brings challenge and excitement to your game world.

## Enemy State Machines

Structure enemy behavior with state-based AI:

```asm
; Enemy states
STATE_IDLE      = 0
STATE_PATROL    = 1
STATE_CHASE     = 2
STATE_ATTACK    = 3
STATE_RETREAT   = 4
STATE_STUNNED   = 5

; Enemy structure (16 bytes)
ENEMY_TYPE      = 0
ENEMY_STATE     = 1
ENEMY_X         = 2
ENEMY_Y         = 3
ENEMY_VX        = 4  ; Velocity X
ENEMY_VY        = 5  ; Velocity Y
ENEMY_HEALTH    = 6
ENEMY_TIMER     = 7  ; State timer
ENEMY_TARGET_X  = 8
ENEMY_TARGET_Y  = 9
ENEMY_SPRITE    = 10
ENEMY_ANIM      = 11 ; Animation frame
ENEMY_FLAGS     = 12
ENEMY_PATROL_X1 = 13 ; Patrol boundary
ENEMY_PATROL_X2 = 14
ENEMY_DATA      = 15 ; Type-specific data

MAX_ENEMIES = 8
enemy_table:
    .res MAX_ENEMIES * 16

update_enemies:
    ldx #0
enemy_loop:
    txa
    asl
    asl
    asl
    asl             ; ×16
    tay
    
    ; Check if enemy active
    lda enemy_table+ENEMY_TYPE,y
    cmp #$FF
    beq next_enemy
    
    ; Update based on state
    lda enemy_table+ENEMY_STATE,y
    
    cmp #STATE_IDLE
    beq update_idle
    cmp #STATE_PATROL
    beq update_patrol
    cmp #STATE_CHASE
    beq update_chase
    cmp #STATE_ATTACK
    beq update_attack
    cmp #STATE_RETREAT
    beq update_retreat
    cmp #STATE_STUNNED
    beq update_stunned
    
next_enemy:
    inx
    cpx #MAX_ENEMIES
    bne enemy_loop
    rts

update_idle:
    jsr enemy_idle_behavior
    jmp next_enemy
    
update_patrol:
    jsr enemy_patrol_behavior
    jmp next_enemy
    
update_chase:
    jsr enemy_chase_behavior
    jmp next_enemy
    
update_attack:
    jsr enemy_attack_behavior
    jmp next_enemy
    
update_retreat:
    jsr enemy_retreat_behavior
    jmp next_enemy
    
update_stunned:
    jsr enemy_stunned_behavior
    jmp next_enemy
```

## Patrol Behavior

Basic movement patterns:

```asm
enemy_patrol_behavior:
    ; Y register has enemy offset
    
    ; Check patrol boundaries
    lda enemy_table+ENEMY_X,y
    cmp enemy_table+ENEMY_PATROL_X1,y
    bcc reverse_patrol
    cmp enemy_table+ENEMY_PATROL_X2,y
    bcs reverse_patrol
    
    ; Continue patrol
    lda enemy_table+ENEMY_VX,y
    clc
    adc enemy_table+ENEMY_X,y
    sta enemy_table+ENEMY_X,y
    
    ; Check for player detection
    jsr check_player_range
    bcc continue_patrol
    
    ; Player detected - switch to chase
    lda #STATE_CHASE
    sta enemy_table+ENEMY_STATE,y
    lda #60         ; Chase duration
    sta enemy_table+ENEMY_TIMER,y
    
continue_patrol:
    rts
    
reverse_patrol:
    ; Hit boundary - reverse direction
    lda enemy_table+ENEMY_VX,y
    eor #$FF
    clc
    adc #1          ; Negate velocity
    sta enemy_table+ENEMY_VX,y
    rts

check_player_range:
    ; Calculate distance to player
    lda player_x
    sec
    sbc enemy_table+ENEMY_X,y
    bpl check_x_positive
    eor #$FF
    clc
    adc #1
    
check_x_positive:
    cmp #40         ; Detection range
    bcs out_of_range
    
    lda player_y
    sec
    sbc enemy_table+ENEMY_Y,y
    bpl check_y_positive
    eor #$FF
    clc
    adc #1
    
check_y_positive:
    cmp #40
    bcs out_of_range
    
    sec             ; In range
    rts
    
out_of_range:
    clc             ; Out of range
    rts
```

## Chase Behavior

Pursue the player intelligently:

```asm
enemy_chase_behavior:
    ; Update timer
    dec enemy_table+ENEMY_TIMER,y
    beq stop_chase
    
    ; Calculate direction to player
    lda player_x
    cmp enemy_table+ENEMY_X,y
    beq check_y_chase
    bcc chase_left
    
chase_right:
    lda #1          ; Chase speed
    sta enemy_table+ENEMY_VX,y
    jmp check_y_chase
    
chase_left:
    lda #$FF        ; -1
    sta enemy_table+ENEMY_VX,y
    
check_y_chase:
    lda player_y
    cmp enemy_table+ENEMY_Y,y
    beq apply_chase
    bcc chase_up
    
chase_down:
    lda #1
    sta enemy_table+ENEMY_VY,y
    jmp apply_chase
    
chase_up:
    lda #$FF
    sta enemy_table+ENEMY_VY,y
    
apply_chase:
    ; Move enemy
    lda enemy_table+ENEMY_X,y
    clc
    adc enemy_table+ENEMY_VX,y
    sta enemy_table+ENEMY_X,y
    
    lda enemy_table+ENEMY_Y,y
    clc
    adc enemy_table+ENEMY_VY,y
    sta enemy_table+ENEMY_Y,y
    
    ; Check if close enough to attack
    jsr check_attack_range
    bcc chase_done
    
    ; Switch to attack
    lda #STATE_ATTACK
    sta enemy_table+ENEMY_STATE,y
    lda #30         ; Attack duration
    sta enemy_table+ENEMY_TIMER,y
    
chase_done:
    rts
    
stop_chase:
    ; Return to patrol
    lda #STATE_PATROL
    sta enemy_table+ENEMY_STATE,y
    lda #0
    sta enemy_table+ENEMY_VX,y
    sta enemy_table+ENEMY_VY,y
    rts

check_attack_range:
    lda player_x
    sec
    sbc enemy_table+ENEMY_X,y
    bpl check_attack_x
    eor #$FF
    clc
    adc #1
    
check_attack_x:
    cmp #8          ; Attack range
    bcs not_in_range
    
    lda player_y
    sec
    sbc enemy_table+ENEMY_Y,y
    bpl check_attack_y
    eor #$FF
    clc
    adc #1
    
check_attack_y:
    cmp #8
    bcs not_in_range
    
    sec             ; In attack range
    rts
    
not_in_range:
    clc
    rts
```

## Smart Movement

Enemies that navigate obstacles:

```asm
; A* pathfinding simplified for 8-bit
find_path_to_player:
    ; Y = enemy offset
    
    ; Get current grid position
    lda enemy_table+ENEMY_X,y
    lsr
    lsr
    lsr             ; /8 for grid
    sta current_gx
    
    lda enemy_table+ENEMY_Y,y
    lsr
    lsr
    lsr
    sta current_gy
    
    ; Get target (player) position
    lda player_x
    lsr
    lsr
    lsr
    sta target_gx
    
    lda player_y
    lsr
    lsr
    lsr
    sta target_gy
    
    ; Simple pathfinding - try direct first
    lda target_gx
    cmp current_gx
    beq check_vertical_path
    bcc try_left_path
    
try_right_path:
    ; Check if right is blocked
    inc current_gx
    jsr check_grid_collision
    bcc path_found_x
    dec current_gx  ; Blocked, restore
    
    ; Try going around
    jsr find_alternate_path
    jmp check_vertical_path
    
try_left_path:
    dec current_gx
    jsr check_grid_collision
    bcc path_found_x
    inc current_gx  ; Blocked, restore
    jsr find_alternate_path
    
path_found_x:
    ; Set horizontal velocity
    lda target_gx
    cmp current_gx
    beq no_x_move
    bcc move_left
    lda #1
    jmp set_x_vel
move_left:
    lda #$FF
set_x_vel:
    sta enemy_table+ENEMY_VX,y
no_x_move:
    
check_vertical_path:
    ; Similar for Y axis
    lda target_gy
    cmp current_gy
    beq path_done
    bcc try_up_path
    
try_down_path:
    inc current_gy
    jsr check_grid_collision
    bcc path_found_y
    dec current_gy
    jsr find_alternate_path
    jmp path_done
    
try_up_path:
    dec current_gy
    jsr check_grid_collision
    bcc path_found_y
    inc current_gy
    jsr find_alternate_path
    
path_found_y:
    lda target_gy
    cmp current_gy
    beq no_y_move
    bcc move_up
    lda #1
    jmp set_y_vel
move_up:
    lda #$FF
set_y_vel:
    sta enemy_table+ENEMY_VY,y
no_y_move:
    
path_done:
    rts

find_alternate_path:
    ; Try perpendicular movement
    lda $D41B       ; Random choice
    and #$01
    beq try_up_alt
    
    lda #1
    sta enemy_table+ENEMY_VY,y
    rts
    
try_up_alt:
    lda #$FF
    sta enemy_table+ENEMY_VY,y
    rts

current_gx: .byte 0
current_gy: .byte 0
target_gx:  .byte 0
target_gy:  .byte 0
```

## Group Behaviors

Coordinated enemy actions:

```asm
; Formation movement
update_formation:
    ; Leader is enemy 0
    lda enemy_table+ENEMY_X
    sta formation_x
    lda enemy_table+ENEMY_Y
    sta formation_y
    
    ; Followers maintain relative positions
    ldx #1
formation_loop:
    txa
    asl
    asl
    asl
    asl
    tay
    
    ; Check if enemy active
    lda enemy_table+ENEMY_TYPE,y
    cmp #$FF
    beq next_formation
    
    ; Calculate desired position
    txa
    asl             ; ×2 for spacing
    asl
    asl
    clc
    adc formation_x
    sta enemy_table+ENEMY_TARGET_X,y
    
    lda formation_y
    sta enemy_table+ENEMY_TARGET_Y,y
    
    ; Move toward position
    jsr move_to_target
    
next_formation:
    inx
    cpx #4          ; 4 enemies in formation
    bne formation_loop
    rts

; Swarm behavior
swarm_attack:
    ; All enemies converge on player
    ldx #0
swarm_loop:
    txa
    asl
    asl
    asl
    asl
    tay
    
    lda enemy_table+ENEMY_TYPE,y
    cmp #$FF
    beq next_swarm
    
    ; Add slight offset to prevent overlap
    lda player_x
    clc
    adc swarm_offsets_x,x
    sta enemy_table+ENEMY_TARGET_X,y
    
    lda player_y
    clc
    adc swarm_offsets_y,x
    sta enemy_table+ENEMY_TARGET_Y,y
    
    ; Aggressive chase
    lda #STATE_CHASE
    sta enemy_table+ENEMY_STATE,y
    
next_swarm:
    inx
    cpx #MAX_ENEMIES
    bne swarm_loop
    rts

swarm_offsets_x:
    .byte 0,8,-8,0,8,-8,4,-4
swarm_offsets_y:
    .byte -8,0,0,8,8,-8,4,-4

formation_x: .byte 0
formation_y: .byte 0
```

## Boss AI

Complex multi-phase behaviors:

```asm
; Boss phases
BOSS_PHASE1 = 0  ; Basic attacks
BOSS_PHASE2 = 1  ; Faster, spawn minions
BOSS_PHASE3 = 2  ; Desperate attacks

boss_state:     .byte 0
boss_phase:     .byte 0
boss_health:    .byte 100
boss_timer:     .byte 0

update_boss:
    ; Check health for phase changes
    lda boss_health
    cmp #66
    bcs phase1
    cmp #33
    bcs phase2
    
    ; Phase 3
    lda #BOSS_PHASE3
    jmp set_phase
    
phase2:
    lda #BOSS_PHASE2
    jmp set_phase
    
phase1:
    lda #BOSS_PHASE1
    
set_phase:
    cmp boss_phase
    beq same_phase
    
    ; Phase change!
    sta boss_phase
    jsr boss_phase_transition
    
same_phase:
    ; Execute phase behavior
    lda boss_phase
    cmp #BOSS_PHASE1
    beq boss_phase1_ai
    cmp #BOSS_PHASE2
    beq boss_phase2_ai
    
boss_phase3_ai:
    ; Desperate attacks
    jsr boss_rapid_fire
    jsr boss_charge_player
    rts
    
boss_phase2_ai:
    ; Spawn minions
    dec boss_timer
    bne boss2_attack
    
    jsr spawn_minion
    lda #90
    sta boss_timer
    
boss2_attack:
    jsr boss_aimed_shot
    rts
    
boss_phase1_ai:
    ; Basic pattern
    jsr boss_move_pattern
    jsr boss_basic_attack
    rts

boss_move_pattern:
    ; Figure-8 movement
    inc boss_timer
    lda boss_timer
    tax
    
    ; X position
    lda figure8_x,x
    sta enemy_table+ENEMY_X
    
    ; Y position
    lda figure8_y,x
    sta enemy_table+ENEMY_Y
    rts

boss_phase_transition:
    ; Special effects for phase change
    lda #60
    sta screen_shake_timer
    
    ; Flash effect
    lda #$01
    sta $D020       ; White border
    
    ; Roar sound
    jsr boss_roar_sound
    
    ; Clear bullets
    jsr clear_all_bullets
    
    ; Invulnerable period
    lda #120
    sta boss_invuln_timer
    rts

; Movement patterns
figure8_x:
    .byte 100,110,120,130,140,130,120,110
    .byte 100,90,80,70,60,70,80,90
figure8_y:
    .byte 60,65,70,75,80,85,90,95
    .byte 100,95,90,85,80,75,70,65
```

## Complete Enemy System

Bringing it all together:

```asm
* = $0801
    .byte $0C,$08,$0A,$00,$9E,$32,$30,$36,$31,$00,$00,$00

* = $080D
main:
    jsr init_screen
    jsr init_enemies
    jsr init_player
    
game_loop:
    jsr wait_frame
    jsr handle_input
    jsr update_player
    jsr update_enemies
    jsr check_enemy_collisions
    jsr render_enemies
    jsr update_difficulty
    jmp game_loop

init_enemies:
    ; Clear enemy table
    ldx #MAX_ENEMIES*16-1
    lda #$FF
clear_enemies:
    sta enemy_table,x
    dex
    bpl clear_enemies
    
    ; Spawn initial enemies
    lda #0          ; Type: basic
    ldx #50
    ldy #60
    jsr spawn_enemy
    
    lda #1          ; Type: fast
    ldx #200
    ldy #60
    jsr spawn_enemy
    rts

spawn_enemy:
    ; A = type, X = x pos, Y = y pos
    pha
    txa
    pha
    tya
    pha
    
    ; Find free slot
    ldx #0
find_enemy_slot:
    txa
    asl
    asl
    asl
    asl
    tay
    
    lda enemy_table+ENEMY_TYPE,y
    cmp #$FF
    beq found_enemy_slot
    
    inx
    cpx #MAX_ENEMIES
    bne find_enemy_slot
    
    ; No free slots
    pla
    pla
    pla
    rts
    
found_enemy_slot:
    pla             ; Y pos
    sta enemy_table+ENEMY_Y,y
    pla             ; X pos
    sta enemy_table+ENEMY_X,y
    pla             ; Type
    sta enemy_table+ENEMY_TYPE,y
    
    ; Set initial state
    lda #STATE_PATROL
    sta enemy_table+ENEMY_STATE,y
    
    ; Set patrol boundaries
    lda enemy_table+ENEMY_X,y
    sec
    sbc #30
    sta enemy_table+ENEMY_PATROL_X1,y
    lda enemy_table+ENEMY_X,y
    clc
    adc #30
    sta enemy_table+ENEMY_PATROL_X2,y
    
    ; Set health
    lda #3
    sta enemy_table+ENEMY_HEALTH,y
    
    ; Set sprite
    txa
    clc
    adc #2          ; Sprites 2+ for enemies
    sta enemy_table+ENEMY_SPRITE,y
    rts

check_enemy_collisions:
    ; Check player bullets vs enemies
    ldx #0
enemy_hit_check:
    txa
    asl
    asl
    asl
    asl
    tay
    
    lda enemy_table+ENEMY_TYPE,y
    cmp #$FF
    beq next_enemy_hit
    
    ; Check collision with player bullets
    jsr check_bullet_collision
    bcc next_enemy_hit
    
    ; Enemy hit!
    dec enemy_table+ENEMY_HEALTH,y
    bne enemy_damaged
    
    ; Enemy destroyed
    jsr destroy_enemy
    jsr add_enemy_score
    jmp next_enemy_hit
    
enemy_damaged:
    ; Flash and stun
    lda #STATE_STUNNED
    sta enemy_table+ENEMY_STATE,y
    lda #30
    sta enemy_table+ENEMY_TIMER,y
    
next_enemy_hit:
    inx
    cpx #MAX_ENEMIES
    bne enemy_hit_check
    rts

update_difficulty:
    ; Increase difficulty over time
    inc difficulty_timer
    bne diff_done
    
    ; Every 256 frames
    inc difficulty_level
    
    ; Spawn harder enemies
    lda difficulty_level
    and #$03
    tax
    lda difficulty_types,x
    ldx #100
    ldy #50
    jsr spawn_enemy
    
diff_done:
    rts

; Variables
difficulty_timer: .byte 0
difficulty_level: .byte 0
boss_invuln_timer: .byte 0
screen_shake_timer: .byte 0

; Data
difficulty_types:
    .byte 0,1,2,3   ; Enemy types

player_x: .byte 100
player_y: .byte 150
```

## Key Concepts

1. **State Machines**: Organize complex behaviors into manageable states
2. **Detection Ranges**: Different ranges for detection, chase, and attack
3. **Pathfinding**: Simple algorithms for navigating obstacles
4. **Group Coordination**: Formations, swarms, and synchronized attacks
5. **Difficulty Scaling**: Progressive challenge through smarter AI
