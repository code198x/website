# Lesson 9: Neon Nexus - Entity Swarms

**Your neon battle arena needs worthy opponents!** In your complete Neon Nexus game from lessons 1-8, you faced single enemies and obstacles. Today, you'll transform your arena into an intense multi-entity battleground with swarms of geometric enemies, multiple projectiles, and dynamic spawning systems. You'll learn arrays and memory management through exciting swarm combat that makes your game dramatically more challenging and engaging!

## Your Entity Swarms Emerge

Let's transform your single-entity combat into an epic swarm battle system using arrays for efficient entity management:

```nasm
; Neon Nexus - Entity Swarms System
* = $0801
        .byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

start:
    ; Initialize complete C64 environment
    lda #$93        ; Clear screen
    jsr $ffd2       ; CHROUT
    
    ; Set up enhanced neon visual environment for swarms
    lda #$00        ; Black background
    sta $d020       ; Border color
    sta $d021       ; Background color
    
    ; Initialize comprehensive game state from lessons 1-8
    jsr init_complete_game_state
    
    ; Initialize new entity swarm systems
    jsr init_entity_swarm_system
    jsr init_enemy_array_system
    jsr init_projectile_array_system
    jsr init_spawning_system
    
    ; Set up swarm-ready arena
    jsr create_swarm_arena
    
    ; Entity Swarm main loop
entity_swarm_loop:
    ; Process all input
    jsr read_complete_input
    
    ; Update player systems
    jsr update_player_movement
    jsr update_player_combat
    
    ; Update entity swarm systems
    jsr update_enemy_swarm
    jsr update_projectile_swarm
    jsr process_swarm_spawning
    
    ; Process all swarm interactions
    jsr process_swarm_collisions
    jsr update_swarm_ai
    
    ; Update visual and audio systems
    jsr update_swarm_effects
    jsr update_audio_systems
    jsr update_score_display
    
    ; Check for swarm victory conditions
    jsr check_swarm_victory
    
    jmp entity_swarm_loop

; Initialize entity swarm system using arrays for efficient management
init_entity_swarm_system:
    ; Set up entity management arrays in extended memory
    jsr init_enemy_entity_arrays
    jsr init_projectile_entity_arrays
    jsr init_effect_entity_arrays
    jsr init_spawn_management_arrays
    
    ; Initialize entity counters and limits
    lda #$00        ; No active entities initially
    sta active_enemy_count
    sta active_projectile_count
    sta active_effect_count
    
    ; Set entity limits for performance
    lda #$08        ; Maximum 8 enemies
    sta max_enemy_count
    lda #$10        ; Maximum 16 projectiles
    sta max_projectile_count
    lda #$06        ; Maximum 6 effects
    sta max_effect_count
    
    rts

; Initialize enemy entity arrays for swarm management
init_enemy_entity_arrays:
    ; Enemy position arrays (8 enemies max)
    ldx #$07        ; Enemy index (0-7)
clear_enemy_positions:
    lda #$00        ; Inactive enemy marker
    sta enemy_x_array,x     ; Clear X position
    sta enemy_y_array,x     ; Clear Y position
    sta enemy_active_array,x ; Clear active flag
    dex
    bpl clear_enemy_positions
    
    ; Enemy state arrays
    ldx #$07        ; Enemy index (0-7)
clear_enemy_states:
    lda #$00        ; Default state
    sta enemy_type_array,x      ; Clear enemy type
    sta enemy_health_array,x    ; Clear enemy health
    sta enemy_ai_state_array,x  ; Clear AI state
    sta enemy_velocity_x_array,x ; Clear X velocity
    sta enemy_velocity_y_array,x ; Clear Y velocity
    dex
    bpl clear_enemy_states
    
    rts

; Initialize projectile entity arrays for efficient bullet management
init_projectile_entity_arrays:
    ; Projectile position arrays (16 projectiles max)
    ldx #$0f        ; Projectile index (0-15)
clear_projectile_positions:
    lda #$00        ; Inactive projectile marker
    sta projectile_x_array,x     ; Clear X position
    sta projectile_y_array,x     ; Clear Y position
    sta projectile_active_array,x ; Clear active flag
    dex
    bpl clear_projectile_positions
    
    ; Projectile state arrays
    ldx #$0f        ; Projectile index (0-15)
clear_projectile_states:
    lda #$00        ; Default state
    sta projectile_type_array,x      ; Clear projectile type
    sta projectile_velocity_x_array,x ; Clear X velocity
    sta projectile_velocity_y_array,x ; Clear Y velocity
    sta projectile_owner_array,x     ; Clear owner (player/enemy)
    dex
    bpl clear_projectile_states
    
    rts

; Create swarm-ready arena environment
create_swarm_arena:
    ; Draw enhanced arena boundaries for swarm combat
    jsr draw_swarm_arena_boundaries
    
    ; Set up spawn points around arena perimeter
    jsr setup_enemy_spawn_points
    
    ; Create strategic cover elements for tactical combat
    jsr create_tactical_cover_elements
    
    ; Initialize swarm-specific visual elements
    jsr init_swarm_visual_elements
    
    rts

; Update enemy swarm using array-based entity management
update_enemy_swarm:
    ; Process all active enemies using array iteration
    ldx #$00        ; Start with enemy 0
    
update_enemy_loop:
    ; Check if this enemy slot is active
    lda enemy_active_array,x
    beq next_enemy  ; Skip if inactive
    
    ; Update this enemy's position
    jsr update_enemy_movement
    
    ; Update this enemy's AI behavior
    jsr update_enemy_ai
    
    ; Update this enemy's visual representation
    jsr update_enemy_display
    
next_enemy:
    inx             ; Next enemy index
    cpx #$08        ; Check all 8 enemy slots
    bne update_enemy_loop
    
    rts

; Update enemy movement using velocity arrays
update_enemy_movement:
    ; X is current enemy index
    ; Update X position
    lda enemy_x_array,x         ; Load current X
    clc
    adc enemy_velocity_x_array,x ; Add X velocity
    sta enemy_x_array,x         ; Store new X
    
    ; Update Y position  
    lda enemy_y_array,x         ; Load current Y
    clc
    adc enemy_velocity_y_array,x ; Add Y velocity
    sta enemy_y_array,x         ; Store new Y
    
    ; Check arena boundaries
    jsr check_enemy_boundaries
    
    rts

; Update enemy AI behavior based on type and state
update_enemy_ai:
    ; X is current enemy index
    ; Load enemy type to determine AI behavior
    lda enemy_type_array,x
    cmp #$01
    beq ai_aggressive_chaser
    cmp #$02  
    beq ai_tactical_flanker
    cmp #$03
    beq ai_defensive_shooter
    
    ; Default AI: simple patrol
    jmp ai_simple_patrol

ai_aggressive_chaser:
    ; Chase player directly with aggressive movement
    lda $80         ; Player X position
    cmp enemy_x_array,x
    bcc chase_move_left
    
    ; Move right toward player
    lda #$01
    sta enemy_velocity_x_array,x
    jmp check_y_chase
    
chase_move_left:
    ; Move left toward player
    lda #$ff        ; -1 in two's complement
    sta enemy_velocity_x_array,x
    
check_y_chase:
    lda $81         ; Player Y position
    cmp enemy_y_array,x
    bcc chase_move_up
    
    ; Move down toward player
    lda #$01
    sta enemy_velocity_y_array,x
    rts
    
chase_move_up:
    ; Move up toward player
    lda #$ff        ; -1 in two's complement
    sta enemy_velocity_y_array,x
    rts

ai_tactical_flanker:
    ; More sophisticated AI: try to flank player
    lda enemy_ai_state_array,x
    cmp #$00
    beq flanker_approach
    cmp #$01
    beq flanker_circle
    
flanker_approach:
    ; Approach player at an angle
    lda $80         ; Player X
    clc
    adc #$10        ; Offset for flanking
    cmp enemy_x_array,x
    bcc flanker_move_left
    
    lda #$01
    sta enemy_velocity_x_array,x
    jmp flanker_check_distance
    
flanker_move_left:
    lda #$ff
    sta enemy_velocity_x_array,x
    
flanker_check_distance:
    ; Check if close enough to start circling
    lda enemy_x_array,x
    sec
    sbc $80         ; Distance from player
    cmp #$08        ; Close enough?
    bcc start_circling
    rts
    
start_circling:
    ; Switch to circling behavior
    lda #$01
    sta enemy_ai_state_array,x
    rts

flanker_circle:
    ; Circle around player
    lda enemy_ai_state_array,x
    clc
    adc #$01
    and #$07        ; Keep in 0-7 range for circular motion
    sta enemy_ai_state_array,x
    
    ; Use sine/cosine-like movement for circling
    ; (Simplified circular motion using lookup table)
    jsr calculate_circular_movement
    rts

ai_defensive_shooter:
    ; Stay at distance and shoot projectiles
    ; Calculate distance from player
    lda enemy_x_array,x
    sec
    sbc $80         ; Player X
    bpl positive_distance
    eor #$ff        ; Make positive (simple abs)
    clc
    adc #$01
    
positive_distance:
    cmp #$18        ; Minimum shooting distance (24 pixels)
    bcc back_away   ; Too close, back away
    cmp #$30        ; Maximum shooting distance (48 pixels)  
    bcs move_closer ; Too far, move closer
    
    ; Perfect distance - shoot!
    jsr enemy_fire_projectile
    rts
    
back_away:
    ; Move away from player
    lda $80         ; Player X
    cmp enemy_x_array,x
    bcc move_right_away
    
    ; Move left (away from player)
    lda #$ff
    sta enemy_velocity_x_array,x
    rts
    
move_right_away:
    ; Move right (away from player)
    lda #$01
    sta enemy_velocity_x_array,x
    rts
    
move_closer:
    ; Move toward player (but not too close)
    lda $80         ; Player X
    cmp enemy_x_array,x
    bcc move_left_closer
    
    ; Move right toward player
    lda #$01
    sta enemy_velocity_x_array,x
    rts
    
move_left_closer:
    ; Move left toward player
    lda #$ff
    sta enemy_velocity_x_array,x
    rts

ai_simple_patrol:
    ; Simple back-and-forth patrol
    lda enemy_ai_state_array,x
    bne patrol_left
    
patrol_right:
    ; Moving right
    lda #$01
    sta enemy_velocity_x_array,x
    
    ; Check if reached right boundary
    lda enemy_x_array,x
    cmp #$25        ; Right patrol limit
    bcc patrol_done
    
    ; Switch direction
    lda #$01
    sta enemy_ai_state_array,x
    rts
    
patrol_left:
    ; Moving left
    lda #$ff
    sta enemy_velocity_x_array,x
    
    ; Check if reached left boundary
    lda enemy_x_array,x
    cmp #$05        ; Left patrol limit
    bcs patrol_done
    
    ; Switch direction
    lda #$00
    sta enemy_ai_state_array,x
    
patrol_done:
    rts

; Process swarm spawning system
process_swarm_spawning:
    ; Check if we should spawn new enemies
    lda active_enemy_count
    cmp max_enemy_count
    bcs no_spawn       ; At maximum, don't spawn
    
    ; Check spawn timer
    dec spawn_timer
    bne no_spawn
    
    ; Reset spawn timer
    lda #$3c        ; 60 frames = 1 second at 60fps
    sta spawn_timer
    
    ; Spawn new enemy
    jsr spawn_new_enemy
    
no_spawn:
    rts

; Spawn new enemy in available slot
spawn_new_enemy:
    ; Find empty enemy slot
    ldx #$00
find_empty_slot:
    lda enemy_active_array,x
    beq found_empty_slot
    inx
    cpx #$08
    bne find_empty_slot
    rts             ; No empty slots
    
found_empty_slot:
    ; Activate this enemy slot
    lda #$01
    sta enemy_active_array,x
    
    ; Set spawn position (random edge of screen)
    jsr generate_spawn_position
    
    ; Set enemy type (random)
    jsr generate_enemy_type
    
    ; Initialize enemy stats
    lda #$03        ; 3 health points
    sta enemy_health_array,x
    lda #$00        ; Initial AI state
    sta enemy_ai_state_array,x
    
    ; Increment active enemy count
    inc active_enemy_count
    
    rts

; Update projectile swarm using array management
update_projectile_swarm:
    ; Process all active projectiles
    ldx #$00        ; Start with projectile 0
    
update_projectile_loop:
    ; Check if this projectile slot is active
    lda projectile_active_array,x
    beq next_projectile
    
    ; Update projectile position
    lda projectile_x_array,x
    clc
    adc projectile_velocity_x_array,x
    sta projectile_x_array,x
    
    lda projectile_y_array,x  
    clc
    adc projectile_velocity_y_array,x
    sta projectile_y_array,x
    
    ; Check projectile boundaries
    jsr check_projectile_boundaries
    
next_projectile:
    inx
    cpx #$10        ; Check all 16 projectile slots
    bne update_projectile_loop
    
    rts

; Process swarm collision detection
process_swarm_collisions:
    ; Check player vs all enemies
    jsr check_player_vs_enemy_swarm
    
    ; Check player projectiles vs all enemies
    jsr check_player_projectiles_vs_enemies
    
    ; Check enemy projectiles vs player
    jsr check_enemy_projectiles_vs_player
    
    ; Check projectile vs projectile (optional advanced feature)
    ; jsr check_projectile_vs_projectile_collisions
    
    rts

; Check player collisions with enemy swarm
check_player_vs_enemy_swarm:
    ldx #$00        ; Enemy index
    
check_enemy_collision_loop:
    ; Skip inactive enemies
    lda enemy_active_array,x
    beq next_enemy_collision
    
    ; Check collision between player and this enemy
    lda $80         ; Player X
    sec
    sbc enemy_x_array,x
    bpl player_enemy_x_positive
    eor #$ff        ; Make positive
    clc
    adc #$01
    
player_enemy_x_positive:
    cmp #$03        ; Collision distance (3 pixels)
    bcs next_enemy_collision
    
    ; Check Y distance
    lda $81         ; Player Y
    sec
    sbc enemy_y_array,x
    bpl player_enemy_y_positive
    eor #$ff        ; Make positive
    clc
    adc #$01
    
player_enemy_y_positive:
    cmp #$03        ; Collision distance
    bcs next_enemy_collision
    
    ; Collision detected!
    jsr handle_player_enemy_collision
    
next_enemy_collision:
    inx
    cpx #$08
    bne check_enemy_collision_loop
    
    rts

; Handle player-enemy collision
handle_player_enemy_collision:
    ; X contains enemy index
    ; Damage player
    dec $84         ; Reduce player lives
    
    ; Create explosion effect
    jsr create_collision_explosion
    
    ; Destroy enemy
    lda #$00
    sta enemy_active_array,x
    dec active_enemy_count
    
    ; Screen flash effect
    jsr create_screen_flash
    
    ; Audio feedback
    jsr play_collision_sound
    
    rts

; Enemy fire projectile system
enemy_fire_projectile:
    ; X contains enemy index
    ; Find empty projectile slot
    ldy #$00
find_projectile_slot:
    lda projectile_active_array,y
    beq found_projectile_slot
    iny
    cpy #$10
    bne find_projectile_slot
    rts             ; No empty slots
    
found_projectile_slot:
    ; Activate projectile
    lda #$01
    sta projectile_active_array,y
    
    ; Set projectile position to enemy position
    lda enemy_x_array,x
    sta projectile_x_array,y
    lda enemy_y_array,x
    sta projectile_y_array,y
    
    ; Set projectile type as enemy projectile
    lda #$02        ; Enemy projectile type
    sta projectile_type_array,y
    lda x           ; Store which enemy fired it
    sta projectile_owner_array,y
    
    ; Calculate velocity toward player
    lda $80         ; Player X
    sec
    sbc enemy_x_array,x
    bpl enemy_fire_x_positive
    
    ; Negative direction
    lda #$ff        ; -1 velocity
    sta projectile_velocity_x_array,y
    jmp enemy_fire_calc_y
    
enemy_fire_x_positive:
    lda #$01        ; +1 velocity
    sta projectile_velocity_x_array,y
    
enemy_fire_calc_y:
    lda $81         ; Player Y
    sec
    sbc enemy_y_array,x
    bpl enemy_fire_y_positive
    
    ; Negative direction
    lda #$ff        ; -1 velocity
    sta projectile_velocity_y_array,y
    rts
    
enemy_fire_y_positive:
    lda #$01        ; +1 velocity
    sta projectile_velocity_y_array,y
    rts

; Display all entities in swarm
update_swarm_effects:
    ; Display all active enemies
    jsr display_enemy_swarm
    
    ; Display all active projectiles
    jsr display_projectile_swarm
    
    ; Display swarm-specific effects
    jsr display_swarm_visual_effects
    
    rts

; Display enemy swarm on screen
display_enemy_swarm:
    ldx #$00        ; Enemy index
    
display_enemy_loop:
    ; Skip inactive enemies
    lda enemy_active_array,x
    beq next_enemy_display
    
    ; Get enemy position
    lda enemy_x_array,x
    sta temp_x
    lda enemy_y_array,x
    sta temp_y
    
    ; Get enemy type for display character
    lda enemy_type_array,x
    clc
    adc #$40        ; Base enemy character (@ABC...)
    sta temp_char
    
    ; Display enemy
    jsr display_character_at_position
    
next_enemy_display:
    inx
    cpx #$08
    bne display_enemy_loop
    
    rts

; Data arrays for entity swarm management
enemy_x_array:              .res 8      ; Enemy X positions
enemy_y_array:              .res 8      ; Enemy Y positions  
enemy_active_array:         .res 8      ; Enemy active flags
enemy_type_array:           .res 8      ; Enemy type IDs
enemy_health_array:         .res 8      ; Enemy health points
enemy_ai_state_array:       .res 8      ; Enemy AI states
enemy_velocity_x_array:     .res 8      ; Enemy X velocities
enemy_velocity_y_array:     .res 8      ; Enemy Y velocities

projectile_x_array:         .res 16     ; Projectile X positions
projectile_y_array:         .res 16     ; Projectile Y positions
projectile_active_array:    .res 16     ; Projectile active flags
projectile_type_array:      .res 16     ; Projectile type IDs
projectile_velocity_x_array: .res 16    ; Projectile X velocities
projectile_velocity_y_array: .res 16    ; Projectile Y velocities
projectile_owner_array:     .res 16     ; Projectile owner IDs

; Entity management variables
active_enemy_count:         .res 1      ; Current active enemy count
active_projectile_count:    .res 1      ; Current active projectile count
active_effect_count:        .res 1      ; Current active effect count
max_enemy_count:            .res 1      ; Maximum enemy limit
max_projectile_count:       .res 1      ; Maximum projectile limit
max_effect_count:           .res 1      ; Maximum effect limit
spawn_timer:                .res 1      ; Enemy spawn timer
temp_x:                     .res 1      ; Temporary X coordinate
temp_y:                     .res 1      ; Temporary Y coordinate
temp_char:                  .res 1      ; Temporary character
```
