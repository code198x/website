# Optimization Techniques: Speed and Efficiency
**Learn to make your C64 programs faster and smaller**

Optimization is the art of making programs run faster or use less memory. On the C64's 1MHz processor, every cycle counts. Understanding common optimization techniques helps you create smooth, responsive games that push the hardware to its limits. The key is knowing when to optimize for speed versus size, and which techniques work best for different situations.

## What You'll Learn

In this lesson, you'll discover:
- **Speed vs size tradeoffs** - When to prioritize each
- **Loop unrolling** - Trading memory for speed
- **Lookup tables** - Precomputing for performance
- **Page alignment** - Avoiding page boundary penalties
- **Common optimizations** - Practical techniques that work
- **Profiling** - Measuring to find bottlenecks

## Speed Optimization Basics

Make your code run faster with these techniques:

```asm
; C64 Memory Layout (64KB total)
;
; $0000-$00FF: Zero Page (256 bytes) - FASTEST ACCESS
;   $00-$01: Processor port (memory configuration)
;   $02-$8F: BASIC/KERNAL variables (avoid in games)
;   $90-$FF: Free for user programs (112 bytes available)
;
; $0100-$01FF: Stack (256 bytes)
;   Hardware stack for subroutine calls
;
; $0200-$02FF: Input buffer and vectors (256 bytes)
;   Some areas can be used carefully
;
; $0300-$03FF: Cassette buffer (256 bytes)
;   Usually safe to use for game data
;
; $0400-$07FF: Default screen RAM (1024 bytes)
;   Character codes displayed on screen
;
; $0800-$9FFF: RAM for programs (38KB)
;   Your main program code goes here
;
; $A000-$BFFF: BASIC ROM (8KB) - can be banked out
; $C000-$CFFF: RAM (4KB) - available when BASIC banked out
; $D000-$DFFF: I/O area (4KB) - VIC, SID, CIA chips
; $E000-$FFFF: KERNAL ROM (8KB) - can be banked out

; Memory configuration register

; Set memory configuration
set_memory_config:
    ; Input: A = configuration byte
    sta MEMORY_CONFIG

; Zero page allocation strategy

; Fast temporary variables (frequently used)
ZP_TEMP1        = $90    ; General purpose temp

; System addresses (hardware registers)
SCREEN_RAM      = $0400  ; Default screen memory
COLOR_RAM       = $D800  ; Color memory
SPRITE_POINTERS = $07F8  ; Sprite pointers (screen + $3F8)

VIC_BASE        = $D000  ; VIC-II chip base
VIC_SPRITES     = $D000  ; Sprite coordinates start
VIC_CONTROL1    = $D011  ; Control register 1

VIC_BORDER      = $D020  ; Border color
```

## Zero Page Optimization

Maximize speed with smart zero page usage:

```asm
; Zero page operations are faster and smaller
; Compare these equivalent operations:

; SLOW: Regular memory access (3 bytes, 4 cycles)
regular_load:
    lda $1234       ; Load from regular memory
    sta $5678       ; Store to regular memory

; FAST: Zero page access (2 bytes, 3 cycles each)
zeropage_load:
    lda ZP_TEMP1    ; Load from zero page
    sta ZP_TEMP2    ; Store to zero page

; Indirect addressing through zero page pointers
; Essential for dynamic memory access

; Set up pointer in zero page
setup_pointer:
    lda #<target_address    ; Low byte
    sta ZP_PTR1
    lda #>target_address    ; High byte
    sta ZP_PTR1+1

    ; Now access memory indirectly
    ldy #0
    lda (ZP_PTR1),y         ; Load from address in pointer

    ; Or with offset
    ldy #5
    sta (ZP_PTR1),y         ; Store 5 bytes from pointer

    rts

; Efficient 16-bit arithmetic using zero page
add_16bit:
    ; Add value in X/Y to 16-bit value at ZP_PTR1
    txa                     ; X = low byte to add
    clc
    adc ZP_PTR1             ; Add to low byte
    sta ZP_PTR1

    tya                     ; Y = high byte to add
    adc ZP_PTR1+1           ; Add with carry to high byte
    sta ZP_PTR1+1

    rts

; Fast memory copy using zero page pointers
fast_memory_copy:
    ; Copy from (ZP_PTR1) to (ZP_PTR2), X bytes
    ; Input: source in ZP_PTR1, dest in ZP_PTR2, count in X

    ldy #0                  ; Index
copy_loop:
    lda (ZP_PTR1),y         ; Load from source
    sta (ZP_PTR2),y         ; Store to destination
    iny                     ; Next byte
    dex                     ; Decrement count
    bne copy_loop           ; Continue if not zero

    rts

; Zero page variable pool management
; Allocate/deallocate temporary variables

ZP_POOL_START   = $A0       ; Start of variable pool
ZP_POOL_SIZE    = 32        ; 32 bytes available
zp_pool_used:   .byte 0     ; Bitmask of used bytes

; Allocate zero page variable
alloc_zp_var:
    ; Output: A = zero page address, or $FF if failed

    ldx #0                  ; Start searching from 0
find_free_zp:
    lda zp_pool_used
    and zp_bit_masks,x      ; Check if bit X is set
    beq found_free_zp       ; Bit clear = available

    inx
    cpx #ZP_POOL_SIZE
    bne find_free_zp

    ; No free variables
    lda #$FF
    rts

found_free_zp:
    ; Mark as used
    lda zp_pool_used
    ora zp_bit_masks,x      ; Set bit X
    sta zp_pool_used

    ; Return address
    txa
    clc
    adc #ZP_POOL_START
    rts

; Free zero page variable
free_zp_var:
    ; Input: A = zero page address to free

    ; Convert address to bit index
    sec
    sbc #ZP_POOL_START
    tax

    ; Clear bit
    lda zp_bit_masks,x
    eor #$FF                ; Invert mask
    and zp_pool_used        ; Clear the bit
    sta zp_pool_used

    rts

; Bit masks for variable allocation
zp_bit_masks:
    .byte %00000001, %00000010, %00000100, %00001000
    .byte %00010000, %00100000, %01000000, %10000000

target_address: .word $1234
```

## Code Organization Strategies

Structure your program for maintainability and efficiency:

```asm
; Game code organization template
; Organize by functionality, not arbitrary divisions

;=============================================================================
; PROGRAM HEADER - Basic stub and initialization
;=============================================================================
* = $0801                   ; Start after zero page/stack/vectors

basic_stub:
    !byte $0B,$08,$0A,$00,$9E
    !text "2061"            ; SYS 2061
    !byte $00,$00,$00

* = $080D                   ; 2061 decimal
program_start:
    jmp main_init           ; Jump to main initialization

;=============================================================================
; MAIN GAME LOOP - Core game logic
;=============================================================================
main_init:
    jsr init_hardware       ; Initialize VIC, SID, CIA
    jsr init_memory         ; Set up memory layout
    jsr init_game_data      ; Initialize game variables
    jsr init_interrupts     ; Set up custom interrupts

main_loop:
    jsr wait_frame          ; Sync to 50/60 Hz
    jsr update_input        ; Read joystick/keyboard
    jsr update_game_logic   ; Core game updates
    jsr update_graphics     ; Render everything
    jsr update_sound        ; Handle music/SFX
    jmp main_loop

;=============================================================================
; INPUT SYSTEM - Centralized input handling
;=============================================================================
update_input:
    jsr read_joystick       ; Read joystick ports
    jsr read_keyboard       ; Read keyboard matrix
    jsr process_input       ; Convert to game actions
    rts

read_joystick:
    lda CIA1_PORTA          ; Read joystick port 2
    sta joystick_current

    ; Calculate new button presses (not held)
    eor joystick_previous   ; XOR with previous state
    and joystick_current    ; AND with current (new presses)
    sta joystick_pressed

    ; Update previous state
    lda joystick_current
    sta joystick_previous
    rts

;=============================================================================
; GAME LOGIC - Core gameplay systems
;=============================================================================
update_game_logic:
    jsr update_player       ; Player movement/actions
    jsr update_enemies      ; Enemy AI and movement
    jsr update_projectiles  ; Bullets, missiles, etc.
    jsr update_collectibles ; Power-ups, coins, etc.
    jsr update_physics      ; Collisions, gravity, etc.
    jsr update_game_state   ; Level progress, win/lose
    rts

;=============================================================================
; GRAPHICS SYSTEM - All rendering code
;=============================================================================
update_graphics:
    jsr clear_sprites       ; Reset sprite positions
    jsr render_background   ; Tiles, scrolling background
    jsr render_sprites      ; Player, enemies, objects
    jsr render_ui           ; Score, health, etc.
    jsr update_screen_fx    ; Special effects
    rts

;=============================================================================
; SOUND SYSTEM - Music and sound effects
;=============================================================================
update_sound:
    jsr update_music        ; Background music
    jsr update_sfx          ; Sound effects
    jsr process_sound_queue ; Play queued sounds
    rts

;=============================================================================
; UTILITY ROUTINES - Commonly used functions
;=============================================================================
wait_frame:
    lda #$FB                ; Wait for raster line 251
wait_raster:
    cmp VIC_CONTROL1+$12    ; Compare with current raster
    bne wait_raster
    rts

multiply_by_40:
    ; Multiply A by 40 (screen row width)
    ; Result in A/X (low/high)

    sta ZP_TEMP1            ; Save original
    asl                     ; * 2
    asl                     ; * 4
    asl                     ; * 8
    tax                     ; Save * 8 in X

    asl                     ; * 16
    asl                     ; * 32
    clc
    adc ZP_TEMP1            ; + original = * 33
    clc
    adc ZP_TEMP1            ; + original = * 34
    clc
    adc ZP_TEMP1            ; + original = * 35
    clc
    adc ZP_TEMP1            ; + original = * 36
    clc
    adc ZP_TEMP1            ; + original = * 37
    clc
    adc ZP_TEMP1            ; + original = * 38
    clc
    adc ZP_TEMP1            ; + original = * 39
    clc
    adc ZP_TEMP1            ; + original = * 40

    ; Handle carry
    bcc mult_done
    inx                     ; Increment high byte

mult_done:
    rts

;=============================================================================
; DATA SECTION - Game data and tables
;=============================================================================

; Game state variables
game_state:         .byte 0     ; Current game state
level_number:       .byte 1     ; Current level
score:              .byte 0,0,0 ; BCD score (3 bytes)
lives:              .byte 3     ; Player lives
health:             .byte 100   ; Player health

; Input state
joystick_current:   .byte 0     ; Current joystick state
joystick_previous:  .byte 0     ; Previous joystick state
joystick_pressed:   .byte 0     ; Newly pressed buttons

; Animation frames
sprite_frames:      .byte 64,65,66,67  ; Player animation
enemy_frames:       .byte 68,69,70,71  ; Enemy animation

; Sound effect queue
sfx_queue:          !fill 8, 0  ; 8 sound effect slots
music_track:        .byte 1     ; Current music track

;=============================================================================
; LOOKUP TABLES - Precomputed values for speed
;=============================================================================

; Screen row addresses (low bytes)
screen_rows_lo:
    .byte <($0400 + 0*40), <($0400 + 1*40), <($0400 + 2*40)
    .byte <($0400 + 3*40), <($0400 + 4*40), <($0400 + 5*40)
    ; ... continue for all 25 rows

; Screen row addresses (high bytes)
screen_rows_hi:
    .byte >($0400 + 0*40), >($0400 + 1*40), >($0400 + 2*40)
    .byte >($0400 + 3*40), >($0400 + 4*40), >($0400 + 5*40)
    ; ... continue for all 25 rows

; Fast screen address calculation
get_screen_address:
    ; Input: A = row (0-24), X = column (0-39)
    ; Output: Address in ZP_PTR1

    tay                     ; Y = row
    lda screen_rows_lo,y    ; Get row address low byte
    clc
    adc x                   ; Add column
    sta ZP_PTR1

    lda screen_rows_hi,y    ; Get row address high byte
    adc #0                  ; Add carry
    sta ZP_PTR1+1

    rts
```

## Memory Banking & Advanced Techniques

Access more memory and optimize layout:

```asm
; Memory banking - access more than 64K
; The C64 can bank out ROMs to access underlying RAM

; Bank switching for extra RAM access
switch_to_ram:
    ; Turn off BASIC and KERNAL ROMs
    lda #CONFIG_ALLRAM      ; All RAM configuration
    sta MEMORY_CONFIG

    ; Now $A000-$BFFF and $E000-$FFFF are RAM
    ; But no KERNAL routines available!
    rts

switch_to_normal:
    ; Restore normal memory map
    lda #CONFIG_DEFAULT     ; BASIC + KERNAL visible
    sta MEMORY_CONFIG
    rts

; Safe ROM banking - preserve important vectors
safe_bank_switch:
    ; Save interrupt vectors before switching
    sei                     ; Disable interrupts

    lda $0314               ; IRQ vector low
    sta saved_irq_lo
    lda $0315               ; IRQ vector high
    sta saved_irq_hi

    ; Switch to RAM
    jsr switch_to_ram

    ; Your code here using extra RAM

    ; Restore vectors and memory
    jsr switch_to_normal

    lda saved_irq_lo
    sta $0314
    lda saved_irq_hi
    sta $0315

    cli                     ; Re-enable interrupts
    rts

; VIC-II memory banking (16K banks)
; Controls where VIC reads screen/character data

set_vic_bank:
    ; Input: A = bank number (0-3)
    ; Bank 0: $0000-$3FFF
    ; Bank 1: $4000-$7FFF
    ; Bank 2: $8000-$BFFF
    ; Bank 3: $C000-$FFFF (default)

    eor #%00000011          ; Invert bits (CIA2 inverts)
    sta CIA2_PORTA          ; Set VIC bank
    rts

; Character set banking
set_character_set:
    ; Input: A = character set location
    ; 0 = $0000-$07FF, 1 = $0800-$0FFF, etc.

    and #%00001110          ; Mask valid bits
    asl                     ; Shift to correct position

    sta ZP_TEMP1            ; Save character bits

    lda VIC_MEMORY          ; Get current memory setup
    and #%11110001          ; Clear character bits
    ora ZP_TEMP1            ; Add new character bits
    sta VIC_MEMORY          ; Set new configuration

    rts

; Dynamic memory allocation system
; Simple heap manager for variable-sized blocks

HEAP_START      = $C000     ; Start of heap (when BASIC ROM off)
HEAP_SIZE       = $2000     ; 8K heap size
BLOCK_HEADER    = 3         ; 3 bytes per block header

; Block header format:
; Byte 0-1: Block size (16-bit)
; Byte 2: Status (0=free, 1=used)

heap_initialized: .byte 0

init_heap:
    lda heap_initialized
    bne heap_already_init

    ; Initialize first free block
    lda #<(HEAP_SIZE-BLOCK_HEADER)
    sta HEAP_START          ; Size low byte
    lda #>(HEAP_SIZE-BLOCK_HEADER)
    sta HEAP_START+1        ; Size high byte
    lda #0
    sta HEAP_START+2        ; Status = free

    lda #1
    sta heap_initialized

heap_already_init:
    rts

; Allocate memory block
malloc:
    ; Input: A/X = size needed (low/high)
    ; Output: ZP_PTR1 = allocated address, or $0000 if failed

    sta requested_size      ; Save requested size
    stx requested_size+1

    ; Search for free block
    lda #<HEAP_START
    sta current_block
    lda #>HEAP_START
    sta current_block+1

search_blocks:
    ; Check if we've reached end of heap
    lda current_block
    cmp #<(HEAP_START+HEAP_SIZE)
    bne not_end_of_heap
    lda current_block+1
    cmp #>(HEAP_START+HEAP_SIZE)
    beq allocation_failed

not_end_of_heap:
    ; Check block status
    ldy #2                  ; Status byte offset
    lda (current_block),y
    bne block_used          ; Skip if used

    ; Check block size
    ldy #0
    lda (current_block),y   ; Size low byte
    cmp requested_size
    ldy #1
    lda (current_block),y   ; Size high byte
    sbc requested_size+1    ; Compare high bytes
    bcc block_too_small     ; Block smaller than needed

    ; Found suitable block
    jmp allocate_block

block_used:
block_too_small:
    ; Move to next block
    ; current_block += BLOCK_HEADER + block_size
    ldy #0
    lda (current_block),y   ; Block size low
    clc
    adc #BLOCK_HEADER
    clc
    adc current_block
    sta current_block

    ldy #1
    lda (current_block),y   ; Block size high
    adc #0
    adc current_block+1
    sta current_block+1

    jmp search_blocks

allocate_block:
    ; Mark block as used
    ldy #2
    lda #1
    sta (current_block),y

    ; Return address (after header)
    lda current_block
    clc
    adc #BLOCK_HEADER
    sta ZP_PTR1
    lda current_block+1
    adc #0
    sta ZP_PTR1+1
    rts

allocation_failed:
    ; Return NULL pointer
    lda #0
    sta ZP_PTR1
    sta ZP_PTR1+1
    rts

; Free memory block
free:
    ; Input: ZP_PTR1 = address to free

    ; Calculate block header address
    lda ZP_PTR1
    sec
    sbc #BLOCK_HEADER
    sta current_block
    lda ZP_PTR1+1
    sbc #0
    sta current_block+1

    ; Mark as free
    ldy #2
    lda #0
    sta (current_block),y

    rts

; Working variables
saved_irq_lo:     .byte 0
saved_irq_hi:     .byte 0
requested_size:   .word 0
current_block:    .word 0
```

## Memory-Mapped Game Architecture

Design games that efficiently use memory mapping:

```asm
; Game architecture using memory mapping
; Different screens/levels in different memory areas

; Memory layout for multi-level game
LEVEL_DATA_BASE = $4000     ; Base for level data (16K bank 1)
SPRITE_DATA     = $6000     ; Sprite graphics
MUSIC_DATA      = $7000     ; Music and sound data

; Screen configurations for different areas
SCREEN_TITLE    = $4400     ; Title screen
SCREEN_GAME     = $0400     ; Main game screen (default)
SCREEN_HISCORE  = $4C00     ; High score screen

; Character set configurations
CHARSET_MENU    = $4800     ; Menu graphics
CHARSET_GAME    = $5000     ; Game graphics
CHARSET_UI      = $5800     ; UI elements

; Level data structure in memory
; Each level: 1K screen + 1K color + sprites + music

init_memory_layout:
    ; Set VIC bank to bank 1 ($4000-$7FFF)
    lda #1
    jsr set_vic_bank

    ; Configure character set
    lda #0                  ; Use charset at $4800 (relative to bank)
    jsr set_character_set

    rts

; Load level into memory
load_level_data:
    ; Input: A = level number

    ; Calculate level data address
    asl                     ; * 2 for word index
    tax
    lda level_pointers,x
    sta source_ptr
    lda level_pointers+1,x
    sta source_ptr+1

    ; Copy level screen data
    lda #<SCREEN_GAME
    sta dest_ptr
    lda #>SCREEN_GAME
    sta dest_ptr+1

    ldx #4                  ; Copy 4 blocks (1K)
    jsr copy_memory_blocks

    ; Copy level color data
    lda #<COLOR_RAM
    sta dest_ptr
    lda #>COLOR_RAM
    sta dest_ptr+1

    ldx #4                  ; Copy 4 blocks (1K)
    jsr copy_memory_blocks

    rts

; Fast memory copy (256 bytes per block)
copy_memory_blocks:
    ; Input: source_ptr, dest_ptr, X = number of blocks

copy_block_loop:
    stx blocks_remaining

    ldy #0
copy_byte_loop:
    lda (source_ptr),y
    sta (dest_ptr),y
    iny
    bne copy_byte_loop      ; Copy 256 bytes

    ; Move to next block
    inc source_ptr+1        ; Next source block
    inc dest_ptr+1          ; Next dest block

    ldx blocks_remaining
    dex
    bne copy_block_loop

    rts

; Screen switching for different game states
switch_to_title:
    ; Switch VIC to read from title screen area
    lda VIC_MEMORY
    and #%00001111          ; Clear screen bits
    ora #%00010000          ; Screen at $4400 (bank offset $400)
    sta VIC_MEMORY

    ; Load title character set
    lda #0                  ; Charset at $4800
    jsr set_character_set

    rts

switch_to_game:
    ; Switch to game screen
    lda VIC_MEMORY
    and #%00001111
    ora #%00000000          ; Screen at $4000 (bank offset $000)
    sta VIC_MEMORY

    ; Load game character set
    lda #2                  ; Charset at $5000
    jsr set_character_set

    rts

; Overlay system - temporary screens over game
show_pause_overlay:
    ; Save current screen configuration
    lda VIC_MEMORY
    sta saved_vic_memory

    ; Switch to overlay screen
    lda VIC_MEMORY
    and #%00001111
    ora #%11000000          ; Screen at $7C00
    sta VIC_MEMORY

    ; Draw pause interface
    jsr draw_pause_interface

    rts

hide_pause_overlay:
    ; Restore previous screen
    lda saved_vic_memory
    sta VIC_MEMORY
    rts

; Dynamic sprite allocation
; Manage sprite graphics in memory

SPRITE_BLOCK_SIZE = 64      ; 64 bytes per sprite

allocate_sprite_gfx:
    ; Input: A = number of sprites needed
    ; Output: X = starting sprite number, or $FF if failed

    sta sprites_needed

    ; Find consecutive free sprites
    ldx #0                  ; Start searching from sprite 0

find_sprites:
    lda sprite_allocated,x
    bne sprite_used

    ; Check if we have enough consecutive sprites
    jsr check_consecutive_sprites
    bcc found_sprites

sprite_used:
    inx
    cpx #8                  ; Only 8 sprites available
    bne find_sprites

    ; No space found
    ldx #$FF
    rts

found_sprites:
    ; Mark sprites as allocated
    ldy sprites_needed
mark_allocated:
    lda #1
    sta sprite_allocated,x
    inx
    dey
    bne mark_allocated

    ; Return starting sprite number
    txa
    sec
    sbc sprites_needed      ; X now points past allocated block
    tax
    rts

check_consecutive_sprites:
    ; Check if sprites_needed consecutive sprites are free
    ; Input: X = starting sprite, sprites_needed set
    ; Output: Carry clear if enough space

    ldy sprites_needed

check_loop:
    cpx #8
    bcs not_enough          ; Past end of sprite table

    lda sprite_allocated,x
    bne not_enough          ; Sprite in use

    inx
    dey
    bne check_loop

    clc                     ; Success
    rts

not_enough:
    sec                     ; Failure
    rts

; Data tables
level_pointers:
    .word level_1_data, level_2_data, level_3_data

sprite_allocated:   !fill 8, 0     ; Sprite allocation table
saved_vic_memory:   .byte 0
```

### Network Optimization for Neon Nexus

Apply optimization techniques to network code:

```asm
; Optimized packet processing for Neon Nexus

; Page-aligned packet buffer for speed
        * = $2000       ; Page boundary
packet_buffer:
        !fill 256, 0

; Fast packet scanner (unrolled)
scan_packets_fast:
        ldx #0
        ; Unroll 4x for speed
!for i, 0, 63 {
        lda packet_buffer+i*4
        beq +           ; Skip empty
        jsr process_packet_fast
+       lda packet_buffer+i*4+1
        beq +
        jsr process_packet_fast
+       lda packet_buffer+i*4+2
        beq +
        jsr process_packet_fast  
+       lda packet_buffer+i*4+3
        beq +
        jsr process_packet_fast
+
}
        rts

; Lookup table for routing (faster than calculation)
route_lookup:
        ; [source][dest] = next_hop
        !byte 0,1,2,1,2,3,1,2  ; From node 0
        !byte 1,0,1,2,3,2,3,1  ; From node 1
        !byte 2,1,0,1,2,3,2,3  ; From node 2
        ; etc...

; Fast route calculation using table
get_route_fast:
        ; Input: A = source, X = dest
        ; Output: A = next hop
        
        ; Calculate table index
        asl
        asl
        asl             ; Source * 8
        stx temp
        clc
        adc temp        ; + dest
        tax
        
        ; Lookup result
        lda route_lookup,x
        rts

; Optimized packet counter
count_packets_fast:
        ; Count set bits in packet_active flags
        ldx #0          ; Counter
        lda packet_active
        
        ; Kernighan's algorithm
count_loop:
        beq count_done
        inx
        tay
        dey
        tya
        and packet_active
        sta packet_active
        jmp count_loop
        
count_done:
        txa             ; Return count
        rts

; Variables
temp: !byte 0
packet_active: !byte 0
```

**Optimized Performance:** Your Neon Nexus network code now runs significantly faster! Page-aligned buffers eliminate boundary checks, unrolled loops reduce overhead, and lookup tables replace expensive calculations.
