# Text Display: Messages, Menus, and PETSCII Art
**Professional text output for games and applications**

Every game needs text—for menus, scores, instructions, dialog. The C64 provides powerful text display capabilities through its KERNAL routines and the unique PETSCII character set. While you've learned to write directly to screen memory, sometimes you need the convenience of formatted output, cursor control, and the special graphic characters that make C64 displays distinctive.

PETSCII (PET Standard Code of Information Interchange) isn't just text—it's a complete graphic system. Those box-drawing characters you see in C64 menus? PETSCII. The card symbols in poker games? PETSCII. The elaborate title screens drawn entirely with characters? All PETSCII. Games like "Zak McKracken" used it for interfaces, "The Bard's Tale" for menus, and "Elite" for its legendary wireframe scanner.

This lesson teaches you professional text display techniques used in every C64 application.

## KERNAL Text Output

### The CHROUT Routine

The fundamental text output routine at $FFD2:

<CodeBlock title="Basic KERNAL Text Output" language="asm" code={`; Using KERNAL CHROUT for text display
; The professional way to output text

        * = $0801
        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

        * = $080d
start:
        ; Clear screen using PETSCII code
        lda #147        ; Clear screen character
        jsr $ffd2       ; CHROUT
        
        ; Print a simple message
        jsr print_hello
        
        ; Position cursor
        jsr position_cursor
        
        ; Print with colors
        jsr colored_text
        
        ; Draw a box
        jsr draw_menu_box
        
        rts

print_hello:
        ldx #0
hello_loop:
        lda hello_msg,x
        beq done_hello  ; Zero terminator
        jsr $ffd2       ; Print character
        inx
        jmp hello_loop
done_hello:
        rts

hello_msg:
        !text "HELLO FROM KERNAL!"
        !byte 13        ; Carriage return
        !byte 0         ; Terminator

position_cursor:
        ; Position cursor at column 10, row 5
        ; Using PETSCII cursor movement codes
        
        ; Home cursor first
        lda #19         ; HOME
        jsr $ffd2
        
        ; Move down 5 rows
        ldy #5
down_loop:
        lda #17         ; Cursor down
        jsr $ffd2
        dey
        bne down_loop
        
        ; Move right 10 columns
        ldy #10
right_loop:
        lda #29         ; Cursor right
        jsr $ffd2
        dey
        bne right_loop
        
        ; Now print at this position
        ldx #0
pos_loop:
        lda positioned_text,x
        beq done_pos
        jsr $ffd2
        inx
        jmp pos_loop
done_pos:
        rts

positioned_text:
        !text "POSITIONED TEXT"
        !byte 0`} />

<CodeBlock title="Creating Boxes and Borders" language="asm" code={`draw_menu_box:

        ; Using PETSCII box drawing characters

        jsr $ffd2

        jsr $ffd2

        ldx #20
top_line:
        lda #195        ; ═ horizontal
        jsr $ffd2

        bne top_line
        
        ; Top right corner
        lda #174        ; ╗ corner
        jsr $ffd2
        
        ; Draw sides
        ldy #5
draw_sides:
        ; Move to next line
        lda #17         ; Down
        jsr $ffd2
        lda #157        ; Left
        jsr $ffd2
        
        ; Right side
        lda #221        ; ║ vertical
        jsr $ffd2

        ldx #21
back:   lda #157        ; Left
        jsr $ffd2

        jsr $ffd2

        jsr $ffd2

bot_line:

        jsr clear_screen
        jsr draw_menu_frame
        jsr display_menu_title

        jsr display_instructions

draw_menu_frame:

## String Handling

### Efficient String Output

<CodeBlock title="String Management Routines" language="asm" code={`; Flexible string printing system

print_string_at:
        ; Input: A/X = string address (lo/hi)
        ;        Y = position index
        
        sta str_ptr
        stx str_ptr+1
        
        ; Look up position
        tya
        asl
        tay
        lda position_table,y
        tax
        lda position_table+1,y
        tay
        jsr goto_xy
        
        ; Print string
        ldy #0
str_loop:
        lda (str_ptr),y
        beq str_done
        jsr $ffd2
        iny
        jmp str_loop
str_done:
        rts

goto_xy:
        ; Input: X = column, Y = row
        ; Position cursor at X,Y
        
        lda #19         ; Home
        jsr $ffd2
        
        ; Move down Y rows
        cpy #0
        beq done_y
move_y: lda #17         ; Down
        jsr $ffd2
        dey
        bne move_y
done_y:
        
        ; Move right X columns
        cpx #0
        beq done_x
move_x: lda #29         ; Right
        jsr $ffd2
        dex
        bne move_x
done_x:
        rts

; Position lookup table (x,y pairs)
position_table:
        !byte 0,0       ; Position 0: top-left
        !byte 20,0      ; Position 1: top-center
        !byte 0,12      ; Position 2: middle-left
        !byte 15,12     ; Position 3: center
        !byte 0,24      ; Position 4: bottom-left
        
str_ptr: !word 0`} />

### Number Display

<CodeBlock title="Displaying Numbers" language="asm" code={`display_decimal:
        ; Display 16-bit number in decimal
        ; Input: number_lo/hi = number to display
        
        ; Convert to decimal digits
        jsr binary_to_decimal
        
        ; Display digits
        ldx #4          ; 5 digits max
        ldy #0          ; Leading zero flag
        
digit_loop:
        lda decimal_buffer,x
        bne non_zero
        cpy #0          ; Skip leading zeros
        beq skip_digit
        
non_zero:
        ldy #1          ; Found non-zero
        ora #$30        ; Convert to ASCII
        jsr $ffd2
        
skip_digit:
        dex
        bpl digit_loop
        
        ; Always show at least one digit
        cpy #0
        bne done_decimal
        lda #'0'
        jsr $ffd2
        
done_decimal:
        rts

binary_to_decimal:
        ; Convert 16-bit binary to decimal
        ; Uses repeated subtraction
        
        ldx #0
        stx decimal_buffer
        stx decimal_buffer+1
        stx decimal_buffer+2
        stx decimal_buffer+3
        stx decimal_buffer+4
        
        ; Ten thousands
conv_10000:
        lda number_lo
        sec
        sbc #<10000
        sta temp_lo
        lda number_hi
        sbc #>10000
        bcc conv_1000
        sta number_hi
        lda temp_lo
        sta number_lo
        inc decimal_buffer+4
        jmp conv_10000
        
        ; Thousands
conv_1000:
        lda number_lo
        sec
        sbc #<1000
        sta temp_lo
        lda number_hi
        sbc #>1000
        bcc conv_100
        sta number_hi
        lda temp_lo
        sta number_lo
        inc decimal_buffer+3
        jmp conv_1000
        
        ; Hundreds
conv_100:
        lda number_lo
        cmp #100
        bcc conv_10
        sbc #100
        sta number_lo
        inc decimal_buffer+2
        jmp conv_100
        
        ; Tens
conv_10:
        lda number_lo
        cmp #10
        bcc conv_1
        sbc #10
        sta number_lo
        inc decimal_buffer+1
        jmp conv_10
        
        ; Ones
conv_1:
        lda number_lo
        sta decimal_buffer
        rts

number_lo: !byte 0
number_hi: !byte 0
temp_lo:   !byte 0
decimal_buffer: !byte 0,0,0,0,0`} />
