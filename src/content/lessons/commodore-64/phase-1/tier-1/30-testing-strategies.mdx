# Testing and Quality Assurance

Build robust, bug-free games with systematic testing approaches adapted for C64 development.

## Automated Testing

Create test suites for your code:

```asm
; Test framework
run_all_tests:
    lda #0
    sta tests_passed
    sta tests_failed
    
    jsr test_collision
    jsr test_movement
    jsr test_scoring
    jsr test_memory
    
    ; Display results
    jsr show_test_results
    rts

test_collision:
    ; Setup test
    lda #100
    sta sprite1_x
    sta sprite1_y
    lda #105
    sta sprite2_x
    sta sprite2_y
    
    ; Run function
    jsr check_collision
    
    ; Verify result
    bcc test_failed_collision
    inc tests_passed
    rts
    
test_failed_collision:
    inc tests_failed
    
    ; Log failure
    lda #<msg_collision_fail
    ldy #>msg_collision_fail
    jsr log_error
    rts

; Assertions
assert_equal:
    ; A = expected, X = actual
    stx temp
    cmp temp
    beq assert_pass
    
    ; Log assertion failure
    jsr log_assertion_fail
    inc tests_failed
    rts
    
assert_pass:
    inc tests_passed
    rts

tests_passed: .byte 0
tests_failed: .byte 0
temp:         .byte 0
```

## Boundary Testing

Test edge cases:

```asm
; Test screen boundaries
test_boundaries:
    ; Test left edge
    lda #0
    sta player_x
    jsr move_left
    lda player_x
    cmp #0          ; Should not wrap
    beq bound_test1_ok
    jsr test_fail
    
bound_test1_ok:
    ; Test right edge
    lda #255
    sta player_x
    jsr move_right
    lda player_x
    cmp #255
    beq bound_test2_ok
    jsr test_fail
    
bound_test2_ok:
    ; Test collision at edges
    lda #0
    sta enemy_x
    lda #7
    sta player_x
    jsr check_collision
    bcs bound_test3_ok
    jsr test_fail
    
bound_test3_ok:
    rts
```

## Stress Testing

Push the system to its limits:

```asm
; Spawn maximum entities
stress_test:
    ; Spawn max sprites
    ldx #0
spawn_all:
    txa
    jsr spawn_enemy
    inx
    cpx #MAX_ENEMIES
    bne spawn_all
    
    ; Max bullets
    ldx #0
fire_all:
    jsr fire_bullet
    inx
    cpx #MAX_BULLETS
    bne fire_all
    
    ; Run for extended time
    ldy #0
stress_loop:
    jsr game_update
    jsr check_memory_corruption
    bcc stress_ok
    
    ; Memory corrupted!
    jsr panic
    
stress_ok:
    iny
    bne stress_loop
    
    rts
```

## Regression Testing

Ensure fixes don't break existing features:

```asm
; Regression test suite
regression_tests:
    ; Test previously fixed bugs
    jsr test_sprite_overflow_fix
    jsr test_score_rollover_fix
    jsr test_collision_miss_fix
    
    ; Verify all features still work
    jsr test_all_features
    rts

; Keep test cases for fixed bugs
test_sprite_overflow_fix:
    ; Bug: Sprite index overflow caused crash
    lda #255
    sta sprite_index
    jsr next_sprite
    lda sprite_index
    cmp #0
    beq test_pass
    jmp test_fail
```

Quality assurance on the C64 requires creativity, but solid testing practices ensure your game runs flawlessly!
