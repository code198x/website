# Projectiles

Fill the screen with blazing laser bolts! We'll build a complete projectile system with rapid-fire bullets, spread patterns, and smart-seeking missiles that hunt down targets.

## Bullet Pool System

Instead of creating/destroying bullets, use a pool for efficiency:

<CodeBlock title="Bullet Management System" code={`; Bullet pool constants
MAX_BULLETS     = 16        ; Maximum bullets on screen
BULLET_SPEED    = 4         ; Pixels per frame

; Bullet structure (5 bytes each)
; Byte 0: Active flag (0=inactive, 1=active)
; Byte 1: X position (low)
; Byte 2: X position (high)
; Byte 3: Y position
; Byte 4: Direction/velocity

bullet_pool:    !fill MAX_BULLETS * 5, 0

; Find free bullet slot
find_free_bullet:
    ldx #0
    ldy #0              ; Pool index
search_loop:
    lda bullet_pool,y
    beq found_free      ; Active flag = 0
    
    ; Move to next bullet
    tya
    clc
    adc #5              ; 5 bytes per bullet
    tay
    inx
    cpx #MAX_BULLETS
    bne search_loop
    
    ; No free slots
    lda #$FF
    rts
    
found_free:
    tya                 ; Return pool offset
    rts

; Spawn bullet from player
spawn_bullet:
    jsr find_free_bullet
    cmp #$FF
    beq spawn_failed
    
    tax                 ; X = pool offset
    
    ; Mark as active
    lda #1
    sta bullet_pool,x
    
    ; Set position (from player sprite)
    lda $D000           ; Player X
    sta bullet_pool+1,x
    lda #0
    sta bullet_pool+2,x ; High byte
    
    lda $D001           ; Player Y
    sec
    sbc #8              ; Above player
    sta bullet_pool+3,x
    
    ; Set upward velocity
    lda #-BULLET_SPEED
    sta bullet_pool+4,x
    
    ; Bullet sound
    lda #$20
    sta $D400
    lda #$40
    sta $D401
    lda #$11            ; Triangle wave
    sta $D404
    
spawn_failed:
    rts`} language="asm" />

## Update All Bullets

Process the entire bullet pool each frame:

<CodeBlock title="Bullet Update Loop" code={`; Update all active bullets
update_bullets:
    ldx #0              ; Bullet counter
    ldy #0              ; Pool offset
    
bullet_loop:
    lda bullet_pool,y
    beq next_bullet     ; Skip inactive
    
    ; Move bullet
    lda bullet_pool+3,y ; Y position
    clc
    adc bullet_pool+4,y ; Add velocity
    sta bullet_pool+3,y
    
    ; Check bounds
    cmp #240            ; Off bottom
    bcs deactivate_bullet
    cmp #20             ; Off top
    bcc deactivate_bullet
    
    ; Check collision with targets
    jsr check_bullet_collision
    bcc next_bullet     ; No hit
    
deactivate_bullet:
    ; Mark as inactive
    lda #0
    sta bullet_pool,y
    
next_bullet:
    ; Move to next bullet
    tya
    clc
    adc #5
    tay
    inx
    cpx #MAX_BULLETS
    bne bullet_loop
    
    rts

; Check if bullet hit any target
check_bullet_collision:
    ; Get bullet position
    lda bullet_pool+1,y ; X
    sta temp_x
    lda bullet_pool+3,y ; Y  
    sta temp_y
    
    ; Check against all enemies
    ldx #0
enemy_check_loop:
    lda enemy_active,x
    beq next_enemy
    
    ; Check X range
    lda enemy_x,x
    sec
    sbc #8
    cmp temp_x
    bcs next_enemy
    lda enemy_x,x
    clc
    adc #8
    cmp temp_x
    bcc next_enemy
    
    ; Check Y range
    lda enemy_y,x
    sec
    sbc #8
    cmp temp_y
    bcs next_enemy
    lda enemy_y,x
    clc
    adc #8
    cmp temp_y
    bcc next_enemy
    
    ; Hit! 
    jsr destroy_enemy
    sec                 ; Carry set = hit
    rts
    
next_enemy:
    inx
    cpx #8              ; Max enemies
    bne enemy_check_loop
    
    clc                 ; No hit
    rts

temp_x: .byte 0
temp_y: .byte 0`} language="asm" />

## Spread Fire Patterns

Create impressive bullet patterns:

<CodeBlock title="Spread Shot System" code={`; Different firing patterns
PATTERN_SINGLE  = 0
PATTERN_DOUBLE  = 1
PATTERN_TRIPLE  = 2
PATTERN_SPREAD  = 3

current_pattern: .byte PATTERN_SINGLE
pattern_timer:   .byte 0

; Spawn bullets based on current pattern
fire_pattern:
    lda current_pattern
    cmp #PATTERN_SINGLE
    beq fire_single
    cmp #PATTERN_DOUBLE
    beq fire_double
    cmp #PATTERN_TRIPLE
    beq fire_triple
    cmp #PATTERN_SPREAD
    beq fire_spread
    rts

fire_single:
    jsr spawn_bullet
    rts

fire_double:
    ; Left bullet
    jsr find_free_bullet
    cmp #$FF
    beq fire_failed
    tax
    lda #1
    sta bullet_pool,x
    lda $D000
    sec
    sbc #4              ; Offset left
    sta bullet_pool+1,x
    lda #0
    sta bullet_pool+2,x
    lda $D001
    sec
    sbc #8
    sta bullet_pool+3,x
    lda #-BULLET_SPEED
    sta bullet_pool+4,x
    
    ; Right bullet
    jsr find_free_bullet
    cmp #$FF
    beq fire_failed
    tax
    lda #1
    sta bullet_pool,x
    lda $D000
    clc
    adc #4              ; Offset right
    sta bullet_pool+1,x
    lda #0
    sta bullet_pool+2,x
    lda $D001
    sec
    sbc #8
    sta bullet_pool+3,x
    lda #-BULLET_SPEED
    sta bullet_pool+4,x
    
fire_failed:
    rts

fire_spread:
    ; Center bullet
    jsr spawn_bullet
    
    ; Left diagonal
    jsr find_free_bullet
    cmp #$FF
    beq spread_done
    tax
    lda #1
    sta bullet_pool,x
    lda $D000
    sec
    sbc #2
    sta bullet_pool+1,x
    lda #0
    sta bullet_pool+2,x
    lda $D001
    sec
    sbc #8
    sta bullet_pool+3,x
    lda #-3             ; Slower, angled
    sta bullet_pool+4,x
    
    ; Right diagonal
    jsr find_free_bullet
    cmp #$FF
    beq spread_done
    tax
    lda #1
    sta bullet_pool,x
    lda $D000
    clc
    adc #2
    sta bullet_pool+1,x
    lda #0
    sta bullet_pool+2,x
    lda $D001
    sec
    sbc #8
    sta bullet_pool+3,x
    lda #-3
    sta bullet_pool+4,x
    
spread_done:
    rts

fire_triple:
    ; Three bullets in quick succession
    jsr fire_single
    
    ; Delay between shots
    inc pattern_timer
    lda pattern_timer
    cmp #3
    bcc triple_done
    
    ; Reset timer, fire again
    lda #0
    sta pattern_timer
    jsr fire_single
    
triple_done:
    rts`} language="asm" />

## Homing Missiles

Create smart projectiles that track targets:

<CodeBlock title="Homing Missile AI" code={`; Missile pool (separate from bullets)
MAX_MISSILES    = 4
MISSILE_SPEED   = 2

; Missile structure (7 bytes each)
; Byte 0: Active flag
; Byte 1: X position
; Byte 2: Y position  
; Byte 3: Target index
; Byte 4: Velocity X
; Byte 5: Velocity Y
; Byte 6: Fuel/lifetime

missile_pool:   !fill MAX_MISSILES * 7, 0

; Spawn homing missile
spawn_missile:
    jsr find_free_missile
    cmp #$FF
    beq missile_failed
    
    tax                 ; Pool offset
    
    ; Activate
    lda #1
    sta missile_pool,x
    
    ; Position
    lda $D000
    sta missile_pool+1,x
    lda $D001
    sta missile_pool+2,x
    
    ; Find nearest target
    jsr find_nearest_enemy
    sta missile_pool+3,x ; Store target index
    
    ; Initial velocity (straight up)
    lda #0
    sta missile_pool+4,x
    lda #-1
    sta missile_pool+5,x
    
    ; Fuel for tracking
    lda #120            ; 2 seconds of guidance
    sta missile_pool+6,x
    
    ; Missile launch sound
    lda #$10
    sta $D400
    lda #$20
    sta $D401
    lda #$21            ; Pulse wave
    sta $D404
    
missile_failed:
    rts

; Update homing missiles
update_missiles:
    ldx #0
    ldy #0
    
missile_update_loop:
    lda missile_pool,y
    beq next_missile
    
    ; Decrement fuel
    dec missile_pool+6,y
    bne has_fuel
    
    ; Out of fuel - explode
    jsr explode_missile
    jmp deactivate_missile
    
has_fuel:
    ; Get target position
    ldx missile_pool+3,y ; Target index
    lda enemy_active,x
    beq find_new_target  ; Target destroyed
    
    ; Calculate direction to target
    lda enemy_x,x
    sec
    sbc missile_pool+1,y ; Target X - missile X
    bpl target_right
    
    ; Target is left
    lda missile_pool+4,y
    sec
    sbc #1              ; Turn left
    cmp #-MISSILE_SPEED
    bcs store_vel_x
    lda #-MISSILE_SPEED
    jmp store_vel_x
    
target_right:
    ; Target is right
    lda missile_pool+4,y
    clc
    adc #1              ; Turn right
    cmp #MISSILE_SPEED
    bcc store_vel_x
    lda #MISSILE_SPEED
    
store_vel_x:
    sta missile_pool+4,y
    
    ; Same for Y velocity
    lda enemy_y,x
    sec
    sbc missile_pool+2,y
    bpl target_below
    
target_above:
    lda missile_pool+5,y
    sec
    sbc #1
    cmp #-MISSILE_SPEED
    bcs store_vel_y
    lda #-MISSILE_SPEED
    jmp store_vel_y
    
target_below:
    lda missile_pool+5,y
    clc
    adc #1
    cmp #MISSILE_SPEED
    bcc store_vel_y
    lda #MISSILE_SPEED
    
store_vel_y:
    sta missile_pool+5,y
    
    ; Apply velocity
    lda missile_pool+1,y
    clc
    adc missile_pool+4,y
    sta missile_pool+1,y
    
    lda missile_pool+2,y
    clc
    adc missile_pool+5,y
    sta missile_pool+2,y
    
    ; Check collision
    jsr check_missile_collision
    bcc missile_bounds
    
    ; Hit target
    jsr explode_missile
    jmp deactivate_missile
    
missile_bounds:
    ; Check screen bounds
    lda missile_pool+1,y
    cmp #255
    bcs deactivate_missile
    lda missile_pool+2,y
    cmp #240
    bcs deactivate_missile
    jmp next_missile
    
find_new_target:
    ; Original target gone, find new one
    jsr find_nearest_enemy
    sta missile_pool+3,y
    jmp next_missile
    
deactivate_missile:
    lda #0
    sta missile_pool,y
    
next_missile:
    tya
    clc
    adc #7              ; 7 bytes per missile
    tay
    inx
    cpx #MAX_MISSILES
    bne missile_update_loop
    
    rts

; Find closest enemy for targeting
find_nearest_enemy:
    lda #0
    sta closest_enemy
    lda #255
    sta closest_distance
    
    ldx #0
distance_check:
    lda enemy_active,x
    beq next_distance
    
    ; Calculate simple distance (Manhattan)
    lda enemy_x,x
    sec
    sbc $D000           ; Player X
    bpl pos_diff_x
    eor #$FF            ; Make positive
    clc
    adc #1
pos_diff_x:
    sta temp_dist
    
    lda enemy_y,x
    sec
    sbc $D001           ; Player Y
    bpl pos_diff_y
    eor #$FF
    clc
    adc #1
pos_diff_y:
    clc
    adc temp_dist       ; Total distance
    
    cmp closest_distance
    bcs next_distance
    
    sta closest_distance
    stx closest_enemy
    
next_distance:
    inx
    cpx #8
    bne distance_check
    
    lda closest_enemy
    rts

closest_enemy:      .byte 0
closest_distance:   .byte 255
temp_dist:          .byte 0`} language="asm" />

## Visual Effects

Make projectiles look amazing:

<CodeBlock title="Projectile Rendering" code={`; Draw all active projectiles
draw_projectiles:
    ; Draw bullets as sprites 4-7
    ldx #4              ; Start sprite index
    ldy #0              ; Pool offset
    
bullet_draw_loop:
    lda bullet_pool,y
    beq skip_bullet
    
    ; Enable sprite
    lda $D015
    ora sprite_masks,x
    sta $D015
    
    ; Position sprite
    lda bullet_pool+1,y ; X position
    sta $D000,x
    sta $D000,x
    lda bullet_pool+3,y ; Y position
    sta $D001,x
    sta $D001,x
    
    ; Set sprite frame
    lda #72             ; Bullet sprite
    sta $07F8,x         ; Sprite pointer
    
    ; Color (yellow)
    lda #7
    sta $D027,x
    
skip_bullet:
    ; Next sprite/bullet
    inx
    inx                 ; Sprite registers are 2 bytes apart
    tya
    clc
    adc #5              ; Next bullet
    tay
    cpx #8              ; Sprites 4-7 only
    bne bullet_draw_loop
    
    ; Draw missiles as character graphics
    ; (More complex visually)
    jsr draw_missiles_char
    rts

; Draw missiles using character mode
draw_missiles_char:
    ldy #0
missile_draw_loop:
    lda missile_pool,y
    beq skip_missile_draw
    
    ; Convert position to character coords
    lda missile_pool+1,y
    lsr
    lsr
    lsr                 ; X/8
    tax
    
    lda missile_pool+2,y
    lsr
    lsr
    lsr                 ; Y/8
    
    ; Calculate screen address
    jsr calc_screen_pos
    
    ; Draw missile character
    lda #42             ; Asterisk
    sta ($FB),0
    
    ; Trail effect
    lda missile_pool+6,y ; Use fuel for trail
    and #$03
    clc
    adc #8              ; Colors 8-11
    sta ($FD),0         ; Color RAM
    
skip_missile_draw:
    tya
    clc
    adc #7
    tay
    cpy #MAX_MISSILES * 7
    bne missile_draw_loop
    rts

; Sprite enable masks
sprite_masks:
    .byte %00010000     ; Sprite 4
    .byte %00100000     ; Sprite 5
    .byte %01000000     ; Sprite 6
    .byte %10000000     ; Sprite 7`} language="asm" />

## Complete Shooting System

Put it together for an epic space battle:

<CodeBlock title="Main Shooting Game Loop" code={`* = $0801
    !byte $0B,$08,$0A,$00,$9E
    !text "2061"
    !byte $00,$00,$00

* = $080D

main:
    jsr init_game
    jsr init_sprites
    jsr spawn_enemies
    
game_loop:
    lda #251
wait_frame:
    cmp $D012
    bne wait_frame
    
    ; Handle input
    jsr handle_player_input
    
    ; Update systems
    jsr update_bullets
    jsr update_missiles
    jsr update_enemies
    
    ; Check firing
    jsr check_fire_button
    
    ; Render
    jsr draw_projectiles
    jsr draw_enemies
    jsr update_hud
    
    jmp game_loop

; Handle player firing
check_fire_button:
    lda $DC00
    and #$10            ; Fire button
    bne not_firing
    
    ; Check fire rate limit
    dec fire_cooldown
    bne not_firing
    
    ; Reset cooldown
    lda #6              ; 10 FPS fire rate
    sta fire_cooldown
    
    ; Fire current pattern
    jsr fire_pattern
    
    ; Special weapon (joystick up + fire)
    lda $DC00
    and #$01            ; Up
    bne not_firing
    
    ; Launch missile
    jsr spawn_missile
    
not_firing:
    rts

; Power-up system
pickup_powerup:
    ; Check what type of powerup
    lda powerup_type
    cmp #0
    beq upgrade_spread
    cmp #1
    beq add_missile
    
upgrade_spread:
    inc current_pattern
    lda current_pattern
    cmp #4
    bcc pattern_ok
    lda #3              ; Max pattern
    sta current_pattern
pattern_ok:
    rts

add_missile:
    ; Give player a missile
    inc missile_count
    rts

; Variables
fire_cooldown:  .byte 0
missile_count:  .byte 3
powerup_type:   .byte 0

; Enemy spawn system
spawn_enemies:
    ldx #0
enemy_spawn_loop:
    lda #1
    sta enemy_active,x
    
    ; Random X position
    lda frame_counter
    clc
    adc #37
    and #$3F
    clc
    adc #40
    sta enemy_x,x
    
    ; Y position (top of screen)
    lda #20
    sta enemy_y,x
    
    ; Simple AI - move down
    lda #1
    sta enemy_vel_y,x
    
    inx
    cpx #8
    bne enemy_spawn_loop
    rts

; Enemy data
enemy_active:   !fill 8, 0
enemy_x:        !fill 8, 0  
enemy_y:        !fill 8, 0
enemy_vel_y:    !fill 8, 0
frame_counter:  .byte 0`} language="asm" />
