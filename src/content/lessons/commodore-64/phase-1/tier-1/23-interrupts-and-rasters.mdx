# Interrupts and Raster Magic

Interrupts let you synchronize with the video beam, create impossible effects, and run multiple tasks simultaneously. Master raster interrupts to unlock the C64's hidden powers.

## Understanding Interrupts

The C64's interrupt system:

```asm
; Interrupt vectors
IRQ_VECTOR = $0314  ; IRQ handler (60Hz)
NMI_VECTOR = $0318  ; NMI handler (RESTORE key)
BRK_VECTOR = $0316  ; BRK instruction handler

; Install custom IRQ handler
install_irq:
    sei             ; Disable interrupts
    
    ; Save old handler
    lda IRQ_VECTOR
    sta old_irq
    lda IRQ_VECTOR+1
    sta old_irq+1
    
    ; Install new handler
    lda #<custom_irq
    sta IRQ_VECTOR
    lda #>custom_irq
    sta IRQ_VECTOR+1
    
    cli             ; Enable interrupts
    rts

custom_irq:
    ; Your code here
    inc $D020       ; Flash border
    
    ; Call original handler
    jmp (old_irq)

old_irq: .word 0

; Direct hardware IRQ (bypassing KERNAL)
install_hardware_irq:
    sei
    
    ; Point hardware vector
    lda #<hardware_irq
    sta $FFFE
    lda #>hardware_irq
    sta $FFFF
    
    ; Disable KERNAL IRQ
    lda #$7F
    sta $DC0D       ; Disable CIA interrupts
    lda $DC0D       ; Clear pending
    
    ; Enable VIC interrupts
    lda #$01
    sta $D01A       ; Enable raster interrupt
    
    ; Set raster line
    lda #50
    sta $D012
    lda $D011
    and #$7F        ; Clear high bit
    sta $D011
    
    cli
    rts

hardware_irq:
    pha             ; Save registers
    txa
    pha
    tya
    pha
    
    ; Your code
    inc $D020
    dec $D020
    
    ; Acknowledge interrupt
    asl $D019
    
    pla             ; Restore registers
    tay
    pla
    tax
    pla
    rti             ; Return from interrupt
```

### VIC-II Interrupt Registers

| Register | Name | Bits | Description |
|----------|------|------|-------------|
| $D012 | Raster Line | Line number (0-311) | Current/target raster line |
| $D011 | Control | RST8\|ECM\|BMM\|DEN\|RSEL\|YSCROLL | Display control register |
| $D019 | IRQ Status | --\|--\|--\|--\|ILP\|IMMC\|IMBC\|IRST | Interrupt status flags |
| $D01A | IRQ Enable | --\|--\|--\|--\|ELP\|EMMC\|EMBC\|ERST | Interrupt enable flags |

## Screen Splitting

Different graphics modes on one screen:

```asm
; Text/bitmap split screen
split_screen_irq:
    pha
    lda $D012
    cmp #100
    beq switch_to_bitmap
    
switch_to_text:
    ; Top half: text mode
    lda #$1B        ; Text mode
    sta $D011
    lda #$15        ; Screen at $0400
    sta $D018
    
    ; Next interrupt at line 100
    lda #100
    sta $D012
    jmp irq_done
    
switch_to_bitmap:
    ; Bottom half: bitmap mode
    lda #$3B        ; Bitmap mode
    sta $D011
    lda #$18        ; Bitmap at $2000
    sta $D018
    
    ; Next interrupt at line 250
    lda #250
    sta $D012
    
irq_done:
    asl $D019
    pla
    rti

; Scrolling split
scroll_split_irq:
    pha
    
    lda split_zone
    beq zone_top
    
zone_bottom:
    ; Static UI area (no scroll)
    lda #$C8        ; No scroll
    sta $D016
    
    lda #0
    sta split_zone
    lda #180        ; UI starts here
    sta $D012
    jmp split_done
    
zone_top:
    ; Scrolling game area
    lda scroll_x
    and #$07
    ora #$C0        ; Add control bits
    sta $D016
    
    lda #1
    sta split_zone
    lda #50         ; Game area
    sta $D012
    
split_done:
    asl $D019
    pla
    rti

split_zone: .byte 0
scroll_x:   .byte 0
```

## Timer Interrupts

CIA timer-based interrupts:

```asm
; Set up CIA timer interrupt
setup_timer_irq:
    sei
    
    ; CIA 1 Timer A
    lda #<timer_irq
    sta $FFFE
    lda #>timer_irq
    sta $FFFF
    
    ; Set timer value (PAL: 985248 Hz)
    ; For 60Hz: 985248/60 = 16420
    lda #<16420
    sta $DC04       ; Timer A low
    lda #>16420
    sta $DC05       ; Timer A high
    
    ; Start timer
    lda #$11        ; Start timer, generate interrupts
    sta $DC0E
    
    ; Enable timer interrupt
    lda #$81
    sta $DC0D
    
    cli
    rts

timer_irq:
    pha
    
    ; This runs at exactly 60Hz
    inc frame_counter
    
    ; Update music/sound
    jsr music_player
    
    ; Acknowledge
    lda $DC0D
    
    pla
    rti

frame_counter: .byte 0

; NMI for pause (RESTORE key)
setup_nmi:
    lda #<pause_nmi
    sta $FFFA
    lda #>pause_nmi
    sta $FFFB
    
    ; NMI is always enabled
    rts

pause_nmi:
    pha
    
    ; Toggle pause
    lda paused
    eor #$01
    sta paused
    
    beq unpause
    
    ; Pause: dim screen
    lda #$00
    sta $D020
    jmp nmi_done
    
unpause:
    ; Restore border
    lda #$0E
    sta $D020
    
nmi_done:
    pla
    rti

paused: .byte 0
```

## Multiplexed Sprites

More than 8 sprites on screen:

```asm
; Sprite multiplexer interrupt
MAX_VIRTUAL = 24    ; 24 virtual sprites

multiplex_irq:
    pha
    txa
    pha
    
    ; Current sprite zone
    ldx current_zone
    
    ; Position hardware sprites for this zone
    ldy #0
position_loop:
    lda sprite_zones,x
    cmp #$FF
    beq zone_done
    
    ; Get virtual sprite index
    tax
    lda virtual_sprite_x,x
    sta $D000,y
    lda virtual_sprite_y,x
    sta $D001,y
    lda virtual_sprite_color,x
    sta $D027,y
    
    iny
    iny
    inc current_zone
    cpy #16         ; 8 sprites × 2
    bne position_loop
    
zone_done:
    ; Set next interrupt
    inc zone_index
    lda zone_index
    cmp #3          ; 3 zones
    bcc set_next_zone
    
    lda #0
    sta zone_index
    
set_next_zone:
    tax
    lda zone_raster_lines,x
    sta $D012
    
    asl $D019
    pla
    tax
    pla
    rti

current_zone:   .byte 0
zone_index:     .byte 0

zone_raster_lines:
    .byte 50,120,190

sprite_zones:
    ; Zone 0: sprites 0-7
    .byte 0,1,2,3,4,5,6,7

    .byte 8,9,10,11,12,13,14,15
    ; Zone 2: sprites 16-23
```

## Timer Interrupts Deep Dive

CIA timers for precise timing independent of video:

```asm
; Music player using CIA timer
; Runs at exactly 50Hz regardless of PAL/NTSC

setup_music_timer:
    sei
    
    ; Disable all CIA interrupts first
    lda #$7F
    sta $DC0D
    lda $DC0D       ; Clear pending
    
    ; Set timer for 50Hz
    ; PAL: 985248 Hz / 50 = 19704 cycles
    ; NTSC: 1022727 Hz / 50 = 20454 cycles
    
    lda system_type
    beq setup_pal
    
setup_ntsc:
    lda #<20454
    sta $DC04
    lda #>20454
    sta $DC05
    jmp timer_ready
    
setup_pal:
    lda #<19704
    sta $DC04
    lda #>19704
    sta $DC05
    
timer_ready:
    ; Install handler
    lda #<music_irq
    sta $FFFE
    lda #>music_irq
    sta $FFFF
    
    ; Start timer
    lda #$11        ; Start timer A, generate IRQ
    sta $DC0E
    
    ; Enable timer interrupt
    lda #$81        ; Enable timer A interrupt
    sta $DC0D
    
    cli
    rts

music_irq:
    pha
    txa
    pha
    tya
    pha
    
    ; Play music
    jsr music_play
    
    ; Update sound effects
    jsr sfx_update
    
    ; Acknowledge interrupt
    lda $DC0D
    
    pla
    tay
    pla
    tax
    pla
    rti

; Dual timer system (music + gameplay)
dual_timer_system:
    ; Timer A: Music at 50Hz
    ; Timer B: Game logic at 60Hz
    
    sei
    
    ; Timer A for music
    lda #<19704
    sta $DC04
    lda #>19704
    sta $DC05
    
    ; Timer B for game
    lda #<16420
    sta $DC06
    lda #>16420
    sta $DC07
    
    ; Combined interrupt handler
    lda #<combined_irq
    sta $FFFE
    lda #>combined_irq
    sta $FFFF
    
    ; Start both timers
    lda #$11
    sta $DC0E       ; Timer A
    lda #$51        ; Timer B continuous
    sta $DC0F
    
    ; Enable both interrupts
    lda #$83        ; Timer A and B
    sta $DC0D
    
    cli
    rts

combined_irq:
    pha
    lda $DC0D       ; Which timer?
    sta irq_source
    
    and #$01        ; Timer A?
    beq check_timer_b
    jsr music_irq_handler
    
check_timer_b:
    lda irq_source
    and #$02        ; Timer B?
    beq irq_done_combined
    jsr game_irq_handler
    
irq_done_combined:
    pla
    rti

irq_source: .byte 0
system_type: .byte 0  ; 0=PAL, 1=NTSC
```

## Complete Interrupt System

Full featured interrupt framework:

```asm
* = $0801
    .byte $0C,$08,$0A,$00,$9E,$32,$30,$36,$31,$00,$00,$00

* = $080D
main:
    jsr init_interrupts
    
    ; Main loop runs while interrupts handle everything
main_loop:
    lda paused
    bne main_loop   ; Do nothing if paused
    
    jsr game_logic
    jmp main_loop

init_interrupts:
    sei
    
    ; Disable CIA interrupts
    lda #$7F
    sta $DC0D
    sta $DD0D
    lda $DC0D
    lda $DD0D
    
    ; Set up raster interrupt chain
    lda #<irq_main
    sta $FFFE
    lda #>irq_main
    sta $FFFF
    
    ; First interrupt at top of screen
    lda #0
    sta $D012
    lda $D011
    and #$7F
    sta $D011
    
    ; Enable raster interrupt
    lda #$01
    sta $D01A
    
    ; Set up NMI for pause
    lda #<nmi_handler
    sta $FFFA
    lda #>nmi_handler
    sta $FFFB
    
    cli
    rts

irq_main:
    ; Save everything
    pha
    txa
    pha
    tya
    pha
    
    ; Which interrupt?
    ldx irq_index
    lda irq_table_lo,x
    sta irq_jump+1
    lda irq_table_hi,x
    sta irq_jump+2
    
irq_jump:
    jsr $0000       ; Modified
    
    ; Next interrupt
    inc irq_index
    lda irq_index
    cmp #NUM_IRQS
    bcc irq_ok
    lda #0
    sta irq_index
    
irq_ok:
    ; Set next raster line
    tax
    lda irq_lines,x
    sta $D012
    
    ; Acknowledge
    asl $D019
    
    ; Restore everything
    pla
    tay
    pla
    tax
    pla
    rti

; IRQ chain
irq_index: .byte 0
NUM_IRQS = 4

irq_table_lo:
```

```asm
; Professional NMI handler system
; RESTORE key detection and recovery

setup_nmi_system:

    lda #<nmi_handler

    ; Save all registers
    pha
    txa
    pha
    tya
    pha
    
    ; Check NMI source
    bit $DD0D       ; CIA 2 interrupt?
    bmi restore_key_pressed
    
    ; Check for BRK instruction
    tsx
    lda $0104,x     ; Get status from stack
    and #$10        ; BRK flag?
    bne brk_handler
    
    jmp nmi_exit

    ; RESTORE key handler
    lda pause_state
    eor #$01
    sta pause_state
    
    beq unpause_game
    
    ; Pause game
    lda #$00        ; Black border
    sta $D020
    
    ; Save SID state
    ldx #$18

    lda $D400,x
    sta sid_backup,x
    dex
    bpl save_sid
    
    ; Mute sound
    lda #0
    sta $D418

    lda #$0E        ; Light blue border
    sta $D020

    ldx #$18

    lda sid_backup,x
    sta $D400,x
    dex
    bpl restore_sid

    pla
    tay
    pla
    tax
    pla
    rti

    ; BRK instruction handler for debugging
    ; Get BRK parameter
    tsx
    lda $0102,x     ; PC low
    sta brk_addr
    lda $0103,x     ; PC high
    sta brk_addr+1
    
    ; Get parameter byte after BRK
    ldy #0
    lda (brk_addr),y
    sta brk_param
    
    ; Handle based on parameter
    cmp #$00
    beq brk_breakpoint
    cmp #$01
    beq brk_assert
    cmp #$02
    beq brk_trace
    
    jmp nmi_exit

    lda #2          ; Red border
    sta $D020
    
    ; Show registers on screen

    txa
    ldx #0
    jsr show_hex
    
    jmp halt_system

    lda #7          ; Yellow border
    sta $D020
    jmp halt_system

    ; Trace point
    inc trace_counter
    jmp nmi_exit

    ; System halted
    jmp halt_system
```

```asm
    sei             ; Disable interrupts
```

## Historical Context: Demo Scene Techniques

The techniques that defined C64 demos:

```asm
; Opening the borders - Discovery by 1001 Crew (1986)
; This "impossible" effect shocked the scene

border_opening_history:
    ; The trick: Change $D011 at exact cycle
    ; to confuse VIC's border logic
    
    ; Top/bottom border
    lda $D012
    cmp #$F9        ; Line 249
    bne no_open
    
    lda $D011
    and #$F7        ; Clear bit 3
    sta $D011
    
    ; Must restore within 63 cycles
    ldx #8
delay_restore:
    dex
    bne delay_restore
    
    lda $D011
    ora #$08
    sta $D011
    
no_open:
    rti

; FLI - Flexible Line Interpretation (Blackmail, 1988)
; Show new colors every line - 16K colors on screen!

fli_effect:
    ; The trick: Force new color fetch every line
    ; by triggering badlines repeatedly
    
    ldx #200
fli_line:
    ; Wait for exact cycle
    lda $D012
    cmp $D012
    beq *+2
    
    ; Force badline
    lda $D011
    and #$F8
    sta fli_temp
    
    ; Calculate line within char
    txa
    and #$07
    ora fli_temp
    sta $D011
    
    ; Color RAM must be updated here!
    ; Only 23 cycles available
    
    dex
    bne fli_line
    rti

fli_temp: .byte 0

; AGSP - Any Given Screen Position (Mayhem in Monsterland)
; Pixel-perfect horizontal scrolling

agsp_scroll:
    ; Combine hardware scroll with VSP
    lda scroll_x
    and #$07
    sta $D016       ; Hardware fine scroll
    
    lda scroll_x
    lsr
    lsr
    lsr             ; Divide by 8
    tax
    
    ; VSP for coarse scroll
    lda $D011
    and #$F8
    sta $D011
    
    ; Delay for VSP offset
agsp_wait:
    dex
    bne agsp_wait
    
    lda $D011
    ora #$07
    sta $D011
    rti

```

```asm
; Bug 1: Interrupt not firing
debug_no_interrupt:

    ; Solution: Double-check stabilization

    cmp $D012

    ; Usually caused by interrupt during screen update
```
