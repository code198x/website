# Memory Organization: Efficient Code Structure
**Learn to organize your C64 programs for maximum efficiency**

The Commodore 64 has 64KB of addressable memory, but not all of it is available for your programs. Understanding the memory map and organizing your code and data efficiently is crucial for creating larger, more complex games. Proper memory organization prevents conflicts, improves performance, and makes your code maintainable.

## What You'll Learn

In this lesson, you'll discover:
- **C64 memory map** - Understanding what goes where
- **Zero page usage** - Fast access to important variables
- **Code organization** - Separating code, data, and resources
- **Memory conflicts** - Avoiding system areas
- **Page alignment** - Optimizing for speed
- **Memory management** - Dynamic allocation strategies

```asm
; Game state constants
STATE_TITLE         = 0
STATE_MENU          = 1
STATE_PLAYING       = 2
STATE_PAUSED        = 3
STATE_GAME_OVER     = 4
STATE_HIGH_SCORES   = 5
STATE_OPTIONS       = 6
STATE_LEVEL_COMPLETE = 7

; Current state and transition
current_state:      .byte STATE_TITLE
next_state:         .byte STATE_TITLE
state_timer:        .byte 0
transition_active:  .byte 0

; Main game state loop
main_state_loop:
    ; Handle state transitions
    jsr handle_state_transition

    ; Process current state
    lda current_state
    cmp #STATE_TITLE
    beq handle_title
    cmp #STATE_MENU
    beq handle_menu
    cmp #STATE_PLAYING
    beq handle_playing
    cmp #STATE_PAUSED
    beq handle_paused
    cmp #STATE_GAME_OVER
    beq handle_game_over
    cmp #STATE_HIGH_SCORES
    beq handle_high_scores
    cmp #STATE_OPTIONS
    beq handle_options
    cmp #STATE_LEVEL_COMPLETE
    beq handle_level_complete

    jmp main_state_loop

; Handle state transitions with effects
handle_state_transition:
    lda transition_active
    beq no_transition

    ; Decrement transition timer
    dec state_timer
    bne transition_in_progress

    ; Transition complete
    lda #0
    sta transition_active

    ; Switch to next state
    lda next_state
    sta current_state

    ; Initialize new state
    jsr init_current_state

    rts

transition_in_progress:
    ; Handle transition effects (fade, slide, etc.)
    jsr update_transition_effect
    rts

no_transition:
    rts

; Change to new state with transition
change_state:
    ; Input: A = new state
    sta next_state

    ; Start transition
    lda #1
    sta transition_active

    ; Set transition duration
    lda #30             ; Half second transition
    sta state_timer

    ; Initialize transition effect
    jsr init_transition_effect

    rts

; Initialize state when entering
init_current_state:
    lda current_state

    cmp #STATE_TITLE
    beq init_title_state
    cmp #STATE_MENU
    beq init_menu_state
    cmp #STATE_PLAYING
    beq init_playing_state
    cmp #STATE_PAUSED
    beq init_paused_state
    cmp #STATE_GAME_OVER
    beq init_game_over_state
    cmp #STATE_HIGH_SCORES
    beq init_high_scores_state
    cmp #STATE_OPTIONS
    beq init_options_state
    cmp #STATE_LEVEL_COMPLETE
    beq init_level_complete_state

    rts

; State initialization routines
init_title_state:
    jsr clear_screen
    jsr display_title_screen
    jsr start_title_music

    ; Reset title animation
    lda #0
    sta title_anim_frame

    rts

init_menu_state:
    jsr display_main_menu

    ; Reset menu selection
    lda #0
    sta menu_selection
    sta menu_blink_timer

    rts

init_playing_state:
    jsr init_game_level
    jsr reset_player_stats
    jsr start_game_music

    rts

init_paused_state:
    jsr display_pause_overlay
    jsr pause_game_music

    rts

init_game_over_state:
    jsr display_game_over_screen
    jsr stop_all_music
    jsr check_high_score

    ; Game over timer
    lda #180            ; 3 seconds
    sta game_over_timer

    rts

; Working variables
title_anim_frame:   .byte 0
menu_selection:     .byte 0
menu_blink_timer:   .byte 0
game_over_timer:    .byte 0
```

## Title Screen System

Create an impressive opening experience:

```asm
; Handle title screen state
handle_title:
    ; Update title screen animation
    jsr update_title_animation

    ; Check for input to proceed
    lda $DC00
    and #$10            ; Fire button
    bne no_title_input

    ; Proceed to main menu
    lda #STATE_MENU
    jsr change_state

no_title_input:
    rts

; Display animated title screen
display_title_screen:
    ; Clear screen with title colors
    lda #0              ; Black background
    sta $D021
    lda #6              ; Blue border
    sta $D020

    ; Display title logo
    jsr draw_title_logo

    ; Display credits
    jsr draw_title_credits

    ; Display "Press Fire" message
    jsr draw_press_fire_message

    rts

; Draw large title logo using custom characters
draw_title_logo:
    ; Position for centered title
    ldx #8              ; Start column
    ldy #6              ; Start row

    ; Draw "SPACE" word
    jsr set_cursor_position
    ldx #0
space_loop:
    lda title_space_chars,x
    beq space_done
    jsr $FFD2
    inx
    jmp space_loop

space_done:
    ; Move to next line for "QUEST"
    ldx #8
    ldy #8
    jsr set_cursor_position

    ldx #0
quest_loop:
    lda title_quest_chars,x
    beq quest_done
    jsr $FFD2
    inx
    jmp quest_loop

quest_done:
    rts

; Animate title elements
update_title_animation:
    inc title_anim_frame

    ; Pulsing title colors
    lda title_anim_frame
    and #$1F
    cmp #16
    bcc title_bright

    ; Dim colors
    ldx #0
dim_title:
    lda #11             ; Dark gray
    sta $D800+6*40+8,x  ; Title line 1
    sta $D800+8*40+8,x  ; Title line 2
    inx
    cpx #24             ; Title width
    bne dim_title
    jmp title_color_done

title_bright:
    ; Bright colors
    ldx #0
bright_title:
    lda title_colors,x
    sta $D800+6*40+8,x
    lda title_colors,x
    sta $D800+8*40+8,x
    inx
    cpx #24
    bne bright_title

title_color_done:
    ; Blinking "Press Fire" message
    lda title_anim_frame
    and #$20
    beq show_press_fire

    ; Hide message
    jsr clear_press_fire_line
    jmp press_fire_done

show_press_fire:
    jsr draw_press_fire_message

press_fire_done:
    rts

; Draw credits and version info
draw_title_credits:
    ; Position at bottom
    ldx #5
    ldy #20
    jsr set_cursor_position

    ldx #0
credits_loop:
    lda credits_text,x
    beq credits_done
    jsr $FFD2
    inx
    jmp credits_loop

credits_done:
    rts

; Set cursor position
set_cursor_position:
    ; Input: X = column, Y = row
    stx cursor_x
    sty cursor_y

    ; Calculate screen position (Y * 40 + X)
    lda #0
    sta screen_pos
    sta screen_pos+1

    ldy cursor_y
    beq pos_calculated

row_loop:
    lda screen_pos
    clc
    adc #40
    sta screen_pos
    lda screen_pos+1
    adc #0
    sta screen_pos+1
    dey
    bne row_loop

pos_calculated:
    lda screen_pos
    clc
    adc cursor_x
    sta screen_pos
    lda screen_pos+1
    adc #0
    sta screen_pos+1

    ; Set cursor (simplified)
    rts

; Title screen data
title_space_chars:
    !text "   SPACE ADVENTURE   ",0

title_quest_chars:
    !text "      QUEST 64       ",0

title_colors:
    .byte 1,7,1,7,1,7,1,7,1,7,1,7,1,7,1,7,1,7,1,7,1,7,1,7

credits_text:
    !text "     PROGRAMMED BY YOU     ",13
    !text "       C64 ASSEMBLY        ",13
    !text "        VERSION 1.0        ",0

press_fire_text:
    !text "    PRESS FIRE TO START    ",0

; Working variables
cursor_x:       .byte 0
cursor_y:       .byte 0
screen_pos:     .word 0
```

## Menu System

Build navigable menus with selection and feedback:

```asm
; Handle main menu state
handle_menu:
    ; Update menu display
    jsr update_menu_display

    ; Handle menu input
    jsr handle_menu_input

    rts

; Display main menu options
display_main_menu:
    jsr clear_screen

    ; Menu title
    ldx #12
    ldy #5
    jsr set_cursor_position

    ldx #0
main_title_loop:
    lda main_menu_title,x
    beq title_displayed
    jsr $FFD2
    inx
    jmp main_title_loop

title_displayed:
    ; Menu options
    jsr draw_menu_options

    ; Instructions
    ldx #8
    ldy #20
    jsr set_cursor_position

    ldx #0
instructions_loop:
    lda menu_instructions,x
    beq instructions_done
    jsr $FFD2
    inx
    jmp instructions_loop

instructions_done:
    rts

; Draw all menu options
draw_menu_options:
    ldx #0              ; Option index
option_loop:
    cpx #NUM_MENU_OPTIONS
    beq options_done

    ; Calculate Y position (8 + option * 2)
    txa
    asl                 ; Multiply by 2
    clc
    adc #8
    tay

    ; Set position
    stx temp_option
    ldx #14             ; Center X
    jsr set_cursor_position
    ldx temp_option

    ; Draw option text
    jsr draw_single_option

    inx
    jmp option_loop

options_done:
    rts

; Draw individual menu option
draw_single_option:
    ; Input: X = option index

    ; Get pointer to option text
    txa
    asl                 ; Word index
    tay

    lda menu_option_ptrs,y
    sta text_ptr
    lda menu_option_ptrs+1,y
    sta text_ptr+1

    ; Check if this is selected option
    cpx menu_selection
    beq draw_selected

    ; Draw normal option
    ldy #0
normal_option_loop:
    lda (text_ptr),y
    beq option_drawn
    jsr $FFD2
    iny
    jmp normal_option_loop

draw_selected:
    ; Draw highlighted option
    lda #18             ; Reverse on
    jsr $FFD2

    ldy #0
selected_option_loop:
    lda (text_ptr),y
    beq selected_done
    jsr $FFD2
    iny
    jmp selected_option_loop

selected_done:
    lda #146            ; Reverse off
    jsr $FFD2

option_drawn:
    rts

; Handle menu input
handle_menu_input:
    lda $DC00
    and #$1F
    cmp #$1F
    beq no_menu_input   ; No input

    ; Check directions
    lda $DC00
    and #$01            ; Up
    beq menu_up

    lda $DC00
    and #$02            ; Down
    beq menu_down

    lda $DC00
    and #$10            ; Fire
    beq menu_select

    jmp no_menu_input

menu_up:
    ; Move selection up
    lda menu_selection
    beq wrap_to_bottom
    dec menu_selection
    jmp menu_moved

wrap_to_bottom:
    lda #NUM_MENU_OPTIONS-1
    sta menu_selection
    jmp menu_moved

menu_down:
    ; Move selection down
    lda menu_selection
    cmp #NUM_MENU_OPTIONS-1
    beq wrap_to_top
    inc menu_selection
    jmp menu_moved

wrap_to_top:
    lda #0
    sta menu_selection
    jmp menu_moved

menu_moved:
    ; Menu selection sound
    lda #$40
    sta $D400
    lda #$20
    sta $D401
    lda #$11
    sta $D404

    ; Debounce delay
    lda #10
    sta input_delay
    jmp no_menu_input

menu_select:
    ; Execute selected option
    lda menu_selection
    jsr execute_menu_option

no_menu_input:
    ; Handle input delay
    lda input_delay
    beq input_ready
    dec input_delay

input_ready:
    rts

; Execute menu option
execute_menu_option:
    ; Input: A = selected option

    cmp #0
    beq start_new_game
    cmp #1
    beq goto_options
    cmp #2
    beq show_high_scores
    cmp #3
    beq show_credits

    rts

start_new_game:
    lda #STATE_PLAYING
    jsr change_state
    rts

goto_options:
    lda #STATE_OPTIONS
    jsr change_state
    rts

show_high_scores:
    lda #STATE_HIGH_SCORES
    jsr change_state
    rts

show_credits:
    ; Show credits screen briefly then return
    jsr display_credits_screen
    lda #120            ; 2 second delay
    sta credits_timer
    rts

; Update menu visual effects
update_menu_display:
    ; Blink selected option
    inc menu_blink_timer

    ; Redraw options every 30 frames
    lda menu_blink_timer
    and #$1E
    bne menu_display_done

    jsr draw_menu_options

menu_display_done:
    rts

; Menu data
NUM_MENU_OPTIONS = 4

main_menu_title:    !text "MAIN MENU",0

menu_option_ptrs:
    .word option_new_game
    .word option_options
    .word option_scores
    .word option_credits

option_new_game:    !text "NEW GAME",0
option_options:     !text "OPTIONS",0
option_scores:      !text "HIGH SCORES",0
option_credits:     !text "CREDITS",0

menu_instructions:  !text "USE JOYSTICK - FIRE TO SELECT",0

; Working variables
temp_option:    .byte 0
text_ptr:       .word 0
input_delay:    .byte 0
credits_timer:  .byte 0
```

## Pause System

Add pause functionality that preserves game state:

```asm
; Handle paused state
handle_paused:
    ; Check for unpause input
    lda $DC00
    and #$10            ; Fire button
    bne check_pause_menu

    ; Resume game
    lda #STATE_PLAYING
    jsr change_state
    jmp pause_handled

check_pause_menu:
    ; Check for pause menu navigation
    jsr handle_pause_menu_input

pause_handled:
    ; Update pause screen effects
    jsr update_pause_effects

    rts

; Display pause overlay
display_pause_overlay:
    ; Save current screen state
    jsr save_game_screen

    ; Draw pause box
    jsr draw_pause_box

    ; Display pause options
    jsr draw_pause_menu

    rts

; Draw pause dialog box
draw_pause_box:
    ; Draw box outline using characters
    ldx #10             ; Start X
    ldy #8              ; Start Y

    ; Top border
    jsr set_cursor_position
    lda #85             ; Top-left corner
    jsr $FFD2

    ldx #20             ; Box width
box_top:
    lda #67             ; Horizontal line
    jsr $FFD2
    dex
    bne box_top

    lda #73             ; Top-right corner
    jsr $FFD2

    ; Middle rows
    ldy #9
    ldx #5              ; Box height
box_middle:
    stx temp_height
    sty temp_y

    ; Left border
    ldx #10
    ldy temp_y
    jsr set_cursor_position
    lda #93             ; Vertical line
    jsr $FFD2

    ; Clear middle
    ldx #20
box_clear:
    lda #32             ; Space
    jsr $FFD2
    dex
    bne box_clear

    ; Right border
    lda #93             ; Vertical line
    jsr $FFD2

    ldy temp_y
    iny
    ldx temp_height
    dex
    bne box_middle

    ; Bottom border
    ldx #10
    ldy #14
    jsr set_cursor_position

    lda #74             ; Bottom-left corner
    jsr $FFD2

    ldx #20
box_bottom:
    lda #67             ; Horizontal line
    jsr $FFD2
    dex
    bne box_bottom

    lda #75             ; Bottom-right corner
    jsr $FFD2

    rts

; Draw pause menu options
draw_pause_menu:
    ; Title
    ldx #16
    ldy #9
    jsr set_cursor_position

    ldx #0
pause_title_loop:
    lda pause_title_text,x
    beq pause_title_done
    jsr $FFD2
    inx
    jmp pause_title_loop

pause_title_done:
    ; Options
    ldx #14
    ldy #11
    jsr set_cursor_position

    ldx #0
pause_options_loop:
    lda pause_options_text,x
    beq pause_options_done
    jsr $FFD2
    inx
    jmp pause_options_loop

pause_options_done:
    rts

; Handle pause menu input
handle_pause_menu_input:
    ; For now, simple resume/quit options
    lda $DC00
    and #$01            ; Up = Quit to menu
    beq quit_to_menu

    rts

quit_to_menu:
    lda #STATE_MENU
    jsr change_state
    rts

; Save current game screen
save_game_screen:
    ; Copy screen RAM to buffer
    ldx #0
save_screen_loop:
    lda $0400,x
    sta screen_backup,x
    lda $0500,x
    sta screen_backup+$100,x
    lda $0600,x
    sta screen_backup+$200,x
    lda $0700,x
    sta screen_backup+$300,x

    ; Copy color RAM too
    lda $D800,x
    sta color_backup,x
    lda $D900,x
    sta color_backup+$100,x
    lda $DA00,x
    sta color_backup+$200,x
    lda $DB00,x
    sta color_backup+$300,x

    inx
    bne save_screen_loop

    rts

restore_game_screen:

    ldx #0
restore_screen_loop:

    sta $0400,x
    lda screen_backup+$100,x
    sta $0500,x
    lda screen_backup+$200,x
    sta $0600,x
    lda screen_backup+$300,x
    sta $0700,x

    lda color_backup,x
    sta $D800,x
    lda color_backup+$100,x
    sta $D900,x
    lda color_backup+$200,x
    sta $DA00,x
    lda color_backup+$300,x
    sta $DB00,x

    inx
    bne restore_screen_loop

    rts

; Update pause visual effects
update_pause_effects:
    ; Pulsing pause text
    inc pause_effect_timer

    lda pause_effect_timer
    and #$10
    beq pause_text_bright

    ; Dim text
    lda #11
    jmp set_pause_color

pause_text_bright:
    lda #1

set_pause_color:
    ; Set color of pause title
    ldx #8
pause_color_loop:

    dex
    bpl pause_color_loop

    rts

; Pause system data
pause_title_text:   !text "PAUSED",0
pause_options_text: !text "FIRE=RESUME UP=QUIT",0

; Screen backup buffers

; Working variables
```

## Game Over & High Scores

Handle game completion and score tracking:

```asm
; Handle game over state
handle_game_over:
    ; Count down game over timer
    dec game_over_timer
    bne game_over_waiting

    ; Time expired - go to high scores or menu
    lda player_made_high_score
    beq return_to_menu

    lda #STATE_HIGH_SCORES
    jsr change_state
    jmp game_over_done

return_to_menu:
    lda #STATE_MENU
    jsr change_state

game_over_done:
game_over_waiting:
    ; Update game over animation
    jsr update_game_over_effects

    rts

; Display game over screen
display_game_over_screen:
    jsr clear_screen

    ; Large "GAME OVER" text
    ldx #13
    ldy #8
    jsr set_cursor_position

    ldx #0
game_over_loop:
    lda game_over_text,x
    beq game_over_text_done
    jsr $FFD2
    inx
    jmp game_over_loop

game_over_text_done:
    ; Show final score
    ldx #10
    ldy #12
    jsr set_cursor_position

    ldx #0
final_score_loop:
    lda final_score_text,x
    beq show_actual_score
    jsr $FFD2
    inx
    jmp final_score_loop

show_actual_score:
    jsr display_current_score

    ; Show high score message if applicable
    lda player_made_high_score
    beq no_high_score_msg

    ldx #8
    ldy #16
    jsr set_cursor_position

    ldx #0
high_score_msg_loop:
    lda new_high_score_text,x
    beq no_high_score_msg
    jsr $FFD2
    inx
    jmp high_score_msg_loop

no_high_score_msg:
    rts

; Check if player made high score
check_high_score:
    ; Compare current score with high scores table
    lda #0
    sta player_made_high_score

    ldx #0              ; High score index
check_high_score_loop:
    ; Compare with high score entry
    lda current_score+2 ; High byte
    cmp high_scores+2,x
    bcc not_high_enough
    bne is_high_score

    lda current_score+1 ; Mid byte
    cmp high_scores+1,x
    bcc not_high_enough
    bne is_high_score

    lda current_score   ; Low byte
    cmp high_scores,x
    bcc not_high_enough

is_high_score:
    ; Insert new high score
    jsr insert_high_score
    lda #1
    sta player_made_high_score
    rts

not_high_enough:
    ; Check next high score slot
    txa
    clc
    adc #4              ; 4 bytes per high score entry
    tax
    cpx #40             ; 10 high scores * 4 bytes
    bne check_high_score_loop

    rts

; Insert high score into table
insert_high_score:
    ; X = insertion point
    ; Shift lower scores down

    ; Start from bottom and move up
    ldy #36             ; Last entry (9 * 4)

shift_scores_loop:
    cpy x
    beq insert_position

    ; Copy score down one position
    lda high_scores,y
    sta high_scores+4,y
    lda high_scores+1,y
    sta high_scores+5,y
    lda high_scores+2,y
    sta high_scores+6,y
    lda high_scores+3,y
    sta high_scores+7,y

    ; Move to previous entry
    tya
    sec
    sbc #4
    tay
    jmp shift_scores_loop

insert_position:
    ; Insert new score at position X
    lda current_score
    sta high_scores,x
    lda current_score+1
    sta high_scores+1,x
    lda current_score+2
    sta high_scores+2,x

    ; Add player name (simplified)
    lda #'A'            ; Default name
    sta high_scores+3,x

    rts

; Handle high scores display
handle_high_scores:
    ; Display high scores table
    jsr display_high_scores_table

    ; Wait for input to return
    lda $DC00
    and #$10
    bne no_scores_input

    lda #STATE_MENU
    jsr change_state

no_scores_input:
    rts

; Display high scores table
display_high_scores_table:
    jsr clear_screen

    ; Title
    ldx #13
    ldy #3
    jsr set_cursor_position

    ldx #0
scores_title_loop:
    lda scores_title_text,x
    beq scores_title_done
    jsr $FFD2
    inx
    jmp scores_title_loop

scores_title_done:
    ; Display each high score
    ldx #0              ; Score index
    ldy #6              ; Start row

display_scores_loop:
    cpx #40             ; 10 scores * 4 bytes
    beq scores_displayed

    ; Position cursor
    stx temp_score_index
    sty temp_row

    ldx #8              ; Column
    ldy temp_row
    jsr set_cursor_position

    ; Display rank number
    lda temp_row
    sec
    sbc #5              ; Convert to 1-based rank
    clc
    adc #'0'
    jsr $FFD2

    lda #'.'
    jsr $FFD2
    lda #' '
    jsr $FFD2

    ; Display score
    ldx temp_score_index
    jsr display_score_at_index

    ; Next score
    ldx temp_score_index
    txa
    clc
    adc #4              ; Next score entry
    tax

    ldy temp_row
    iny                 ; Next row
    jmp display_scores_loop

scores_displayed:
    rts

; Game over data and variables
game_over_text:         !text "GAME OVER",0
final_score_text:       !text "FINAL SCORE: ",0
new_high_score_text:    !text "NEW HIGH SCORE!",0
scores_title_text:      !text "HIGH SCORES",0

; High scores table (10 entries * 4 bytes each)
; Format: score_low, score_mid, score_high, player_initial
high_scores:            !fill 40, 0

; Current game data
current_score:          .byte 0, 0, 0
player_made_high_score: .byte 0
temp_score_index:       .byte 0
temp_row:               .byte 0
```

## Complete State System

Put it all together for smooth game flow:

```asm
* = $0801
    !byte $0B,$08,$0A,$00,$9E
    !text "2061"
    !byte $00,$00,$00

* = $080D

main:
    jsr init_state_system

    ; Start with title screen
    lda #STATE_TITLE
    jsr change_state

game_state_main:
    lda #251
wait_state:
    cmp $D012
    bne wait_state

    ; Main state loop
    jsr main_state_loop

    inc global_timer
    jmp game_state_main

; Initialize state system
init_state_system:
    ; Initialize variables
    lda #STATE_TITLE
    sta current_state
    sta next_state

    lda #0
    sta transition_active
    sta state_timer

    ; Initialize high scores with default values
    jsr init_default_high_scores

    ; Setup hardware
    jsr init_hardware

    rts

; Initialize default high scores
init_default_high_scores:
    ldx #0
    ldy #9              ; 10 scores (0-9)

init_score_loop:
    ; Calculate decreasing scores
    tya
    asl
    asl
    asl                 ; * 8
    asl                 ; * 16
    asl                 ; * 32
    clc
    adc #50             ; Base score

    sta high_scores,x   ; Low byte
    lda #0
    sta high_scores+1,x ; Mid byte
    sta high_scores+2,x ; High byte
    lda #'*'
    sta high_scores+3,x ; Initial

    ; Next entry
    txa
    clc
    adc #4
    tax

    dey
    bpl init_score_loop

    rts

; Quick pause check (can be called from any state)
check_pause_request:
    lda current_state
    cmp #STATE_PLAYING
    bne not_playing

    ; Check RUN/STOP key
    lda $91
    cmp #$7F
    bne no_pause_request

    ; Pause requested
    lda #STATE_PAUSED
    jsr change_state

no_pause_request:
not_playing:
    rts

; Transition effects
init_transition_effect:
    ; Initialize fade or other effect
    lda #0
    sta transition_step
    rts

update_transition_effect:
    ; Simple fade effect
    inc transition_step

    ; Darken screen gradually
    lda transition_step
    lsr
    lsr
    and #$0F
    eor #$0F            ; Invert for fade in
    sta $D020           ; Border color

    rts

; Global variables
global_timer:       .byte 0
transition_step:    .byte 0
```

### Network Memory Layout for Neon Nexus

Apply memory organization to structure network data:

```asm
; Neon Nexus memory layout
; $0800-$0FFF: Main program code
; $1000-$1FFF: Network node data
; $2000-$2FFF: Packet buffers
; $3000-$3FFF: Security protocols
; $4000-$7FFF: Level data
; $8000-$9FFF: Sprite data
; $A000-$BFFF: Music/SFX
; $C000-$CFFF: Character set

; Zero page allocation for network
network_zp_start = $60
packet_ptr      = $60  ; 2 bytes
node_ptr        = $62  ; 2 bytes
route_ptr       = $64  ; 2 bytes
security_flags  = $66  ; 1 byte
connection_state = $67 ; 1 byte
bandwidth_level = $68  ; 1 byte
packet_count    = $69  ; 1 byte

; Fast zero page packet processing
process_packet_fast:
        ldy #0
        lda (packet_ptr),y      ; Fast indirect access
        sta packet_type
        iny
        lda (packet_ptr),y
        sta packet_dest
        
        ; Route packet using zero page pointers
        lda packet_dest
        asl
        tay
        lda (route_ptr),y
        sta next_hop
        iny
        lda (route_ptr),y
        sta next_hop+1
        
        rts

; Page-aligned network buffers
        * = $2000
packet_buffer:
        !fill 256, 0    ; Page-aligned for speed
        
transmit_buffer:
        !fill 256, 0
        
receive_buffer:
        !fill 256, 0

; Network node structure (page-aligned)
        * = $1000
node_table:
        ; 16 nodes, 16 bytes each = 256 bytes (full page)
        !fill 256, 0

; Working variables
packet_type:    !byte 0
packet_dest:    !byte 0
next_hop:       !word 0
```

**Memory Efficiency:** Your Neon Nexus now has optimized memory layout! Network data structures are page-aligned for speed, zero page is used for critical pointers, and memory regions are clearly defined to prevent conflicts.
