## Complete Main Game Loop

Build the core game architecture that ties everything together:

<CodeBlock title="Master Game Loop" code={`* = $0801
    !byte $0B,$08,$0A,$00,$9E
    !text "2061"
    !byte $00,$00,$00

* = $080D

;=============================================================================
; MAIN PROGRAM ENTRY
;=============================================================================

main:
    ; Initialize everything
    jsr init_hardware
    jsr init_game_data
    jsr init_graphics
    jsr init_sound
    
    ; Start with title screen
    lda #STATE_TITLE
    sta current_state
    jsr init_title_screen
    
;=============================================================================
; MASTER GAME LOOP - 50Hz synchronized
;=============================================================================

master_loop:
    ; Wait for vertical sync (50 Hz)
    jsr wait_frame
    
    ; Update frame counter
    inc frame_counter
    
    ; Handle current game state
    lda current_state
    cmp #STATE_TITLE
    beq handle_title_state
    cmp #STATE_PLAYING  
    beq handle_playing_state
    cmp #STATE_PAUSED
    beq handle_paused_state
    cmp #STATE_GAME_OVER
    beq handle_game_over_state
    
    jmp master_loop

;=============================================================================
; STATE HANDLERS
;=============================================================================

handle_title_state:
    jsr update_title_screen
    jsr read_input
    
    ; Fire button starts game
    lda joy_pressed
    and #$10
    beq title_continue
    
    ; Transition to game
    lda #STATE_PLAYING
    sta current_state
    jsr init_game_level
    
title_continue:
    jsr render_title_screen
    jmp master_loop

handle_playing_state:
    ; Core game updates
    jsr read_input
    jsr handle_pause_check
    jsr update_player
    jsr update_enemies
    jsr update_bullets
    jsr update_powerups
    jsr update_particles
    jsr update_background
    jsr check_collisions
    jsr update_game_logic
    jsr update_sound_effects
    
    ; Render everything
    jsr render_game
    jmp master_loop

handle_paused_state:
    jsr read_input
    
    ; Fire button unpauses
    lda joy_pressed
    and #$10
    beq pause_continue
    
    lda #STATE_PLAYING
    sta current_state
    
pause_continue:
    jsr render_pause_screen
    jmp master_loop

handle_game_over_state:
    jsr update_game_over
    jsr read_input
    
    ; Return to title after delay or input
    lda game_over_timer
    bne game_over_continue
    
    lda joy_pressed
    and #$10
    beq game_over_continue
    
    ; Back to title
    lda #STATE_TITLE
    sta current_state
    jsr init_title_screen
    
game_over_continue:
    jsr render_game_over
    jmp master_loop

;=============================================================================
; HARDWARE INITIALIZATION
;=============================================================================

init_hardware:
    ; Setup VIC-II
    lda #$1B            ; Enable screen, 25 rows
    sta $D011
    lda #$C8            ; Multicolor off, 40 columns
    sta $D016
    
    ; Set colors
    lda #0              ; Black background
    sta $D021
    lda #0              ; Black border
    sta $D020
    
    ; Enable sprites
    lda #$FF            ; All sprites on
    sta $D015
    
    ; Setup SID
    lda #$0F            ; Max volume
    sta $D418
    
    ; Initialize CIA timers for sound
    jsr init_cia_timers
    
    rts

init_cia_timers:
    ; Setup timer for music tempo
    lda #$FF            ; Timer value
    sta $DC04           ; Timer A low
    lda #$FF
    sta $DC05           ; Timer A high
    
    lda #%00010001      ; Start timer A, continuous
    sta $DC0E
    
    rts

;=============================================================================
; GAME DATA INITIALIZATION  
;=============================================================================

init_game_data:
    ; Clear all game arrays
    jsr clear_bullets
    jsr clear_enemies
    jsr clear_powerups  
    jsr clear_particles
    
    ; Initialize player
    lda #160
    sta player_x
    lda #200
    sta player_y
    lda #PLAYER_MAX_HEALTH
    sta player_health
    lda #1
    sta player_weapon
    
    ; Initialize game state
    lda #PLAYER_START_LIVES
    sta lives
    lda #1
    sta level
    
    ; Clear score
    lda #0
    sta score
    sta score+1
    sta score+2
    
    rts

clear_bullets:
    ldx #MAX_PLAYER_BULLETS + MAX_ENEMY_BULLETS
    lda #0
clear_bullets_loop:
    sta bullet_active,x
    dex
    bpl clear_bullets_loop
    rts

clear_enemies:
    ldx #MAX_ENEMIES
    lda #0
clear_enemies_loop:
    sta enemy_active,x
    dex
    bpl clear_enemies_loop
    rts

clear_powerups:
    ldx #MAX_POWERUPS
    lda #0
clear_powerups_loop:
    sta powerup_active,x
    dex
    bpl clear_powerups_loop
    rts

clear_particles:
    ldx #MAX_PARTICLES
    lda #0
clear_particles_loop:
    sta particle_active,x
    dex
    bpl clear_particles_loop
    rts`} language="asm" />

## Complete Player System

Implement the full player experience with smooth controls and feedback:

<CodeBlock title="Advanced Player System" code={`;=============================================================================
; PLAYER SYSTEM - Smooth movement, shooting, collision
;=============================================================================

update_player:
    ; Skip if invulnerable (flashing)
    lda player_invulnerable
    beq player_can_move
    
    dec player_invulnerable
    ; Flash effect during invulnerability
    lda frame_counter
    and #$04
    beq hide_player
    
    lda #$01            ; White (visible)
    sta $D027           ; Sprite 0 color
    jmp player_movement_done
    
hide_player:
    lda #$00            ; Black (invisible)
    sta $D027
    jmp player_movement_done
    
player_can_move:
    ; Normal player color
    lda #$0E            ; Light blue
    sta $D027
    
    ; Handle movement input
    jsr handle_player_movement
    
    ; Handle shooting
    jsr handle_player_shooting
    
    ; Update shield effect
    jsr update_player_shield
    
player_movement_done:
    ; Position player sprite
    lda player_x
    sta $D000           ; Sprite 0 X
    lda player_y
    sta $D001           ; Sprite 0 Y
    
    ; Set sprite pointer (player ship graphic)
    lda #64             ; Sprite frame 64
    sta $07F8
    
    rts

handle_player_movement:
    lda joy_current
    
    ; Check left movement
    and #$04
    bne check_right
    
    lda player_x
    sec
    sbc #PLAYER_SPEED
    cmp #SCREEN_LEFT
    bcc check_right     ; Don't move past left edge
    sta player_x
    
check_right:
    lda joy_current
    and #$08
    bne check_up
    
    lda player_x
    clc
    adc #PLAYER_SPEED
    cmp #SCREEN_RIGHT
    bcs check_up        ; Don't move past right edge
    sta player_x
    
check_up:
    lda joy_current
    and #$01
    bne check_down
    
    lda player_y
    sec
    sbc #PLAYER_SPEED
    cmp #SCREEN_TOP
    bcc check_down      ; Don't move past top edge
    sta player_y
    
check_down:
    lda joy_current
    and #$02
    bne movement_done
    
    lda player_y
    clc
    adc #PLAYER_SPEED
    cmp #SCREEN_BOTTOM
    bcs movement_done   ; Don't move past bottom edge
    sta player_y
    
movement_done:
    rts

handle_player_shooting:
    ; Decrement fire rate timer
    lda player_fire_timer
    beq can_fire
    dec player_fire_timer
    rts
    
can_fire:
    ; Check fire button
    lda joy_current
    and #$10
    bne no_shooting
    
    ; Fire bullet based on weapon level
    jsr fire_player_weapon
    
    ; Set fire rate delay
    lda #PLAYER_FIRE_RATE
    sta player_fire_timer
    
no_shooting:
    rts

fire_player_weapon:
    lda player_weapon
    cmp #1
    beq fire_single
    cmp #2
    beq fire_double
    cmp #3
    beq fire_triple
    
fire_single:
    ; Single bullet straight up
    lda player_x
    ldx player_y
    ldy #-4             ; Upward velocity
    jsr spawn_player_bullet
    rts
    
fire_double:
    ; Two bullets
    lda player_x
    sec
    sbc #4              ; Left bullet
    ldx player_y
    ldy #-4
    jsr spawn_player_bullet
    
    lda player_x
    clc
    adc #4              ; Right bullet
    ldx player_y
    ldy #-4
    jsr spawn_player_bullet
    rts
    
fire_triple:
    ; Three bullets (center + angled)
    jsr fire_single     ; Center bullet
    
    lda player_x
    sec
    sbc #8              ; Left angled
    ldx player_y
    ldy #-3             ; Slightly slower
    jsr spawn_player_bullet
    
    lda player_x
    clc
    adc #8              ; Right angled
    ldx player_y  
    ldy #-3
    jsr spawn_player_bullet
    rts

update_player_shield:
    lda player_shield_timer
    beq no_shield
    
    dec player_shield_timer
    
    ; Shield visual effect
    lda frame_counter
    and #$02
    beq shield_color_1
    
    lda #$0D            ; Light green shield
    jmp set_shield_color
    
shield_color_1:
    lda #$05            ; Green shield
    
set_shield_color:
    sta $D027           ; Override player color when shielded
    
no_shield:
    rts

; Player takes damage
damage_player:
    ; Input: A = damage amount
    
    ; Check if invulnerable or shielded
    lda player_invulnerable
    bne damage_blocked
    lda player_shield_timer
    bne damage_blocked
    
    ; Apply damage
    sta ZP_TEMP1        ; Save damage amount
    lda player_health
    sec
    sbc ZP_TEMP1
    bpl health_ok
    lda #0              ; Clamp to 0
    
health_ok:
    sta player_health
    
    ; Set invulnerability period
    lda #90             ; 1.5 seconds at 60fps
    sta player_invulnerable
    
    ; Check if dead
    lda player_health
    bne still_alive
    
    jsr player_died
    
still_alive:
    ; Damage effects
    jsr screen_flash_red
    jsr screen_shake_effect
    jsr play_damage_sound
    
damage_blocked:
    rts

player_died:
    ; Lose a life
    dec lives
    
    ; Check game over
    lda lives
    bne respawn_player
    
    ; Game over
    lda #STATE_GAME_OVER
    sta current_state
    jsr init_game_over
    rts
    
respawn_player:
    ; Reset player
    lda #PLAYER_MAX_HEALTH
    sta player_health
    lda #160
    sta player_x
    lda #200
    sta player_y
    
    ; Clear screen of enemies/bullets
    jsr clear_enemies
    jsr clear_enemy_bullets
    
    rts`} language="asm" />

## Enemy AI & Wave Management

Create intelligent opposition with varied attack patterns:

<CodeBlock title="Complete Enemy System" code={`;=============================================================================
; ENEMY SYSTEM - Multiple types, AI patterns, wave spawning
;=============================================================================

update_enemies:
    ldx #0
enemy_update_loop:
    lda enemy_active,x
    beq next_enemy
    
    ; Update based on enemy type
    lda enemy_type,x
    cmp #ENEMY_SCOUT
    beq update_scout
    cmp #ENEMY_FIGHTER
    beq update_fighter
    cmp #ENEMY_BOMBER
    beq update_bomber
    
next_enemy:
    inx
    cpx #MAX_ENEMIES
    bne enemy_update_loop
    
    ; Spawn new enemies based on wave pattern
    jsr update_enemy_spawning
    
    rts

update_scout:
    ; Fast, weaving movement
    stx current_enemy
    
    ; Move down screen
    lda enemy_y,x
    clc
    adc #3              ; Fast movement
    sta enemy_y,x
    
    ; Weaving motion
    lda frame_counter
    clc
    adc current_enemy   ; Each enemy different phase
    and #$1F
    tay
    lda sine_table,y
    lsr                 ; Reduce amplitude
    lsr
    clc
    adc enemy_start_x,x ; Add to starting X
    sta enemy_x,x
    
    jmp check_enemy_bounds

update_fighter:
    ; Balanced enemy - occasional direction changes
    stx current_enemy
    
    ; Decrement AI timer
    dec enemy_ai_timer,x
    bne fighter_move_down
    
    ; Change direction randomly
    lda frame_counter
    eor current_enemy
    and #$60            ; Reset timer
    clc
    adc #30
    sta enemy_ai_timer,x
    
    ; Choose movement pattern
    and #$03
    cmp #0
    beq fighter_move_left
    cmp #1
    beq fighter_move_right
    ; Otherwise move down
    
fighter_move_down:
    lda enemy_y,x
    clc
    adc #2
    sta enemy_y,x
    jmp fighter_maybe_shoot
    
fighter_move_left:
    lda enemy_x,x
    sec
    sbc #2
    cmp #SCREEN_LEFT
    bcc fighter_move_down
    sta enemy_x,x
    jmp fighter_maybe_shoot
    
fighter_move_right:
    lda enemy_x,x
    clc
    adc #2
    cmp #SCREEN_RIGHT
    bcs fighter_move_down
    sta enemy_x,x
    
fighter_maybe_shoot:
    ; Random shooting
    lda frame_counter
    eor current_enemy
    and #$3F
    bne check_enemy_bounds
    
    ; Fire at player
    jsr enemy_fire_at_player
    jmp check_enemy_bounds

update_bomber:
    ; Slow, tough, shoots frequently
    stx current_enemy
    
    ; Slow movement
    lda enemy_y,x
    clc
    adc #1
    sta enemy_y,x
    
    ; Frequent shooting
    lda frame_counter
    and #$0F
    bne check_enemy_bounds
    
    jsr enemy_fire_at_player
    
check_enemy_bounds:
    ldx current_enemy
    
    ; Remove if off screen
    lda enemy_y,x
    cmp #SCREEN_BOTTOM+20
    bcc enemy_bounds_ok
    
    ; Deactivate enemy
    lda #0
    sta enemy_active,x
    
enemy_bounds_ok:
    rts

enemy_fire_at_player:
    ; Current enemy in X register
    ; Fire bullet toward player position
    
    lda enemy_x,x
    sta bullet_spawn_x
    lda enemy_y,x
    clc
    adc #8              ; Fire from bottom of enemy
    sta bullet_spawn_y
    
    ; Calculate direction to player
    lda player_x
    sec
    sbc enemy_x,x
    bpl enemy_fire_right
    
    lda #-2             ; Fire left
    jmp enemy_fire_y
    
enemy_fire_right:
    lda #2              ; Fire right
    
enemy_fire_y:
    sta bullet_spawn_vx
    
    ; Y velocity (always downward toward player)
    lda #3
    sta bullet_spawn_vy
    
    ; Spawn enemy bullet
    jsr spawn_enemy_bullet
    
    ; Enemy fire sound
    lda #$C0
    sta $D407           ; Voice 3 frequency low
    lda #$08
    sta $D408           ; Voice 3 frequency high
    lda #$21            ; Pulse wave, gate on
    sta $D40B
    
    rts

update_enemy_spawning:
    ; Decrement spawn timer
    dec enemy_spawn_timer
    bne no_enemy_spawn
    
    ; Reset timer based on difficulty
    lda level
    asl
    asl
    sta ZP_TEMP1
    lda #120            ; Base spawn rate
    sec
    sbc ZP_TEMP1        ; Faster spawning at higher levels
    cmp #30
    bcs spawn_rate_ok
    lda #30             ; Minimum spawn rate
    
spawn_rate_ok:
    sta enemy_spawn_timer
    
    ; Spawn enemy wave
    jsr spawn_enemy_wave
    
no_enemy_spawn:
    rts

spawn_enemy_wave:
    ; Spawn pattern based on level
    lda level
    and #$03
    cmp #0
    beq spawn_scouts
    cmp #1  
    beq spawn_mixed
    cmp #2
    beq spawn_formation
    ; Level 3+ - bombers
    
spawn_bombers:
    lda #ENEMY_BOMBER
    ldx #80             ; Left side
    jsr try_spawn_enemy
    
    lda #ENEMY_BOMBER
    ldx #240            ; Right side
    jsr try_spawn_enemy
    rts
    
spawn_scouts:
    ; Fast scouts across screen
    ldx #0
scout_spawn_loop:
    lda #ENEMY_SCOUT
    lda scout_spawn_x,x
    tax
    jsr try_spawn_enemy
    
    ldx ZP_TEMP1        ; Restore loop counter
    inx
    stx ZP_TEMP1
    cpx #4
    bne scout_spawn_loop
    rts
    
spawn_mixed:
    ; Mix of enemy types
    lda #ENEMY_FIGHTER
    ldx #120
    jsr try_spawn_enemy
    
    lda #ENEMY_SCOUT
    ldx #200
    jsr try_spawn_enemy
    rts
    
spawn_formation:
    ; Formation of fighters
    lda #ENEMY_FIGHTER
    ldx #80
    jsr try_spawn_enemy
    
    lda #ENEMY_FIGHTER
    ldx #120
    jsr try_spawn_enemy
    
    lda #ENEMY_FIGHTER
    ldx #160
    jsr try_spawn_enemy
    rts

try_spawn_enemy:
    ; Input: A = enemy type, X = X position
    sta spawn_type
    stx spawn_x
    
    ; Find free enemy slot
    ldx #0
find_enemy_slot:
    lda enemy_active,x
    beq found_enemy_slot
    inx
    cpx #MAX_ENEMIES
    bne find_enemy_slot
    rts                 ; No free slots
    
found_enemy_slot:
    ; Initialize enemy
    lda #1
    sta enemy_active,x
    lda spawn_type
    sta enemy_type,x
    lda spawn_x
    sta enemy_x,x
    sta enemy_start_x,x ; Save for weaving
    lda #30             ; Start at top
    sta enemy_y,x
    
    ; Set health based on type
    ldy spawn_type
    lda enemy_health_table,y
    sta enemy_health,x
    
    ; Initialize AI timer
    lda #60
    sta enemy_ai_timer,x
    
    rts

; Data tables
enemy_health_table:
    .byte 1, 3, 5       ; Scout, Fighter, Bomber health

scout_spawn_x:
    .byte 60, 120, 180, 240

; Working variables
current_enemy:      .byte 0
enemy_spawn_timer:  .byte 60
spawn_type:         .byte 0
spawn_x:            .byte 0
bullet_spawn_x:     .byte 0
bullet_spawn_y:     .byte 0
bullet_spawn_vx:    .byte 0
bullet_spawn_vy:    .byte 0

; Enemy data arrays
enemy_active:       !fill MAX_ENEMIES, 0
enemy_type:         !fill MAX_ENEMIES, 0
enemy_x:            !fill MAX_ENEMIES, 0
enemy_y:            !fill MAX_ENEMIES, 0
enemy_health:       !fill MAX_ENEMIES, 0
enemy_ai_timer:     !fill MAX_ENEMIES, 0
enemy_start_x:      !fill MAX_ENEMIES, 0`} language="asm" />

## Complete Visual Effects System

Add the juice that makes the game feel amazing:

<CodeBlock title="Polish & Effects System" code={`;=============================================================================
; VISUAL EFFECTS - Screen shake, particles, flashes, juice!
;=============================================================================

update_particles:
    ldx #0
particle_loop:
    lda particle_active,x
    beq next_particle
    
    ; Update position
    lda particle_x,x
    clc
    adc particle_vx,x
    sta particle_x,x
    
    lda particle_y,x
    clc
    adc particle_vy,x
    sta particle_y,x
    
    ; Apply gravity/friction
    inc particle_vy,x   ; Gravity
    
    ; Fade particle
    dec particle_life,x
    bne particle_alive
    
    ; Particle died
    lda #0
    sta particle_active,x
    jmp next_particle
    
particle_alive:
    ; Draw particle using character mode
    jsr draw_particle_char
    
next_particle:
    inx
    cpx #MAX_PARTICLES
    bne particle_loop
    rts

draw_particle_char:
    ; Draw particle X as character on screen
    
    ; Convert pixel position to character position
    lda particle_x,x
    lsr
    lsr
    lsr                 ; Divide by 8
    tay                 ; Y = screen column
    
    lda particle_y,x
    lsr
    lsr
    lsr                 ; Divide by 8
    
    ; Calculate screen address (row * 40 + column)
    jsr multiply_by_40
    sty ZP_TEMP1        ; Save column
    clc
    adc ZP_TEMP1        ; Add column
