# Level Design and Data Management

Great level design transforms simple mechanics into engaging experiences. On the C64's limited memory, efficient data structures and compression techniques let you pack rich, varied levels into just a few kilobytes.

## Level Data Structures

Organize level data efficiently:

```asm
; Level header structure
LEVEL_WIDTH     = 0  ; Level width in tiles
LEVEL_HEIGHT    = 1  ; Level height in tiles  
LEVEL_START_X   = 2  ; Player start X
LEVEL_START_Y   = 3  ; Player start Y
LEVEL_TIME      = 4  ; Time limit (BCD)
LEVEL_TILESET   = 5  ; Tileset ID
LEVEL_MUSIC     = 6  ; Music track
LEVEL_DATA_PTR  = 7  ; Pointer to compressed data (2 bytes)

; Current level data
current_level:  .byte 1
level_header:   .res 9

; Level 1 data
level1_header:
    .byte 40,25     ; Size: 40×25
    .byte 2,22      ; Start position
    .byte $90       ; 90 seconds
    .byte 0         ; Forest tileset
    .byte 1         ; Track 1
    .word level1_data

level1_data:
    ; Compressed level data (RLE encoded)
    .byte $20,40    ; 40 spaces (empty top row)
    .byte $A0,1     ; 1 wall
    .byte $20,38    ; 38 spaces
    .byte $A0,1     ; 1 wall
    ; ... continue for entire level

load_level:
    ; A = level number
    tax
    
    ; Calculate header offset
    lda level_table_lo,x
    sta temp_ptr
    lda level_table_hi,x
    sta temp_ptr+1
    
    ; Copy header
    ldy #0
copy_header:
    lda (temp_ptr),y
    sta level_header,y
    iny
    cpy #9
    bne copy_header
    
    ; Decompress level data
    jsr decompress_level
    
    ; Set up tileset
    lda level_header+LEVEL_TILESET
    jsr load_tileset
    
    ; Position player
    lda level_header+LEVEL_START_X
    asl
    asl
    asl             ; ×8 for pixels
    sta player_x
    lda level_header+LEVEL_START_Y
    asl
    asl
    asl
    sta player_y
    
    rts

level_table_lo:
    .byte <level1_header,<level2_header,<level3_header
level_table_hi:
    .byte >level1_header,>level2_header,>level3_header
```

## RLE Compression

Run-Length Encoding saves memory:

```asm
decompress_level:
    ; Get data pointer
    lda level_header+LEVEL_DATA_PTR
    sta src_ptr
    lda level_header+LEVEL_DATA_PTR+1
    sta src_ptr+1
    
    ; Destination is screen RAM
    lda #<$0400
    sta dst_ptr
    lda #>$0400
    sta dst_ptr+1
    
    ldy #0
decompress_loop:
    lda (src_ptr),y ; Get tile type
    beq decomp_done ; 0 = end marker
    sta current_tile
    
    iny
    lda (src_ptr),y ; Get count
    tax
    
    ; Write tiles
write_tiles:
    lda current_tile
    sta (dst_ptr),y
    
    ; Increment destination
    inc dst_ptr
    bne no_carry
    inc dst_ptr+1
no_carry:
    
    dex
    bne write_tiles
    
    ; Next RLE pair
    iny
    jmp decompress_loop
    
decomp_done:
    rts

current_tile: .byte 0
src_ptr:      .word 0
dst_ptr:      .word 0
```

## Tile-Based Levels

Design levels using reusable tiles:

```asm
; Tile properties
TILE_SOLID      = %10000000
TILE_PLATFORM   = %01000000
TILE_HAZARD     = %00100000
TILE_COLLECTABLE = %00010000
TILE_ANIMATED   = %00001000
TILE_TRIGGER    = %00000100

; Tile definitions
tile_properties:
    .byte 0                 ; $20 Space - empty
    .byte TILE_SOLID        ; $A0 Block - solid
    .byte TILE_PLATFORM     ; $5F Platform - one-way
    .byte TILE_HAZARD       ; $2A Spike - dangerous
    .byte TILE_COLLECTABLE  ; $53 Coin - pickup
    .byte TILE_ANIMATED|TILE_HAZARD ; $7E Lava
    .byte TILE_TRIGGER      ; $54 Switch
    .byte TILE_SOLID|TILE_ANIMATED ; $42 Moving block

check_tile_collision:
    ; Convert pixel position to tile
    lda player_x
    lsr
    lsr
    lsr             ; /8
    sta tile_x
    
    lda player_y
    lsr
    lsr
    lsr
    sta tile_y
    
    ; Calculate screen offset
    lda tile_y
    asl
    asl
    asl             ; ×8
    asl
    asl             ; ×32
    clc
    adc tile_y      ; ×40 total
    adc tile_x
    tay
    
    ; Get tile at position
    lda $0400,y
    tax
    
    ; Look up properties
    lda tile_properties,x
    sta tile_flags
    
    ; Check collision types
    and #TILE_SOLID
    beq check_platform
    
    ; Solid collision
    jsr handle_solid_collision
    rts
    
check_platform:
    lda tile_flags
    and #TILE_PLATFORM
    beq check_hazard
    
    ; Platform collision (only from above)
    lda player_vy
    bmi check_hazard ; Moving up, ignore
    jsr handle_platform_collision
    
check_hazard:
    lda tile_flags
    and #TILE_HAZARD
    beq check_collectable
    
    ; Hazard hit!
    jsr player_damage
    
check_collectable:
    lda tile_flags
    and #TILE_COLLECTABLE
    beq check_done
    
    ; Collect item
    lda #$20        ; Replace with empty
    sta $0400,y
    jsr collect_tile_item
    
check_done:
    rts

tile_x:     .byte 0
tile_y:     .byte 0
tile_flags: .byte 0
```

## Screen Transitions

Move between connected screens:

```asm
; Screen connections
SCREEN_NORTH = 0
SCREEN_EAST  = 1
SCREEN_SOUTH = 2
SCREEN_WEST  = 3

screen_connections:
    ; Screen 0 connections (N,E,S,W)
    .byte $FF,1,4,$FF  ; No north, east=1, south=4, no west
    ; Screen 1
    .byte $FF,2,5,0    ; East=2, south=5, west=0
    ; ... more screens

check_screen_exit:
    ; Check right edge
    lda player_x
    cmp #248
    bcc check_left
    
    ; Exit right
    lda current_screen
    asl
    asl             ; ×4 for connection offset
    clc
    adc #SCREEN_EAST
    tax
    lda screen_connections,x
    cmp #$FF
    beq check_left  ; No connection
    
    ; Load new screen
    sta current_screen
    jsr load_screen
    
    ; Position player on left
    lda #8
    sta player_x
    rts
    
check_left:
    lda player_x
    cmp #8
    bcs check_top
    
    ; Exit left
    lda current_screen
    asl
    asl
    clc
    adc #SCREEN_WEST
    tax
    lda screen_connections,x
    cmp #$FF
    beq check_top
    
    sta current_screen
    jsr load_screen
    lda #240
    sta player_x
    rts
    
check_top:
    ; Similar for top/bottom
    rts

current_screen: .byte 0
```

## Dynamic Level Elements

Create interactive environments:

```asm
; Moving platforms
platform_data:
    ; x1,y1,x2,y2,speed,current_pos
    .byte 50,100,150,100,1,0   ; Horizontal platform
    .byte 200,50,200,150,2,0   ; Vertical platform

update_platforms:
    ldx #0
platform_loop:
    ; Get platform data offset
    txa
    asl
    asl
    asl             ; ×8 bytes per platform
    tay
    
    ; Update position
    inc platform_data+5,y
    lda platform_data+5,y
    
    ; Check bounds
    cmp #100        ; Max position
    bcc move_platform
    
    ; Reverse direction
    lda #0
    sta platform_data+5,y
    
move_platform:
    ; Calculate actual position
    lsr             ; /2 for slower movement
    sta temp
    
    ; Interpolate between points
    lda platform_data+0,y   ; x1
    sta platform_x
    lda platform_data+1,y   ; y1
    sta platform_y
    
    ; Draw platform sprite
    lda platform_x
    sta $D008       ; Sprite 4 X
    lda platform_y
    sta $D009       ; Sprite 4 Y
    
    inx
    cpx #2          ; 2 platforms
    bne platform_loop
    rts

; Destructible blocks
destroy_block:
    ; Y = screen offset of block
    lda #$20        ; Empty space
    sta $0400,y
    
    ; Spawn debris particles
    jsr spawn_debris
    
    ; Play break sound
    jsr block_break_sound
    
    ; Check for secrets
    lda $D41B       ; Random
    and #$07
    cmp #$01        ; 1 in 8 chance
    bne no_secret
    
    ; Spawn bonus item
    lda #TYPE_DIAMOND
    jsr spawn_item_at_tile
    
no_secret:
    rts

temp: .byte 0
platform_x: .byte 0
platform_y: .byte 0
```

## Level Progression

Manage multiple levels and objectives:

```asm
; Level objectives
OBJECTIVE_COLLECT_ALL = 0
OBJECTIVE_REACH_EXIT  = 1
OBJECTIVE_DEFEAT_BOSS = 2
OBJECTIVE_SURVIVE     = 3

level_objective:    .byte 0
objective_counter:  .byte 0
objective_target:   .byte 0

check_level_complete:
    lda level_objective
    
    cmp #OBJECTIVE_COLLECT_ALL
    beq check_collection
    cmp #OBJECTIVE_REACH_EXIT
    beq check_exit_reached
    cmp #OBJECTIVE_DEFEAT_BOSS
    beq check_boss_defeated
    cmp #OBJECTIVE_SURVIVE
    beq check_survival
    rts
    
check_collection:
    ; Count remaining collectables
    ldx #0
    ldy #0
count_loop:
    lda $0400,x
    cmp #$53        ; Coin tile
    bne not_coin
    iny             ; Count it
not_coin:
    inx
    cpx #0          ; Full screen
    bne count_loop
    
    cpy #0          ; All collected?
    bne not_complete
    jmp level_complete
    
check_exit_reached:
    ; Check if player at exit
    lda player_x
    cmp exit_x
    bne not_complete
    lda player_y
    cmp exit_y
    bne not_complete
    jmp level_complete
    
check_boss_defeated:
    lda boss_health
    bne not_complete
    jmp level_complete
    
check_survival:
    lda timer_min
    bne not_complete
    lda timer_sec
    bne not_complete
    jmp level_complete
    
not_complete:
    rts
    
level_complete:
    ; Show completion message
    lda #<msg_complete
    ldy #>msg_complete
    jsr show_message
    
    ; Calculate score bonus
    jsr calculate_time_bonus
    jsr calculate_perfect_bonus
    
    ; Advance to next level
    inc current_level
    lda current_level
    cmp #MAX_LEVELS
    bcc load_next
    
    ; Game complete!
    jmp game_victory
    
load_next:
    jsr load_level
    rts

msg_complete: .text "LEVEL CLEAR!"
              .byte 0
              
exit_x: .byte 230
exit_y: .byte 100
```

## Procedural Generation

Create infinite variety:

```asm
generate_random_level:
    ; Seed based on level number
    lda current_level
    sta random_seed
    
    ; Clear screen
    jsr clear_screen
    
    ; Generate terrain
    jsr generate_terrain
    
    ; Place platforms
    jsr place_platforms
    
    ; Add enemies
    jsr spawn_level_enemies
    
    ; Place items
    jsr place_items
    rts

generate_terrain:
    ; Generate ground height
    lda #20         ; Start height
    sta ground_height
    
    ldx #0
terrain_loop:
    ; Vary height
    jsr random
    and #$03
    sec
    sbc #1          ; -1 to +2
    clc
    adc ground_height
    
    ; Clamp height
    cmp #18
    bcs too_high
    cmp #23
    bcc height_ok
too_high:
    lda #20
height_ok:
    sta ground_height
    
    ; Fill column
    tay             ; Height in Y
fill_column:
    txa
    pha
    
    ; Calculate screen position
    tya
    asl
    asl
    asl
    asl
    asl             ; ×40
    adc temp_x
    tax
    
    lda #$A0        ; Solid block
    sta $0400,x
    
    pla
    tax
    
    iny
    cpy #25         ; Bottom of screen
    bne fill_column
    
    inx
    stx temp_x
    cpx #40         ; Screen width
    bne terrain_loop
    rts

place_platforms:
    ; Place 3-5 random platforms
    jsr random
    and #$03
    clc
    adc #3
    sta platform_count
    
    ldx #0
plat_loop:
    ; Random position
    jsr random
    and #$1F
    clc
    adc #5          ; Margin
    sta plat_x
    
    jsr random
    and #$0F
    clc
    adc #5
    sta plat_y
    
    ; Random width (3-8)
    jsr random
    and #$07
    clc
    adc #3
    sta plat_width
    
    ; Draw platform
    jsr draw_platform
    
    inx
    cpx platform_count
    bne plat_loop
    rts

random:
    ; Simple PRNG
    lda random_seed
    asl
    bcc no_eor
    eor #$1D
no_eor:
    sta random_seed
    rts

random_seed:    .byte $42
ground_height:  .byte 20
temp_x:         .byte 0
platform_count: .byte 0
plat_x:         .byte 0
plat_y:         .byte 0
plat_width:     .byte 0
```

## Complete Level System

Full level management:

```asm
* = $0801
    .byte $0C,$08,$0A,$00,$9E,$32,$30,$36,$31,$00,$00,$00

* = $080D
main:
    jsr init_game
    lda #1
    jsr load_level
    
game_loop:
    jsr wait_frame
    jsr handle_input
    jsr update_player
    jsr update_platforms
    jsr check_tile_collision
    jsr check_screen_exit
    jsr check_level_complete
    jsr render_level
    jmp game_loop

init_game:
    ; Initialize level system
    lda #1
    sta current_level
    
    lda #0
    sta player_deaths
    sta total_score
    sta total_score+1
    rts

render_level:
    ; Update animated tiles
    inc anim_timer
    lda anim_timer
    and #$0F
    bne no_anim
    
    ; Cycle water animation
    ldx #0
water_anim:
    lda $0400,x
    cmp #$7E        ; Water tile
    bne next_tile
    
    inc $0400,x     ; Next frame
    lda $0400,x
    cmp #$82        ; Last frame?
    bcc next_tile
    lda #$7E        ; Reset
    sta $0400,x
    
next_tile:
    inx
    cpx #0
    bne water_anim
    
no_anim:
    rts

; Level transition
transition_effect:
    ; Fade out
    ldx #15
fade_out:
    stx $D020       ; Border
    stx $D021       ; Background
    
    ldy #5
fade_delay:
    jsr wait_frame
    dey
    bne fade_delay
    
    dex
    bpl fade_out
    
    ; Load new level
    jsr load_level
    
    ; Fade in
    ldx #0
fade_in:
    stx $D020
    stx $D021
    
    ldy #5
fade_delay2:
    jsr wait_frame
    dey
    bne fade_delay2
    
    inx
    cpx #16
    bne fade_in
    
    ; Reset colors
    lda #$0E
    sta $D020
    lda #$06
    sta $D021
    rts

; Variables
anim_timer:     .byte 0
player_deaths:  .byte 0
total_score:    .word 0
MAX_LEVELS      = 10
```

## Key Concepts

1. **Data Compression**: RLE and other techniques maximize level content
2. **Tile Systems**: Reusable tiles with properties create rich environments
3. **Screen Management**: Smooth transitions between connected areas
4. **Dynamic Elements**: Moving platforms and interactive objects
5. **Procedural Generation**: Algorithmic content for variety
