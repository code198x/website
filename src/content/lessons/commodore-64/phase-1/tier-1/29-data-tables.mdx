# Data Tables: Efficient Game Data Organization
**Learn to structure game information for speed and maintainability**

Games contain vast amounts of data: level layouts, enemy patterns, item properties, animation frames, and more. Organizing this data efficiently is crucial for both performance and maintainability. Data tables let you separate game content from game logic, making it easy to modify levels, adjust difficulty, or add new content without changing code.

## What You'll Learn

In this lesson, you'll discover:
- **Data structures** - Organizing related information efficiently
- **Lookup tables** - Trading memory for computation speed
- **Level data** - Storing complete game levels compactly
- **Enemy waves** - Defining spawn patterns and formations
- **Compression** - Reducing data size with encoding techniques
- **Table-driven code** - Separating data from logic

<CodeBlock title="Enemy Wave System" code={`; Enemy wave entry structure (8 bytes each)
; Byte 0: Enemy type
; Byte 1: Spawn X position  
; Byte 2: Spawn Y position
; Byte 3: Movement pattern
; Byte 4: Speed
; Byte 5: Health

    .byte 1, 60, 30, 3, 1, 10, 310, %00000010   ; Left follower
    .byte 1, 100, 30, 3, 1, 10, 320, %00000100  ; Right follower
    
    .byte WAVE_END_MARKER ; End of wave data

enemy_data_2:
    .byte 3, 40, 20, 4, 3, 25, 0, 0     ; Tough enemy
    .byte 2, 120, 15, 1, 2, 15, 30, 0  ; Flying support
    .byte 2, 160, 25, 1, 2, 15, 60, 0  ; Another flyer
    .byte 1, 200, 40, 0, 2, 10, 120, 0 ; Fast walker
    .byte WAVE_END_MARKER

    ; Use enemy_spawn_ptr loaded from network topology data
    ldy #0
    
spawn_wave_loop:
    lda (enemy_spawn_ptr),y
    cmp #WAVE_END_MARKER
    beq wave_spawn_done
    
    ; Create enemy spawn event
    jsr create_enemy_spawn_event
    
    ; Move to next wave entry
    tya
    clc
    adc #WAVE_ENTRY_SIZE
    tay
    jmp spawn_wave_loop
    
wave_spawn_done:
    rts

; Create enemy spawn event
create_enemy_spawn_event:
    ; Y = offset in security protocol wave data
    
    ; Find free spawn event slot
    ldx #0
find_spawn_slot:
    lda spawn_events_active,x
    beq found_spawn_slot
    inx
    cpx #MAX_SPAWN_EVENTS
    bne find_spawn_slot
    rts ; No free slots
    
found_spawn_slot:
    ; Copy wave data to spawn event
    lda (enemy_spawn_ptr),y ; Enemy type
    sta spawn_events_type,x
    
    iny
    lda (enemy_spawn_ptr),y ; Spawn X
    sta spawn_events_x,x
    
    iny
    lda (enemy_spawn_ptr),y ; Spawn Y
    sta spawn_events_y,x
    
    iny
    lda (enemy_spawn_ptr),y ; Movement pattern
    sta spawn_events_pattern,x
    
    iny
    lda (enemy_spawn_ptr),y ; Speed
    sta spawn_events_speed,x
    
    iny
    lda (enemy_spawn_ptr),y ; Health
    sta spawn_events_health,x
    
    iny
    lda (enemy_spawn_ptr),y ; Spawn delay
    sta spawn_events_timer,x
    
    iny
    lda (enemy_spawn_ptr),y ; Formation flags
    sta spawn_events_flags,x
    
    ; Mark as active
    lda #1
    sta spawn_events_active,x
    
    dey ; Restore Y to point to formation flags
    rts

; Spawn event system
MAX_SPAWN_EVENTS = 16
spawn_events_active:  !fill MAX_SPAWN_EVENTS, 0
spawn_events_type:    !fill MAX_SPAWN_EVENTS, 0
spawn_events_x:       !fill MAX_SPAWN_EVENTS, 0
spawn_events_y:       !fill MAX_SPAWN_EVENTS, 0
spawn_events_pattern: !fill MAX_SPAWN_EVENTS, 0
spawn_events_speed:   !fill MAX_SPAWN_EVENTS, 0
spawn_events_health:  !fill MAX_SPAWN_EVENTS, 0
spawn_events_timer:   !fill MAX_SPAWN_EVENTS, 0
spawn_events_flags:   !fill MAX_SPAWN_EVENTS, 0

; Update spawn events
update_spawn_events:
    ldx #0
spawn_event_loop:
    lda spawn_events_active,x
    beq next_spawn_event
    
    ; Decrement timer
    dec spawn_events_timer,x
    bne next_spawn_event
    
    ; Time to spawn!
    jsr execute_enemy_spawn
    
    ; Deactivate event
    lda #0
    sta spawn_events_active,x
    
next_spawn_event:
    inx
    cpx #MAX_SPAWN_EVENTS
    bne spawn_event_loop
    rts

; Execute the actual enemy spawn
execute_enemy_spawn:
    ; X = spawn event index
    
    ; Find free enemy slot
    jsr find_free_enemy
    cpy #$FF
    beq spawn_failed
    
    ; Y = enemy slot index
    ; Copy spawn data to enemy
    lda spawn_events_type,x
    sta enemy_type,y
    
    lda spawn_events_x,x
    sta enemy_x,y
    
    lda spawn_events_y,x
    sta enemy_y,y
    
    lda spawn_events_pattern,x
    sta enemy_ai_pattern,y
    
    lda spawn_events_speed,x
    sta enemy_speed,y
    
    lda spawn_events_health,x
    sta enemy_health,y
    
    lda spawn_events_flags,x
    sta enemy_formation,y
    
    ; Activate enemy
    lda #1
    sta enemy_active,y
    
spawn_failed:
    rts`} language="asm" />

<CodeBlock title="Data Compression System" code={`; Run-length encoding for background tiles
; Format: [count][tile] pairs, 0 = end of data

; Compressed background data
bg_data_1:
    .byte 20, 32    ; 20 spaces
    .byte 5, 102    ; 5 wall blocks
    .byte 15, 32    ; 15 spaces  
    .byte 3, 42     ; 3 asterisks

    ldy #0          ; Source offset
    ldx #0          ; Destination offset

    lda (bg_data_ptr),y ; Get count
    beq decompress_done

    iny
    lda (bg_data_ptr),y ; Get tile value
    sta tile_value
    iny

    rts

    .byte $80       ; Dictionary index 0 = "THE "
    .byte 'Q', 'U', 'E', 'S', 'T', ' '  ; Literal "QUEST "
    .byte $81       ; Dictionary index 1 = "AND "
    .byte $84       ; Dictionary index 4 = "HAVE "
    .byte 'F', 'U', 'N', '!'            ; Literal "FUN!"

    ; Output: Decompressed text in text_buffer

    beq decomp_text_done

    tay

    sta dict_ptr

    sta dict_ptr+1

    lda (dict_ptr),y
    beq dict_word_done
    sta text_buffer,x
    inx
    iny
    jmp copy_dict_word
    
dict_word_done:
    ; Restore Y to compressed data
    ; (Complex bookkeeping needed here)
    jmp continue_decompress
    
literal_char:
    ; Copy literal character
    sta text_buffer,x
    inx
    iny
    
continue_decompress:
    jmp decomp_text_loop
    
decomp_text_done:
    lda #0
    sta text_buffer,x ; Null terminator
    rts

; Store only differences from previous values

    .byte 2, 0      ; Right 2
    .byte 1, -1     ; Right 1, Up 1  
    .byte 0, -2     ; Up 2
    .byte -1, -1    ; Left 1, Up 1
    .byte -2, 0     ; Left 2
    .byte 0, 2      ; Down 2
    .byte $80, $80  ; End marker

    ; Input: Starting position in start_x, start_y
    ; Output: Absolute positions in position_buffer
    
    lda start_x
    sta current_x
    lda start_y
    sta current_y

    ldx #0          ; Destination index
    
delta_loop:
    lda movement_deltas,y
    cmp #$80
    beq delta_done
    
    ; Apply X delta
    clc
    adc current_x
    sta current_x
    sta position_buffer,x
    inx
    
    iny
    lda movement_deltas,y ; Y delta
    clc
    adc current_y
    sta current_y
    sta position_buffer,x
    inx
    
    iny
    jmp delta_loop
    
delta_done:
    rts

position_buffer: !fill 32, 0

set_enemy_flag:
    ; Input: X = enemy index, A = flag mask
    
    stx enemy_index
    sta flag_mask
    
    ; Calculate byte and bit position
    txa
    lsr
    lsr
    lsr             ; Divide by 8
    tay             ; Y = byte index
    
    lda enemy_index
    and #$07        ; Bit position within byte
    tax
    
    ; Create bit mask for position
    lda #1
shift_bit:
    cpx #0
    beq bit_positioned
    asl
    dex
    jmp shift_bit
    
bit_positioned:
    and flag_mask   ; Combine with desired flag
    ora enemy_flags_packed,y
    sta enemy_flags_packed,y
    
    rts

; Test enemy flag
test_enemy_flag:
    ; Input: X = enemy index, A = flag mask
    ; Output: Zero flag set if flag is clear
    
    stx enemy_index
    sta flag_mask
    
    txa
    lsr
    lsr
    lsr
    tay
    
    lda enemy_index
    and #$07
    tax
    
    lda #1
shift_test:
    cpx #0
    beq test_positioned
    asl
    dex
    jmp shift_test
    
test_positioned:
    and flag_mask
    and enemy_flags_packed,y
    
    rts

; Hierarchical data structures
; Parent-child relationships for complex objects

; Boss with multiple parts
boss_main:
    .byte 1         ; Active
    .byte 160       ; X position
    .byte 100       ; Y position
    .byte 2         ; Number of parts
    .word boss_parts ; Pointer to parts array

boss_parts:
    ; Left arm
    .byte 1         ; Active
    .byte -20       ; Relative X offset
    .byte -10       ; Relative Y offset
    .byte 50        ; Health
    
    ; Right arm  
    .byte 1         ; Active
    .byte 20        ; Relative X offset
    .byte -10       ; Relative Y offset
    .byte 50        ; Health

; Update hierarchical object
update_boss:
    ; Update main boss
    lda boss_main   ; Check if active
    beq boss_inactive
    
    ; Update boss parts
    lda boss_main+4 ; Number of parts
    tax
    ldy #0          ; Parts array offset
    
update_parts_loop:
    lda boss_parts,y ; Part active?
    beq skip_part
    
    ; Calculate absolute position
    lda boss_main+1 ; Boss X
    clc
    adc boss_parts+1,y ; Part X offset
    sta part_abs_x
    
    lda boss_main+2 ; Boss Y
    clc
    adc boss_parts+2,y ; Part Y offset
    sta part_abs_y
    
    ; Update part at absolute position
    jsr update_boss_part
    
skip_part:
    ; Move to next part (4 bytes each)
    tya
    clc
    adc #4
    tay
    
    dex
    bne update_parts_loop
    
boss_inactive:
    rts

; Working variables
enemy_index:    .byte 0
flag_mask:      .byte 0
part_abs_x:     .byte 0
part_abs_y:     .byte 0
enemy_velocities_x: !fill 32, 0`} language="asm" />
