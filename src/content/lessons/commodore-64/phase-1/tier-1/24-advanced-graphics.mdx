# Advanced Graphics Techniques

Push the VIC-II chip beyond its limits. Create high-resolution graphics, multicolor bitmaps, and effects that seem impossible on 1982 hardware.

## Bitmap Mode

Full screen pixel control:

```asm
; Enable bitmap mode
; Bitmap at $2000, screen at $0400
setup_bitmap:
    ; Set bitmap address
    lda #$3B        ; Bitmap mode on
    sta $D011
    
    lda #$18        ; Bitmap at $2000, screen at $0400
    sta $D018
    
    ; Clear bitmap (8000 bytes)
    lda #0
    ldx #0
clear_bitmap:
    sta $2000,x
    sta $2100,x
    sta $2200,x
    sta $2300,x
    sta $2400,x
    sta $2500,x
    sta $2600,x
    sta $2700,x
    inx
    bne clear_bitmap
    
    ; Continue for all 8000 bytes...
    rts

; Plot pixel in bitmap
; X = x coordinate (0-319)
; Y = y coordinate (0-199)
plot_pixel:
    ; Save coordinates
    stx plot_x
    sty plot_y
    
    ; Calculate byte address
    ; Address = $2000 + (Y & $F8) * 40 + (Y & 7) + (X & $1F8)
    
    tya
    and #$F8        ; Y & $F8
    sta temp
    asl
    asl             ; ×4
    clc
    adc temp        ; ×5
    asl
    asl
    asl             ; ×40 total
    sta bitmap_addr
    
    tya
    and #$07        ; Y & 7
    clc
    adc bitmap_addr
    sta bitmap_addr
    
    txa
    and #$F8        ; X & $1F8
    clc
    adc bitmap_addr
    sta bitmap_addr
    lda #$20        ; High byte
    adc #0
    sta bitmap_addr+1
    
    ; Calculate bit position
    txa
    and #$07        ; X & 7
    tax
    lda bit_table,x
    
    ; Set pixel
    ldy #0
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts

bit_table:
    .byte $80,$40,$20,$10,$08,$04,$02,$01

plot_x:       .byte 0
plot_y:       .byte 0
bitmap_addr:  .word 0
temp:         .byte 0
```

## Line Drawing

Bresenham's algorithm for lines:

```asm
; Draw line from (x0,y0) to (x1,y1)
draw_line:
    ; Calculate deltas
    lda x1
    sec
    sbc x0
    bpl dx_positive
    
    ; dx is negative
    eor #$FF
    clc
    adc #1
    sta dx
    lda #$FF
    sta x_inc
    jmp calc_dy
    
dx_positive:
    sta dx
    lda #1
    sta x_inc
    
calc_dy:
    lda y1
    sec
    sbc y0
    bpl dy_positive
    
    ; dy is negative
    eor #$FF
    clc
    adc #1
    sta dy
    lda #$FF
    sta y_inc
    jmp start_line
    
dy_positive:
    sta dy
    lda #1
    sta y_inc
    
start_line:
    ; Bresenham's algorithm
    lda dx
    cmp dy
    bcs x_major
    
y_major:
    ; dy > dx
    lda dy
    lsr
    sta error
    
y_loop:
    ldx x0
    ldy y0
    jsr plot_pixel
    
    lda y0
    cmp y1
    beq line_done
    
    lda error
    sec
    sbc dx
    sta error
    bpl no_x_step_y
    
    lda error
    clc
    adc dy
    sta error
    
    lda x0
    clc
    adc x_inc
    sta x0
    
no_x_step_y:
    lda y0
    clc
    adc y_inc
    sta y0
    jmp y_loop
    
x_major:
    ; dx >= dy
    lda dx
    lsr
    sta error
    
x_loop:
    ldx x0
    ldy y0
    jsr plot_pixel
    
    lda x0
    cmp x1
    beq line_done
    
    lda error
    sec
    sbc dy
    sta error
    bpl no_y_step_x
    
    lda error
    clc
    adc dx
    sta error
    
    lda y0
    clc
    adc y_inc
    sta y0
    
no_y_step_x:
    lda x0
    clc
    adc x_inc
    sta x0
    jmp x_loop
    
line_done:
    rts

x0:     .byte 0
y0:     .byte 0
x1:     .byte 100
y1:     .byte 50
dx:     .byte 0
dy:     .byte 0
x_inc:  .byte 0
y_inc:  .byte 0
error:  .byte 0
```

## Multicolor Bitmap

160×200 with 4 colors per cell:

```asm
; Multicolor bitmap mode
setup_multicolor:
    lda #$3B        ; Bitmap mode
    sta $D011
    
    lda #$18        ; Multicolor on
    sta $D016
    
    lda #$18        ; Bitmap at $2000
    sta $D018
    
    ; Set up color RAM
    ; Each cell can have:
    ; - Background color ($D021)
    ; - Upper nibble of screen RAM
    ; - Lower nibble of screen RAM  
    ; - Color RAM
    
    ldx #0
set_colors:
    lda #$12        ; Colors 1 and 2
    sta $0400,x
    sta $0500,x
    sta $0600,x
    sta $06E8,x
    
    lda #$07        ; Color 3
    sta $D800,x
    sta $D900,x
    sta $DA00,x
    sta $DAE8,x
    
    inx
    bne set_colors
    rts

; Plot multicolor pixel (2×1 pixels)
plot_multicolor:
    ; X = x coordinate (0-159)
    ; Y = y coordinate (0-199)
    ; A = color (0-3)
    
    pha             ; Save color
    
    ; Calculate byte position
    txa
    asl             ; ×2 for hires coordinates
    tax
    jsr calculate_bitmap_addr
    
    ; Get bit pair position
    txa
    lsr
    and #$03        ; Which pair (0-3)
    tax
    
    pla             ; Get color
    asl
    asl
    cpx #3
    beq pair3
    cpx #2
    beq pair2
    cpx #1
    beq pair1
    
pair0:
    ; Bits 6-7
    asl
    asl
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts
    
pair1:
    ; Bits 4-5
    asl
    asl
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts
    
pair2:
    ; Bits 2-3
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts
    
pair3:
    ; Bits 0-1
    lsr
    lsr
    ora (bitmap_addr),y
    sta (bitmap_addr),y
    rts
```

## Character Mode Graphics

Extended color and multicolor text:

```asm
; Extended color mode (4 background colors)
extended_color_mode:
    lda $D011
    ora #$40        ; ECM bit
    sta $D011
    
    ; Set 4 background colors
    lda #$00        ; Black
    sta $D021
    lda #$01        ; White
    sta $D022
    lda #$02        ; Red
    sta $D023
    lda #$05        ; Green
    sta $D024
    
    ; Character codes 0-63 use different backgrounds
    ; 00-3F: Background 0
    ; 40-7F: Background 1
    ; 80-BF: Background 2
    ; C0-FF: Background 3
    rts

; Multicolor character mode
multicolor_char_mode:
    lda $D016
    ora #$10        ; MCM bit
    sta $D016
    
    ; Characters with color > 7 are multicolor
    ldx #0
setup_mc_chars:
    lda #$08        ; Make multicolor
    sta $D800,x
    sta $D900,x
    sta $DA00,x
    sta $DAE8,x
    inx
    bne setup_mc_chars
    
    ; Set shared colors
    lda #$01        ; White
    sta $D022
    lda #$02        ; Red  
    sta $D023
    rts
```

## Sprite Overlays

Combine sprites for high-res overlays:

```asm
; Overlay sprites for extra colors
sprite_overlay:
    ; Use two sprites at same position
    ; Different colors = more colors!
    
    ; Sprite 0: Base layer
    lda #100
    sta $D000       ; X position
    lda #100
    sta $D001       ; Y position
    lda #$01        ; White
    sta $D027
    
    ; Sprite 1: Overlay (use OR/AND logic)
    lda #100
    sta $D002
    lda #100
    sta $D003
    lda #$02        ; Red
    sta $D028
    
    ; Priority: sprite 1 behind sprite 0
    lda #$01
    sta $D01B
    
    ; Create composite colors
    rts

; Sprite-based text overlay
sprite_text:
    ; 8 sprites = 24×21 pixels each
    ; Can create 192×21 pixel text banner
    
    ldx #0
position_text_sprites:
    txa
    asl
    asl
    asl             ; ×8
    asl
    asl             ; ×32
    clc
    adc #24         ; Start X
    sta $D000,x
    
    lda #50         ; Y position
    sta $D001,x
    
    ; Point to character sprite
    txa
    clc
    adc #$80        ; Sprite pointers
    sta $07F8,x
    
    inx
    inx             ; Next sprite
    cpx #16
    bne position_text_sprites
    
    ; Enable all sprites
    lda #$FF
    sta $D015
    rts
```

## Fill Algorithms

Fast area filling:

```asm
; Stack-based flood fill
flood_fill:
    ; X,Y = start position
    ; A = fill color
    
    sta fill_color
    
    ; Get original color
    jsr get_pixel
    sta original_color
    cmp fill_color
    beq fill_done   ; Already filled
    
    ; Push starting point
    txa
    pha
    tya
    pha
    
fill_loop:
    ; Pop point from stack
    tsx
    cpx #$FF
    beq fill_done   ; Stack empty
    
    pla
    tay
    pla
    tax
    
    ; Check if we should fill
    jsr get_pixel
    cmp original_color
    bne fill_loop   ; Different color
    
    ; Fill this pixel
    lda fill_color
    jsr set_pixel
    
    ; Push adjacent pixels
    ; Right
    inx
    cpx #160        ; Boundary check
    bcs skip_right
    txa
    pha
    tya
    pha
skip_right:
    
    ; Left
    dex
    dex
    bpl push_left
    inx
    jmp skip_left
push_left:
    txa
    pha
    tya
    pha
skip_left:
    inx             ; Restore X
    
    ; Down
    iny
    cpy #200
    bcs skip_down
    txa
    pha
    tya
    pha
skip_down:
    
    ; Up
    dey
    dey
    bpl push_up
    iny
    jmp fill_loop
push_up:
    txa
    pha
    tya
    pha
    iny             ; Restore Y
    
    jmp fill_loop
    
fill_done:
    rts

fill_color:     .byte 0
original_color: .byte 0
```

## 3D Wireframe

Simple 3D projection:

```asm
; 3D to 2D projection
; Project 3D point (x,y,z) to screen
project_3d:
    ; Perspective projection
    ; screen_x = x * 128 / (z + 128) + 160
    ; screen_y = y * 128 / (z + 128) + 100
    
    ; Calculate divisor (z + 128)
    lda z_coord
    clc
    adc #128
    sta divisor
    
    ; Project X
    lda x_coord
    ldx #128
    jsr multiply
    ldx divisor
    jsr divide
    clc
    adc #160        ; Center X
    sta screen_x
    
    ; Project Y
    lda y_coord
    ldx #128
    jsr multiply
    ldx divisor
    jsr divide
    clc
    adc #100        ; Center Y
    sta screen_y
    
    rts

; Draw wireframe cube
draw_cube:
    ; Define cube vertices
    ; Front face
    lda #-20
    sta x_coord
    sta y_coord
    lda #20
    sta z_coord
    jsr project_3d
    ldx screen_x
    ldy screen_y
    jsr plot_pixel
    
    ; ... project and draw all 8 vertices
    
    ; Draw edges
    ; ... draw 12 edges connecting vertices
    
    rts

; Rotate cube
rotate_cube:
    ; Simple rotation around Y axis
    ldx #0
rotate_vertex:
    ; x' = x * cos - z * sin
    ; z' = x * sin + z * cos
    
    lda cube_x,x
    sta old_x
    lda cube_z,x
    sta old_z
    
    ; Simplified with fixed angle
    lda old_x
    sec
    sbc old_z
    lsr             ; Approximate cos/sin
    sta cube_x,x
    
    lda old_z
    clc
    adc old_x
    lsr
    sta cube_z,x
    
    inx
    cpx #8          ; 8 vertices
    bne rotate_vertex
    rts

x_coord:    .byte 0
y_coord:    .byte 0
z_coord:    .byte 0
divisor:    .byte 0
screen_x:   .byte 0
screen_y:   .byte 0
old_x:      .byte 0
old_z:      .byte 0

cube_x: .res 8
cube_y: .res 8
cube_z: .res 8
```

Putting it all together:

```asm

    jmp draw_square

    lda #$00

    asl
```

## Key Concepts

1. **Bitmap Mode**: Full pixel control at 320×200
2. **Multicolor Mode**: Trade resolution for more colors
3. **Line Algorithms**: Efficient line drawing
4. **3D Projection**: Transform 3D coordinates to 2D
5. **Fill Algorithms**: Fast area filling techniques
