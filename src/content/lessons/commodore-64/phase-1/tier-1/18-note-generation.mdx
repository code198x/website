# Note Generation: Musical Tones and Scales
**From beeps to Bach - making real music on the SID**

The difference between a beep and a note is frequency. When you played that first sound in Lesson 17, you were already making music—you just didn't know which note it was. The SID chip can produce any frequency from deep bass to ultrasonic, but certain frequencies correspond to musical notes. Middle C is 261.63 Hz. A above middle C is 440 Hz. These aren't random—they follow mathematical patterns discovered centuries ago.

Today you'll learn to speak music to the SID chip. You'll convert note names to frequencies, play scales that sound right, and create the simple melodies that make games memorable. That victory fanfare when you complete a level? That's just a sequence of notes. The coin collection sound in "Super Mario"? Two notes. The death jingle in "Pac-Man"? A descending scale. Music is patterns, and you're about to master them.

This is where games gain their musical voice—from simple jingles to memorable themes.

## Understanding Musical Frequencies

### The Mathematics of Music

Music follows mathematical relationships:

```text
Musical Note Frequencies:
Note  Frequency  SID Value (Freq × 0.0596)
C0    16.35 Hz   $0003
C1    32.70 Hz   $0007  
C2    65.41 Hz   $000F
C3    130.81 Hz  $001F
C4    261.63 Hz  $03F4 (Middle C)
C5    523.25 Hz  $07E8
C6    1046.50 Hz $0FD0
C7    2093.00 Hz $1FA0
C8    4186.01 Hz $3F40

The Pattern:
- Each octave doubles the frequency
- 12 semitones per octave
- Multiply by 12th root of 2 (≈1.059) for next semitone

SID Frequency Formula:
SID_value = Frequency_Hz × (16777216 / 1022727)
         ≈ Frequency_Hz × 16.4

For PAL machines:
SID_value = Frequency_Hz × 17.03
```

## Playing Musical Notes

### Simple Note Player

```asm
; Musical note player for SID
; Play any note by name

        * = $0801
        !byte $0b,$08,$0a,$00,$9e,$32,$30,$36,$31,$00,$00,$00

        * = $080d
start:
        ; Initialize
        jsr init_music
        
        ; Play middle C
        lda #NOTE_C4
        jsr play_note
        
        ; Wait
        jsr wait_duration
        
        ; Play E
        lda #NOTE_E4
        jsr play_note
        
        jsr wait_duration
        
        ; Play G
        lda #NOTE_G4
        jsr play_note
        
        jsr wait_duration
        
        ; Stop
        jsr stop_note
        
        rts

init_music:
        ; Clear SID
        ldx #$18
        lda #0
clear_music:
        sta $d400,x
        dex
        bpl clear_music
        
        ; Set volume
        lda #15
        sta $d418
        
        ; Set envelope for musical notes
        lda #$00        ; Fast attack, no decay
        sta $d405
        lda #$f0        ; Full sustain, no release
        sta $d406
        
        ; Set waveform (triangle for pure tone)
        lda #$10
        sta $d404
        
        rts

play_note:
        ; Input: A = note number (0-95)
        ; Play the specified note
        
        tax
        
        ; Set frequency low byte
        lda note_freq_lo,x
        sta $d400
        
        ; Set frequency high byte
        lda note_freq_hi,x
        sta $d401
        
        ; Gate on
        lda #$11        ; Triangle + gate
        sta $d404
        
        rts

stop_note:
        ; Gate off
        lda #$10
        sta $d404
        rts

wait_duration:
        ; Wait for note duration
        ldx #30
duration_loop:
        lda #250
dw:     cmp $d012
        bne dw
        dex
        bne duration_loop
        rts

; Note name constants (index into frequency table)
NOTE_C0  = 0
NOTE_CS0 = 1
NOTE_D0  = 2
NOTE_DS0 = 3
NOTE_E0  = 4
NOTE_F0  = 5
NOTE_FS0 = 6
NOTE_G0  = 7
NOTE_GS0 = 8
NOTE_A0  = 9
NOTE_AS0 = 10
NOTE_B0  = 11

NOTE_C4  = 48   ; Middle C
NOTE_CS4 = 49
NOTE_D4  = 50
NOTE_DS4 = 51
NOTE_E4  = 52
NOTE_F4  = 53
NOTE_FS4 = 54
NOTE_G4  = 55
NOTE_GS4 = 56
NOTE_A4  = 57
NOTE_AS4 = 58
NOTE_B4  = 59
```

## Playing Scales

### Major and Minor Scales

```asm
play_major_scale:
        ; Play C major scale ascending
        
        ldx #0
major_loop:
        lda c_major_scale,x
        cmp #$ff        ; End marker
        beq major_done
        
        jsr play_note
        jsr wait_duration
        jsr stop_note
        
        ; Brief pause between notes
        ldy #5
        jsr wait_frames
        
        inx
        jmp major_loop
        
major_done:
        rts

play_minor_scale:
        ; Play A minor scale
        
        ldx #0
minor_loop:
        lda a_minor_scale,x
        cmp #$ff
        beq minor_done
        
        jsr play_note
        jsr wait_duration
        jsr stop_note
        
        ldy #5
        jsr wait_frames
        
        inx
        jmp minor_loop
        
minor_done:
        rts

play_chromatic_scale:
        ; Play all 12 semitones
        
        lda #NOTE_C4
        sta current_note
        
        ldx #12
chromatic_loop:
        lda current_note
        jsr play_note
        
        ldy #15         ; Shorter duration
        jsr wait_frames
        
        jsr stop_note
        
        ldy #2
        jsr wait_frames
        
        inc current_note
        
        dex
        bne chromatic_loop
        
        rts

play_pentatonic:
        ; Play pentatonic scale (black keys)
        ; Popular in game music
        
        ldx #0
pent_loop:
        lda pentatonic_scale,x
        cmp #$ff
        beq pent_done
        
        jsr play_note
        jsr wait_duration
        jsr stop_note
        
        ldy #5
        jsr wait_frames
        
        inx
        jmp pent_loop
        
pent_done:
        rts

wait_frames:
        ; Wait Y frames
wf:     lda #250
        cmp $d012
        bne *-3
        dey
        bne wf
        rts

; Scale data
c_major_scale:
        !byte NOTE_C4, NOTE_D4, NOTE_E4, NOTE_F4
        !byte NOTE_G4, NOTE_A4, NOTE_B4, NOTE_C5
        !byte $ff

a_minor_scale:
        !byte NOTE_A4, NOTE_B4, NOTE_C5, NOTE_D5
        !byte NOTE_E5, NOTE_F5, NOTE_G5, NOTE_A5
        !byte $ff

pentatonic_scale:
        !byte NOTE_C4, NOTE_D4, NOTE_F4
        !byte NOTE_G4, NOTE_A4, NOTE_C5
        !byte $ff

current_note: !byte 0
```

## Arpeggios and Chords

### Creating Harmonies

```asm
play_arpeggio:
        ; Fast arpeggio creates chord illusion
        
        lda #0
        sta arp_index
        
        ldx #60         ; Duration
arp_loop:
        txa
        pha
        
        ; Get current note
        ldx arp_index
        lda arpeggio_notes,x
        
        ; Play it
        jsr play_note
        
        ; Very brief
        ldy #2
        jsr wait_frames
        
        ; Next note
        inc arp_index
        lda arp_index
        cmp #3          ; 3 notes in arpeggio
        bne arp_continue
        lda #0
        sta arp_index
        
arp_continue:
        pla
        tax
        dex
        bne arp_loop
        
        jsr stop_note
        rts

play_broken_chord:
        ; Slower broken chord
        
        ldx #0
broken_loop:
        lda chord_notes,x
        cmp #$ff
        beq broken_done
        
        jsr play_note
        
        ldy #15
        jsr wait_frames
        
        inx
        jmp broken_loop
        
broken_done:
        jsr stop_note
        rts

play_power_chord:
        ; Alternating root and fifth
        
        ldx #8
power_loop:
        ; Root
        lda #NOTE_C3
        jsr play_note
        ldy #10
        jsr wait_frames
        
        ; Fifth
        lda #NOTE_G3
        jsr play_note
        ldy #10
        jsr wait_frames
        
        dex
        bne power_loop
        
        jsr stop_note
        rts

; Arpeggio data
arpeggio_notes:
        !byte NOTE_C4, NOTE_E4, NOTE_G4  ; C major

chord_notes:
        !byte NOTE_C4, NOTE_E4, NOTE_G4, NOTE_C5
        !byte NOTE_G4, NOTE_E4, NOTE_C4
        !byte $ff

arp_index: !byte 0
```

### Network Handshake Communication

```asm
network_handshake:
        ; Three-way handshake protocol sound
        ; SYN, SYN-ACK, ACK sequence
        
        ; SYN - Outgoing request
        lda #NOTE_C4
        jsr play_note
        ldy #10
        jsr wait_frames
        jsr stop_note
        
        ; Brief pause
        ldy #8
        jsr wait_frames
        
        ; SYN-ACK - Response with acknowledgment
        lda #NOTE_G4
        jsr play_note
        ldy #8
        jsr wait_frames
        jsr stop_note
        
        ldy #4
        jsr wait_frames
        
        lda #NOTE_E4
        jsr play_note
        ldy #8
        jsr wait_frames
        jsr stop_note
        
        ; Brief pause
        ldy #8
        jsr wait_frames
        
        ; ACK - Final acknowledgment
        lda #NOTE_C5
        jsr play_note
        ldy #12
        jsr wait_frames
        jsr stop_note
        
        ; Connection established
        rts

network_protocol_sequence:
        ; Complex multi-step protocol negotiation
        ; Represents advanced network communication
        
        ldx #0
protocol_loop:
        lda protocol_sequence,x
        cmp #$ff
        beq protocol_done
        
        cmp #REST
        beq protocol_rest
        
        jsr play_note
        jmp protocol_timing
        
protocol_rest:
        jsr stop_note

        ; Get timing for this step
        lda protocol_timings,x
        tay
        jsr wait_frames
        
        inx
        jmp protocol_loop
        
protocol_done:
        jsr stop_note
        rts

        ; Alternating tones with perfect intervals
        
        ; Establish connection
        lda #NOTE_C4
        jsr play_note
        ldy #8
        jsr wait_frames

        lda #NOTE_G4        ; Perfect fifth
        jsr play_note
        ldy #6
        jsr wait_frames

        lda #NOTE_E4        ; Major third
        jsr play_note
        ldy #6
        jsr wait_frames
        
        ; Transfer complete
        lda #NOTE_C5        ; Octave resolution
        jsr play_note
        ldy #10
        jsr wait_frames
        
        jsr stop_note
        rts

        ; Periodic network status check tones
        ; Subtle background communication sounds
        
        ; Quick status ping
        lda #NOTE_A4

        ldy #3
        jsr wait_frames

        jsr wait_frames

        jsr wait_frames

        bne sync_loop
```

```asm
        beq use_data_stream
        cmp #2
```

```text
Minor: W-H-W-W-H-W-W
Pentatonic: W-W-m3-W-m3

Tempo Guidelines:
Fast: 5-10 frames per note
Medium: 15-20 frames per note
Slow: 30-40 frames per note

Chord Arpeggios:
Major: Root-3rd-5th (C-E-G)
Minor: Root-b3rd-5th (C-Eb-G)
7th: Root-3rd-5th-7th (C-E-G-B)
Power: Root-5th (C-G)

Tips:
- Use triangle for pure tones
- Use pulse for hollow sounds
- Use sawtooth for bright leads
- Keep melodies simple
- Repeat memorable phrases
```
