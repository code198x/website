## Conceptual Overview

### Understanding Value Comparison

When you compare two numbers in your head, you're actually performing a subtraction and looking at the result. If you want to know if 7 is greater than 5, you calculate 7 - 5 = 2 (positive), so yes, 7 is greater. The 6502 processor does exactly the same thing with the CMP (Compare) instruction - it subtracts one value from another without storing the result, just setting flags based on what happened.

The genius of this approach is that your original values remain untouched. The processor just updates its status flags to tell you about the relationship between the numbers.

### How It Works on the Commodore 64

The 6502 has three comparison instructions:
- **CMP** - Compare with Accumulator
- **CPX** - Compare with X Register
- **CPY** - Compare with Y Register

Each performs a subtraction and sets flags:
- **Zero Flag (Z)** - Set if values are equal
- **Carry Flag (C)** - Set if first value ≥ second value
- **Negative Flag (N)** - Set if result would be negative

```asm
; Basic comparison pattern
    lda player_score
    cmp #100           ; Compare player_score with 100
    bcs extra_life     ; Branch if score >= 100
    ; Score is less than 100
    jmp continue
extra_life:
    ; Award extra life
    inc player_lives
continue:
```

### Core Implementation

Now let's implement the main concept:

```asm
main:
    jsr clear_screen

    ; Example 1: Simple equality check
    lda #10
    sta player_score

    lda player_score
    cmp #10
    bne not_ten
    ; Score is exactly 10
    lda #'='
    sta SCREEN
    jmp test_greater
not_ten:
    lda #'!'
    sta SCREEN

test_greater:
    ; Example 2: Greater than or equal
    lda player_score
    cmp #5            ; Is score >= 5?
    bcc too_low       ; Branch if < 5
    ; Score is >= 5
    lda #'>'
    sta SCREEN+1
    jmp test_range
too_low:
    lda #'<'
    sta SCREEN+1

test_range:
    ; Example 3: Range checking (5 <= score <= 15)
    lda player_score
    cmp #5            ; Check lower bound
    bcc out_of_range  ; Branch if < 5
    cmp #16           ; Check upper bound + 1
    bcs out_of_range  ; Branch if >= 16
    ; Score is in range [5,15]
    lda #GREEN
    sta COLOR
    jmp continue_tests
out_of_range:
    lda #RED
    sta COLOR

continue_tests:
    ; Example 4: Compare two variables
    lda player_x
    cmp enemy_x       ; Who is further right?
    bcs player_ahead  ; Branch if player_x >= enemy_x
    ; Enemy is ahead
    lda #'E'
    sta SCREEN+40
    jmp done
player_ahead:
    lda #'P'
    sta SCREEN+40

done:
    rts
```

## Common Pitfalls

Watch out for these common mistakes:

1. **Forgetting CMP doesn't change the accumulator**: The value in A remains unchanged after CMP. If you need to use the subtraction result, use SBC instead.

2. **Confusing BCS/BCC logic**: BCS means "greater than or equal" after CMP, not just "greater than". For strictly greater than, use BEQ to skip the equal case.

3. **Wrong comparison order**: CMP #10 means "compare A with 10", subtracting 10 from A internally. The flags reflect A's relationship to 10, not the other way around.

## Deep Dive: 16-Bit Comparisons

For those who want to understand more deeply:

Games often need to compare 16-bit values (0-65535) for scores, positions, or timers. This requires comparing two bytes, starting with the high byte:

```asm
; 16-bit comparison technique
; Compare score_low/score_high with goal_low/goal_high

    lda score_high
    cmp goal_high
    bcc score_less      ; High byte is less
    bne score_greater    ; High byte is greater
    ; High bytes equal, check low bytes
    lda score_low
    cmp goal_low
    bcc score_less
    ; Score >= goal
score_greater:
    ; Handle score >= goal
    jmp continue
score_less:
    ; Handle score < goal
continue:
```

## Complete Working Example

Here's the full code bringing everything together:

```asm
; Complete Comparison Demo - High Score System
; Shows all comparison techniques in action

* = $0801
    !byte $0c,$08,$0a,$00,$9e,$32
    !byte $30,$36,$34,$00,$00,$00
* = $0810

; Constants
SCREEN = $0400
COLOR = $d800
BORDER = $d020
WHITE = 1
RED = 2
GREEN = 5
YELLOW = 7
BLUE = 6

; Variables
player_score:    !byte 0
high_score:      !byte 25
bonus_threshold: !byte 50
lives:           !byte 3
level:           !byte 1

main:
    jsr clear_screen
    jsr display_status

game_loop:
    ; Simulate scoring
    jsr wait_frame
    inc player_score

    ; Check for new high score
    lda player_score
    cmp high_score
    bcc check_bonus      ; Not a new high score
    beq check_bonus      ; Equal to high score
    ; NEW HIGH SCORE!
    sta high_score       ; Update high score
    lda #YELLOW
    sta BORDER           ; Flash border
    jsr display_high_score_message

check_bonus:
    ; Check for bonus life at 50 points
    lda player_score
    cmp bonus_threshold
    bne check_level      ; Not at bonus threshold
    ; Award bonus life!
    inc lives
    lda #GREEN
    sta BORDER
    jsr display_bonus_message
    ; Increase threshold for next bonus
    lda bonus_threshold
    clc
    adc #50
    sta bonus_threshold

check_level:
    ; Increase difficulty every 20 points
    lda player_score
    and #%00010011       ; Check if divisible by 20(ish)
    bne update_display
    inc level

update_display:
    jsr display_status

    ; Check game over at 100 points
    lda player_score
    cmp #100
    bcc game_loop

    ; Game complete!
    jsr game_complete
    rts

; Display routines
display_status:
    ; Show score
    lda #'S'
    sta SCREEN
    lda #':'
    sta SCREEN+1
    lda player_score
    jsr display_number

    ; Show high score
    lda #'H'
    sta SCREEN+40
    lda #':'
    sta SCREEN+41
    lda high_score
    jsr display_number_line2

    ; Show lives
    lda #'L'
    sta SCREEN+80
    lda #':'
    sta SCREEN+81
    lda lives
    clc
    adc #'0'
    sta SCREEN+82

    ; Show level
    lda #'L'
    sta SCREEN+120
    lda #'V'
    sta SCREEN+121
    lda #':'
    sta SCREEN+122
    lda level
    clc
    adc #'0'
    sta SCREEN+123
    rts

display_number:
    ; Simple 2-digit display at SCREEN+2
    tax
    lda #'0'
    sta SCREEN+2
    sta SCREEN+3
    txa
    cmp #10
    bcc single_digit

    ; Count tens
    ldx #'0'
count_tens:
    cmp #10
    bcc show_ones
    inx
    sec
    sbc #10
    jmp count_tens
show_ones:
    stx SCREEN+2
single_digit:
    clc
    adc #'0'
    sta SCREEN+3
    rts

display_number_line2:
    ; Display on line 2
    tax
    lda #'0'
    sta SCREEN+42
    sta SCREEN+43
    txa
    cmp #10
    bcc single_digit2

    ldx #'0'
count_tens2:
    cmp #10
    bcc show_ones2
    inx
    sec
    sbc #10
    jmp count_tens2
show_ones2:
    stx SCREEN+42
single_digit2:
    clc
    adc #'0'
    sta SCREEN+43
    rts

display_high_score_message:
    ldx #0
high_loop:
    lda high_msg,x
    beq high_done
    sta SCREEN+200,x
    lda #YELLOW
    sta COLOR+200,x
    inx
    jmp high_loop
high_done:
    rts

display_bonus_message:
    ldx #0
bonus_loop:
    lda bonus_msg,x
    beq bonus_done
    sta SCREEN+280,x
    lda #GREEN
    sta COLOR+280,x
    inx
    jmp bonus_loop
bonus_done:
    rts

game_complete:
    ldx #0
complete_loop:
    lda complete_msg,x
    beq complete_done
    sta SCREEN+480,x
    lda #WHITE
    sta COLOR+480,x
    inx
    jmp complete_loop
complete_done:
    rts

wait_frame:
    ; Wait for raster
    lda #251
wait:
    cmp $d012
    bne wait
    rts

clear_screen:
    ldx #0
    lda #' '
clear_loop:
    sta SCREEN,x
    sta SCREEN+256,x
    sta SCREEN+512,x
    sta SCREEN+768,x
    inx
    bne clear_loop

    ; Reset border
    lda #BLUE
    sta BORDER
    rts

; Messages
high_msg:     !scr "new high score!",0
bonus_msg:    !scr "bonus life!",0
complete_msg: !scr "amazing! 100 points!",0
```

## Visual Guide

```
CMP Operation Visualization:

    LDA #10     [A = 10]
    CMP #7      [Internally: 10 - 7 = 3]

    Results:
    Z Flag: 0 (not zero, so not equal)
    C Flag: 1 (no borrow needed, so A >= 7)
    N Flag: 0 (result positive)

    Branches available:
    BEQ - Won't branch (Z=0)
    BNE - Will branch (Z=0)
    BCS - Will branch (C=1, meaning >=)
    BCC - Won't branch (C=1)
```

## Quick Reference

### Key Commands Used
- `CMP`: Compare accumulator with memory/immediate
- `CPX`: Compare X register with memory/immediate
- `CPY`: Compare Y register with memory/immediate
- `BCS`: Branch if Carry Set (&gt;=)
- `BCC`: Branch if Carry Clear (&lt;)
- `BEQ`: Branch if Equal (Z=1)
- `BNE`: Branch if Not Equal (Z=0)

### Memory Locations
- `$0400`: Screen memory start
- `$D800`: Color memory start
- `$D020`: Border color register

### Comparison Patterns
- Equal: `CMP #n, BEQ`
- Not Equal: `CMP #n, BNE`
- Greater or Equal: `CMP #n, BCS`
- Less Than: `CMP #n, BCC`
- Greater Than: `CMP #n, BEQ skip, BCS target`
