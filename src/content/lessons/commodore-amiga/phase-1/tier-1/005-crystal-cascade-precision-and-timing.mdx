# Lesson 5: Crystal Cascade - Precision and Timing

Time to achieve arcade-perfect precision! In this lesson, you'll master the timing systems that separate amateur games from professional ones. You'll implement fixed-point mathematics, frame-perfect synchronization, and deterministic gameplay that feels smooth and responsive.

## Understanding Amiga Timing Systems

Precision on the Amiga requires understanding hardware timing:

1. **50Hz PAL / 60Hz NTSC** - VBlank provides stable frame timing
2. **Copper List Timing** - Cycle-accurate display synchronization  
3. **CPU Timing** - 7MHz 68000 with predictable instruction cycles
4. **DMA Slots** - Hardware access patterns affect timing
5. **Fixed-Point Math** - Sub-pixel precision without floating point

## Step 1: Fixed-Point Mathematics System

Create precise sub-pixel movement:

```m68k
; Crystal Cascade - Lesson 5
; Precision and timing systems

        include "hardware/custom.i"
        include "hardware/dmabits.i"

; Fixed-point format: 16.16 (16 integer bits, 16 fractional bits)
FIXED_SHIFT     = 16
FIXED_ONE       = 1<<FIXED_SHIFT        ; 1.0 in fixed point
FIXED_HALF      = FIXED_ONE>>1          ; 0.5 in fixed point

; Convert integer to fixed point
; d0 = integer value
; Returns: d0 = fixed point value
int_to_fixed:
        lsl.l   #FIXED_SHIFT,d0
        rts

; Convert fixed point to integer
; d0 = fixed point value  
; Returns: d0 = integer value
fixed_to_int:
        asr.l   #FIXED_SHIFT,d0
        rts

; Fixed point multiplication
; d0 = fixed point A
; d1 = fixed point B
; Returns: d0 = A * B (fixed point)
fixed_mul:
        move.l  d0,d2
        muls.l  d1,d0                   ; 32x32 = 64 bit result
        move.l  d0,d3                   ; Low 32 bits
        swap    d0                      ; High 32 bits to low
        move.w  d0,d3                   ; Combine for 32-bit result
        move.l  d3,d0
        asr.l   #FIXED_SHIFT,d0         ; Adjust for fixed point
        rts

; Fixed point division
; d0 = fixed point dividend
; d1 = fixed point divisor
; Returns: d0 = dividend / divisor (fixed point)
fixed_div:
        lsl.l   #FIXED_SHIFT,d0         ; Scale up dividend
        divs.l  d1,d0                   ; Divide
        rts

; Enhanced sprite structure with fixed-point precision
        section sprite_data,data

; Player sprite with sub-pixel precision
player_sprite:
        dc.l    160<<FIXED_SHIFT        ; X position (fixed point)
        dc.l    100<<FIXED_SHIFT        ; Y position (fixed point)
        dc.l    0                       ; X velocity (fixed point)
        dc.l    0                       ; Y velocity (fixed point)
        dc.l    0                       ; X acceleration (fixed point)
        dc.l    0                       ; Y acceleration (fixed point)
        dc.w    0                       ; Animation frame
        dc.w    0                       ; Animation timer
        dc.w    0                       ; State flags

; Movement constants (fixed point)
GRAVITY         = $0000C000             ; 0.75 in fixed point
JUMP_STRENGTH   = $FFF40000             ; -12.0 in fixed point  
FRICTION        = $0000F000             ; 0.9375 friction coefficient
MAX_VELOCITY    = $00050000             ; 5.0 maximum velocity
ACCELERATION    = $00008000             ; 0.5 acceleration

        section code,code

; Frame-perfect input handling
handle_precise_input:
        lea     custom,a5
        
        ; Read current joystick state
        move.w  joy1dat(a5),d0
        move.w  d0,joy_current_frame
        
        ; Calculate input delta (what changed this frame)
        move.w  joy_previous_frame,d1
        eor.w   d0,d1                   ; XOR to find changes
        move.w  d1,joy_changed_frame
        
        ; Store current as previous for next frame
        move.w  d0,joy_previous_frame
        
        ; Process movement with fixed-point precision
        move.l  #player_sprite,a0
        
        ; Check horizontal movement
        btst    #1,d0                   ; Joy right
        beq     check_left
        
        ; Accelerate right
        move.l  8(a0),d1                ; Current X velocity
        add.l   #ACCELERATION,d1
        cmp.l   #MAX_VELOCITY,d1
        ble     set_x_velocity
        move.l  #MAX_VELOCITY,d1
        bra     set_x_velocity
        
check_left:
        btst    #9,d0                   ; Joy left  
        beq     apply_friction
        
        ; Accelerate left
        move.l  8(a0),d1                ; Current X velocity
        sub.l   #ACCELERATION,d1
        move.l  #MAX_VELOCITY,d2
        neg.l   d2                      ; -MAX_VELOCITY
        cmp.l   d2,d1
        bge     set_x_velocity
        move.l  d2,d1
        bra     set_x_velocity
        
apply_friction:
        ; Apply friction when no input
        move.l  8(a0),d1                ; Current X velocity
        move.l  d1,d0
        move.l  #FRICTION,d2
        bsr     fixed_mul
        move.l  d0,d1
        
set_x_velocity:
        move.l  d1,8(a0)                ; Store X velocity
        
        ; Check for jump (fire button)
        move.w  $dff016,d0              ; CIA port A
        btst    #7,d0                   ; Fire button
        bne     no_jump
        
        ; Check if this is a new press (edge detection)
        btst    #7,joy_changed_frame
        beq     no_jump
        
        ; Jump only if on ground (Y velocity near zero and at bottom)
        move.l  4(a0),d0                ; Y position
        move.l  #224<<FIXED_SHIFT,d1    ; Ground level
        cmp.l   d1,d0
        blt     no_jump
        
        ; Apply jump velocity
        move.l  #JUMP_STRENGTH,12(a0)   ; Y velocity
        
no_jump:
        rts

; Frame variables
joy_current_frame:      dc.w    0
joy_previous_frame:     dc.w    0
joy_changed_frame:      dc.w    0
```

## Step 2: Precise Physics Integration

Implement accurate physics with proper integration:

```m68k
; Physics update with precise integration
update_sprite_physics:
        move.l  #player_sprite,a0
        
        ; Apply gravity to Y velocity
        move.l  12(a0),d0               ; Current Y velocity
        add.l   #GRAVITY,d0             ; Add gravity
        move.l  d0,12(a0)               ; Store new Y velocity
        
        ; Integrate velocity to position (Euler integration)
        ; X position += X velocity
        move.l  (a0),d0                 ; X position
        add.l   8(a0),d0                ; Add X velocity
        move.l  d0,(a0)                 ; Store new X position
        
        ; Y position += Y velocity  
        move.l  4(a0),d1                ; Y position
        add.l   12(a0),d1               ; Add Y velocity
        move.l  d1,4(a0)                ; Store new Y position
        
        ; Boundary checking with precise collision
        ; Left boundary
        tst.l   d0                      ; X position < 0?
        bpl     check_right_boundary
        clr.l   (a0)                    ; Clamp to 0
        clr.l   8(a0)                   ; Stop X velocity
        
check_right_boundary:
        move.l  #304<<FIXED_SHIFT,d2    ; Right boundary
        cmp.l   d2,d0
        ble     check_ground
        move.l  d2,(a0)                 ; Clamp to right edge
        clr.l   8(a0)                   ; Stop X velocity
        
check_ground:
        move.l  #224<<FIXED_SHIFT,d2    ; Ground level
        cmp.l   d2,d1
        ble     check_ceiling
        move.l  d2,4(a0)                ; Clamp to ground
        clr.l   12(a0)                  ; Stop Y velocity
        
check_ceiling:
        move.l  #16<<FIXED_SHIFT,d2     ; Ceiling
        cmp.l   d2,d1
        bge     physics_done
        move.l  d2,4(a0)                ; Clamp to ceiling
        clr.l   12(a0)                  ; Stop Y velocity
        
physics_done:
        rts

; Precise collision detection with sub-pixel accuracy
check_precise_collision:
        ; Check collision between player and enemies
        move.l  #player_sprite,a0
        lea     enemy_pool,a1
        move.w  #MAX_ENEMIES-1,d7
        
collision_loop:
        tst.w   ENEMY_STATE(a1)         ; Skip inactive enemies
        beq     next_collision
        
        ; Get positions in fixed point
        move.l  (a0),d0                 ; Player X (fixed)
        move.l  4(a0),d1                ; Player Y (fixed)
        
        ; Convert enemy position to fixed point
        move.w  ENEMY_X(a1),d2
        ext.l   d2
        lsl.l   #FIXED_SHIFT,d2         ; Enemy X (fixed)
        
        move.w  ENEMY_Y(a1),d3
        ext.l   d3
        lsl.l   #FIXED_SHIFT,d3         ; Enemy Y (fixed)
        
        ; Calculate distance (fixed point)
        sub.l   d2,d0                   ; dx = player_x - enemy_x
        bpl     dx_positive
        neg.l   d0
dx_positive:
        sub.l   d3,d1                   ; dy = player_y - enemy_y  
        bpl     dy_positive
        neg.l   d1
dy_positive:
        
        ; Check collision (16 pixel radius in fixed point)
        move.l  #16<<FIXED_SHIFT,d4
        cmp.l   d4,d0
        bgt     next_collision
        cmp.l   d4,d1
        bgt     next_collision
        
        ; Collision detected - handle with precise response
        bsr     handle_precise_collision
        
next_collision:
        add.w   #ENEMY_SIZE,a1
        dbf     d7,collision_loop
        
        rts

; Precise collision response with momentum conservation
handle_precise_collision:
        ; Calculate collision normal and apply response
        move.l  (a0),d0                 ; Player X
        move.l  4(a0),d1                ; Player Y
        
        ; Convert enemy position to fixed point
        move.w  ENEMY_X(a1),d2
        ext.l   d2
        lsl.l   #FIXED_SHIFT,d2
        move.w  ENEMY_Y(a1),d3
        ext.l   d3
        lsl.l   #FIXED_SHIFT,d3
        
        ; Calculate collision vector
        sub.l   d2,d0                   ; dx
        sub.l   d3,d1                   ; dy
        
        ; Normalize collision vector (simplified)
        move.l  d0,d4
        bpl     norm_dx_pos
        neg.l   d4
norm_dx_pos:
        move.l  d1,d5
        bpl     norm_dy_pos
        neg.l   d5
norm_dy_pos:
        add.l   d5,d4                   ; Manhattan distance
        
        ; Avoid division by zero
        tst.l   d4
        beq     collision_done
        
        ; Calculate reflection vector
        move.l  d0,d2
        move.l  #FIXED_ONE,d3
        bsr     fixed_div               ; Normalize dx
        move.l  d0,d2                   ; Normalized dx
        
        move.l  d1,d0
        move.l  d4,d1
        bsr     fixed_div               ; Normalize dy
        move.l  d0,d3                   ; Normalized dy
        
        ; Apply reflection to velocity
        move.l  8(a0),d0                ; Player X velocity
        move.l  d2,d1                   ; Normalized dx
        bsr     fixed_mul
        lsl.l   #1,d0                   ; 2 * dot product
        move.l  d2,d1
        bsr     fixed_mul
        sub.l   d0,8(a0)                ; Reflect X velocity
        
        move.l  12(a0),d0               ; Player Y velocity
        move.l  d3,d1                   ; Normalized dy
        bsr     fixed_mul
        lsl.l   #1,d0                   ; 2 * dot product
        move.l  d3,d1
        bsr     fixed_mul
        sub.l   d0,12(a0)               ; Reflect Y velocity
        
collision_done:
        rts
```

## Step 3: Frame-Rate Independent Timing

Create systems that work at any frame rate:

```m68k
; Frame timing system
        section timing_data,data

; Timing variables
frame_counter:          dc.l    0
delta_time:             dc.l    FIXED_ONE/50    ; 1/50th second for PAL
target_fps:             dc.w    50              ; PAL refresh rate
last_frame_time:        dc.l    0
current_frame_time:     dc.l    0

; Timer system for frame-rate independent effects
timer_pool:
        dcb.l   16*3,0                  ; 16 timers, 3 longs each

; Timer structure offsets
TIMER_DURATION  = 0     ; How long to run (fixed point seconds)
TIMER_ELAPSED   = 4     ; How much time has passed
TIMER_CALLBACK  = 8     ; Function to call when complete

        section code,code

; Update timing system
update_frame_timing:
        ; Increment frame counter
        addq.l  #1,frame_counter
        
        ; Calculate frame time (simplified - using VBlank)
        move.l  frame_counter,d0
        move.l  #FIXED_ONE,d1
        move.w  target_fps,d2
        ext.l   d2
        bsr     fixed_div
        move.l  d0,current_frame_time
        
        ; Calculate delta time since last frame
        move.l  current_frame_time,d0
        sub.l   last_frame_time,d0
        move.l  d0,delta_time
        
        ; Store current time as last time
        move.l  current_frame_time,last_frame_time
        
        ; Update all active timers
        bsr     update_timers
        
        rts

; Update timer system
update_timers:
        lea     timer_pool,a0
        move.w  #15,d7                  ; 16 timers
        
timer_loop:
        ; Check if timer is active
        move.l  TIMER_DURATION(a0),d0
        beq     next_timer              ; Skip inactive timers
        
        ; Add delta time to elapsed time
        move.l  TIMER_ELAPSED(a0),d1
        add.l   delta_time,d1
        move.l  d1,TIMER_ELAPSED(a0)
        
        ; Check if timer is complete
        cmp.l   d0,d1
        blt     next_timer
        
        ; Timer complete - call callback if set
        move.l  TIMER_CALLBACK(a0),d2
        beq     clear_timer
        
        ; Call callback function
        move.l  d2,a1
        jsr     (a1)
        
clear_timer:
        ; Clear timer
        clr.l   TIMER_DURATION(a0)
        clr.l   TIMER_ELAPSED(a0)
        clr.l   TIMER_CALLBACK(a0)
        
next_timer:
        add.w   #12,a0                  ; Next timer (3 longs)
        dbf     d7,timer_loop
        
        rts

; Start a timer
; d0 = duration (fixed point seconds)
; a1 = callback function (or 0 for none)
; Returns: a0 = timer slot (or 0 if no free slots)
start_timer:
        lea     timer_pool,a0
        move.w  #15,d7
        
find_timer_slot:
        tst.l   TIMER_DURATION(a0)      ; Check if slot is free
        beq     found_timer_slot
        add.w   #12,a0
        dbf     d7,find_timer_slot
        
        ; No free slots
        move.l  #0,a0
        rts
        
found_timer_slot:
        move.l  d0,TIMER_DURATION(a0)
        clr.l   TIMER_ELAPSED(a0)
        move.l  a1,TIMER_CALLBACK(a0)
        rts

; Frame-rate independent animation system
animate_sprite_precise:
        move.l  #player_sprite,a0
        
        ; Animation timing (independent of frame rate)
        move.w  18(a0),d0               ; Current animation timer
        ext.l   d0
        lsl.l   #FIXED_SHIFT,d0         ; Convert to fixed point
        
        add.l   delta_time,d0           ; Add frame time
        
        ; Check if animation frame should advance
        move.l  #FIXED_ONE,d1           ; 1 second per frame
        lsr.l   #3,d1                   ; Divide by 8 = 1/8 second per frame
        
        cmp.l   d1,d0
        blt     store_anim_timer
        
        ; Advance animation frame
        move.w  16(a0),d2               ; Current frame
        addq.w  #1,d2
        cmp.w   #8,d2                   ; 8 frames total
        blt     store_anim_frame
        clr.w   d2                      ; Wrap to frame 0
        
store_anim_frame:
        move.w  d2,16(a0)               ; Store new frame
        sub.l   d1,d0                   ; Subtract frame time
        
store_anim_timer:
        asr.l   #FIXED_SHIFT,d0         ; Convert back to integer
        move.w  d0,18(a0)               ; Store timer
        
        rts
```

## Step 4: Deterministic Game Logic

```m68k
; Deterministic random number generator
        section random_data,data

; Linear congruential generator state
rng_seed:       dc.l    $12345678
rng_a:          dc.l    $41C64E6D       ; Multiplier
rng_c:          dc.l    $00003039       ; Increment

        section code,code

; Generate deterministic random number
; Returns: d0 = random 32-bit value
random_deterministic:
        move.l  rng_seed,d0
        move.l  rng_a,d1
        mulu.l  d1,d0                   ; seed * a
        add.l   rng_c,d0                ; + c
        move.l  d0,rng_seed             ; Store new seed
        rts

; Generate random number in range
; d0 = maximum value (exclusive)
; Returns: d0 = random value [0, max)
random_range:
        move.l  d0,d1                   ; Save max
        bsr     random_deterministic
        
        ; Use modulo to get range
        divu.l  d1,d0                   ; Divide by max
        swap    d0                      ; Get remainder
        ext.l   d0                      ; Sign extend
        rts

; Seed random number generator
; d0 = seed value
seed_random:
        move.l  d0,rng_seed
        rts

; Deterministic enemy behavior
update_enemy_deterministic:
        lea     enemy_pool,a0
        move.w  #MAX_ENEMIES-1,d7
        
det_enemy_loop:
        tst.w   ENEMY_STATE(a0)
        beq     det_next_enemy
        
        ; Use enemy position as seed for deterministic behavior
        move.w  ENEMY_X(a0),d0
        lsl.w   #8,d0
        add.w   ENEMY_Y(a0),d0
        add.l   frame_counter,d0        ; Add frame for variation
        bsr     seed_random
        
        ; Generate deterministic movement decision
        move.l  #100,d0                 ; 100 possible actions
        bsr     random_range
        
        cmp.l   #20,d0                  ; 20% chance
        blt     enemy_turn_left
        cmp.l   #40,d0                  ; 20% chance  
        blt     enemy_turn_right
        cmp.l   #45,d0                  ; 5% chance
        blt     enemy_jump
        ; Otherwise continue straight
        
        bra     det_next_enemy
        
enemy_turn_left:
        subq.w  #1,ENEMY_VX(a0)
        bra     det_next_enemy
        
enemy_turn_right:
        addq.w  #1,ENEMY_VX(a0)
        bra     det_next_enemy
        
enemy_jump:
        move.w  #-20,ENEMY_VY(a0)
        
det_next_enemy:
        add.w   #ENEMY_SIZE,a0
        dbf     d7,det_enemy_loop
        
        rts

; Precise timing for power-ups and events
update_game_events:
        ; Check for power-up spawn (every 5 seconds exactly)
        move.l  frame_counter,d0
        move.l  #250,d1                 ; 5 seconds * 50 fps
        divu.l  d1,d0
        swap    d0                      ; Get remainder
        tst.w   d0
        bne     check_enemy_wave
        
        ; Spawn power-up at precise timing
        bsr     spawn_powerup
        
check_enemy_wave:
        ; Enemy wave every 10 seconds
        move.l  frame_counter,d0
        move.l  #500,d1                 ; 10 seconds * 50 fps
        divu.l  d1,d0
        swap    d0
        tst.w   d0
        bne     events_done
        
        ; Spawn enemy wave
        bsr     spawn_enemy_wave
        
events_done:
        rts

; Precise score calculation
calculate_precise_score:
        ; Score based on precise timing and performance
        move.l  frame_counter,d0
        
        ; Base score: points per second alive
        move.l  #50,d1                  ; 50 fps
        divu.l  d1,d0                   ; Seconds alive
        mulu.l  #10,d0                  ; 10 points per second
        
        ; Bonus for efficiency (crystals collected per second)
        move.w  crystal_count,d1
        ext.l   d1
        mulu.l  #100,d1                 ; 100 points per crystal
        add.l   d1,d0
        
        ; Time bonus (faster completion = higher score)
        move.l  frame_counter,d1
        move.l  #1500,d2                ; Target: 30 seconds (1500 frames)
        sub.l   d1,d2                   ; Time remaining
        bpl     add_time_bonus
        clr.l   d2                      ; No negative bonus
        
add_time_bonus:
        add.l   d2,d0                   ; Add time bonus
        
        move.l  d0,current_score
        rts

current_score:  dc.l    0
```

## Understanding Amiga Precision

### Fixed-Point Mathematics
Essential for smooth gameplay without floating point:
- **16.16 format**: 16 integer bits, 16 fractional bits
- **No rounding errors**: Consistent, predictable calculations
- **Hardware friendly**: Uses integer ALU for speed

### Hardware Timing
The Amiga provides precise timing references:
- **VBlank interrupt**: 50Hz PAL, 60Hz NTSC timing
- **Copper timing**: Cycle-accurate display synchronization
- **Raster position**: Exact display timing information

### Deterministic Behavior
Critical for fair, consistent gameplay:
- **Reproducible results**: Same inputs = same outputs
- **Network compatibility**: Synchronized multiplayer possible
- **Debug friendly**: Predictable behavior aids testing
