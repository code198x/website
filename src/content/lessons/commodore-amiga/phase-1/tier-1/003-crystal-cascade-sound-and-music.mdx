# Lesson 3: Crystal Cascade - Sound and Music

Time to bring your Amiga to life with sound! In this lesson, you'll master Paula's revolutionary 4-channel audio system. You'll create dynamic sound effects that respond to your sprite's actions and add atmospheric music that showcases the Amiga's audio superiority.

## Understanding Paula's Audio System

The Amiga's audio hardware is groundbreaking for 1985:

1. **4 Independent Channels** - Each with DMA, volume, and period control
2. **8-bit Samples** - Any sample rate from 20Hz to ~28kHz  
3. **Audio DMA** - Hardware plays samples without CPU intervention
4. **Channel Pairing** - Channels can be paired for stereo effects
5. **Audio Interrupts** - Perfect timing for music and effects

## Step 1: Basic Audio Setup

Let's start with simple sound effect playback:

```m68k
; Crystal Cascade - Lesson 3
; Paula audio programming

        include "hardware/custom.i"
        include "hardware/dmabits.i"
        include "hardware/intbits.i"

; Audio channel assignments
CHAN_MUSIC1     = 0     ; Left music channel
CHAN_MUSIC2     = 1     ; Right music channel  
CHAN_SFX1       = 2     ; Left sound effects
CHAN_SFX2       = 3     ; Right sound effects

        section audio_data,data,chip

; Sound effect samples (8-bit signed)
crystal_hit_sample:
        ; Short crystal collision sound (1000 samples)
        dc.b    0,32,64,96,127,96,64,32,0,-32,-64,-96,-127,-96,-64,-32
        dc.b    0,28,56,84,112,84,56,28,0,-28,-56,-84,-112,-84,-56,-28
        dc.b    0,24,48,72,96,72,48,24,0,-24,-48,-72,-96,-72,-48,-24
        ; ... (continue with decay pattern)
        dcb.b   970,0           ; Pad to 1000 samples

crystal_collect_sample:
        ; Pleasant collection sound (800 samples)
        dc.b    0,16,32,48,64,48,32,16,0,-16,-32,-48,-64,-48,-32,-16
        ; Rising pitch effect
        dc.b    0,20,40,60,80,60,40,20,0,-20,-40,-60,-80,-60,-40,-20
        dc.b    0,24,48,72,96,72,48,24,0,-24,-48,-72,-96,-72,-48,-24
        dcb.b   736,0           ; Pad to 800 samples

jump_sample:
        ; Quick jump sound (400 samples)
        dc.b    127,120,100,80,60,40,20,0,-20,-40,-60,-80,-100,-120,-127
        dc.b    100,90,70,50,30,10,-10,-30,-50,-70,-90,-100
        dcb.b   376,0           ; Pad to 400 samples

; Music sample - simple bass line
bass_sample:
        ; 2000 sample bass note
        dc.b    127,124,118,109,97,82,65,46,25,3,-19,-40,-60,-78,-94,-108
        dc.b    -119,-127,-127,-124,-118,-109,-97,-82,-65,-46,-25,-3,19,40,60,78
        ; Repeat pattern for full sample
        dcb.b   1936,0          ; Simplified - full sample would continue

; Sample information structure
sample_info:
crystal_hit_info:
        dc.l    crystal_hit_sample      ; Sample address
        dc.w    1000                    ; Sample length in words
        dc.w    400                     ; Period (sample rate)
        dc.w    64                      ; Volume (0-64)

crystal_collect_info:
        dc.l    crystal_collect_sample
        dc.w    800
        dc.w    300                     ; Higher pitch
        dc.w    50

jump_info:
        dc.l    jump_sample
        dc.w    400
        dc.w    200                     ; Even higher pitch
        dc.w    45

bass_info:
        dc.l    bass_sample
        dc.w    2000
        dc.w    800                     ; Low bass frequency
        dc.w    32

        section code,code

; Initialize audio system
init_audio:
        lea     custom,a5
        
        ; Disable audio DMA
        move.w  #DMAF_AUDIO,dmacon(a5)
        
        ; Clear audio interrupt enables
        move.w  #INTF_AUD0|INTF_AUD1|INTF_AUD2|INTF_AUD3,intena(a5)
        
        ; Set maximum volume for all channels
        move.w  #64,aud0vol(a5)
        move.w  #64,aud1vol(a5)
        move.w  #64,aud2vol(a5)
        move.w  #64,aud3vol(a5)
        
        ; Install audio interrupt handlers
        bsr     install_audio_interrupts
        
        ; Enable audio DMA
        move.w  #DMAF_SETCLR|DMAF_MASTER|DMAF_AUDIO,dmacon(a5)
        
        rts

; Play sound effect on specified channel
; d0 = channel number (0-3)
; a0 = sample info structure
play_sound_effect:
        move.l  a0,a1                   ; Save sample info
        
        ; Calculate channel base address
        lea     custom,a5
        mulu.w  #16,d0                  ; Each channel is 16 bytes apart
        lea     aud0lc(a5,d0.w),a2      ; Point to channel registers
        
        ; Set sample address
        move.l  (a1),d1                 ; Get sample address
        move.w  d1,4(a2)                ; AUDxLCL
        swap    d1
        move.w  d1,(a2)                 ; AUDxLCH
        
        ; Set sample length
        move.w  4(a1),6(a2)             ; AUDxLEN
        
        ; Set period (sample rate)
        move.w  6(a1),8(a2)             ; AUDxPER
        
        ; Set volume
        move.w  8(a1),10(a2)            ; AUDxVOL
        
        ; Start playback by setting length register
        move.w  4(a1),6(a2)             ; AUDxLEN (triggers DMA)
        
        rts
```

## Step 2: Sound Effect Integration

Connect sound effects to game events:

```m68k
; Enhanced collision detection with audio
check_collisions_with_audio:
        lea     custom,a5
        
        ; Read collision registers
        move.w  clxdat(a5),d0
        
        ; Check sprite-to-sprite collisions
        btst    #0,d0
        beq     check_sprite_pf_audio
        
        ; Play crystal hit sound
        move.w  #CHAN_SFX1,d0
        lea     crystal_hit_info,a0
        bsr     play_sound_effect
        
        bsr     handle_sprite_collision
        
check_sprite_pf_audio:
        ; Check sprite-to-playfield collisions
        btst    #8,d0
        beq     check_collection
        
        ; Play softer bump sound
        move.w  #CHAN_SFX2,d0
        lea     jump_info,a0            ; Reuse jump sound
        bsr     play_sound_effect
        
        bsr     handle_pf1_collision
        
check_collection:
        ; Check for crystal collection (custom logic)
        bsr     check_crystal_collection
        
        ; Clear collision registers
        move.w  #$7FFF,clxdat(a5)
        rts

; Enhanced movement with jump sounds
update_sprite_movement_with_audio:
        ; Store previous joystick state
        move.w  joy_previous,d7
        
        ; Read current joystick
        lea     custom,a5
        move.w  joy1dat(a5),d0
        move.w  d0,joy_current
        
        ; Check for fire button press (jump)
        move.w  $dff016,d1              ; Read CIA port A
        btst    #7,d1                   ; Fire button
        bne     no_jump                 ; Button not pressed
        
        ; Check if this is a new press
        btst    #7,joy_previous
        beq     no_jump                 ; Was already pressed
        
        ; Play jump sound
        move.w  #CHAN_SFX1,d0
        lea     jump_info,a0
        bsr     play_sound_effect
        
        ; Add jump velocity
        move.w  #-40,sprite_vy          ; Negative = up
        
no_jump:
        ; Store current state as previous
        move.w  joy_current,joy_previous
        
        ; Continue with normal movement
        bra     apply_movement

; Crystal collection system
check_crystal_collection:
        ; Simple collection logic - check sprite position
        move.w  sprite_x,d0
        move.w  sprite_y,d1
        
        ; Check if near center of screen (collection area)
        cmp.w   #150,d0
        blt     no_collection
        cmp.w   #170,d0
        bgt     no_collection
        cmp.w   #90,d1
        blt     no_collection
        cmp.w   #110,d1
        bgt     no_collection
        
        ; Crystal collected!
        move.w  #CHAN_SFX2,d0
        lea     crystal_collect_info,a0
        bsr     play_sound_effect
        
        ; Add to score
        addq.w  #1,crystal_count
        
no_collection:
        rts

; Variables for audio system
joy_current:    dc.w    0
joy_previous:   dc.w    0
crystal_count:  dc.w    0
```

## Step 3: Multi-Channel Music System

Create atmospheric background music:

```m68k
; Music system with multiple channels
start_background_music:
        ; Set up bass line on channel 0
        move.w  #CHAN_MUSIC1,d0
        lea     bass_info,a0
        bsr     setup_looped_channel
        
        ; Set up melody on channel 1 (different sample)
        move.w  #CHAN_MUSIC2,d0
        lea     melody_info,a0
        bsr     setup_looped_channel
        
        ; Enable audio interrupts for music timing
        lea     custom,a5
        move.w  #INTF_SETCLR|INTF_INTEN|INTF_AUD0|INTF_AUD1,intena(a5)
        
        rts

; Set up channel for looped playback
setup_looped_channel:
        move.l  a0,a1                   ; Save sample info
        
        ; Calculate channel base
        lea     custom,a5
        mulu.w  #16,d0
        lea     aud0lc(a5,d0.w),a2
        
        ; Set up sample
        move.l  (a1),d1                 ; Sample address
        move.w  d1,4(a2)                ; AUDxLCL
        swap    d1
        move.w  d1,(a2)                 ; AUDxLCH
        
        move.w  4(a1),6(a2)             ; Length
        move.w  6(a1),8(a2)             ; Period
        move.w  8(a1),10(a2)            ; Volume
        
        ; Enable channel interrupt for looping
        move.w  d0,d1
        move.w  #1,d2
        lsl.w   d1,d2                   ; Create interrupt mask
        move.w  d2,d1
        or.w    #INTF_SETCLR|INTF_INTEN,d1
        move.w  d1,intena(a5)
        
        ; Start playback
        move.w  4(a1),6(a2)             ; Trigger DMA
        
        rts

; Audio interrupt handlers for music looping
audio0_interrupt:
        ; Channel 0 finished - restart bass
        lea     custom,a5
        
        ; Acknowledge interrupt
        move.w  #INTF_AUD0,intreq(a5)
        
        ; Restart bass sample
        move.l  bass_sample,aud0lc(a5)
        move.w  #2000,aud0len(a5)       ; Restart
        
        rte

audio1_interrupt:
        ; Channel 1 finished - restart melody
        lea     custom,a5
        
        move.w  #INTF_AUD1,intreq(a5)
        
        ; Change melody note (simple progression)
        lea     melody_notes,a0
        move.w  current_note,d0
        lsl.w   #1,d0                   ; Word offset
        move.w  (a0,d0.w),d1
        cmp.w   #-1,d1                  ; End marker?
        bne     set_note
        
        ; Reset to beginning
        move.w  #0,current_note
        move.w  (a0),d1
        
set_note:
        addq.w  #1,current_note
        
        ; Set new period for melody
        move.w  d1,aud1per(a5)
        
        ; Restart melody sample
        move.l  melody_sample,aud1lc(a5)
        move.w  #1000,aud1len(a5)
        
        rte

; Simple melody progression (periods for different notes)
melody_notes:
        dc.w    400,350,300,350,400,450,400,350,300,250,300,350,400,-1

current_note:   dc.w    0

; Additional sample for melody
melody_sample:
        ; Different waveform for melody
        dc.b    64,88,108,120,124,120,108,88,64,40,20,8,4,8,20,40
        dc.b    64,80,92,100,104,100,92,80,64,48,36,28,24,28,36,48
        dcb.b   936,0           ; Pad to 1000 samples
```

## Step 4: Audio-Visual Synchronization

Synchronize sound with visual effects:

```m68k
; Enhanced flash effect with audio sync
handle_sprite_collision_with_sync:
        ; Play collision sound
        move.w  #CHAN_SFX1,d0
        lea     crystal_hit_info,a0
        bsr     play_sound_effect
        
        ; Bounce sprite
        neg.w   sprite_vx
        neg.w   sprite_vy
        
        ; Start synchronized flash effect
        move.w  #15,flash_timer         ; Flash duration
        move.w  #3,flash_rate           ; Flash frequency
        
        rts

; Audio-synchronized flash update
update_flash_with_audio:
        tst.w   flash_timer
        beq     no_flash_audio
        
        subq.w  #1,flash_timer
        
        ; Flash color based on audio timing
        move.w  flash_timer,d0
        and.w   flash_rate,d0           ; Create rhythm
        bne     flash_off
        
        ; Flash on - bright white
        lea     custom,a5
        move.w  #$0FFF,color17(a5)      ; White
        move.w  #$0FFF,color18(a5)      ; White
        bra     check_flash_end
        
flash_off:
        ; Flash off - normal colors
        lea     custom,a5
        move.w  #$0F00,color17(a5)      ; Red
        move.w  #$00F0,color18(a5)      ; Green
        
check_flash_end:
        tst.w   flash_timer
        bne     no_flash_audio
        
        ; Flash ended - restore normal colors
        lea     custom,a5
        move.w  #$0F00,color17(a5)
        move.w  #$00F0,color18(a5)
        
no_flash_audio:
        rts

; Dynamic volume control based on sprite position
update_dynamic_audio:
        ; Adjust music volume based on sprite height
        move.w  sprite_y,d0
        sub.w   #32,d0                  ; Normalize to 0-192
        bpl     volume_positive
        move.w  #0,d0
        
volume_positive:
        cmp.w   #192,d0
        ble     volume_in_range
        move.w  #192,d0
        
volume_in_range:
        ; Convert to volume (0-64)
        lsr.w   #2,d0                   ; Divide by 4
        and.w   #63,d0                  ; Ensure 0-63
        
        ; Set music volumes
        lea     custom,a5
        move.w  d0,aud0vol(a5)          ; Bass volume
        move.w  d0,aud1vol(a5)          ; Melody volume
        
        ; Adjust stereo panning with horizontal position
        move.w  sprite_x,d1
        sub.w   #160,d1                 ; Center at screen middle
        bpl     pan_right
        
        ; Sprite on left - emphasize left channel
        neg.w   d1
        lsr.w   #2,d1                   ; Scale down
        move.w  #64,d0                  ; Full left
        sub.w   d1,d0                   ; Reduce right
        bpl     set_right_vol
        move.w  #0,d0
        
set_right_vol:
        move.w  #64,aud0vol(a5)         ; Left full
        move.w  d0,aud1vol(a5)          ; Right reduced
        bra     audio_update_done
        
pan_right:
        ; Sprite on right - emphasize right channel
        lsr.w   #2,d1
        move.w  #64,d0                  ; Full right
        sub.w   d1,d0                   ; Reduce left
        bpl     set_left_vol
        move.w  #0,d0
        
set_left_vol:
        move.w  d0,aud0vol(a5)          ; Left reduced
        move.w  #64,aud1vol(a5)         ; Right full
        
audio_update_done:
        rts
```

## Step 5: Advanced Audio Effects

Create sophisticated audio effects:

```m68k
; Doppler effect for moving sprites
update_doppler_effect:
        ; Calculate sprite velocity magnitude
        move.w  sprite_vx,d0
        bpl     vx_positive
        neg.w   d0
        
vx_positive:
        move.w  sprite_vy,d1
        bpl     vy_positive
        neg.w   d1
        
vy_positive:
        add.w   d1,d0                   ; Simple velocity magnitude
        
        ; Convert to period adjustment
        lsr.w   #2,d0                   ; Scale down
        
        ; Adjust sound effect periods based on velocity
        lea     sfx_base_periods,a0
        move.w  (a0),d1                 ; Base period
        sub.w   d0,d1                   ; Higher velocity = higher pitch
        cmp.w   #100,d1                 ; Don't go too high
        bge     period_ok
        move.w  #100,d1
        
period_ok:
        move.w  d1,current_sfx_period
        rts

; Echo effect using multiple channels
play_echo_effect:
        ; Play main sound on channel 2
        move.w  #CHAN_SFX1,d0
        lea     crystal_hit_info,a0
        bsr     play_sound_effect
        
        ; Set up echo on channel 3 with delay
        move.w  #30,echo_delay          ; 30 frame delay
        move.w  #CHAN_SFX2,echo_channel
        move.l  a0,echo_sample_info
        
        rts

; Process echo delay
update_echo:
        tst.w   echo_delay
        beq     no_echo
        
        subq.w  #1,echo_delay
        bne     no_echo
        
        ; Play delayed echo
        move.w  echo_channel,d0
        move.l  echo_sample_info,a0
        
        ; Reduce volume for echo
        move.w  8(a0),d1                ; Original volume
        lsr.w   #1,d1                   ; Half volume
        move.w  d1,8(a0)                ; Set reduced volume
        
        bsr     play_sound_effect
        
        ; Restore original volume
        move.w  8(a0),d1
        lsl.w   #1,d1
        move.w  d1,8(a0)
        
no_echo:
        rts

; Variables for advanced effects
sfx_base_periods:       dc.w    400,300,200
current_sfx_period:     dc.w    400
echo_delay:             dc.w    0
echo_channel:           dc.w    0
echo_sample_info:       dc.l    0
```

```m68k
; Enhanced VBlank with full audio system
vblank_handler_with_audio:
        movem.l d0-d7/a0-a6,-(sp)
        
        ; Update sprite animation
        bsr     animate_sprite

        bsr     update_doppler_effect
        bsr     update_echo

        lea     custom,a5
        move.w  #INTF_VERTB,intreq(a5)
        
        movem.l (sp)+,d0-d7/a0-a6
        rte

; Install all interrupt handlers
install_audio_interrupts:
        move.l  4.w,a6
        jsr     -132(a6)                ; Forbid()
        
        ; Save old vectors
        move.l  $70,old_audio0_int
        move.l  $74,old_audio1_int
        move.l  $78,old_audio2_int
        move.l  $7c,old_audio3_int
        
        ; Install new handlers
        move.l  #audio0_interrupt,$70
        move.l  #audio1_interrupt,$74
        move.l  #audio2_interrupt,$78
        move.l  #audio3_interrupt,$7c
        
        jsr     -138(a6)                ; Permit()
        rts

restore_audio_interrupts:
        move.l  4.w,a6
        jsr     -132(a6)
        
        ; Disable audio interrupts
        lea     custom,a5
        move.w  #INTF_AUD0|INTF_AUD1|INTF_AUD2|INTF_AUD3,intena(a5)
        
        ; Restore old vectors
        move.l  old_audio0_int,$70
        move.l  old_audio1_int,$74
        move.l  old_audio2_int,$78
        move.l  old_audio3_int,$7c
        
        jsr     -138(a6)
        rts

; Interrupt vector storage
old_audio0_int: dc.l    0
old_audio1_int: dc.l    0
old_audio2_int: dc.l    0
old_audio3_int: dc.l    0
```

## Understanding Paula's Audio Architecture

### DMA-Based Playback
Paula's audio system is revolutionary:
- **Hardware DMA**: Samples play without CPU intervention
- **Independent channels**: Each channel has its own volume, period, and sample
- **Interrupt-driven**: Perfect timing for music and complex effects

### Sample Format
- **8-bit signed**: Values from -128 to +127
- **Any sample rate**: From 20Hz to ~28kHz (limited by DMA timing)
- **Looping**: Hardware supports automatic sample looping

### Channel Architecture
- **Channel pairing**: 0+1 and 2+3 can be paired for stereo
- **Audio filters**: Optional low-pass filter for each channel
- **Volume control**: 0-64 levels per channel

## Quick Reference

**Audio Registers:**
- AUDxLCH/L: Sample data pointer (high/low)
- AUDxLEN: Sample length in words
- AUDxPER: Sample period (frequency)
- AUDxVOL: Channel volume (0-64)

**Audio DMA Control:**
- DMAF_AUD0-3: Individual channel enables
- DMAF_AUDIO: All audio channels

**Interrupt Vectors:**
- $70: Audio channel 0
- $74: Audio channel 1  
- $78: Audio channel 2
- $7C: Audio channel 3
