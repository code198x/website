# Lesson 15: Vector Forge - Audio Synthesis

**Your vector world needs professional sound!** In your environmental effects from Lesson 14, you created immersive visual atmospheres with particle systems. Today, you'll add dynamic audio synthesis with procedural sound effects, musical patterns, and responsive audio feedback that transforms your visual effects into a complete audio-visual experience. You'll master professional sound programming through authentic Paula chip techniques that create engaging, high-quality musical vector gameplay!

; Convert musical note to Paula frequency
convert_note_to_paula_frequency:
        ; d0 contains note number (0-127, covering 10+ octaves)
        ; Returns Paula period in d0
        
        ; Clamp note to reasonable range for Paula
        cmp.l   #96,d0          ; Maximum practical note
        ble.s   note_in_paula_range
        move.l  #96,d0          ; Clamp to maximum
        
note_in_paula_range:
        ; Get base period from table
        move.l  d0,d1
        and.l   #15,d1          ; Get note within octave (0-15 for chromatic)
        lea     paula_base_period_table,a0
        lsl.l   #1,d1           ; Convert to word offset
        move.w  (a0,d1.l),d2    ; Load base period
        
        ; Calculate octave shift
        lsr.l   #4,d0           ; Divide by 16 to get octave
        beq.s   paula_frequency_complete
        
        ; Shift period down by octaves (higher frequency)
        move.l  d0,d1           ; Octave count
paula_octave_shift_loop:
        lsr.w   #1,d2           ; Halve period (double frequency)
        subq.l  #1,d1
        bne.s   paula_octave_shift_loop
        
paula_frequency_complete:

; Create vector sound effect with specified type and priority
create_vector_sound_effect:
        ; d0 = sound type, d1 = priority
        ; Find available sound effect slot
        bsr     find_available_vector_sound_effect_slot
        cmp.l   #-1,d0
        beq.s   vector_sound_effect_creation_complete

        lea     sound_effect_active_array,a0
        move.b  #1,(a0,d0.l)    ; Sound effect active
        
        ; Set sound effect properties
        lea     sound_effect_priority_array,a0
        move.b  d1,(a0,d0.l)    ; Priority
        
        ; Get sound effect parameters from type
        move.l  sound_effect_type,d2

        move.b  d2,(a0,d0.l)    ; Type
        
        ; Set frequency from type table

        lsl.l   #1,d2           ; Convert to word offset
        move.w  (a0,d2.l),d3    ; Load frequency
        lea     sound_effect_frequency_array,a0
        lsl.l   #1,d0           ; Convert to word offset
        move.w  d3,(a0,d0.l)    ; Store frequency
        
        ; Set duration from type table
        lsr.l   #1,d0           ; Convert back to byte offset
        move.l  sound_effect_type,d2
        lea     vector_sound_effect_duration_table,a0
        lsl.l   #1,d2           ; Convert to word offset
