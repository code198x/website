    STA $2007           ; Write palette (absolute addressing)
    LDA #$30            ; White conductors (immediate addressing)
    STA $2007           ; Write palette (absolute addressing)

InitAPUSystem:

clear_apu:

    STA $4000,X         ; Clear APU register (absolute indexed addressing)
    INX                 ; Next register
    CPX #$20            ; Check all APU registers (immediate addressing)
    BNE clear_apu       ; Continue clearing
    
    ; Enable APU channels for electronic effects
    LDA #$0F            ; Enable pulse, triangle, noise (immediate addressing)
    STA $4015           ; APU status register (absolute addressing)
    
    ; Set up Pulse Channel 1 for electrical current sounds
    LDA #$BF            ; Duty cycle, constant volume (immediate addressing)
    STA $4000           ; Pulse 1 control (absolute addressing)
    
    LDA #$08            ; Medium volume (immediate addressing)
    STA $4001           ; Pulse 1 sweep (absolute addressing)
    
    ; Set up Noise Channel for electrical sparks
    LDA #$30            ; Constant volume, medium level (immediate addressing)
    STA $400C           ; Noise control (absolute addressing)

    LDA #$81            ; Linear counter, length (immediate addressing)
    STA $4008           ; Triangle control (absolute addressing)

    STA $83             ; Spark X position (zero page addressing)
    STA $84             ; Spark Y position (zero page addressing)
    
    LDA #$20            ; Spark duration counter (immediate addressing)
    STA $85             ; Spark timer (zero page addressing)
    
    ; Initialize spark sound parameters
    LDA #$C0            ; High frequency base (immediate addressing)
    STA $86             ; Spark frequency low (zero page addressing)
    LDA #$08            ; High frequency base (immediate addressing)
    STA $87             ; Spark frequency high (zero page addressing)
    
    RTS

CreateSparkPattern:
    ; Set up spark trigger zones in circuit
    LDA #$20            ; Point to name table (immediate addressing)
    STA $2006           ; PPU address high (absolute addressing)
    LDA #$43            ; Spark junction position (immediate addressing)
    STA $2006           ; PPU address low (absolute addressing)
    
    LDA #$03            ; Junction/contact tile (immediate addressing)
    STA $2007           ; Place spark trigger (absolute addressing)
    
    ; Set up multiple spark zones
    LDA #$20            ; Point to name table (immediate addressing)
    STA $2006           ; PPU address high (absolute addressing)
    LDA #$67            ; Another junction position (immediate addressing)
    STA $2006           ; PPU address low (absolute addressing)
    
    LDA #$03            ; Junction/contact tile (immediate addressing)
    STA $2007           ; Place spark trigger (absolute addressing)
    
    RTS

; Check for electrical contact and spark generation
CheckElectricalContact:
    ; Load current conductor position
    LDA $80             ; Load X position (zero page addressing)
    STA $88             ; Temporary X storage (zero page addressing)
    LDA $81             ; Load Y position (zero page addressing)
    STA $89             ; Temporary Y storage (zero page addressing)
    
    ; Calculate conductor screen position for collision detection
    LDA $89             ; Load Y position (zero page addressing)
    CMP #$03            ; Check if at junction row (immediate addressing)
    BNE no_spark_contact ; Not at junction height
    
    LDA $88             ; Load X position (zero page addressing)
    CMP #$43            ; Check junction column 1 (immediate addressing)
    BEQ trigger_spark   ; At junction - create spark!
    CMP #$67            ; Check junction column 2 (immediate addressing)
    BEQ trigger_spark   ; At junction - create spark!
    
no_spark_contact:
    RTS                 ; No electrical contact
    
trigger_spark:
    ; Trigger electrical spark effect
    LDA #$01            ; Spark active (immediate addressing)
    STA $82             ; Set spark flag (zero page addressing)
    
    LDA $88             ; Load contact X position (zero page addressing)
    STA $83             ; Store spark X (zero page addressing)
    LDA $89             ; Load contact Y position (zero page addressing)
    STA $84             ; Store spark Y (zero page addressing)
    
    LDA #$20            ; Spark duration (immediate addressing)
    STA $85             ; Set spark timer (zero page addressing)
    
    ; Trigger electrical spark sound
    JSR PlaySparkSound
    
    RTS

; Play electrical spark sound effect
PlaySparkSound:
    ; Use Noise Channel for electrical spark
    LDA #$30            ; Constant volume, short noise (immediate addressing)
    STA $400C           ; Noise control (absolute addressing)
    
    LDA #$07            ; High pitch noise for spark (immediate addressing)
    STA $400E           ; Noise period (absolute addressing)
    
    LDA #$08            ; Short duration (immediate addressing)
    STA $400F           ; Noise length (absolute addressing)
    
    ; Add Pulse Channel accent for electrical zing
    LDA #$8F            ; Strong attack, full volume (immediate addressing)
    STA $4000           ; Pulse 1 control (absolute addressing)
    
    LDA $86             ; Load spark frequency low (zero page addressing)
    STA $4002           ; Pulse 1 frequency low (absolute addressing)
    LDA $87             ; Load spark frequency high (zero page addressing)
    STA $4003           ; Pulse 1 frequency high (absolute addressing)
    
    RTS

; Update spark visual and audio effects
UpdateSparkEffects:
    ; Check if spark is active
    LDA $82             ; Load spark flag (zero page addressing)
    BEQ no_active_spark ; No spark to update
    
    ; Update spark visual effect
    JSR DrawSparkEffect
    
    ; Update spark audio effect
    JSR UpdateSparkAudio
    
    ; Decrement spark timer
    DEC $85             ; Decrement timer (zero page addressing)
    BNE spark_continue  ; Continue if timer > 0
    
    ; Spark finished - clean up
    LDA #$00            ; Spark inactive (immediate addressing)
    STA $82             ; Clear spark flag (zero page addressing)
    
    ; Stop spark sounds
    JSR StopSparkSound
    
spark_continue:
no_active_spark:
    RTS

; Draw visual spark effect
DrawSparkEffect:
    ; Point to spark screen position
    LDA #$20            ; Name table high (immediate addressing)
    STA $2006           ; PPU address high (absolute addressing)
    
    ; Calculate spark position
    LDA $84             ; Load spark Y (zero page addressing)
    ASL A               ; Multiply by 2 (immediate addressing)
    ASL A               ; Multiply by 4 (immediate addressing)
    ASL A               ; Multiply by 8 (immediate addressing)
    ASL A               ; Multiply by 16 (immediate addressing)
    ASL A               ; Multiply by 32 (immediate addressing)
    CLC                 ; Clear carry for addition
    ADC $83             ; Add spark X position (zero page addressing)
    STA $2006           ; PPU address low (absolute addressing)
    
    ; Draw spark character based on timer for animation
    LDA $85             ; Load spark timer (zero page addressing)
    AND #$03            ; Mask to 4 animation frames (immediate addressing)
    TAX                 ; Use as index (transfer addressing)
    LDA SparkFrames,X   ; Load spark frame (absolute indexed addressing)
    STA $2007           ; Draw spark (absolute addressing)
    
    RTS

UpdateSparkAudio:
    ; Modulate spark frequency for electrical crackle effect
    LDA $85             ; Load spark timer (zero page addressing)
    AND #$07            ; Mask to 8 frequency steps (immediate addressing)
    TAX                 ; Use as frequency index (transfer addressing)
    
    ; Load frequency from table
    LDA SparkFreqLow,X  ; Load low frequency (absolute indexed addressing)
    STA $4002           ; Set pulse frequency low (absolute addressing)
    LDA SparkFreqHigh,X ; Load high frequency (absolute indexed addressing)
    STA $4003           ; Set pulse frequency high (absolute addressing)
    
    ; Update noise for electrical crackling
    LDA $85             ; Load timer (zero page addressing)
    LSR A               ; Divide by 2 for slower noise change (immediate addressing)
    AND #$0F            ; Mask to noise range (immediate addressing)
    STA $400E           ; Update noise period (absolute addressing)
    
    RTS

StopSparkSound:

    LDA #$30            ; Zero volume (immediate addressing)
    STA $4000           ; Stop pulse 1 (absolute addressing)

    LDA #$30            ; Zero volume (immediate addressing)

    STA $400B           ; Triangle frequency high (absolute addressing)
    
    RTS

    ; Stop triangle channel
    LDA #$00            ; Stop triangle (immediate addressing)
    STA $4008           ; Triangle control (absolute addressing)
    RTS

; Update current flow audio feedback
UpdateCurrentAudio:
    ; Create subtle audio feedback for current movement
    LDA $80             ; Load current X position (zero page addressing)
    AND #$1F            ; Mask to frequency range (immediate addressing)
    CLC                 ; Clear carry
    ADC #$80            ; Add base frequency (immediate addressing)
    STA $4002           ; Set current flow frequency (absolute addressing)
    
    ; Very quiet volume for background current
    LDA #$31            ; Very low volume pulse (immediate addressing)
    STA $4000           ; Subtle current sound (absolute addressing)
    
    RTS

; Enhanced current conductor placement (from Lesson 2)
PlaceCurrentConductor:
    ; Calculate screen address from conductor position
    LDA $81             ; Load Y position (zero page addressing)
    CMP #$00            ; Check if at row 0 (immediate addressing)
    BEQ CalculateX      ; Skip Y calculation if zero
    
    ; Multiply Y by 32 (NES screen width)
    TAX                 ; Transfer Y to X register
    LDA #$00            ; Initialize result (immediate addressing)
    CLC                 ; Clear carry
MultiplyY:
    ADC #$20            ; Add 32 for each row (immediate addressing)
    DEX                 ; Decrement row counter
    BNE MultiplyY       ; Continue until Y rows processed
    
CalculateX:
    CLC                 ; Clear carry for X addition
    ADC $80             ; Add X position (zero page addressing)
    TAY                 ; Use Y as offset
    
    ; Point to screen memory plus calculated position
    LDA #$20            ; Screen base high byte (immediate addressing)
    STA $2006           ; PPU address high (absolute addressing)
    STY $2006           ; Screen base + position (zero page addressing)
    
    ; Place current conductor character with spark check
    LDA $82             ; Check if spark active (zero page addressing)
    BEQ normal_conductor ; No spark, normal conductor
    
    ; Enhanced conductor during spark
    LDA #$05            ; Bright conductor tile (immediate addressing)
    JMP place_conductor
    
normal_conductor:
    LDA #$04            ; Normal conductor tile (immediate addressing)
    
place_conductor:
    STA $2007           ; Write to screen (absolute addressing)
    
    RTS

## What Just Happened? The Magic Behind Console Audio Control

Now that you've heard your electronic sparks, let's understand how **addressing modes** orchestrated this precise console audio experience.

### Addressing Modes: Your Console Audio Control System

The most important thing to understand: **addressing modes determine how efficiently you control real-time audio**. In your electronic spark code:

**Immediate Addressing (#value)**: Loading audio constants
```nasm
LDA #$0F            ; Load APU enable mask (immediate addressing)
LDA #$BF            ; Load pulse duty cycle (immediate addressing)
LDA #$30            ; Load noise volume level (immediate addressing)
```

**Absolute Addressing (address)**: Controlling APU hardware registers
```nasm
STA $4015           ; APU status register (absolute addressing)
STA $4000           ; Pulse 1 control register (absolute addressing)
STA $400C           ; Noise channel control (absolute addressing)
```

**Zero Page Addressing ($zp)**: Lightning-fast audio timing and state
```nasm
STA $82             ; Store spark flag (zero page addressing - fastest!)
LDA $85             ; Load spark timer (zero page addressing - fastest!)
DEC $85             ; Decrement timer (zero page addressing - fastest!)
```

**Absolute Indexed Addressing (address,X)**: Dynamic audio data access
```nasm
LDA SparkFrames,X   ; Load animation frame (absolute indexed addressing)
LDA SparkFreqLow,X  ; Load frequency from table (absolute indexed addressing)
STA $4000,X         ; Clear APU registers in loop (absolute indexed addressing)
```

### The NES APU: Your Console Audio Engine

**NES APU Register Map - Your Electronic Audio Control Panel:**
- **$4000-$4003**: Pulse Channel 1 (electrical current sounds)
- **$4004-$4007**: Pulse Channel 2 (harmonic effects)
- **$4008-$400B**: Triangle Channel (background electrical hum)
- **$400C-$400F**: Noise Channel (electrical sparks and crackling)
- **$4015**: APU Status (enable/disable audio channels)

When you write `STA $4000`, you're directly controlling the legendary NES APU that defined console audio!

### Why These Addressing Modes Matter for Console Audio

**Immediate Addressing** - Perfect for audio constants:
```nasm
LDA #$BF            ; Pulse wave duty cycle (known at programming time)
LDA #$0F            ; APU channel enable mask (constant value)
```

**Absolute Addressing** - Essential for APU hardware control:
```nasm
STA $4015           ; APU status register (hardware location)
STA $4002           ; Frequency register (hardware location)
```

**Zero Page Addressing** - Critical for real-time audio timing:
```nasm
DEC $85             ; Spark timer (needs maximum speed for precise timing)
LDA $82             ; Spark state (checked every frame for audio sync)
```

**Absolute Indexed Addressing** - Efficient for dynamic audio data:
```nasm
LDA SparkFreqLow,X  ; Frequency table lookup (dynamic audio data)
STA $4000,X         ; APU register array clearing (efficient loops)
```

**Why this matters for console games**: Audio feedback happens 60 times per second - efficient addressing ensures perfect timing and responsiveness.
