CheckPowerConsumptionEfficiencyEvents:
    ; Calculate total power consumption across all active conductors
    LDA #$00            ; Clear total power consumption
    STA TempPowerTotal
    
    LDX #$00            ; Start with conductor 0
CalculatePowerConsumptionLoop:
    ; Check if conductor is active
    LDA ConductorActiveArray,X
    BEQ NextPowerConsumptionCheck
    
    ; Calculate power consumption for this conductor
    JSR CalculateConductorPowerConsumption
    
    ; Add to total power consumption
    LDA TempPowerTotal
    CLC
    ADC TempConductorPower
    STA TempPowerTotal
    BCC NoPowerOverflow
    
    ; Handle power consumption overflow
    LDA #$FF            ; Cap at maximum
    STA TempPowerTotal

NoPowerOverflow:

NextPowerConsumptionCheck:
    INX                 ; Next conductor
    CPX #$06            ; Check all conductors
    BNE CalculatePowerConsumptionLoop
    
    ; Store total power consumption
    LDA TempPowerTotal
    STA PowerConsumptionTotal
    
    ; Calculate power efficiency (inverse of consumption)

    STA TempEfficiencyValue

; Calculate conductor power consumption using voltage and current
CalculateConductorPowerConsumption:
    ; X contains conductor index
    ; Power = Voltage × Current (simplified electrical formula)
    LDA ConductorVoltageArray,X     ; Load voltage
    LDY ConductorCurrentArray,X     ; Load current
    
    ; Multiply voltage × current using repeated addition (NES-friendly)
    LDA #$00            ; Clear result
    CPY #$00            ; Check if current is zero
    BEQ StoreConductorPower
    
PowerMultiplicationLoop:
    CLC
    ADC ConductorVoltageArray,X     ; Add voltage
    BCC NoMultiplicationOverflow
    LDA #$FF            ; Cap at maximum
    JMP StoreConductorPower
    
NoMultiplicationOverflow:
    DEY                 ; Decrement current counter
    BNE PowerMultiplicationLoop

StoreConductorPower:
    ; Apply conductor type modifier
    LDY ConductorTypeArray,X        ; Load conductor type
    CPY #$02            ; High-efficiency conductor?
    BCC NormalPowerConsumption
    
    ; High-efficiency conductor - reduce power consumption
    LSR                 ; Divide by 2
    
NormalPowerConsumption:
    STA TempConductorPower
    
    RTS

UpdateConductorEfficiencyCalculations:
    ; Calculate efficiency for each active conductor

    LDA #$00            ; Clear total efficiency
    STA TempTotalEfficiency
    
CalculateConductorEfficiencyLoop:
    ; Check if conductor is active
    LDA ConductorActiveArray,X
    BEQ NextConductorEfficiency
    
    ; Calculate conductor efficiency: (Voltage + Current) / Power Consumption
    LDA ConductorVoltageArray,X
    CLC
    ADC ConductorCurrentArray,X     ; Add current
    STA TempConductorEffectiveness
    
    ; Get power consumption for this conductor
    JSR CalculateConductorPowerConsumption
    
    ; Calculate efficiency: effectiveness / power consumption
    LDA TempConductorEffectiveness
    CMP TempConductorPower
    BCC LowEfficiencyConductor
    
    ; High efficiency conductor
    LDA TempConductorEffectiveness
    SEC
    SBC TempConductorPower          ; Subtract power consumption
    JMP StoreConductorEfficiency
    
LowEfficiencyConductor:
    ; Low efficiency - minimal score
    LDA #$01            ; Minimum efficiency

StoreConductorEfficiency:
    ; Add to total efficiency
    CLC
    ADC TempTotalEfficiency
    STA TempTotalEfficiency
    BCC NextConductorEfficiency
    
    ; Handle efficiency overflow
    LDA #$FF            ; Cap at maximum
    STA TempTotalEfficiency

NextConductorEfficiency:

    CPX #$06            ; Check all conductors
    BNE CalculateConductorEfficiencyLoop
    
    ; Store total conductor efficiency
    LDA TempTotalEfficiency
    STA ConductorEfficiencyTotal
    
    RTS

; Apply efficiency calculations with multipliers and bonuses
ApplyEfficiencyCalculations:
    ; Apply efficiency multiplier to temp value
    LDA TempEfficiencyValue
    LDX EfficiencyMultiplier
    CPX #$01            ; No multiplication needed?
    BEQ NoEfficiencyMultiplier
    
    ; Apply multiplier using repeated addition
    LDY TempEfficiencyValue         ; Save original value
    LDA #$00            ; Clear result
    
EfficiencyMultiplierLoop:
    CLC
    ADC Y               ; Add original value
    BCC NoEfficiencyOverflow
    LDA #$FF            ; Cap at maximum
    JMP StoreMultipliedEfficiency
    
NoEfficiencyOverflow:
    DEX                 ; Decrement multiplier
    BNE EfficiencyMultiplierLoop

StoreMultipliedEfficiency:
    STA TempEfficiencyValue

NoEfficiencyMultiplier:
    ; Add efficiency to total score
    LDA ElectricalEfficiencyScore
    CLC
    ADC TempEfficiencyValue
    STA ElectricalEfficiencyScore
    BCC NoTotalEfficiencyOverflow
    
    ; Handle total efficiency overflow
    LDA #$FF            ; Cap at maximum
    STA ElectricalEfficiencyScore

NoTotalEfficiencyOverflow:
    RTS

; Update efficiency streak with timing calculations
UpdateEfficiencyStreakCalculations:
    ; Check if efficiency event occurred this frame
    LDA TempEfficiencyValue
    CMP #$05            ; Minimum efficiency for streak
    BCC NoEfficiencyStreakUpdate
    
    ; Check if streak timer is active
    LDA StreakTimer
    BNE ContinueEfficiencyStreak
    
    ; Start new efficiency streak
    LDA #$01            ; Initialize streak
    STA EfficiencyStreak
    LDA #$78            ; 120 frames = 2 seconds
    STA StreakTimer
    JMP EfficiencyStreakComplete

ContinueEfficiencyStreak:
    ; Continue existing streak
    INC EfficiencyStreak
    
    ; Check for streak overflow
    LDA EfficiencyStreak
    CMP #$10            ; Cap at 16 streak
    BCC ResetEfficiencyStreakTimer
    LDA #$0F            ; Maximum 15 streak
    STA EfficiencyStreak
    
ResetEfficiencyStreakTimer:
    LDA #$78            ; Reset timer
    STA StreakTimer
    
    ; Calculate streak multiplier (streak / 4 + 1)
    LDA EfficiencyStreak
    LSR                 ; Divide by 2
    LSR                 ; Divide by 4
    CLC
    ADC #$01            ; Add 1
    CMP #$05            ; Cap at 5x multiplier
    BCC StoreEfficiencyStreakMultiplier
    LDA #$05            ; Maximum 5x
    
StoreEfficiencyStreakMultiplier:
    STA EfficiencyMultiplier

EfficiencyStreakComplete:

    BEQ EfficiencyStreakUpdated

    BNE EfficiencyStreakUpdated

    STA EfficiencyMultiplier

    ; Convert to 3-digit decimal display
    
    ; Extract hundreds digit

    CMP #$64            ; >= 100?

    SBC #$64            ; Subtract 100
    INX                 ; Increment hundreds

    BCS HundredsLoop
