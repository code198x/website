    SBC ElectricalEntityYArray,X
    BPL SurgeDistanceYPositive

    CLC
    ADC #$01
    
SurgeDistanceYPositive:
    STA TempElectricalDistanceY
    
    ; Move toward conductor with electrical attraction
    LDA ConductorXPosition

    ; Move right toward conductor
    INC ElectricalEntityXArray,X
    JMP SurgeCheckYMovement

    DEC ElectricalEntityXArray,X

    JMP SurgeMovementComplete

    ; Get current decision node type
    LDY CurrentDecisionNode
    LDA ElectricalDecisionNodeTypes,Y
    
    CMP #$00            ; Check electrical distance to conductor
    BEQ EvaluateElectricalDistanceDecision
    CMP #$01            ; Check conductor electrical activity
    BEQ EvaluateConductorElectricalDecision
    CMP #$02            ; Check electrical energy status
    BEQ EvaluateElectricalEnergyDecision
    CMP #$03            ; Check electrical network load

    RTS
    
EvaluateElectricalDistanceDecision:
    ; Calculate electrical distance to conductor

    SEC

    SEC

    BPL ElectricalDistanceYPositive

    CLC

    CLC
    ADC TempElectricalDistanceX
    
    ; Compare with decision threshold
    LDY CurrentDecisionNode
    CMP ElectricalDecisionThresholds,Y
    BCC ElectricalDistanceClose
    
    ; Conductor far - follow left branch
    LDA ElectricalDecisionLeftBranches,Y
    STA CurrentDecisionNode
    JMP ContinueElectricalTraversal
    
ElectricalDistanceClose:
    ; Conductor close - follow right branch
    LDA ElectricalDecisionRightBranches,Y
    STA CurrentDecisionNode
    
ContinueElectricalTraversal:
    ; Check if we've reached a leaf node
    LDY CurrentDecisionNode
    LDA ElectricalDecisionNodeTypes,Y
    CMP #$FF            ; Leaf node marker
    BEQ ElectricalTraversalComplete
    
    ; Continue traversal
    JMP TraverseElectricalDecisionTree
    
ElectricalTraversalComplete:

    STA LastDecisionResult
    RTS

    BCC ElectricalActivityLow
    
    ; High electrical activity - defensive response
    LDA ElectricalDecisionRightBranches,Y
    STA CurrentDecisionNode
    JMP ContinueElectricalTraversal
    
ElectricalActivityLow:
    ; Low electrical activity - aggressive response
    LDA ElectricalDecisionLeftBranches,Y

    JMP ContinueElectricalTraversal
    
EvaluateElectricalEnergyDecision:

    STA CurrentDecisionNode
    JMP ContinueElectricalTraversal
    
EvaluateElectricalNetworkDecision:
