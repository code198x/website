# Lesson 15: Prism Quest - Audio Synthesis

**Your dimensional world needs dynamic sound!** In your environmental effects from Lesson 14, you created immersive visual atmospheres with particle systems. Today, you'll add dynamic audio synthesis with procedural sound effects, musical patterns, and responsive audio feedback that transforms your visual effects into a complete audio-visual experience. You'll master creative sound programming through innovative ZX Spectrum beeper techniques that create engaging, musical dimensional gameplay!

generate_dimensional_musical_frame:

; Generate dimensional melody note using beeper
generate_dimensional_melody_note:
    ; Get current melody pattern
    LD A, (current_melody_pattern)
    LD H, A             ; Save pattern
    
    ; Get current beat position
    LD A, (musical_pattern_timer)
    SRL A               ; Divide by musical tempo
    SRL A
    AND $07             ; 8 beats per pattern
    LD L, A             ; Save beat
    
    ; Calculate pattern table offset
    LD A, H             ; Restore pattern
    ADD A, A            ; Pattern * 2
    ADD A, A            ; Pattern * 4
    ADD A, A            ; Pattern * 8
    ADD A, L            ; Add beat offset
    
    ; Get note from pattern table
    LD HL, dimensional_melody_pattern_table
    ADD A, L
    LD L, A
    JR NC, no_melody_table_carry
    INC H
no_melody_table_carry:
    LD A, (HL)          ; Load note
    
    ; Convert note to beeper frequency
    CALL convert_note_to_beeper_frequency
    
    ; Play note using beeper
    CALL play_beeper_note
    
    RET

; Convert musical note to beeper frequency
convert_note_to_beeper_frequency:
    ; A contains note number (0-47, covering 4 octaves for Spectrum)
    ; Returns beeper timing value in A
    
    ; Limit note range for Spectrum capabilities
    CP $30              ; Maximum note (48)
    JR C, note_in_range
    LD A, $2F           ; Clamp to maximum
    
note_in_range:
    ; Get base timing from table
    LD HL, beeper_frequency_table
    ADD A, L
    LD L, A
    JR NC, no_freq_table_carry
    INC H
no_freq_table_carry:
    LD A, (HL)          ; Load beeper timing
    
    LD (current_beeper_frequency), A
    RET

; Play beeper note with calculated frequency
play_beeper_note:
    ; Get note frequency and duration
    LD A, (current_beeper_frequency)
    LD B, A             ; Beeper timing
    LD A, (musical_tempo)
    ADD A, A            ; Double for note duration
    LD C, A             ; Note duration
    
    ; Play note using optimized beeper routine
    CALL optimized_beeper_note
    
    RET

; Optimized beeper note routine for Spectrum
optimized_beeper_note:
    ; B = frequency timing, C = duration
    LD A, C             ; Duration counter
    OR A                ; Check if zero
    RET Z               ; Return if no duration
    
beeper_note_loop:
    ; Generate beeper tone
    LD D, B             ; Load frequency timing
    
beeper_tone_loop:
    ; Toggle beeper bit
    LD A, $10           ; Beeper bit
    OUT ($FE), A        ; Turn on beeper
    
    ; Frequency delay
    LD E, D             ; Load timing
beeper_delay_1:
    DEC E
    JR NZ, beeper_delay_1
    
    ; Toggle beeper off
    LD A, $00           ; No beeper
    OUT ($FE), A        ; Turn off beeper
    
    ; Frequency delay
    LD E, D             ; Load timing
beeper_delay_2:
    DEC E
    JR NZ, beeper_delay_2
    
    ; Continue tone
    DEC C               ; Decrement duration
    JR NZ, beeper_note_loop
    
    RET

create_dimensional_sound_effect:

    JR Z, sound_effect_creation_complete
