# Lesson 5: Prism Quest - Attribute Magic

**Your dimensional maze needs spectacular visual magic!** In the last lesson, you added interactive puzzles and maze challenges to your color-shifting world. Today, you'll unleash the Spectrum's most distinctive feature - attribute magic - creating dynamic color cycling, pattern effects, and visual spectacle that transforms constraints into stunning gameplay advantages!

## Your Attribute Magic Awakens

Let's start with the magic - adding spectacular attribute manipulation to your dimensional world:

```nasm
; Prism Quest - Attribute Magic Creation
    ORG $8000
    
start:
    ; Set up enhanced dimensional maze from Lessons 1-4
    CALL clear_screen
    CALL draw_dimensional_maze
    
    ; Initialize dimensional navigator from previous lessons
    LD BC, $0205        ; Navigator position: B=X(2), C=Y(5)
    LD (navigator_x), BC ; Store in memory
    
    ; Initialize maze challenges from Lesson 4
    CALL init_puzzle_system
    CALL create_maze_obstacles

    CALL init_attribute_magic

    ; Main attribute magic loop
main_loop:

    CALL update_navigator_position

    CALL update_color_cycling

    CALL update_visual_effects
    JR main_loop

; Initialize attribute magic system
init_attribute_magic:
    ; Set up attribute magic variables
    LD A, $00           ; Magic effect counter
    LD (magic_counter), A
    
    LD A, $01           ; Magic effect direction
    LD (magic_direction), A
    
    LD A, $20           ; Magic effect speed
    LD (magic_speed), A
    
    ; Initialize color cycling parameters
    LD A, $00           ; Cycle position
    LD (cycle_position), A
    
    LD A, $08           ; Number of colors in cycle
    LD (cycle_length), A
    
    ; Set up attribute pattern data structure
    LD HL, attribute_patterns ; Point to pattern data
    LD (pattern_pointer), HL  ; Store pointer
    
    ; Initialize effect timing structure
    LD HL, effect_timers     ; Point to timer data
    LD (timer_pointer), HL   ; Store pointer
    
    RET

; Create dynamic color cycling patterns
create_color_patterns:
    ; Set up base color pattern in attribute memory
    LD HL, $5800        ; Attribute memory start
    LD BC, $0300        ; 768 bytes (32x24 attributes)
    
    ; Create initial color pattern
    LD A, $47           ; White on black
    
fill_base_pattern:
    LD (HL), A          ; Set base attribute
    INC HL              ; Next attribute
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, fill_base_pattern
    
    ; Create special magic areas
    CALL create_magic_zones
    
    ; Set up color cycle table
    LD HL, color_cycle_table ; Point to cycle data
    LD DE, active_colors     ; Point to active colors
    LD BC, $0008            ; Copy 8 colors
    LDIR                    ; Copy color cycle
    
    RET

; Create special magic zones in the maze
create_magic_zones:
    ; Create Magic Zone 1 (top-left corner)
    LD HL, $5800 + (4*32) + 6  ; Zone position
    LD B, $03           ; Zone height
    
magic_zone_1_row:
    LD C, $04           ; Zone width
    
magic_zone_1_col:
    LD A, $45           ; Bright magenta on black
    LD (HL), A          ; Set magic attribute
    INC HL              ; Next position
    DEC C               ; Decrement width
    JR NZ, magic_zone_1_col ; Continue column
    
    LD DE, $001C        ; Move to next row (32-4)
    ADD HL, DE          ; Advance to next row
    DEC B               ; Decrement height
    JR NZ, magic_zone_1_row ; Continue rows
    
    ; Create Magic Zone 2 (center area)
    LD HL, $5800 + (12*32) + 14 ; Center zone position
    LD B, $02           ; Zone height
    
magic_zone_2_row:
    LD C, $05           ; Zone width
    
magic_zone_2_col:
    LD A, $46           ; Bright yellow on black
    LD (HL), A          ; Set magic attribute
    INC HL              ; Next position
    DEC C               ; Decrement width
    JR NZ, magic_zone_2_col ; Continue column
    
    LD DE, $001B        ; Move to next row (32-5)
    ADD HL, DE          ; Advance to next row
    DEC B               ; Decrement height
    JR NZ, magic_zone_2_row ; Continue rows
    
    ; Create Magic Zone 3 (bottom-right corner)
    LD HL, $5800 + (18*32) + 25 ; Bottom zone position
    LD B, $03           ; Zone height
    
magic_zone_3_row:
    LD C, $03           ; Zone width
    
magic_zone_3_col:
    LD A, $43           ; Bright cyan on black
    LD (HL), A          ; Set magic attribute
    INC HL              ; Next position
    DEC C               ; Decrement width
    JR NZ, magic_zone_3_col ; Continue column
    
    LD DE, $001D        ; Move to next row (32-3)
    ADD HL, DE          ; Advance to next row
    DEC B               ; Decrement height
    JR NZ, magic_zone_3_row ; Continue rows
    
    RET

; Initialize visual effect data structures
init_effect_structures:
    ; Set up wave effect structure
    LD HL, wave_effect_data
    LD A, $00           ; Wave position
    LD (HL), A          ; Store wave position
    INC HL
    LD A, $01           ; Wave direction
    LD (HL), A          ; Store wave direction
    INC HL
    LD A, $04           ; Wave speed
    LD (HL), A          ; Store wave speed
    INC HL
    LD A, $10           ; Wave amplitude
    LD (HL), A          ; Store wave amplitude
    
    ; Set up pulse effect structure

    INC HL
    LD A, $08           ; Pulse frequency

    INC HL
    LD A, $02           ; Pulse intensity

    ; Set up spiral effect structure
    LD HL, spiral_effect_data

    LD A, $10           ; Spiral center X
    LD (HL), A          ; Store center X
    INC HL
    LD A, $0C           ; Spiral center Y
    LD (HL), A          ; Store center Y
    INC HL
    LD A, $02           ; Spiral rotation speed
    LD (HL), A          ; Store rotation speed
    
    RET

; Update attribute magic effects
update_attribute_magic:
    ; Update magic counter for timing
    LD A, (magic_counter)  ; Load current counter
    LD B, A                ; Save for calculations
    ADD A, (magic_direction) ; Add direction
    LD (magic_counter), A  ; Store updated counter
    
    ; Check for counter overflow/underflow
    CP $FF              ; Check maximum
    JR Z, reverse_magic_direction
    CP $00              ; Check minimum
    JR Z, reverse_magic_direction
    JR continue_magic_update
    
reverse_magic_direction:
    LD A, (magic_direction) ; Load current direction
    NEG                     ; Negate direction
    LD (magic_direction), A ; Store reversed direction
    
continue_magic_update:
    ; Apply magic effects based on counter
    CALL apply_wave_effect
    CALL apply_pulse_effect
    CALL apply_spiral_effect
    
    RET

; Apply wave effect to attributes
apply_wave_effect:
    ; Load wave effect structure
    LD HL, wave_effect_data
    LD A, (HL)          ; Load wave position
    INC HL
    LD B, (HL)          ; Load wave direction
    INC HL
    LD C, (HL)          ; Load wave speed
    INC HL
    LD D, (HL)          ; Load wave amplitude
    
    ; Update wave position
    ADD A, C            ; Add speed to position
    AND $1F             ; Wrap to 32 positions
    LD (wave_effect_data), A ; Store updated position
    
    ; Apply wave to magic zones
    LD E, A             ; Use position as offset
    LD HL, $5800 + (4*32) + 6  ; Magic zone 1
    
    ; Calculate wave intensity
    LD A, E             ; Load position
    ADD A, D            ; Add amplitude
    AND $07             ; Mask to color range
    ADD A, $40          ; Add bright bit
    
    ; Apply to zone
    LD B, $03           ; Zone height
wave_apply_row:
    LD C, $04           ; Zone width
wave_apply_col:
    LD (HL), A          ; Apply wave color
    INC HL              ; Next position
    DEC C               ; Decrement width
    JR NZ, wave_apply_col ; Continue column
    
    LD IX, $001C        ; Offset to next row
    ADD HL, IX          ; Move to next row
    DEC B               ; Decrement height
    JR NZ, wave_apply_row ; Continue rows
    
    RET

; Apply pulse effect to attributes
apply_pulse_effect:
    ; Load pulse effect structure
    LD HL, pulse_effect_data
    LD A, (HL)          ; Load pulse phase
    INC HL
    LD B, (HL)          ; Load pulse frequency
    INC HL
    LD C, (HL)          ; Load pulse intensity
    
    ; Update pulse phase
    ADD A, B            ; Add frequency to phase
    AND $3F             ; Wrap phase
    LD (pulse_effect_data), A ; Store updated phase
    
    ; Calculate pulse brightness
    LD D, A             ; Copy phase
    SRL D               ; Divide by 2
    SRL D               ; Divide by 4
    SRL D               ; Divide by 8 (0-7 range)
    
    ; Apply to center magic zone
    LD HL, $5800 + (12*32) + 14 ; Center zone
    LD A, D             ; Load brightness
    ADD A, $40          ; Add base color + bright
    
    LD B, $02           ; Zone height
pulse_apply_row:
    LD C, $05           ; Zone width
pulse_apply_col:
    LD (HL), A          ; Apply pulse color
    INC HL              ; Next position
    DEC C               ; Decrement width
    JR NZ, pulse_apply_col ; Continue column
    
    LD IX, $001B        ; Offset to next row
    ADD HL, IX          ; Move to next row
    DEC B               ; Decrement height
    JR NZ, pulse_apply_row ; Continue rows
    
    RET

; Apply spiral effect to attributes
apply_spiral_effect:
    ; Load spiral effect structure
    LD HL, spiral_effect_data
    LD A, (HL)          ; Load spiral angle
    INC HL
    LD B, (HL)          ; Load center X
    INC HL
    LD C, (HL)          ; Load center Y
    INC HL
    LD D, (HL)          ; Load rotation speed
    
    ; Update spiral angle
    ADD A, D            ; Add rotation speed
    AND $0F             ; Wrap angle (0-15)
    LD (spiral_effect_data), A ; Store updated angle
    
    ; Apply spiral pattern to bottom zone
    LD E, A             ; Use angle for pattern
    LD HL, $5800 + (18*32) + 25 ; Bottom zone
    
    ; Create spiral color pattern
    LD A, E             ; Load angle
    SRL A               ; Divide by 2
    ADD A, $41          ; Add base color (bright blue)
    
    LD B, $03           ; Zone height
spiral_apply_row:
    LD C, $03           ; Zone width
spiral_apply_col:
    ; Vary color based on position in spiral
    PUSH AF             ; Save color
    LD A, E             ; Load angle
    ADD A, B            ; Add row offset
    ADD A, C            ; Add column offset
    AND $07             ; Mask to color range
    ADD A, $40          ; Add bright bit
    LD (HL), A          ; Apply spiral color
    POP AF              ; Restore color
    
    INC HL              ; Next position
    DEC C               ; Decrement width
    JR NZ, spiral_apply_col ; Continue column
    
    LD IX, $001D        ; Offset to next row
    ADD HL, IX          ; Move to next row
    DEC B               ; Decrement height
    JR NZ, spiral_apply_row ; Continue rows
    
    RET

; Update color cycling system
update_color_cycling:
    ; Load cycle position
    LD A, (cycle_position) ; Load current position
    INC A                  ; Advance position
    LD B, A                ; Save for comparison
    LD A, (cycle_length)   ; Load cycle length
    CP B                   ; Compare with position
    JR NZ, store_cycle_position ; Store if within cycle
    
    ; Reset cycle
    LD B, $00              ; Reset to beginning
    
store_cycle_position:
    LD A, B                ; Load position
    LD (cycle_position), A ; Store updated position
    
    ; Apply cycling to maze borders
    CALL cycle_border_colors
    
    ; Apply cycling to navigation trails
    CALL cycle_trail_colors
    
    RET

; Cycle border colors around the maze
cycle_border_colors:
    ; Load current cycle position
    LD A, (cycle_position)    ; Load position
    LD HL, color_cycle_table  ; Point to color table
    LD B, $00                 ; Clear high byte
    LD C, A                   ; Use position as offset
    ADD HL, BC                ; Point to current color
    LD A, (HL)                ; Load cycle color
    
    ; Apply to top border
    LD HL, $5800              ; Top row of attributes
    LD B, $20                 ; 32 columns
apply_top_border:
    LD (HL), A                ; Set border color
    INC HL                    ; Next position
    DJNZ apply_top_border     ; Continue until row complete
    
    ; Apply to bottom border
    LD HL, $5800 + (23*32)    ; Bottom row of attributes
    LD B, $20                 ; 32 columns
apply_bottom_border:
    LD (HL), A                ; Set border color
    INC HL                    ; Next position
    DJNZ apply_bottom_border  ; Continue until row complete
    
    ; Apply to left border
    LD HL, $5800              ; Start of attributes
    LD B, $18                 ; 24 rows
apply_left_border:
    LD (HL), A                ; Set border color
    LD DE, $0020              ; Offset to next row
    ADD HL, DE                ; Move to next row
    DJNZ apply_left_border    ; Continue until column complete
    
    ; Apply to right border
    LD HL, $5800 + $001F      ; Right edge of attributes
    LD B, $18                 ; 24 rows
apply_right_border:
    LD (HL), A                ; Set border color
    LD DE, $0020              ; Offset to next row
    ADD HL, DE                ; Move to next row
    DJNZ apply_right_border   ; Continue until column complete
    
    RET

; Cycle trail colors behind navigator
cycle_trail_colors:
    ; Load current cycle position
    LD A, (cycle_position)    ; Load position
    LD HL, trail_color_table  ; Point to trail colors
    LD B, $00                 ; Clear high byte
    LD C, A                   ; Use position as offset
    ADD HL, BC                ; Point to current trail color
    LD A, (HL)                ; Load trail color
    
    ; Apply trail effect around navigator
    LD BC, (navigator_x)      ; Load navigator position
    
    ; Calculate attribute position
    LD A, C                   ; Load Y position
    ADD A, A                  ; Multiply by 2
    ADD A, A                  ; Multiply by 4
    ADD A, A                  ; Multiply by 8
    ADD A, A                  ; Multiply by 16
    ADD A, A                  ; Multiply by 32
    LD H, A                   ; High byte of offset
    LD L, B                   ; X position as low byte
    LD BC, $5800              ; Attribute memory base
    ADD HL, BC                ; Final attribute position
    
    ; Apply trail color around navigator position
    LD A, (trail_color_table) ; Load trail color
    LD (HL), A                ; Set navigator trail
    
    ; Create fading trail effect
    DEC HL                    ; Left of navigator
    LD A, (trail_color_table+1) ; Fading color
    LD (HL), A                ; Set fading trail
    
    INC HL                    ; Back to navigator
    INC HL                    ; Right of navigator
    LD A, (trail_color_table+1) ; Fading color
    LD (HL), A                ; Set fading trail
    
    RET

; Check magic interactions with navigator
check_magic_interactions:
    ; Load navigator position
    LD BC, (navigator_x)      ; Load position
    
    ; Check if navigator is in magic zone 1
    LD A, B                   ; Load X position
    CP $06                    ; Check left boundary
    JR C, check_zone_2        ; Skip if too far left
    CP $0A                    ; Check right boundary
    JR NC, check_zone_2       ; Skip if too far right
    
    LD A, C                   ; Load Y position
    CP $04                    ; Check top boundary
    JR C, check_zone_2        ; Skip if too far up
    CP $07                    ; Check bottom boundary
    JR NC, check_zone_2       ; Skip if too far down
    
    ; Navigator is in magic zone 1!
    CALL trigger_magic_effect_1
    JR magic_interaction_done
    
check_zone_2:
    ; Check if navigator is in magic zone 2
    LD A, B                   ; Load X position
    CP $0E                    ; Check left boundary
    JR C, check_zone_3        ; Skip if too far left
    CP $13                    ; Check right boundary
    JR NC, check_zone_3       ; Skip if too far right
    
    LD A, C                   ; Load Y position
    CP $0C                    ; Check top boundary
    JR C, check_zone_3        ; Skip if too far up
    CP $0E                    ; Check bottom boundary
    JR NC, check_zone_3       ; Skip if too far down
    
    ; Navigator is in magic zone 2!
    CALL trigger_magic_effect_2
    JR magic_interaction_done
    
check_zone_3:
    ; Check if navigator is in magic zone 3
    LD A, B                   ; Load X position
    CP $19                    ; Check left boundary
    JR C, magic_interaction_done ; Skip if too far left
    CP $1C                    ; Check right boundary
    JR NC, magic_interaction_done ; Skip if too far right
    
    LD A, C                   ; Load Y position
    CP $12                    ; Check top boundary
    JR C, magic_interaction_done ; Skip if too far up
    CP $15                    ; Check bottom boundary
    JR NC, magic_interaction_done ; Skip if too far down
    
    ; Navigator is in magic zone 3!
    CALL trigger_magic_effect_3
    
magic_interaction_done:
    RET

; Trigger magic effect 1 (teleportation boost)
trigger_magic_effect_1:
    ; Flash border with teleport colors
    LD A, $05                 ; Magenta border
    OUT ($FE), A              ; Set border
    
    ; Brief delay
    LD BC, $0100              ; Delay counter
teleport_delay:
    DEC BC                    ; Decrement counter
    LD A, B                   ; Check if BC = 0
    OR C
    JR NZ, teleport_delay     ; Continue until zero
    
    ; Restore border
    LD A, $00                 ; Black border
    OUT ($FE), A              ; Restore border
    
    ; Grant teleportation ability (simplified effect)
    LD A, $10                 ; Teleport range
    LD (teleport_range), A    ; Grant teleport ability
    
    RET

    OUT ($FE), A              ; Set border
    
    ; Brief delay
    LD BC, $0150              ; Delay counter
vision_delay:

    LD A, B                   ; Check if BC = 0
    OR C
    JR NZ, vision_delay       ; Continue until zero
    
    ; Restore border

    ; Brief delay
    LD BC, $0200              ; Delay counter
power_delay:

; Update all visual effects
update_visual_effects:
    ; Update magic particle effects
    CALL update_magic_particles
    
    ; Update color transition effects
    CALL update_color_transitions
    
    ; Update dynamic brightness effects
    CALL update_brightness_effects

; Update magic particle effects
update_magic_particles:
    ; Create sparkling particle effect in magic zones
    LD A, (magic_counter)     ; Load magic timing
    AND $03                   ; Mask to 4 states
    JR Z, create_particle     ; Create particle every 4th frame
    RET                       ; Skip this frame
    
create_particle:
    ; Generate random position in magic zones
    LD A, (magic_counter)     ; Use counter as pseudo-random
    AND $1F                   ; Mask to screen width
    LD B, A                   ; X position
    
    LD A, (magic_counter)     ; Use counter again
    RLA                       ; Rotate for variation
    AND $17                   ; Mask to screen height
    LD C, A                   ; Y position
    
    ; Calculate attribute position
    LD A, C                   ; Load Y position
    ADD A, A                  ; Multiply by 2
    ADD A, A                  ; Multiply by 4
    ADD A, A                  ; Multiply by 8
    ADD A, A                  ; Multiply by 16
    ADD A, A                  ; Multiply by 32
    LD H, A                   ; High byte
    LD L, B                   ; X position
    LD BC, $5800              ; Attribute base
    ADD HL, BC                ; Final position
    
    ; Set sparkling attribute
    LD A, $47                 ; Bright white on black
    LD (HL), A                ; Create sparkle
    
    RET

; Update color transition effects
update_color_transitions:
    ; Smooth color transitions between zones
    LD HL, transition_zones   ; Point to transition data
    LD B, $05                 ; Number of transition zones
    
transition_loop:
    PUSH BC                   ; Save counter
    
    ; Load transition zone data
    LD A, (HL)                ; Load zone X
    INC HL
    LD B, A                   ; Store X
    LD A, (HL)                ; Load zone Y
    INC HL
    LD C, A                   ; Store Y
    LD A, (HL)                ; Load transition color
    INC HL
    
    ; Calculate attribute position
    LD D, A                   ; Save color
    LD A, C                   ; Load Y
    ADD A, A                  ; Multiply by 32
    ADD A, A
    ADD A, A
    ADD A, A
    ADD A, A
    LD E, A                   ; High byte
    LD A, B                   ; Load X
    ADD A, E                  ; Add offset
    LD E, A                   ; Final offset
    
    ; Apply transition color
    LD BC, $5800              ; Attribute base
    ADD BC, DE                ; Add offset (using DE as 16-bit)
    EX DE, HL                 ; Exchange for memory access
    LD (HL), D                ; Set transition color (D has color)
    EX DE, HL                 ; Exchange back
    
    POP BC                    ; Restore counter
    DJNZ transition_loop      ; Continue until all zones processed
    
    RET

; Update dynamic brightness effects
update_brightness_effects:
    ; Create pulsing brightness effect
    LD A, (magic_counter)     ; Load timing
    AND $1F                   ; Mask to 32 values
    CP $10                    ; Check midpoint
    JR C, dimming_phase       ; First half - dimming
    
brightening_phase:
    ; Second half - brightening
    LD A, $08                 ; Bright bit
    JR apply_brightness
    
dimming_phase:
    ; First half - dimming
    LD A, $00                 ; No bright bit
    
apply_brightness:
    ; Apply brightness to specific areas
    LD B, A                   ; Save brightness
    LD HL, $5800 + (8*32) + 12 ; Center area
    LD A, $47                 ; Base color
    OR B                      ; Add brightness
    LD (HL), A                ; Apply brightness
    
    RET

; Reveal hidden paths (for enhanced vision)
reveal_hidden_paths:
    ; Show secret passages in different colors
    LD HL, $5800 + (6*32) + 20  ; Hidden path 1
    LD A, $44                    ; Bright green
    LD (HL), A                   ; Reveal path
    
    LD HL, $5800 + (14*32) + 8   ; Hidden path 2
    LD A, $44                    ; Bright green
    LD (HL), A                   ; Reveal path
    
    LD HL, $5800 + (20*32) + 15  ; Hidden path 3
    LD A, $44                    ; Bright green
    LD (HL), A                   ; Reveal path
    
    RET

; Enhance navigator abilities (for power surge)
enhance_navigator:
    ; Increase navigator movement speed (visual feedback)
    LD HL, $5800 + (navigator_y*32) + navigator_x ; Navigator position
    LD A, $4F                    ; Bright white (powered up)
    LD (HL), A                   ; Show enhancement
    
    RET

; Enhanced navigator movement from previous lessons
read_navigator_input:
    ; Read keyboard input
    LD A, $FD           ; Select keyboard row
    IN A, ($FE)         ; Read keyboard port
    
    ; Check for movement keys (simplified)
    BIT 0, A            ; Check key state
    JR NZ, no_input     ; No key pressed
    
    ; Process movement
    CALL move_navigator_right
    
no_input:
    RET

move_navigator_right:
    LD A, (navigator_x) ; Load current X
    INC A               ; Move right
    CP 31               ; Check boundary
    JR NC, no_move      ; Don't move if at edge
    LD (navigator_x), A ; Store new position
no_move:
    RET

update_navigator_position:
    ; Load navigator position
    LD BC, (navigator_x) ; Load X,Y position
    
    ; Calculate screen position
    CALL calc_navigator_screen_position
    
    ; Place navigator character
    LD A, $81           ; Navigator character
    LD (HL), A          ; Place navigator
    
    RET

calc_navigator_screen_position:
    LD A, C             ; Load Y position
    ADD A, A            ; Multiply by 32
    ADD A, A
    ADD A, A
    ADD A, A
    ADD A, A
    LD H, A             ; High byte
    LD L, B             ; X position as low byte
    LD BC, $4000        ; Screen base
    ADD HL, BC          ; Final screen position
    RET

; Enhanced screen clearing from previous lessons
clear_screen:
    ; Clear character memory
    LD HL, $4000        ; Character memory start
    LD BC, $1800        ; 6144 bytes
    LD A, $00           ; Clear character
    
clear_char_loop:
    LD (HL), A          ; Clear character
    INC HL              ; Next position
    DEC BC              ; Decrement counter
    LD A, B             ; Check if BC = 0
    OR C
    JR NZ, clear_char_loop
```
